/**
 * 复制以下代码到浏览器控制台中运行
 * 立即可用的定时器调试工具
 */

// 创建立即可用的调试工具
(function() {
  
class InstantTimerDebugger {
  constructor() {
    this.API_BASE = '/api/v1';
    this.log('🔧 立即调试工具已加载', 'success');
    this.log('现在可以使用 window.timerDebug 进行调试');
  }

  log(message, type = 'info') {
    const styles = {
      info: 'color: #2196F3',
      success: 'color: #4CAF50; font-weight: bold',
      error: 'color: #F44336; font-weight: bold',
      warning: 'color: #FF9800; font-weight: bold'
    };
    console.log(`%c[定时器调试] ${message}`, styles[type] || styles.info);
  }

  // 检查认证状态
  checkAuth() {
    this.log('=== 检查认证状态 ===');
    
    const token = localStorage.getItem('token');
    if (token) {
      this.log('✅ 找到认证token', 'success');
      try {
        const payload = JSON.parse(atob(token.split('.')[1]));
        const now = Date.now() / 1000;
        if (payload.exp && payload.exp > now) {
          this.log('✅ Token有效', 'success');
          return true;
        } else {
          this.log('❌ Token已过期', 'error');
          return false;
        }
      } catch (e) {
        this.log('⚠️ 无法解析token', 'warning');
        return false;
      }
    } else {
      this.log('❌ 未找到认证token', 'error');
      this.log('建议：先登录应用或运行 window.timerDebug.quickLogin()');
      return false;
    }
  }

  // 检查定时器状态
  async checkTimer() {
    this.log('=== 检查定时器状态 ===');
    
    if (!this.checkAuth()) {
      return false;
    }

    const token = localStorage.getItem('token');
    try {
      const response = await fetch(`${this.API_BASE}/timer/current`, {
        headers: { 'Authorization': `Bearer ${token}` }
      });

      if (response.ok) {
        const data = await response.json();
        this.log(`定时器状态: ${JSON.stringify(data, null, 2)}`);
        
        if (data.is_running) {
          this.log(`✅ 定时器运行中: ${data.task_title} (${data.formatted_time})`, 'success');
        } else {
          this.log('⏹️ 定时器未运行');
        }
        
        return data;
      } else {
        this.log(`❌ API请求失败: ${response.status}`, 'error');
        return false;
      }
    } catch (error) {
      this.log(`❌ 检查定时器失败: ${error.message}`, 'error');
      return false;
    }
  }

  // 检查React组件状态
  checkComponents() {
    this.log('=== 检查React组件状态 ===');
    
    // 检查React是否加载
    if (typeof React !== 'undefined') {
      this.log(`✅ React版本: ${React.version}`, 'success');
    } else {
      this.log('❌ React未检测到', 'error');
    }

    // 检查根元素
    const root = document.getElementById('root');
    if (root) {
      this.log('✅ React根元素存在', 'success');
      this.log(`根元素内容长度: ${root.innerHTML.length}`);
    } else {
      this.log('❌ React根元素不存在', 'error');
    }

    // 检查定时器相关组件
    const timerElements = {
      'FloatingTimer': document.querySelectorAll('.floating-timer'),
      'TimerCard': document.querySelectorAll('.timer-card'),
      'TimerButton': document.querySelectorAll('[class*="timer"]'),
      'LoadingSpinner': document.querySelectorAll('.ant-spin')
    };

    Object.entries(timerElements).forEach(([name, elements]) => {
      this.log(`${name}: 找到 ${elements.length} 个元素`);
    });

    // 检查localStorage状态
    try {
      const timerState = localStorage.getItem('globalTimerState');
      if (timerState) {
        const parsed = JSON.parse(timerState);
        this.log('✅ localStorage定时器状态:', 'success');
        this.log(JSON.stringify(parsed, null, 2));
      } else {
        this.log('⚠️ localStorage中无定时器状态', 'warning');
      }
    } catch (error) {
      this.log('❌ localStorage读取失败', 'error');
    }

    // 检查控制台错误
    this.log('请检查浏览器控制台是否有其他错误信息');
  }

  // 尝试修复常见问题
  async fixIssues() {
    this.log('=== 尝试修复常见问题 ===');
    
    // 1. 清理localStorage
    this.log('清理localStorage...');
    localStorage.removeItem('globalTimerState');
    localStorage.removeItem('floatingTimerPosition');
    localStorage.removeItem('floatingTimerMinimized');
    localStorage.removeItem('floatingTimerHidden');
    this.log('✅ localStorage已清理', 'success');

    // 2. 重新获取定时器状态
    this.log('重新获取定时器状态...');
    await this.checkTimer();

    // 3. 触发React重新渲染
    this.log('尝试触发组件重新渲染...');
    window.dispatchEvent(new Event('storage'));
    this.log('✅ 已触发storage事件', 'success');

    this.log('🎉 修复完成！请观察应用是否正常工作');
  }

  // 快速登录
  async quickLogin() {
    this.log('尝试快速登录...');
    try {
      const response = await fetch(`${this.API_BASE}/auth/login`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username: 'admin', password: 'password123' })
      });

      const data = await response.json();
      if (data.success && data.data && data.data.token) {
        localStorage.setItem('token', data.data.token);
        this.log('✅ 快速登录成功', 'success');
        window.location.reload();
        return true;
      } else {
        this.log('❌ 快速登录失败', 'error');
        return false;
      }
    } catch (error) {
      this.log(`❌ 登录错误: ${error.message}`, 'error');
      return false;
    }
  }

  // 运行完整诊断
  async runDiagnostics() {
    this.log('🚀 开始运行完整诊断...', 'success');
    
    this.checkAuth();
    await this.checkTimer();
    this.checkComponents();
    
    this.log('🎉 诊断完成！', 'success');
    this.log('如果发现问题，运行 window.timerDebug.fixIssues() 尝试修复');
  }

  // 获取帮助信息
  help() {
    this.log('=== 帮助信息 ===');
    this.log('可用方法:');
    this.log('• runDiagnostics() - 运行完整诊断');
    this.log('• checkTimer() - 检查定时器状态');
    this.log('• checkComponents() - 检查React组件');
    this.log('• checkAuth() - 检查认证状态');
    this.log('• fixIssues() - 修复常见问题');
    this.log('• quickLogin() - 快速登录');
    this.log('• help() - 显示帮助');
  }
}

// 在全局暴露调试工具
window.timerDebug = new InstantTimerDebugger();
window.timerDiag = window.timerDebug; // 兼容旧调用

})();
