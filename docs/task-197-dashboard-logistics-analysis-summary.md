```typescript
interface LogisticsStats {
  consignmentStats: ConsignmentStats;
  waybillStats: WaybillStats;
  inventoryStats: InventoryStats;
  deliveryStats: DeliveryStats;
}

interface LogisticsStatus {
  total: number;
  inTransit: number;
  arrived: number;
  avgDeliveryTime: number;
  avgDistance: number;
  activeCarriers: number;
  onlineWarehouses: number;
}

// 趋势和时段数据类型
interface TrendData {
  month: string;
  value: number;
}

interface HourlyData {
  hour: string;
  consignments: number;
  deliveries: number;
}

interface RankingData {
  name: string;
  value: string;
}
```
