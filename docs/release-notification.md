```javascript
// 监控配置示例
const errorThresholds = {
  warning: 0.5,  // 0.5%
  critical: 1.0, // 1.0%
  emergency: 5.0 // 5.0%
};

// 告警规则
if (errorRate > errorThresholds.emergency) {
  // 立即回滚
  alert('EMERGENCY: 错误率超过5%,立即执行回滚!');
  executeRollback();
} else if (errorRate > errorThresholds.critical) {
  // 紧急通知
  alert('CRITICAL: 错误率超过1%,请立即处理!');
  notifyTeam(['tech-lead', 'on-call']);
} else if (errorRate > errorThresholds.warning) {
  // 警告通知
  alert('WARNING: 错误率超过0.5%,请关注');
  notifyTeam(['dev-team']);
}
```
