🔧 手动测试 - 批量操作修复验证

⚠️ CSP限制说明

由于Content Security Policy限制，我们采用手动复制粘贴的方式进行测试。

不要担心，这种方式同样有效且更安全！

1打开项目页面

🚀 打开项目页面

在新标签页中打开项目页面，然后继续下面的步骤。

2选择任务

在项目页面的任务列表中，勾选任意一个任务的复选框。勾选后，批量操作区域将出现。

3打开浏览器控制台

在项目页面按 F12 打开开发者工具，点击 Console 标签。

4运行测试脚本 A - 基础检查

复制下面的脚本，粘贴到控制台中按回车运行：

console.log('🔧 === 批量操作按钮基础检查 ==='); // 查找批量操作按钮 const batchButton = Array.from(document.querySelectorAll('button')).find(btn => btn.textContent && btn.textContent.trim() === '批量操作' ); if (batchButton) { console.log('✅ 找到批量操作按钮!'); console.log(' 按钮文本:', '"' + batchButton.textContent + '"'); console.log(' 按钮尺寸:', batchButton.offsetWidth + 'x' + batchButton.offsetHeight + 'px'); console.log(' 按钮可见:', batchButton.offsetParent !== null); // 检查是否有下拉箭头 const hasDownIcon = batchButton.innerHTML.includes('down') || batchButton.innerHTML.includes('Down') || batchButton.innerHTML.includes('anticon'); console.log(' 包含下拉箭头:', hasDownIcon ? '✅ 是' : '❌ 否'); console.log('🎯 基础检查通过！请继续运行脚本B'); } else { console.log('❌ 未找到"批量操作"按钮'); // 检查是否还有分离的按钮 const separateButtons = Array.from(document.querySelectorAll('button')).filter(btn => { const text = btn.textContent || ''; return text.includes('批量更新状态') || text.includes('批量设置优先级') || text.includes('批量删除'); }); if (separateButtons.length > 0) { console.log('❌ 发现分离的批量操作按钮 - 修复未生效:'); separateButtons.forEach((btn, index) => { console.log(' ' + (index + 1) + '. "' + btn.textContent + '" (' + btn.offsetWidth + 'x' + btn.offsetHeight + 'px)'); }); console.log('💡 修复可能未正确应用'); } else { console.log('🔍 显示所有可见按钮:'); const allButtons = Array.from(document.querySelectorAll('button')); const visibleButtons = allButtons.filter(btn => btn.offsetParent !== null); console.log('页面共有' + visibleButtons.length + '个可见按钮'); visibleButtons.slice(0, 10).forEach((btn, index) => { const text = btn.textContent ? btn.textContent.trim() : '[无文本]'; console.log(' ' + (index + 1) + '. "' + text + '"'); }); } }

5运行测试脚本 B - 功能测试

如果脚本A显示找到了批量操作按钮，请运行这个脚本测试点击功能：

console.log('🖱️ === 批量操作按钮功能测试 ==='); // 重新获取按钮引用 const batchButton = Array.from(document.querySelectorAll('button')).find(btn => btn.textContent && btn.textContent.trim() === '批量操作' ); if (batchButton) { console.log('🎯 开始点击测试...'); // 点击按钮 batchButton.click(); // 等待下拉菜单出现 setTimeout(() => { const dropdown = document.querySelector('.ant-dropdown:not(.ant-dropdown-hidden)'); if (dropdown) { console.log('✅ 下拉菜单成功出现!'); const menuItems = dropdown.querySelectorAll('.ant-dropdown-menu-item'); console.log(' 菜单项数量:', menuItems.length); console.log(' 菜单项列表:'); menuItems.forEach((item, index) => { console.log(' ' + (index + 1) + '. ' + item.textContent.trim()); }); // 验证预期菜单项 const expectedItems = ['批量更新状态', '批量设置优先级', '更改父任务', '批量删除']; const actualItems = Array.from(menuItems).map(item => item.textContent.trim()); console.log('\\n📋 菜单项验证:'); let allFound = true; expectedItems.forEach(expected => { const found = actualItems.some(actual => actual.includes(expected)); console.log(' ' + expected + ':', found ? '✅ 存在' : '❌ 缺失'); if (!found) allFound = false; }); // 关闭菜单 document.body.click(); // 最终结果 console.log('\\n🎉 === 最终测试结果 ==='); if (allFound && menuItems.length >= 4) { console.log('🎊 批量操作按钮修复完全成功！'); console.log('✅ 单一下拉按钮: 正常工作'); console.log('✅ 菜单功能: 完整可用'); console.log('✅ 所有菜单项: 全部存在'); } else { console.log('⚠️ 修复部分成功，存在细节问题'); console.log('菜单完整性:', allFound ? '✅' : '❌'); console.log('菜单项数量:', menuItems.length + '/4'); } } else { console.log('❌ 下拉菜单未出现'); console.log('🔍 检查页面其他dropdown元素...'); const allDropdowns = document.querySelectorAll('.ant-dropdown'); console.log('找到' + allDropdowns.length + '个dropdown元素'); console.log('💡 可能的问题: 点击事件处理或菜单定位异常'); } }, 800); } else { console.log('❌ 未找到批量操作按钮，无法进行功能测试'); console.log('💡 请先运行脚本A确认按钮存在'); }
