document.querySelectorAll('*').forEach(el => { if (el.textContent && el.textContent.includes('批量')) { console.log('找到批量元素:', el.tagName, el.textContent.substring(0, 50)); } });

🔍 调试命令3：选择任务测试

const checkbox = document.querySelector('.ant-checkbox-input'); if (checkbox) { checkbox.click(); console.log('✅ 已选择第一个任务'); setTimeout(() => { document.querySelectorAll('*').forEach(el => { if (el.textContent && (el.textContent.includes('批量操作') || el.textContent.includes('已选择'))) { console.log('选择后找到:', el.textContent.trim()); } }); checkbox.click(); // 取消选择 }, 1000); } else { console.log('❌ 未找到复选框'); }

📊 期望的调试结果

正常情况下应该看到：

  • 复选框数量 > 0
  • 选择任务后出现"已选择 1 项"文本
  • 选择任务后出现"批量操作"按钮或相关元素

如果出现问题：

  • 复选框数量 = 0 → 表格渲染问题
  • 选择后无"已选择"文本 → React状态问题
  • 有"已选择"但无"批量操作" → 条件渲染问题
