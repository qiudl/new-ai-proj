🔍 批量操作菜单调试工具

专门用于调试批量操作下拉菜单中"更改父任务"选项缺失问题

🎯 问题描述

用户反馈：只看到批量删除按钮，没看到批量操作按钮

预期行为：应该显示带下拉菜单的"批量操作"按钮，包含"更改父任务"和"批量删除"两个选项

🔗 快速访问

📋 调试步骤

  1. 首先在新标签页打开项目任务列表
  2. 选择一个或多个任务（勾选复选框）
  3. 返回这个页面，点击"开始菜单调试"
  4. 查看调试结果，找出问题原因

🛠️ 修复建议

🧪 手动检查代码

请在项目任务列表页面的Console中运行以下代码：

// 1. 检查是否有任务被选中
const selectedCheckboxes = document.querySelectorAll('input[type="checkbox"]:checked');
console.log('选中的任务数量:', selectedCheckboxes.length);

// 2. 查找批量操作相关元素
const batchButton = Array.from(document.querySelectorAll('button')).find(btn => 
    btn.textContent.includes('批量操作')
);
console.log('批量操作按钮:', batchButton);

// 3. 检查下拉菜单
const dropdownTrigger = document.querySelector('.ant-dropdown-trigger');
console.log('下拉菜单触发器:', dropdownTrigger);

// 4. 如果找到批量操作按钮，点击它查看菜单
if (batchButton) {
    console.log('找到批量操作按钮，尝试点击...');
    batchButton.click();
    
    // 等待菜单出现
    setTimeout(() => {
        const menuItems = document.querySelectorAll('.ant-dropdown-menu-item');
        console.log('菜单项数量:', menuItems.length);
        console.log('菜单项内容:');
        menuItems.forEach((item, index) => {
            console.log(`  ${index + 1}. ${item.textContent}`);
        });
        
        // 查找"更改父任务"选项
        const changeParentItem = Array.from(menuItems).find(item => 
            item.textContent.includes('更改父任务')
        );
        console.log('更改父任务选项:', changeParentItem ? '✅ 找到' : '❌ 未找到');
        
        // 再次点击关闭菜单
        batchButton.click();
    }, 500);
} else {
    console.log('❌ 未找到批量操作按钮');
    
    // 检查是否有其他批量操作相关按钮
    const allButtons = Array.from(document.querySelectorAll('button'));
    const batchRelatedButtons = allButtons.filter(btn => 
        btn.textContent.includes('批量') || 
        btn.textContent.includes('删除') ||
        btn.textContent.includes('选择')
    );
    
    console.log('批量相关按钮:');
    batchRelatedButtons.forEach(btn => {
        console.log(`  - "${btn.textContent}" (类名: ${btn.className})`);
    });
}

// 5. 检查React组件props (需要React DevTools)
if (window.__REACT_DEVTOOLS_GLOBAL_HOOK__) {
    console.log('React DevTools可用，可以检查组件props');
} else {
    console.log('React DevTools不可用');
}
