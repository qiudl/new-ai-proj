📋 步骤1: 获取批量操作按钮引用

console.log('=== 获取批量操作按钮 ==='); // 先选择任务 const checkbox = document.querySelector('.ant-table-selection-column .ant-checkbox-input'); if (checkbox && !checkbox.checked) { console.log('选择任务...'); checkbox.click(); } // 等待React状态更新 setTimeout(() => { // 获取批量操作按钮 window.batchButton = Array.from(document.querySelectorAll('button')).find(btn => { // 检查按钮或其子元素是否包含"批量"或"操作" const text = btn.textContent || ''; const innerHTML = btn.innerHTML || ''; return text.includes('批量') || text.includes('操作') || innerHTML.includes('批量') || innerHTML.includes('操作'); }); if (window.batchButton) { console.log('✅ 找到批量操作按钮:', window.batchButton); console.log('按钮textContent:', `"${window.batchButton.textContent}"`); console.log('按钮innerHTML:', window.batchButton.innerHTML); } else { console.log('❌ 未找到批量操作按钮'); // 尝试通过其他方式查找 const allButtons = Array.from(document.querySelectorAll('button')); console.log('所有按钮:', allButtons.map(btn => ({ text: btn.textContent, html: btn.innerHTML, width: btn.offsetWidth }))); } }, 1500);

🔍 步骤2: 深度分析按钮结构

console.log('=== 深度分析按钮结构 ==='); if (window.batchButton) { console.log('按钮基本信息:'); console.log(' - tagName:', window.batchButton.tagName); console.log(' - className:', window.batchButton.className); console.log(' - id:', window.batchButton.id); console.log(' - textContent:', `"${window.batchButton.textContent}"`); console.log(' - innerText:', `"${window.batchButton.innerText}"`); console.log(' - innerHTML:', window.batchButton.innerHTML); console.log(' - outerHTML:', window.batchButton.outerHTML); console.log('\n子元素分析:'); console.log(' - children.length:', window.batchButton.children.length); console.log(' - childNodes.length:', window.batchButton.childNodes.length); // 分析每个子节点 Array.from(window.batchButton.childNodes).forEach((node, index) => { console.log(` 子节点${index + 1}:`, { nodeType: node.nodeType, nodeName: node.nodeName, textContent: node.textContent, innerHTML: node.innerHTML || '(no innerHTML)', className: node.className || '(no className)' }); }); } else { console.log('❌ 没有找到批量操作按钮，无法进行结构分析'); }

🔍 步骤3: 查找Dropdown相关元素

console.log('=== 查找Dropdown相关元素 ==='); // 查找所有可能的Dropdown相关按钮 const potentialButtons = Array.from(document.querySelectorAll('button')).filter(btn => { const width = btn.offsetWidth; const height = btn.offsetHeight; // 查找宽度在100-130px之间，高度在20-30px之间的按钮 return width > 100 && width < 130 && height > 20 && height < 30; }); console.log('符合尺寸条件的按钮数量:', potentialButtons.length); potentialButtons.forEach((btn, index) => { console.log(`候选按钮${index + 1}:`); console.log(' - 尺寸:', `${btn.offsetWidth}x${btn.offsetHeight}px`); console.log(' - 文本:', `"${btn.textContent}"`); console.log(' - HTML:', btn.innerHTML); console.log(' - 类名:', btn.className); // 检查是否有下拉箭头图标 const hasDownIcon = btn.innerHTML.includes('down') || btn.innerHTML.includes('Down') || btn.innerHTML.includes('anticon'); console.log(' - 有下拉图标:', hasDownIcon); if (hasDownIcon) { console.log('🎯 这可能就是我们要找的批量操作按钮!'); window.suspectButton = btn; } });

🔍 步骤4: 分析Ant Design组件渲染

console.log('=== 分析Ant Design组件渲染 ==='); // 查找所有包含anticon的元素（Ant Design图标） const anticons = document.querySelectorAll('[class*="anticon"]'); console.log('页面上的Ant Design图标数量:', anticons.length); // 查找下拉箭头图标 const downIcons = Array.from(anticons).filter(icon => icon.className.includes('down') || icon.className.includes('Down') ); console.log('下拉箭头图标数量:', downIcons.length); downIcons.forEach((icon, index) => { console.log(`下拉图标${index + 1}:`); console.log(' - 类名:', icon.className); console.log(' - 父元素:', icon.parentElement.tagName); console.log(' - 父元素文本:', `"${icon.parentElement.textContent}"`); console.log(' - 父元素HTML:', icon.parentElement.innerHTML); // 检查父元素是否是按钮 if (icon.parentElement.tagName === 'BUTTON') { console.log('🎯 找到包含下拉图标的按钮!'); window.iconButton = icon.parentElement; } }); // 如果找到了包含图标的按钮，进一步分析 if (window.iconButton) { console.log('\n包含图标的按钮详细分析:'); console.log(' - 完整HTML:', window.iconButton.outerHTML); console.log(' - 文本内容:', `"${window.iconButton.textContent}"`); console.log(' - 子元素数量:', window.iconButton.children.length); // 分析每个子元素 Array.from(window.iconButton.children).forEach((child, index) => { console.log(` 子元素${index + 1}:`); console.log(' - 标签:', child.tagName); console.log(' - 类名:', child.className); console.log(' - 文本:', `"${child.textContent}"`); console.log(' - HTML:', child.innerHTML); }); }

🔍 步骤5: 检查React组件props

console.log('=== 检查React组件props ==='); // 尝试获取React组件的内部状态 function getReactProps(element) { // React DevTools style property access const reactKey = Object.keys(element).find(key => key.startsWith('__reactInternalInstance') || key.startsWith('__reactFiber') ); if (reactKey) { return element[reactKey]; } return null; } // 检查我们找到的可疑按钮 const targetButton = window.iconButton || window.suspectButton || window.batchButton; if (targetButton) { console.log('检查目标按钮的React props...'); const reactData = getReactProps(targetButton); if (reactData) { console.log('React内部数据:', reactData); } else { console.log('无法获取React内部数据'); } // 尝试触发按钮点击看看会发生什么 console.log('尝试点击按钮进行测试...'); try { targetButton.click(); console.log('✅ 按钮点击成功'); } catch (error) { console.log('❌ 按钮点击失败:', error.message); } } else { console.log('❌ 没有找到可以分析的目标按钮'); }
