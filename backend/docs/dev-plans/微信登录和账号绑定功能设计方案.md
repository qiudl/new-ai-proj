# 微信登录和账号绑定功能设计方案

## 一、功能概述

为AI项目管理系统实现完整的微信生态集成，包括：
1. **微信账号绑定**：现有用户可以绑定微信账号
2. **微信扫码登录**：未注册用户可直接使用微信扫码注册并登录
3. **微信授权登录**：已绑定用户可使用微信快速登录

## 二、技术方案

### 2.1 微信开放平台接入

#### 选择接入方式
- **网站应用**：使用微信开放平台（open.weixin.qq.com）
- **移动应用**：使用微信开放平台移动应用
- **公众号**：如需在微信内使用，需接入微信公众平台

本方案以**网站应用**为主。

#### 申请流程
1. 注册微信开放平台账号
2. 创建网站应用
3. 填写应用信息（名称、简介、官网地址等）
4. 配置授权回调域名
5. 获取AppID和AppSecret

```go
type WechatService struct {
    config *config.WechatConfig
    db     *gorm.DB

// GetAuthURL 生成微信授权URL
func (s *WechatService) GetAuthURL(state string) string {
    return fmt.Sprintf(
        "https://open.weixin.qq.com/connect/qrconnect?appid=%s&redirect_uri=%s&response_type=code&scope=%s&state=%s#wechat_redirect",
        s.config.AppID,
        url.QueryEscape(s.config.RedirectURI),
        s.config.Scope,
        state,
    )
}

// GetAccessToken 通过code获取access_token
func (s *WechatService) GetAccessToken(code string) (*WechatAccessTokenResponse, error) {
    apiURL := fmt.Sprintf(
        "https://api.weixin.qq.com/sns/oauth2/access_token?appid=%s&secret=%s&code=%s&grant_type=authorization_code",
        s.config.AppID,
        s.config.AppSecret,
        code,
    )

    resp, err := http.Get(apiURL)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var result WechatAccessTokenResponse
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, err
    }

    if result.ErrCode != 0 {
        return nil, fmt.Errorf("wechat api error: %s", result.ErrMsg)
    }

    return &result, nil
}

// GetUserInfo 获取微信用户信息
func (s *WechatService) GetUserInfo(accessToken, openID string) (*WechatUserInfo, error) {
    apiURL := fmt.Sprintf(
        "https://api.weixin.qq.com/sns/userinfo?access_token=%s&openid=%s",
        accessToken,
        openID,
    )

    resp, err := http.Get(apiURL)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var userInfo WechatUserInfo
    if err := json.NewDecoder(resp.Body).Decode(&userInfo); err != nil {
        return nil, err
    }

    return &userInfo, nil
}

// BindWechat 绑定微信账号
func (s *WechatService) BindWechat(userID uint, wechatInfo *WechatUserInfo) error {
    now := time.Now()
    updates := map[string]interface{}{
        "wechat_openid":   wechatInfo.OpenID,
        "wechat_unionid":  wechatInfo.UnionID,
        "wechat_nickname": wechatInfo.Nickname,
        "wechat_avatar":   wechatInfo.HeadImgURL,
        "wechat_bound_at": now,
    }

    return s.db.Model(&models.User{}).Where("id = ?", userID).Updates(updates).Error
}

// UnbindWechat 解绑微信账号
func (s *WechatService) UnbindWechat(userID uint) error {
    updates := map[string]interface{}{
        "wechat_openid":   nil,
        "wechat_unionid":  nil,
        "wechat_nickname": nil,
        "wechat_avatar":   nil,
        "wechat_bound_at": nil,
    }

    return s.db.Model(&models.User{}).Where("id = ?", userID).Updates(updates).Error
}

// FindUserByOpenID 通过OpenID查找用户
func (s *WechatService) FindUserByOpenID(openID string) (*models.User, error) {
    var user models.User
    err := s.db.Where("wechat_openid = ?", openID).First(&user).Error
    return &user, err
}
```

#### 2.3.4 Handlers（handlers/wechat_auth_handler.go）

```go
type WechatAuthHandler struct {
    wechatService *services.WechatService
    authService   *services.AuthService
}

// GetWechatAuthURL 获取微信授权URL
// @Summary 获取微信授权URL
// @Tags 认证
// @Produce json
// @Param action query string false "操作类型: login|bind" default(login)
// @Success 200 {object} map[string]string
// @Router /api/v1/auth/wechat/url [get]
func (h *WechatAuthHandler) GetWechatAuthURL(c *gin.Context) {
    action := c.DefaultQuery("action", "login") // login or bind
    state := generateState(action) // 生成包含action的state

    // 将state存入Redis，设置5分钟过期
    // redis.Set("wechat:state:"+state, action, 5*time.Minute)

    authURL := h.wechatService.GetAuthURL(state)

    c.JSON(200, gin.H{
        "success": true,
        "data": gin.H{
            "auth_url": authURL,
            "state":    state,
        },
    })
}

// WechatCallback 微信授权回调
// @Summary 微信授权回调
// @Tags 认证
// @Produce json
// @Param code query string true "授权code"
// @Param state query string true "状态码"
// @Success 200 {object} map[string]interface{}
// @Router /api/v1/auth/wechat/callback [get]
func (h *WechatAuthHandler) WechatCallback(c *gin.Context) {
    code := c.Query("code")
    state := c.Query("state")

    if code == "" || state == "" {
        c.JSON(400, gin.H{"success": false, "error": "缺少必要参数"})
        return
    }

    // 验证state（从Redis获取并删除）
    // action, err := redis.GetDel("wechat:state:" + state)
    action := parseActionFromState(state)

    // 1. 获取access_token
    tokenResp, err := h.wechatService.GetAccessToken(code)
    if err != nil {
        c.JSON(500, gin.H{"success": false, "error": "获取访问令牌失败"})
        return
    }

    // 2. 获取用户信息
    wechatUserInfo, err := h.wechatService.GetUserInfo(tokenResp.AccessToken, tokenResp.OpenID)
    if err != nil {
        c.JSON(500, gin.H{"success": false, "error": "获取用户信息失败"})
        return
    }

    if action == "bind" {
        // 绑定操作：需要用户已登录
        userID := c.GetUint("user_id")
        if userID == 0 {
            c.JSON(401, gin.H{"success": false, "error": "未登录"})
            return
        }

        // 检查该微信是否已绑定其他账号
        existingUser, err := h.wechatService.FindUserByOpenID(wechatUserInfo.OpenID)
        if err == nil && existingUser.ID != userID {
            c.JSON(400, gin.H{"success": false, "error": "该微信已绑定其他账号"})
            return
        }

        // 执行绑定
        if err := h.wechatService.BindWechat(userID, wechatUserInfo); err != nil {
            c.JSON(500, gin.H{"success": false, "error": "绑定失败"})
            return
        }

        c.JSON(200, gin.H{
            "success": true,
            "message": "绑定成功",
        })

    } else {
        // 登录操作
        user, err := h.wechatService.FindUserByOpenID(wechatUserInfo.OpenID)

        if err != nil {
            // 用户不存在，自动注册
            user = &models.User{
                Username:       "wx_" + wechatUserInfo.OpenID[:10],
                WechatOpenID:   &wechatUserInfo.OpenID,
                WechatUnionID:  &wechatUserInfo.UnionID,
                WechatNickname: &wechatUserInfo.Nickname,
                WechatAvatar:   &wechatUserInfo.HeadImgURL,
                UserType:       "wechat",
                Status:         "active",
            }

            if err := h.wechatService.CreateWechatUser(user); err != nil {
                c.JSON(500, gin.H{"success": false, "error": "注册失败"})
                return
            }
        }

        // 生成JWT token
        token, err := h.authService.GenerateToken(user)
        if err != nil {
            c.JSON(500, gin.H{"success": false, "error": "生成令牌失败"})
            return
        }

        c.JSON(200, gin.H{
            "success": true,
            "data": gin.H{
                "token": token,
                "user":  user,
            },
        })
    }
}

// UnbindWechat 解绑微信
// @Summary 解绑微信账号
// @Tags 用户
// @Security BearerAuth
// @Produce json
// @Success 200 {object} map[string]bool
// @Router /api/v1/user/wechat/unbind [post]
func (h *WechatAuthHandler) UnbindWechat(c *gin.Context) {
    userID := c.GetUint("user_id")

    if err := h.wechatService.UnbindWechat(userID); err != nil {
        c.JSON(500, gin.H{"success": false, "error": "解绑失败"})
        return
    }

    c.JSON(200, gin.H{
        "success": true,
        "message": "解绑成功",
    })
}

// GetWechatBindStatus 获取微信绑定状态
// @Summary 获取微信绑定状态
// @Tags 用户
// @Security BearerAuth
// @Produce json
// @Success 200 {object} map[string]interface{}
// @Router /api/v1/user/wechat/status [get]
func (h *WechatAuthHandler) GetWechatBindStatus(c *gin.Context) {
    userID := c.GetUint("user_id")

    var user models.User
    if err := h.db.Select("wechat_openid, wechat_nickname, wechat_avatar, wechat_bound_at").
        Where("id = ?", userID).First(&user).Error; err != nil {
        c.JSON(500, gin.H{"success": false, "error": "查询失败"})
        return
    }

    isBound := user.WechatOpenID != nil && *user.WechatOpenID != ""

    c.JSON(200, gin.H{
        "success": true,
        "data": gin.H{
            "is_bound":       isBound,
            "wechat_nickname": user.WechatNickname,
            "wechat_avatar":   user.WechatAvatar,
            "bound_at":        user.WechatBoundAt,
        },
    })
}
```

```go
    wechat := r.Group("/auth/wechat")
```

#### 2.4.1 API Service（services/wechatAuthService.ts）

```typescript
import apiClient from './apiClient';

export interface WechatAuthURLResponse {
  auth_url: string;
  state: string;
}

export interface WechatBindStatus {
  is_bound: boolean;
  wechat_nickname?: string;
  wechat_avatar?: string;
  bound_at?: string;
}

class WechatAuthService {
  // 获取微信授权URL
  async getAuthURL(action: 'login' | 'bind' = 'login'): Promise<WechatAuthURLResponse> {
    const response = await apiClient.get('/auth/wechat/url', {
      params: { action }
    });
    return response.data;
  }

  // 获取绑定状态
  async getBindStatus(): Promise<WechatBindStatus> {
    const response = await apiClient.get('/user/wechat/status');
    return response.data;
  }

  // 解绑微信
  async unbindWechat(): Promise<void> {
    await apiClient.post('/user/wechat/unbind');
  }

  // 打开微信登录窗口
  openWechatLogin(returnUrl?: string): void {
    this.getAuthURL('login').then(data => {
      const width = 800;
      const height = 600;
      const left = (window.screen.width - width) / 2;
      const top = (window.screen.height - height) / 2;

      const popup = window.open(
        data.auth_url,
        'wechat_login',
        `width=${width},height=${height},left=${left},top=${top},toolbar=no,menubar=no,location=no`
      );

      // 监听回调
      window.addEventListener('message', (event) => {
        if (event.data.type === 'wechat_login_success') {
          popup?.close();
          // 处理登录成功
          if (returnUrl) {
            window.location.href = returnUrl;
          } else {
            window.location.reload();
          }
        }
      });
    });
  }

  // 打开微信绑定窗口
  openWechatBind(onSuccess?: () => void): void {
    this.getAuthURL('bind').then(data => {
      const width = 800;
      const height = 600;
      const left = (window.screen.width - width) / 2;
      const top = (window.screen.height - height) / 2;

      const popup = window.open(
        data.auth_url,
        'wechat_bind',
        `width=${width},height=${height},left=${left},top=${top},toolbar=no,menubar=no,location=no`
      );

      window.addEventListener('message', (event) => {
        if (event.data.type === 'wechat_bind_success') {
          popup?.close();
          onSuccess?.();
        }
      });
    });
  }
}

export default new WechatAuthService();
```

```typescript
import React, { useEffect, useState } from 'react';
import { Card, Button, Avatar, Space, message, Modal } from 'antd';
import { WechatOutlined, DisconnectOutlined } from '@ant-design/icons';
import wechatAuthService, { WechatBindStatus } from '../services/wechatAuthService';

const WechatBindCard: React.FC = () => {
  const [bindStatus, setBindStatus] = useState<WechatBindStatus | null>(null);
  const [loading, setLoading] = useState(false);

  const loadBindStatus = async () => {
    try {
      const status = await wechatAuthService.getBindStatus();
      setBindStatus(status);
    } catch (error) {
      message.error('获取绑定状态失败');
    }
  };

  useEffect(() => {
    loadBindStatus();
  }, []);

  const handleBind = () => {
    wechatAuthService.openWechatBind(() => {
      message.success('绑定成功');
      loadBindStatus();
    });
  };

  const handleUnbind = () => {
    Modal.confirm({
      title: '确认解绑',
      content: '解绑后将无法使用微信快捷登录，确定要解绑吗？',
      onOk: async () => {
        setLoading(true);
        try {
          await wechatAuthService.unbindWechat();
          message.success('解绑成功');
          loadBindStatus();
        } catch (error) {
          message.error('解绑失败');
        } finally {
          setLoading(false);
        }
      },
    });
  };

  if (!bindStatus) return null;

  return (
    <Card title="微信账号" style={{ marginTop: 16 }}>
      {bindStatus.is_bound ? (
        <Space align="center">
          <Avatar src={bindStatus.wechat_avatar} icon={<WechatOutlined />} />

            <div>{bindStatus.wechat_nickname}</div>
            <div style={{ fontSize: 12, color: '#999' }}>

            danger
            icon={<DisconnectOutlined />}
            onClick={handleUnbind}
            loading={loading}
          >
            解绑

        </Space>
      ) : (
        <div>
          <p>绑定微信后，可使用微信快捷登录</p>

            type="primary"
            icon={<WechatOutlined />}
            onClick={handleBind}
            style={{ backgroundColor: '#07c160', borderColor: '#07c160' }}
          >
            绑定微信
          </Button>
        </div>
      )}
    </Card>
  );
};
```

#### 2.4.4 回调页面（pages/WechatCallback.tsx）

```typescript
import React, { useEffect } from 'react';
import { useNavigate, useSearchParams } from 'react-router-dom';
import { Spin, Result } from 'antd';
import { useDispatch } from 'react-redux';
import { setUser, setToken } from '../store/authSlice';

const WechatCallback: React.FC = () => {
  const [searchParams] = useSearchParams();
  const navigate = useNavigate();
  const dispatch = useDispatch();
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const code = searchParams.get('code');
    const state = searchParams.get('state');

    if (!code || !state) {
      setError('缺少必要参数');
      return;
    }

    // 调用后端callback接口
    fetch(`/api/v1/auth/wechat/callback?code=${code}&state=${state}`)
      .then(res => res.json())
      .then(data => {
        if (data.success) {
          if (data.data.token) {
            // 登录成功
            dispatch(setToken(data.data.token));
            dispatch(setUser(data.data.user));

            // 通知父窗口
            if (window.opener) {
              window.opener.postMessage({ type: 'wechat_login_success' }, '*');
            } else {
              navigate('/');
            }
          } else {
            // 绑定成功
            if (window.opener) {
              window.opener.postMessage({ type: 'wechat_bind_success' }, '*');
            } else {
              navigate('/profile');
            }
          }
        } else {
          setError(data.error || '操作失败');
        }
      })
      .catch(() => {
        setError('网络错误');
      });
  }, []);

  if (error) {
    return (
      <Result
        status="error"
        title="操作失败"
        subTitle={error}
      />
    );
  }

  return (
    <div style={{
      display: 'flex',
      justifyContent: 'center',
      alignItems: 'center',
      height: '100vh'
    }}>
      <Spin size="large" tip="处理中..." />
    </div>
  );
};

export default WechatCallback;
```

## 三、安全考虑

### 3.1 CSRF防护
- 使用随机state参数
- state存储在Redis中，设置短暂过期时间
- 回调时验证state有效性

### 3.2 OpenID存储安全
- OpenID加密存储（可选）
- 定期审计绑定日志
- 限制绑定/解绑频率

### 3.3 防止恶意绑定
- 一个微信只能绑定一个账号
- 绑定前检查是否已被其他账号绑定
- 记录审计日志

### 3.4 网络安全
- 使用HTTPS
- 验证微信服务器证书
- 防止中间人攻击

## 四、用户体验优化

### 4.1 二维码展示
- 使用微信官方二维码样式
- 支持刷新二维码
- 显示扫码倒计时

### 4.2 移动端适配
- 在微信内打开自动跳转到微信授权
- 非微信浏览器显示提示信息

### 4.3 错误处理
- 友好的错误提示
- 授权失败自动重试
- 提供客服联系方式
