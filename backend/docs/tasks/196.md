  const loadParentTaskInfo = async (parentId: number) => {
    if (!projectId || parentId <= 0) return;

    try {
      setLoadingParentTask(true);
      const response = await TaskService.getTask(projectId,
  parentId);
      if (response) {
        setSelectedParentTask(response);
      }
    } catch (error) {

  error);

      setLoadingParentTask(false);

  我已经成功分析并修复了父任务选择弹窗的两个关键问题：

  问题1: 当前父任务不显示

  根本原因: 异步竞态条件
  - useEffect试图在搜索完成前从空的searchResults.tasks中找父
  任务
  - 解决方案: 分离搜索初始化和父任务选择逻辑，添加独立的effec
  t监听搜索结果

  问题2: 无法看到其他任务

  根本原因: API参数不匹配
  - 前端发送exclude，后端期望exclude_task_id
  - 前端使用offset/limit，后端期望page/page_size
  - 解决方案: 修正API参数映射，确保前后端通信正确
