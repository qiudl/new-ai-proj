```kotlin
class MarkdownEditorFragment : Fragment() {
    private lateinit var editorEditText: EditText
    private lateinit var previewTextView: TextView
    private lateinit var markwon: Markwon

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        // 初始化Markwon
        markwon = Markwon.builder(requireContext())
            .usePlugin(SyntaxHighlightPlugin.create(Prism4j(), Prism4jThemeDefault()))
            .usePlugin(GlideImagesPlugin.create(requireContext()))
            .usePlugin(TablePlugin.create(requireContext()))
            .build()

        // 监听编辑器文本变化
        editorEditText.addTextChangedListener(object : TextWatcher {
            override fun afterTextChanged(s: Editable?) {
                s?.let {
                    updatePreview(it.toString())
                }
            }
            override fun beforeTextChanged(s: CharSequence?, start: Int, count: Int, after: Int) {}
            override fun onTextChanged(s: CharSequence?, start: Int, before: Int, count: Int) {}
        })
    }

    private fun updatePreview(markdown: String) {
        markwon.setMarkdown(previewTextView, markdown)
    }
}
```

#### 2.3.3 同步滚动实现

```kotlin
class SyncScrollListener(
    private val editorScrollView: NestedScrollView,
    private val previewScrollView: NestedScrollView
) : NestedScrollView.OnScrollChangeListener {

    private var isSyncing = false

    override fun onScrollChange(
        v: NestedScrollView,
        scrollX: Int,
        scrollY: Int,
        oldScrollX: Int,
        oldScrollY: Int
    ) {
        if (isSyncing) return

        isSyncing = true

        val targetScrollView = if (v == editorScrollView) previewScrollView else editorScrollView

        // 计算滚动比例
        val maxScroll = v.getChildAt(0).height - v.height
        val scrollRatio = if (maxScroll > 0) scrollY.toFloat() / maxScroll else 0f

        // 同步滚动
        val targetMaxScroll = targetScrollView.getChildAt(0).height - targetScrollView.height
        val targetScrollY = (targetMaxScroll * scrollRatio).toInt()
        targetScrollView.scrollTo(0, targetScrollY)

        isSyncing = false
    }
}
```

#### 2.4.2 权限检查

```kotlin
class PermissionManager(private val context: Context) {

    fun canDeleteNote(note: WorkNote): Boolean {
        val currentUser = getCurrentUser() ?: return false

        return when {
            // 超级管理员
            currentUser.isSuperAdmin -> true

            // 笔记所有者
            note.ownerId == currentUser.id -> true

            // 团队笔记且有删除权限
            note.visibility == "team" && hasPermission("work_note:delete") -> true

            else -> false
        }
    }

    private fun hasPermission(permission: String): Boolean {
        val userPermissions = getStoredPermissions()
        return permission in userPermissions
    }
}
```

### 3.4 离线支持

**同步策略**:
1. **下拉刷新**: 从服务器拉取最新数据
2. **自动同步**: App启动时同步
3. **冲突解决**: 服务器时间戳优先

**实现示例**:
```kotlin
class WorkNoteSyncManager(
    private val api: WorkNoteApi,
    private val database: WorkNoteDatabase
) {
    suspend fun sync() {
        // 1. 上传本地未同步的修改
        val unsyncedNotes = database.workNoteDao().getUnsyncedNotes()
        unsyncedNotes.forEach { note ->
            try {
                api.updateWorkNote(note.id, note.toRequest())
                database.workNoteDao().markAsSynced(note.id)
            } catch (e: Exception) {
                // 记录同步失败
                logSyncError(note.id, e)
            }
        }

        // 2. 下载服务器最新数据
        val serverNotes = api.getAllWorkNotes()
        database.workNoteDao().insertAll(serverNotes.map { it.toEntity() })
    }
}
```
