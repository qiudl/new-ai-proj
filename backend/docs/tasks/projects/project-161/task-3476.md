# 账户管理功能设计文档

## 1. 功能概述

账户管理是资金流水模块的基础功能，负责管理系统中所有类型的账户，包括客户账户、供应商账户、公司账户和员工账户等。本文档详细定义账户的完整生命周期管理、余额管理、风控策略等核心功能。

### 1.1 核心功能

- **账户创建与管理**：支持多种类型账户的创建、更新、查询
- **余额管理**：实时余额查询、冻结/解冻、余额调整
- **状态管理**：账户启用、冻结、关闭等状态流转
- **风控管理**：额度管理、风险等级评估、交易限制
- **查询统计**：多维度查询、余额统计、交易汇总

### 1.2 业务场景

| 场景 | 描述 | 参与角色 |
|------|------|---------|
| 客户注册开户 | 客户注册时自动创建账户 | 系统、客户 |
| 供应商入驻 | 供应商入驻时创建对公账户 | 系统、供应商 |
| 余额充值 | 客户账户充值 | 客户、财务 |
| 账户冻结 | 风控或人工冻结异常账户 | 财务、风控 |
| 余额调整 | 财务人员手动调整账户余额 | 财务 |
| 账户对账 | 定期核对账户余额 | 财务 |

```go
func (b *AccountBiz) Create(db *gorm.DB, req *ReqCreateAccount) (*RespCreateAccount, error) {
    // 1. 参数验证
    if err := validateAccountType(req.AccountType); err != nil {
        return nil, err
    }
    
    if err := validateOwnerType(req.OwnerType); err != nil {
        return nil, err
    }
    
    // 2. 检查账户编号是否已存在
    var exists int64
    db.Model(&Account{}).Where("account_no = ?", req.AccountNo).Count(&exists)
    if exists > 0 {
        return nil, errors.New("账户编号已存在")
    }
    
    // 3. 检查所有者是否已有账户
    db.Model(&Account{}).
        Where("owner_type = ? AND owner_id = ? AND account_type = ?", 
              req.OwnerType, req.OwnerID, req.AccountType).
        Count(&exists)
    if exists > 0 {
        return nil, errors.New("该所有者已存在相同类型的账户")
    }
    
    // 4. 创建账户
    account := &Account{
        AccountNo:       req.AccountNo,
        AccountName:     req.AccountName,
        AccountType:     req.AccountType,
        OwnerType:       req.OwnerType,
        OwnerID:         req.OwnerID,
        Currency:        req.Currency,
        Balance:         decimal.Zero,
        FrozenAmount:    decimal.Zero,
        AvailableBalance: decimal.Zero,
        CreditLimit:     req.CreditLimit,
        DailyLimit:      req.DailyLimit,
        MonthlyLimit:    req.MonthlyLimit,
        Status:          "active",
        RiskLevel:       "normal",
    }
    
    if err := db.Create(account).Error; err != nil {
        return nil, err
    }
    
    return &RespCreateAccount{
        ID:          account.ID,
        AccountNo:   account.AccountNo,
        AccountName: account.AccountName,
```

```go
func (b *AccountBiz) List(db *gorm.DB, req *ReqListAccounts) (*RespListAccounts, error) {
    query := db.Model(&Account{})
    
    // 筛选条件
    if req.AccountNo != "" {
        query = query.Where("account_no = ?", req.AccountNo)
    }
    if req.AccountName != "" {
        query = query.Where("account_name LIKE ?", "%"+req.AccountName+"%")
    }
    if req.AccountType != "" {
        query = query.Where("account_type = ?", req.AccountType)
    }
    if req.OwnerType != "" {
        query = query.Where("owner_type = ?", req.OwnerType)
    }
    if req.OwnerID > 0 {
        query = query.Where("owner_id = ?", req.OwnerID)
    }
    if req.Status != "" {
        query = query.Where("status = ?", req.Status)
    }
    if req.RiskLevel != "" {
        query = query.Where("risk_level = ?", req.RiskLevel)
    }
    
    // 统计总数
    var total int64
    query.Count(&total)
    
    // 排序
    sortBy := "created_at"
    if req.SortBy != "" {
        sortBy = req.SortBy
    }
    sortOrder := "DESC"
    if req.SortOrder == "asc" {
        sortOrder = "ASC"
    }
    query = query.Order(fmt.Sprintf("%s %s", sortBy, sortOrder))
    
    // 分页
    offset := (req.Page - 1) * req.PageSize
    query = query.Offset(offset).Limit(req.PageSize)
    
    var accounts []Account
    if err := query.Find(&accounts).Error; err != nil {
        return nil, err
    }
    
    return &RespListAccounts{
        Data:  accounts,
        Total: total,
        Page:  req.Page,
        PageSize: req.PageSize,
    }, nil
}
```

```go
func (b *AccountBiz) Freeze(db *gorm.DB, accountID uint, req *ReqFreezeAccount, operatorID uint) error {
    return db.Transaction(func(tx *gorm.DB) error {
        var account Account
        
        // 悲观锁查询
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&account, accountID).Error; err != nil {
            return err
        }
        
        // 检查账户状态
        if account.Status == "frozen" {
            return errors.New("账户已冻结")
        }
        if account.Status == "closed" {
            return errors.New("账户已关闭，无法冻结")
        }
        
        // 更新状态
        account.Status = "frozen"
        if err := tx.Save(&account).Error; err != nil {
            return err
        }
        
        // 记录操作日志
        log := &AccountOperationLog{
            AccountID:   accountID,
            OperationType: "freeze",
            Reason:      req.Reason,
            OperatorID:  operatorID,
            BeforeStatus: "active",
            AfterStatus:  "frozen",
        }
        if err := tx.Create(log).Error; err != nil {
            return err
        }
        
        return nil
    })
}
```

```go
func (b *AccountBiz) AdjustBalance(db *gorm.DB, accountID uint, req *ReqAdjustBalance, operatorID uint) error {
    return db.Transaction(func(tx *gorm.DB) error {
        var account Account
        
        // 悲观锁
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&account, accountID).Error; err != nil {
            return err
        }
        
        // 检查账户状态
        if account.Status != "active" {
            return errors.New("账户状态异常，无法调整余额")
        }
        
        beforeBalance := account.Balance
        var afterBalance decimal.Decimal
        var direction string
        
        // 计算新余额
        if req.AdjustType == "increase" {
            afterBalance = beforeBalance.Add(req.Amount)
            direction = "income"
        } else {
            afterBalance = beforeBalance.Sub(req.Amount)
            if afterBalance.LessThan(decimal.Zero) {
                return errors.New("调整后余额不能为负数")
            }
            direction = "expense"
        }
        
        // 更新账户余额
        account.Balance = afterBalance
        account.AvailableBalance = afterBalance.Sub(account.FrozenAmount)
        
        if direction == "income" {
            account.TotalIncome = account.TotalIncome.Add(req.Amount)
        } else {
            account.TotalExpense = account.TotalExpense.Add(req.Amount)
        }
        
        account.LastTransactionAt = timeNow()
        
        if err := tx.Save(&account).Error; err != nil {
            return err
        }
        
        // 创建交易流水
        transactionNo := generateTransactionNo()
        transaction := &TransactionRecord{
            TransactionNo:   transactionNo,
            AccountID:       accountID,
            AccountNo:       account.AccountNo,
            TransactionType: TransactionTypeAdjustment,
            BusinessType:    BusinessTypeAdjustment,
            Direction:       direction,
            Amount:          req.Amount,
            BeforeBalance:   beforeBalance,
            AfterBalance:    afterBalance,
            Description:     req.Reason,
            Remark:          req.Remark,
            Status:          "success",
            OperatorID:      operatorID,
            AuditorID:       req.ApproverID,
        }
        
        if err := tx.Create(transaction).Error; err != nil {
            return err
        }
        
        return nil
    })
}
```

```go
func (b *AccountBiz) Statistics(db *gorm.DB, req *ReqAccountStatistics) (*RespAccountStatistics, error) {
    query := db.Model(&Account{})
    
    // 筛选条件
    if req.AccountType != "" {
        query = query.Where("account_type = ?", req.AccountType)
    }
    if req.OwnerType != "" {
        query = query.Where("owner_type = ?", req.OwnerType)
    }
    if req.Status != "" {
        query = query.Where("status = ?", req.Status)
    }
    
    // 基础统计
    var stats struct {
        TotalAccounts  int
        ActiveAccounts int
        FrozenAccounts int
        TotalBalance   decimal.Decimal
        TotalFrozen    decimal.Decimal
        TotalAvailable decimal.Decimal
        TotalIncome    decimal.Decimal
        TotalExpense   decimal.Decimal
    }
    
    query.Select(`
        COUNT(*) as total_accounts,
        SUM(CASE WHEN status = 'active' THEN 1 ELSE 0 END) as active_accounts,
        SUM(CASE WHEN status = 'frozen' THEN 1 ELSE 0 END) as frozen_accounts,
        COALESCE(SUM(balance), 0) as total_balance,
        COALESCE(SUM(frozen_amount), 0) as total_frozen,
        COALESCE(SUM(available_balance), 0) as total_available,
        COALESCE(SUM(total_income), 0) as total_income,
        COALESCE(SUM(total_expense), 0) as total_expense
    `).Scan(&stats)
    
    resp := &RespAccountStatistics{
        TotalAccounts:  stats.TotalAccounts,
        ActiveAccounts: stats.ActiveAccounts,
        FrozenAccounts: stats.FrozenAccounts,
        TotalBalance:   stats.TotalBalance,
        TotalFrozen:    stats.TotalFrozen,
        TotalAvailable: stats.TotalAvailable,
        TotalIncome:    stats.TotalIncome,
        TotalExpense:   stats.TotalExpense,
    }
    
    // 分组统计
    if req.GroupBy != "" {
        var groups []AccountGroup
        query.Select(fmt.Sprintf(`
            %s as key,
            COUNT(*) as account_count,
            COALESCE(SUM(balance), 0) as total_balance,
            COALESCE(SUM(available_balance), 0) as total_available
        `, req.GroupBy)).
        Group(req.GroupBy).
        Scan(&groups)
        
        resp.Groups = groups
    }
    
    return resp, nil
}
```

## 3. 风控管理

### 3.1 额度检查

```go
func (b *AccountBiz) CheckLimit(db *gorm.DB, accountID uint, amount decimal.Decimal) error {
    var account Account
    if err := db.First(&account, accountID).Error; err != nil {
        return err
    }
    
    // 检查可用余额
    if account.AvailableBalance.LessThan(amount) {
        return errors.New("可用余额不足")
    }
    
    // 检查日限额
    if !account.DailyLimit.IsZero() {
        var dailyExpense decimal.Decimal
        db.Model(&TransactionRecord{}).
            Where("account_id = ? AND direction = 'expense' AND DATE(created_at) = CURRENT_DATE", accountID).
            Select("COALESCE(SUM(amount), 0)").
            Scan(&dailyExpense)
        
        if dailyExpense.Add(amount).GreaterThan(account.DailyLimit) {
            return errors.New("超过日交易限额")
        }
    }
    
    // 检查月限额
    if !account.MonthlyLimit.IsZero() {
        var monthlyExpense decimal.Decimal
        db.Model(&TransactionRecord{}).
            Where("account_id = ? AND direction = 'expense' AND DATE_TRUNC('month', created_at) = DATE_TRUNC('month', CURRENT_DATE)", accountID).
            Select("COALESCE(SUM(amount), 0)").
            Scan(&monthlyExpense)
        
        if monthlyExpense.Add(amount).GreaterThan(account.MonthlyLimit) {
            return errors.New("超过月交易限额")
        }
    }
    
    return nil
}
```

### 3.2 风险等级评估

```go
func (b *AccountBiz) EvaluateRiskLevel(db *gorm.DB, accountID uint) (string, error) {
    var account Account
    if err := db.First(&account, accountID).Error; err != nil {
        return "", err
    }
    
    riskScore := 0
    
    // 1. 余额异常（余额过高或过低）
    if account.Balance.GreaterThan(decimal.NewFromInt(1000000)) {
        riskScore += 20
    }
    if account.Balance.LessThan(decimal.Zero) {
        riskScore += 50
    }
    
    // 2. 频繁交易
    var recentTransactionCount int64
    db.Model(&TransactionRecord{}).
        Where("account_id = ? AND created_at > ?", accountID, time.Now().Add(-24*time.Hour)).
        Count(&recentTransactionCount)
    
    if recentTransactionCount > 100 {
        riskScore += 30
    }
    
    // 3. 大额交易
    var largeTransactionCount int64
    db.Model(&TransactionRecord{}).
        Where("account_id = ? AND amount > ? AND created_at > ?", 
              accountID, 50000, time.Now().Add(-7*24*time.Hour)).
        Count(&largeTransactionCount)
    
    if largeTransactionCount > 5 {
        riskScore += 25
    }
    
    // 评级
    riskLevel := "normal"
    if riskScore >= 70 {
        riskLevel = "high"
    } else if riskScore >= 40 {
        riskLevel = "medium"
    } else {
        riskLevel = "low"
    }
    
    // 更新风险等级
    db.Model(&Account{}).Where("id = ?", accountID).Update("risk_level", riskLevel)
    
    return riskLevel, nil
}
```

```go
// 账户信息缓存（Redis）
func (b *AccountBiz) GetWithCache(db *gorm.DB, accountID uint) (*Account, error) {
    cacheKey := fmt.Sprintf("account:%d", accountID)
    
    // 1. 尝试从缓存获取
    var account Account
    cached, err := redis.Get(cacheKey).Result()
    if err == nil {
        json.Unmarshal([]byte(cached), &account)
        return &account, nil
    }
    
    // 2. 从数据库查询
    if err := db.First(&account, accountID).Error; err != nil {
        return nil, err
    }
    
    // 3. 写入缓存（5分钟过期）
    data, _ := json.Marshal(account)
    redis.Set(cacheKey, data, 5*time.Minute)
    
    return &account, nil
}

// 更新时清除缓存
func (b *AccountBiz) UpdateAndClearCache(db *gorm.DB, account *Account) error {
    if err := db.Save(account).Error; err != nil {
        return err
    }
    
    cacheKey := fmt.Sprintf("account:%d", account.ID)
    redis.Del(cacheKey)
    
    return nil
}
```

```go
// 批量查询账户余额
func (b *AccountBiz) BatchGetBalance(db *gorm.DB, accountIDs []uint) (map[uint]decimal.Decimal, error) {
    var accounts []Account
    if err := db.Select("id, balance").Where("id IN ?", accountIDs).Find(&accounts).Error; err != nil {

    }
    
    result := make(map[uint]decimal.Decimal)

    }
```
