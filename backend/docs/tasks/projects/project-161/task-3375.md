# 出库管理功能设计文档

## 1. 概述

### 1.1 文档目的

本文档详细描述仓库库存模块的出库管理功能设计，包括销售出库、调拨出库、退供出库、盘亏出库等核心业务流程的API接口设计、业务逻辑、拣货策略、库存预留机制、波次拣货、异常处理和性能优化。

### 1.2 设计原则

- **库存安全**: 严格预留机制，防止超卖
- **高效拣货**: 支持波次拣货、批量拣货优化
- **流程规范**: 标准化出库流程，支持多种出库类型
- **可追溯**: 完整记录出库操作轨迹和库存变更历史
- **并发控制**: 使用乐观锁/悲观锁防止库存扣减冲突

### 1.3 适用范围

- 仓库管理人员：执行日常出库操作
- 销售人员：查看销售订单出库进度
- 拣货员：执行拣货任务
- 打包员：执行打包发货
- 系统管理员：配置出库规则和拣货策略

## 2. 出库业务场景

### 2.1 出库类型

#### 2.1.1 销售出库 (Sales Outbound)

**业务描述**: 根据销售订单发货给客户

**关键字段**:
- 销售订单号
- 客户信息
- 配送地址
- 物流要求

**业务流程**:
```
1. 创建出库单 → 2. 预留库存 → 3. 分配拣货任务 → 4. 拣货 → 5. 复核 → 6. 打包 → 7. 发货 → 8. 完成出库
```

#### 2.1.2 调拨出库 (Transfer Outbound)

**业务描述**: 将商品调拨到其他仓库

**关键字段**:
- 调拨单号
- 目标仓库信息
- 调拨原因

**业务流程**:
```
1. 创建调拨出库单 → 2. 预留库存 → 3. 拣货 → 4. 打包 → 5. 发运 → 6. 完成出库
```

#### 2.1.3 退供出库 (Return to Supplier Outbound)

**业务描述**: 将不合格商品退回供应商

**关键字段**:
- 退货单号
- 供应商信息
- 退货原因
- 原采购单号

**业务流程**:
```
1. 创建退供出库单 → 2. 拣货（不良品） → 3. 打包 → 4. 发运 → 5. 完成出库
```

#### 2.1.4 盘亏出库 (Shortage Outbound)

**业务描述**: 盘点发现实际库存少于账面库存

**关键字段**:
- 盘点单号
- 盘亏数量
- 盘亏原因

**业务流程**:
```
1. 创建盘亏出库单 → 2. 审核 → 3. 扣减库存 → 4. 完成出库
```

### 2.2 出库状态流转

```
待出库 (Pending)
  ↓
已预留 (Reserved) ← [库存预留]
  ↓
拣货中 (Picking)
  ↓
待复核 (Checking)
  ↓
打包中 (Packing)
  ↓
待发货 (Ready to Ship)
  ↓
已发货 (Shipped)
  ↓
已完成 (Completed)
  ↓
已取消 (Cancelled) ← [异常取消]
```

```go
func (b *OutboundBiz) CreateSalesOutbound(db *gorm.DB, req *ReqCreateSalesOutbound, userID uint) (*RespCreateOutbound, error) {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 生成出库单号
        orderNumber := b.generateOrderNumber("SO") // SO-20250106-00001
        
        // 2. 计算总数量
        totalQty := 0
        for _, item := range req.Items {
            totalQty += item.PlannedQty
        }
        
        // 3. 序列化配送地址
        addressJSON, _ := json.Marshal(req.ShippingAddress)
        
        // 4. 创建出库单主记录
        order := &OutboundOrder{
            OrderNumber:     orderNumber,
            OutboundType:    "sales",
            Status:          "pending",
            WarehouseID:     req.WarehouseID,
            TargetType:      "customer",
            TargetID:        &req.CustomerID,
            SourceOrderNo:   req.SalesOrderNo,
            ShippingAddress: string(addressJSON),
            ShippingMethod:  req.ShippingMethod,
            PlannedDate:     req.PlannedDate,
            TotalQuantity:   totalQty,
            Priority:        req.Priority,
            Remark:          req.Remark,
            CreatedBy:       userID,
        }
        
        if err := tx.Create(order).Error; err != nil {
            return err
        }
        
        // 5. 创建出库单明细
        for _, itemReq := range req.Items {
            // 验证SKU存在
            var sku Sku
            if err := tx.First(&sku, itemReq.SkuID).Error; err != nil {
                return fmt.Errorf("SKU %d 不存在", itemReq.SkuID)
            }
            
            item := &OutboundOrderItem{
                OutboundOrderID: order.ID,
                SkuID:           itemReq.SkuID,
                SkuCode:         sku.Code,
                SkuName:         sku.Name,
                PlannedQty:      itemReq.PlannedQty,
                Remark:          itemReq.Remark,
            }
            
            if err := tx.Create(item).Error; err != nil {
                return err
            }
        }
        
        // 6. 记录操作日志
        log := &OutboundOrderLog{
            OutboundOrderID: order.ID,
            Action:          "created",
            ToStatus:        "pending",
            OperatorID:      userID,
            Description:     fmt.Sprintf("创建销售出库单，订单号: %s", req.SalesOrderNo),
        }
        tx.Create(log)
        
        return &RespCreateOutbound{
            ID:           order.ID,
            OrderNumber:  order.OrderNumber,
            OutboundType: order.OutboundType,
            Status:       order.Status,
            CreatedAt:    order.CreatedAt,
        }, nil
    })
}
```

**业务逻辑**:

```go
func (b *OutboundBiz) ReserveInventory(db *gorm.DB, orderID uint, req *ReqReserveInventory, userID uint) (*RespReserveInventory, error) {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 查询出库单
        var order OutboundOrder
        if err := tx.Preload("Items").First(&order, orderID).Error; err != nil {
            return nil, err
        }
        
        // 2. 验证状态
        if order.Status != "pending" {
            return nil, errors.New("只有待出库状态可以预留库存")
        }
        
        totalReserved := 0
        totalShortage := 0
        
        // 3. 为每个明细预留库存
        for _, item := range order.Items {
            reserved, err := b.reserveForItem(tx, &order, item, req.ReserveStrategy)
            if err != nil {
                return nil, err
            }
            
            item.ReservedQty = reserved
            tx.Save(item)
            
            totalReserved += reserved
            if reserved < item.PlannedQty {
                totalShortage += (item.PlannedQty - reserved)
            }
        }
        
        // 4. 更新出库单状态
        order.ReservedQty = totalReserved
        order.ReservedBy = &userID
        
        if totalReserved > 0 {
            order.Status = "reserved"
        }
        
        tx.Save(&order)
        
        // 5. 记录日志
        log := &OutboundOrderLog{
            OutboundOrderID: order.ID,
            Action:          "reserve",
            FromStatus:      "pending",
            ToStatus:        order.Status,
            OperatorID:      userID,
            Description:     fmt.Sprintf("预留库存: %d/%d", totalReserved, order.TotalQuantity),
        }
        tx.Create(log)
        
        return &RespReserveInventory{
            OrderID:     order.ID,
            ReservedQty: totalReserved,
            TotalQty:    order.TotalQuantity,
            IsComplete:  totalShortage == 0,
            ShortageQty: totalShortage,
        }, nil
    })
}

// 为单个明细预留库存
func (b *OutboundBiz) reserveForItem(tx *gorm.DB, order *OutboundOrder, item *OutboundOrderItem, strategy string) (int, error) {
    // 查询可用库存（按策略排序）
    query := tx.Model(&Inventory{}).
        Where("warehouse_id = ? AND sku_id = ? AND available_quantity > 0", order.WarehouseID, item.SkuID)
    
    // 应用预留策略
    switch strategy {
    case "fifo":
        query = query.Order("created_at ASC") // 先进先出
    case "fefo":
        query = query.Order("expiry_date ASC") // 先过期先出
    case "nearest":
        query = query.Joins("JOIN warehouse_locations ON warehouse_locations.id = inventories.location_id").
            Order("warehouse_locations.code ASC") // 按库位编码（就近原则）
    default:
        query = query.Order("created_at ASC") // 默认FIFO
    }
    
    var inventories []Inventory
    if err := query.Find(&inventories).Error; err != nil {
        return 0, err
    }
    
    needQty := item.PlannedQty
    reservedQty := 0
    
    // 逐个库存记录预留
    for _, inv := range inventories {
        if needQty <= 0 {
            break
        }
        
        // 计算可预留数量
        canReserve := inv.AvailableQuantity
        if canReserve > needQty {
            canReserve = needQty
        }
        
        // 使用悲观锁更新库存
        result := tx.Model(&Inventory{}).
            Where("id = ? AND available_quantity >= ?", inv.ID, canReserve).
            Updates(map[string]interface{}{
                "reserved_quantity":  gorm.Expr("reserved_quantity + ?", canReserve),
                "available_quantity": gorm.Expr("available_quantity - ?", canReserve),
            })
        
        if result.Error != nil {
            return 0, result.Error
        }
        
        if result.RowsAffected > 0 {
            // 创建库存预留记录
            reservation := &InventoryReservation{
                InventoryID:     inv.ID,
                OrderType:       "outbound",
                OrderID:         order.ID,
                SkuID:           item.SkuID,
                ReservedQty:     canReserve,
                Status:          "reserved",
            }
            tx.Create(reservation)
            
            reservedQty += canReserve
            needQty -= canReserve
        }
    }
    
    return reservedQty, nil
}
```

```go
func (b *OutboundBiz) PickGoods(db *gorm.DB, orderID uint, req *ReqPickGoods, userID uint) (*RespPickGoods, error) {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 查询出库单
        var order OutboundOrder
        if err := tx.Preload("Items").First(&order, orderID).Error; err != nil {
            return nil, err
        }
        
        // 2. 验证状态
        if order.Status != "picking" {
            return nil, errors.New("出库单状态不正确")
        }
        
        totalPicked := 0
        
        // 3. 逐个处理拣货明细
        for _, pickReq := range req.PickingDetails {
            var pickDetail PickingDetail
            if err := tx.First(&pickDetail, pickReq.PickingDetailID).Error; err != nil {
                return nil, err
            }
            
            // 验证数量
            if pickReq.ActualQty > pickDetail.PlannedQty {
                return nil, fmt.Errorf("实际拣货数量不能超过计划数量 %d", pickDetail.PlannedQty)
            }
            
            // 更新拣货明细
            pickDetail.ActualQty = pickReq.ActualQty
            pickDetail.Status = "picked"
            pickDetail.PickedBy = &userID
            now := time.Now()
            pickDetail.PickedAt = &now
            
            if err := tx.Save(&pickDetail).Error; err != nil {
                return nil, err
            }
            
            // 扣减库存
            err := b.reduceInventory(tx, pickDetail.InventoryID, pickReq.ActualQty)
            if err != nil {
                return nil, err
            }
            
            totalPicked += pickReq.ActualQty
        }
        
        // 4. 更新出库单明细
        for _, item := range order.Items {
            var itemPicked int
            tx.Model(&PickingDetail{}).
                Where("outbound_order_id = ? AND item_id = ? AND status = ?", orderID, item.ID, "picked").
                Select("SUM(actual_qty)").
                Scan(&itemPicked)
            
            item.PickedQty = itemPicked
            tx.Save(item)
        }
        
        // 5. 更新出库单拣货汇总
        order.PickedQty = totalPicked
        order.PickedBy = &userID
        now := time.Now()
        order.PickedDate = &now
        
        // 6. 判断是否拣货完成
        isComplete := totalPicked >= order.ReservedQty
        
        if isComplete {
            order.Status = "checking" // 进入复核
        }
        
        tx.Save(&order)
        
        // 7. 记录日志
        log := &OutboundOrderLog{
            OutboundOrderID: order.ID,
            Action:          "pick_goods",
            FromStatus:      "picking",
            ToStatus:        order.Status,
            OperatorID:      userID,
            Description:     fmt.Sprintf("拣货完成: %d件", totalPicked),
        }
        tx.Create(log)
        
        return &RespPickGoods{
            OrderID:    order.ID,
            PickedQty:  totalPicked,
            TotalQty:   order.ReservedQty,
            IsComplete: isComplete,
        }, nil
    })
}

// 扣减库存（预留转实际出库）
func (b *OutboundBiz) reduceInventory(tx *gorm.DB, inventoryID uint, quantity int) error {
    result := tx.Model(&Inventory{}).
        Where("id = ? AND reserved_quantity >= ?", inventoryID, quantity).
        Updates(map[string]interface{}{
            "quantity":          gorm.Expr("quantity - ?", quantity),
            "reserved_quantity": gorm.Expr("reserved_quantity - ?", quantity),
        })
    
    if result.Error != nil {
        return result.Error
    }
    
    if result.RowsAffected == 0 {
        return errors.New("库存扣减失败，预留数量不足")
    }
    
    // 更新预留记录状态
    tx.Model(&InventoryReservation{}).
        Where("inventory_id = ?", inventoryID).
        Update("status", "consumed")
    
    return nil
}
```

## 6. 性能优化

### 6.1 库存预留并发控制

```go
// 使用SELECT FOR UPDATE实现悲观锁
func (b *OutboundBiz) reserveWithLock(tx *gorm.DB, inventoryID uint, quantity int) error {
    var inv Inventory
    
    // 悲观锁
    err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
        Where("id = ?", inventoryID).
        First(&inv).Error
    
    if err != nil {
        return err
    }
    
    if inv.AvailableQuantity < quantity {
        return errors.New("库存不足")
    }
    
    // 更新库存
    inv.AvailableQuantity -= quantity
    inv.ReservedQuantity += quantity
    
    return tx.Save(&inv).Error
}
```

```go
// 批量扣减库存
func (b *OutboundBiz) batchReduceInventory(tx *gorm.DB, reductions []InventoryReduction) error {
    for _, red := range reductions {
        sql := `
            UPDATE inventories 
            SET quantity = quantity - ?, 
                reserved_quantity = reserved_quantity - ?
            WHERE id = ? AND reserved_quantity >= ?
        `
        
        result := tx.Exec(sql, red.Quantity, red.Quantity, red.InventoryID, red.Quantity)
        if result.Error != nil {
            return result.Error
        }
        
        if result.RowsAffected == 0 {
            return fmt.Errorf("库存 %d 扣减失败", red.InventoryID)
        }
    }
    
    return nil
}
```

```go
func (b *OutboundBiz) rollbackReservation(tx *gorm.DB, orderID uint) error {
    // 查询预留记录
    var reservations []InventoryReservation
    tx.Where("order_type = ? AND order_id = ?", "outbound", orderID).Find(&reservations)
    
    // 恢复库存
    for _, res := range reservations {
        tx.Model(&Inventory{}).Where("id = ?", res.InventoryID).
            Updates(map[string]interface{}{
                "reserved_quantity":  gorm.Expr("reserved_quantity - ?", res.ReservedQty),

    tx.Delete(&InventoryReservation{}, "order_type = ? AND order_id = ?", "outbound", orderID)
```
