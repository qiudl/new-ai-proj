func TestCalculateCommission_Percentage(t *testing.T) {
	biz := NewPerformanceBiz()
	
	// 创建测试模板
	template := &model.CommissionTemplate{
		TemplateType:    "sales",
		CalculateMethod: "percentage",
	}
	
	rule := &model.CommissionRule{
		CommissionRate: decimal.NewFromFloat(0.05),
	}
	
	// 创建业绩记录
	record := &model.PerformanceRecord{
		SalesAmount: decimal.NewFromInt(10000),
	}
	
	// 计算佣金
	amount := biz.calculateCommission(template, rule, record)
	
	assert.Equal(t, decimal.NewFromInt(500), amount)
}

func TestCalculateTieredCommission(t *testing.T) {
	biz := NewPerformanceBiz()
	
	rules := []model.CommissionRule{
		{MinAmount: decimal.Zero, MaxAmount: decimal.NewFromInt(10000), CommissionRate: decimal.NewFromFloat(0.03)},
		{MinAmount: decimal.NewFromInt(10000), MaxAmount: decimal.NewFromInt(50000), CommissionRate: decimal.NewFromFloat(0.05)},
		{MinAmount: decimal.NewFromInt(50000), MaxAmount: decimal.Zero, CommissionRate: decimal.NewFromFloat(0.08)},
	}
	
	// 测试金额：30000
	amount := biz.calculateTieredCommission(decimal.NewFromInt(30000), rules)
	
	// 预期：10000*0.03 + 20000*0.05 = 300 + 1000 = 1300
	assert.Equal(t, decimal.NewFromInt(1300), amount)
}
```

## 3. 集成测试

### 3.1 完整业绩流程测试

func TestPerformanceFlow_Complete(t *testing.T) {
	db := setupTestDB(t)
	defer cleanupTestDB(t, db)
	
	biz := NewPerformanceBiz()
	
	// 1. 创建模板
	template := createTestTemplate(t, db)
	
	// 2. 创建订单
	order := createTestOrder(t, db)
	order.Status = "completed"
	order.CompletedAt = timePtr(time.Now())
	db.Save(order)
	
	// 3. 记录业绩
	err := biz.RecordPerformance(db, order)
	assert.NoError(t, err)
	
	// 4. 验证业绩记录
	var record model.PerformanceRecord
	db.Where("order_id = ?", order.ID).First(&record)
	assert.Equal(t, order.SalespersonID, record.UserID)
	assert.Equal(t, order.TotalAmount, record.SalesAmount)
	
	// 5. 验证佣金记录
	var commission model.CommissionRecord
	db.Where("performance_record_id = ?", record.ID).First(&commission)
	assert.NotZero(t, commission.CommissionAmount)
	
	// 6. 验证统计更新
	var stats model.PerformanceStatistics
	db.Where("user_id = ? AND stat_period = ?", order.SalespersonID, record.StatPeriod).First(&stats)
	assert.Equal(t, 1, stats.OrderCount)
}
```
