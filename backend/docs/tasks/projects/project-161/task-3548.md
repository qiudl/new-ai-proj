#### **方案3：订单操作组件**

```typescript
// paas-system/web/src/modules/business/components/Order/OrderActions.tsx
import React from 'react';
import { Space, Button, Dropdown, Menu } from 'antd';
import { MoreOutlined } from '@ant-design/icons';
import { usePermission } from '@/hooks/usePermission';
import { ORDER_ACTIONS_CONFIG } from '../../config/orderActions';
import { Order } from '../../types/order';

interface OrderActionsProps {
  order: Order;
  onActionComplete?: () => void;
  maxVisibleActions?: number; // 最多显示几个按钮，其余放到"更多"菜单
}

export const OrderActions: React.FC<OrderActionsProps> = ({
  order,
  onActionComplete,
  maxVisibleActions = 3
}) => {
  const { hasPermission } = usePermission();
  
  // 获取当前状态的可用操作
  const actions = ORDER_ACTIONS_CONFIG[order.status] || [];
  
  // 过滤有权限的操作
  const availableActions = actions.filter(action => {
    if (action.permission && !hasPermission(action.permission)) {
      return false;
    }
    if (action.disabled && action.disabled(order)) {
      return false;
    }
    return true;
  });
  
  // 执行操作
  const handleAction = async (action: OrderAction) => {
    try {
      await action.handler(order);
      onActionComplete?.();
    } catch (error) {
      message.error(`操作失败: ${error.message}`);
    }
  };
  
  // 分离主要操作和次要操作
  const primaryActions = availableActions.slice(0, maxVisibleActions);
  const moreActions = availableActions.slice(maxVisibleActions);
  
  return (
    <Space size="small">
      {primaryActions.map(action => (
        <Button
          key={action.key}
          type={action.type}
          icon={action.icon}
          size="small"
          onClick={() => handleAction(action)}
        >
          {action.label}
        </Button>
      ))}
      
      {moreActions.length > 0 && (
        <Dropdown
          overlay={
            <Menu>
              {moreActions.map(action => (
                <Menu.Item
                  key={action.key}
                  icon={action.icon}
                  danger={action.type === 'danger'}
                  onClick={() => handleAction(action)}
                >
                  {action.label}
                </Menu.Item>
              ))}
            </Menu>
          }
        >
          <Button size="small" icon={<MoreOutlined />} />
        </Dropdown>
      )}
    </Space>
  );
};
```

#### **使用示例**


#### **创建 BatchActionBar 组件**

```typescript
// paas-system/web/src/components/common/BatchActionBar/index.tsx
import React from 'react';
import { Space, Button, Typography, Affix } from 'antd';
import { CloseOutlined } from '@ant-design/icons';
import styles from './index.less';

interface BatchActionBarProps {
  selectedCount: number;
  onCancel: () => void;
  actions?: Array<{
    key: string;
    label: string;
    icon?: React.ReactNode;
    type?: 'primary' | 'default' | 'danger';
    disabled?: boolean;
    onClick: () => void;
  }>;
}

export const BatchActionBar: React.FC<BatchActionBarProps> = ({
  selectedCount,
  onCancel,
  actions = []
}) => {
  if (selectedCount === 0) return null;
  
  return (
    <Affix offsetBottom={0}>
      <div className={styles.batchActionBar}>
        <Space className={styles.info}>
          <Typography.Text>
            已选择 <span className={styles.count}>{selectedCount}</span> 项
          </Typography.Text>
          <Button
            type="link"
            size="small"
            icon={<CloseOutlined />}
            onClick={onCancel}
          >
            取消选择
          </Button>
        </Space>
        
        <Space className={styles.actions}>
          {actions.map(action => (
            <Button
              key={action.key}
              type={action.type}
              icon={action.icon}
              disabled={action.disabled}
              onClick={action.onClick}
            >
              {action.label}
            </Button>
          ))}
        </Space>
      </div>
    </Affix>
  );
};
```

```typescript
// paas-system/web/src/modules/business/pages/Order/OrderList.tsx
import { BatchActionBar } from '@/components/common/BatchActionBar';

const OrderList = () => {
  const [selectedRows, setSelectedRows] = useState<Order[]>([]);
  
  // 批量删除
  const handleBatchDelete = async () => {
    Modal.confirm({
      title: '确认删除',
      content: `确定要删除选中的 ${selectedRows.length} 个订单吗？`,
      onOk: async () => {
        await batchDeleteOrders(selectedRows.map(r => r.id));
        message.success('删除成功');
        setSelectedRows([]);
        refetch();
      }
    });
  };
  
  // 批量发货
  const handleBatchShip = () => {
    // 检查是否所有订单都可以发货
    const canShip = selectedRows.every(order => order.status === OrderStatus.PAID);
    if (!canShip) {
      message.warning('只有已支付的订单才能发货');
      return;
    }
    
    // 打开批量发货Modal
    setBatchShipModalVisible(true);
  };
  
  return (
    <>
      <ProTable
        rowSelection={{
          selectedRowKeys: selectedRows.map(r => r.id),
          onChange: (_, rows) => setSelectedRows(rows)
        }}
      />
      
      <BatchActionBar
        selectedCount={selectedRows.length}
        onCancel={() => setSelectedRows([])}
        actions={[
          {
            key: 'ship',
            label: '批量发货',
            type: 'primary',
            disabled: selectedRows.some(r => r.status !== OrderStatus.PAID),
            onClick: handleBatchShip
          },
          {
            key: 'export',
            label: '导出数据',
            onClick: () => handleBatchExport(selectedRows)
          },
          {
            key: 'delete',
            label: '批量删除',
            type: 'danger',
            onClick: handleBatchDelete
          }
        ]}
      />
    </>
  );
};
```

```typescript
// paas-system/web/src/modules/business/pages/Order/OrderDetail.tsx
import React, { useEffect, useState } from 'react';
import { useParams } from 'react-router-dom';
import { PageContainer } from '@ant-design/pro-components';
import { Spin, Space } from 'antd';
import { getOrderDetail } from '../../api/order';
import {
  BaseInfoCard,
  ReceiverInfoCard,
  ProductInfoCard,
  CostInfoCard,
  OperationHistoryCard
} from '../../components/Order/OrderDetailCards';

const OrderDetail: React.FC = () => {
  const { id } = useParams<{ id: string }>();
  const [order, setOrder] = useState<Order>();
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    loadOrder();
  }, [id]);
  
  const loadOrder = async () => {
    setLoading(true);
    try {
      const data = await getOrderDetail(id);
      setOrder(data);
    } finally {
      setLoading(false);
    }
  };
  
  if (loading) {
    return <Spin />;
  }
  
  if (!order) {
    return <div>订单不存在</div>;
  }
  
  return (
    <PageContainer
      title={`订单详情 - ${order.orderNo}`}
      extra={<OrderActions order={order} onActionComplete={loadOrder} />}
    >
      <Space direction="vertical" size="large" style={{ width: '100%' }}>
        <BaseInfoCard order={order} />
        <ReceiverInfoCard order={order} />
        <ProductInfoCard order={order} />
        <CostInfoCard order={order} />
        <OperationHistoryCard order={order} />
      </Space>
    </PageContainer>
  );
};

export default OrderDetail;
```

```typescript
      description: order.paidAt,
      status: OrderStatus.PAID
    },
    {
      title: '订单发货',
      description: order.shippedAt,
      status: OrderStatus.SHIPPED
    },
    {
      title: '确认收货',
      description: order.deliveredAt,
      status: OrderStatus.DELIVERED
    },
    {
      title: '完成评价',
      description: order.completedAt,
      status: OrderStatus.COMPLETED
    }
  ];
  
  const steps = mode === 'erp' ? erpSteps : ecommerceSteps;
  
  // 计算当前步骤
  const getCurrentStep = () => {
    for (let i = steps.length - 1; i >= 0; i--) {
      if (order.status >= steps[i].status) {
        return i;
      }
    }
    return 0;
  };
  
  // 判断步骤状态
  const getStepStatus = () => {
    if (order.status === OrderStatus.CANCELLED) {
      return 'error';
    }
    if (order.status === OrderStatus.COMPLETED) {
      return 'finish';
    }
    return 'process';
  };
  
  return (
    <Steps
      current={getCurrentStep()}
      status={getStepStatus()}
      style={{ marginBottom: 24 }}
    >
      {steps.map((step, index) => (
        <Steps.Step
          key={index}
          title={step.title}
          description={
            step.description
              ? dayjs(step.description).format('MM-DD HH:mm')
              : undefined
          }
        />
      ))}
    </Steps>
  );
};
```

```typescript
// paas-system/web/src/components/common/ProTable/AdvancedProTable.tsx
import React, { useState, useEffect } from 'react';
import { ProTable } from './index';
import { TableDensity, ColumnSetting } from './types';

export interface AdvancedProTableProps<T> extends ProTableProps<T> {
  // 列配置持久化
  persistenceKey?: string;
  
  // 密度切换
  density?: TableDensity;
  onDensityChange?: (density: TableDensity) => void;
  
  // 全屏模式
  fullscreen?: boolean;
  onFullscreenChange?: (fullscreen: boolean) => void;
}

export const AdvancedProTable = <T extends Record<string, any>>({
  persistenceKey,
  columns: initialColumns,
  density: controlledDensity,
  onDensityChange,
  fullscreen: controlledFullscreen,
  onFullscreenChange,
  ...restProps
}: AdvancedProTableProps<T>) => {
  // 列配置状态
  const [columns, setColumns] = useState(initialColumns);
  const [density, setDensity] = useState<TableDensity>(controlledDensity || 'middle');
  const [fullscreen, setFullscreen] = useState(controlledFullscreen || false);
  
  // 从 localStorage 恢复列配置
  useEffect(() => {
    if (persistenceKey) {
      const saved = localStorage.getItem(`protable_columns_${persistenceKey}`);
      if (saved) {
        try {
          const columnSettings: ColumnSetting[] = JSON.parse(saved);
          const restoredColumns = initialColumns.map(col => {
            const setting = columnSettings.find(s => s.dataIndex === col.dataIndex);
            if (setting) {
              return {
                ...col,
                fixed: setting.fixed,
                width: setting.width,
                hideInTable: !setting.visible
              };
            }
            return col;
          });
          setColumns(restoredColumns);
        } catch (e) {
          console.error('Failed to restore column settings:', e);
        }
      }
    }
  }, [persistenceKey]);
  
  // 保存列配置
  const handleColumnsChange = (newColumns: ProTableColumn<T>[]) => {
    setColumns(newColumns);
    
    if (persistenceKey) {
      const columnSettings: ColumnSetting[] = newColumns.map(col => ({
        dataIndex: col.dataIndex as string,
        visible: !col.hideInTable,
        fixed: col.fixed,
        width: col.width
      }));
      localStorage.setItem(
        `protable_columns_${persistenceKey}`,
        JSON.stringify(columnSettings)
      );
    }
  };
  
  // 密度变化
  const handleDensityChange = (newDensity: TableDensity) => {
    setDensity(newDensity);
    onDensityChange?.(newDensity);
  };
  
  // 全屏切换
  const handleFullscreenChange = (isFullscreen: boolean) => {
    setFullscreen(isFullscreen);
    onFullscreenChange?.(isFullscreen);
  };
  
  return (
    <div className={fullscreen ? 'protable-fullscreen' : undefined}>
      <ProTable
        {...restProps}
        columns={columns}
        onColumnsChange={handleColumnsChange}
        size={density}
        toolbar={{
          ...restProps.toolbar,
          settings: [
            ...(restProps.toolbar?.settings || []),
            {
              key: 'density',
              icon: <DensityIcon />,
              tooltip: '密度',
              onClick: () => {
                // 显示密度选择菜单
              }
            },
            {
              key: 'fullscreen',
              icon: fullscreen ? <FullscreenExitOutlined /> : <FullscreenOutlined />,
              tooltip: fullscreen ? '退出全屏' : '全屏',
              onClick: () => handleFullscreenChange(!fullscreen)
            }
          ]
        }}
      />
    </div>
  );
};
```

```typescript
// paas-system/web/src/hooks/usePermission.ts
import { useAuth } from '@/contexts/AuthContext';

export const usePermission = () => {
  const { currentUser } = useAuth();
  
  const hasPermission = (permission: string | string[]): boolean => {
    if (!currentUser) return false;
    
    const permissions = currentUser.permissions || [];
    
    if (Array.isArray(permission)) {
      return permission.some(p => permissions.includes(p));
    }
    
    return permissions.includes(permission);
  };
  
  const hasRole = (role: string | string[]): boolean => {
    if (!currentUser) return false;
    
    const roles = currentUser.roles || [];
    
    if (Array.isArray(role)) {
      return role.some(r => roles.includes(r));
    }
    
    return roles.includes(role);
  };
  
  return { hasPermission, hasRole };
};

// 权限组件
export const PermissionGuard: React.FC<{
  permission?: string | string[];
  role?: string | string[];
  fallback?: React.ReactNode;
  children: React.ReactNode;
}> = ({ permission, role, fallback = null, children }) => {
  const { hasPermission, hasRole } = usePermission();
  
  if (permission && !hasPermission(permission)) {
    return <>{fallback}</>;
  }
  
  if (role && !hasRole(role)) {
    return <>{fallback}</>;
  }
  
  return <>{children}</>;
};
```
