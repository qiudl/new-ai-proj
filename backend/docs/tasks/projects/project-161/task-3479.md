# 结算流程功能设计文档

## 1. 功能概述

结算流程是资金流水模块的重要功能，负责管理与客户、供应商之间的定期结算业务。通过结算功能，系统可以自动汇总一段时间内的交易数据，生成结算单，并跟踪结算支付进度。

### 1.1 核心功能

- **客户结算**：与客户的应收账款结算
- **供应商结算**：与供应商的应付账款结算  
- **结算周期管理**：支持月结、周结等多种结算周期
- **结算单生成**：自动或手动生成结算单
- **结算审核**：多级审核流程
- **结算支付**：支付/收款处理
- **结算报表**：结算统计分析

### 1.2 业务场景

| 场景 | 结算类型 | 描述 |
|------|---------|------|
| 月度客户结算 | customer | 每月与客户结算应收账款 |
| 月度供应商结算 | supplier | 每月与供应商结算应付账款 |
| 临时结算 | ad_hoc | 临时发起的结算 |
| 预付款结算 | prepayment | 预付款抵扣结算 |

func (b *SettlementBiz) CreateCustomerSettlement(db *gorm.DB, req *ReqCreateCustomerSettlement, userID uint) (*RespCreateSettlement, error) {
    return db.Transaction(func(tx *gorm.DB) (*RespCreateSettlement, error) {
        // 1. 查询客户账户
        var customerAccount Account
        if err := tx.First(&customerAccount, req.CustomerAccountID).Error; err != nil {
            return nil, errors.New("客户账户不存在")
        }
        
        // 2. 查询结算周期内的订单
        var orders []Order
        if err := tx.Where("customer_id = ? AND DATE(created_at) BETWEEN ? AND ? AND status IN ?",
            req.CustomerID, req.SettlementPeriodStart, req.SettlementPeriodEnd, 
            []string{"completed", "shipped"}).
            Find(&orders).Error; err != nil {
            return nil, err
        }
        
        if len(orders) == 0 {
            return nil, errors.New("结算周期内无可结算订单")
        }
        
        // 3. 计算结算金额
        var orderAmount decimal.Decimal
        orderIDs := make([]uint, len(orders))
        
        for i, order := range orders {
            orderAmount = orderAmount.Add(order.TotalAmount)
            orderIDs[i] = order.ID
        }
        
        // 4. 查询退款金额
        var refundAmount decimal.Decimal
        tx.Model(&RefundOrder{}).
            Where("related_type = 'order' AND related_id IN ? AND status = 'success'", orderIDs).
            Select("COALESCE(SUM(refund_amount), 0)").
            Scan(&refundAmount)
        
        // 5. 查询已支付金额
        var paidAmount decimal.Decimal
        tx.Model(&TransactionRecord{}).
            Where("account_id = ? AND related_type = 'order' AND related_id IN ? AND direction = 'income' AND status = 'success'",
                req.CustomerAccountID, orderIDs).
            Select("COALESCE(SUM(amount), 0)").
            Scan(&paidAmount)
        
        // 6. 计算应结算金额
        totalAmount := orderAmount.Sub(refundAmount)
        unpaidAmount := totalAmount.Sub(paidAmount)
        
        if unpaidAmount.LessThanOrEqual(decimal.Zero) {
            return nil, errors.New("无需结算，订单已全部付款")
        }
        
        // 7. 获取公司账户
        var companyAccount Account
        if err := tx.Where("account_type = 'company' AND owner_type = 'company'").
            First(&companyAccount).Error; err != nil {
            return nil, errors.New("未找到公司账户")
        }
        
        // 8. 创建结算单
        settlementNo := generateSettlementNo()
        settlement := &Settlement{
            SettlementNo:          settlementNo,
            SettlementType:        "customer",
            PartyType:             "customer",
            PartyID:               req.CustomerID,
            PartyName:             customerAccount.AccountName,
            PartyAccountID:        req.CustomerAccountID,
            CompanyAccountID:      companyAccount.ID,
            SettlementPeriodStart: parseDate(req.SettlementPeriodStart),
            SettlementPeriodEnd:   parseDate(req.SettlementPeriodEnd),
            TotalAmount:           totalAmount,
            PaidAmount:            paidAmount,
            UnpaidAmount:          unpaidAmount,
            OrderAmount:           orderAmount,
            RefundAmount:          refundAmount,
            Status:                "pending",
            Description:           req.Description,
            CreatorID:             userID,
        }
        
        if err := tx.Create(settlement).Error; err != nil {
            return nil, err
        }
        
        // 9. 创建结算明细
        for _, order := range orders {
            item := &SettlementItem{
                SettlementID: settlement.ID,
                RelatedType:  "order",
                RelatedID:    order.ID,
                RelatedNo:    order.OrderNo,
                Amount:       order.TotalAmount,
                Status:       "pending",
            }
            tx.Create(item)
        }
        
        return &RespCreateSettlement{
            SettlementNo: settlementNo,
            TotalAmount:  totalAmount,
            OrderAmount:  orderAmount,
            RefundAmount: refundAmount,
            UnpaidAmount: unpaidAmount,
            Status:       settlement.Status,
            OrderCount:   len(orders),
        }, nil
    })
}

func (b *SettlementBiz) Approve(db *gorm.DB, settlementID uint, req *ReqApproveSettlement, auditorID uint) error {
    return db.Transaction(func(tx *gorm.DB) error {
        var settlement Settlement
        
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&settlement, settlementID).Error; err != nil {
            return err
        }
        
        if settlement.Status != "pending" {
            return errors.New("结算单状态不允许审核")
        }
        
        if req.Approved {
            settlement.Status = "approved"
        } else {
            settlement.Status = "rejected"
        }
        
        settlement.AuditorID = auditorID
        settlement.AuditedAt = timeNow()
        settlement.AuditRemark = req.Remark
        
        return tx.Save(&settlement).Error
    })
}

func (b *SettlementBiz) ReceivePayment(db *gorm.DB, settlementID uint, req *ReqReceiveSettlementPayment, userID uint) error {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 锁定结算单
        var settlement Settlement
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&settlement, settlementID).Error; err != nil {
            return err
        }
        
        if settlement.Status != "approved" {
            return errors.New("结算单未审核通过")
        }
        
        // 2. 检查收款金额
        if req.Amount.GreaterThan(settlement.UnpaidAmount) {
            return errors.New("收款金额超过未付金额")
        }
        
        // 3. 锁定公司账户（收款方）
        var companyAccount Account
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&companyAccount, settlement.CompanyAccountID).Error; err != nil {
            return err
        }
        
        // 4. 增加公司账户余额
        beforeBalance := companyAccount.Balance
        companyAccount.Balance = companyAccount.Balance.Add(req.Amount)
        companyAccount.AvailableBalance = companyAccount.AvailableBalance.Add(req.Amount)
        companyAccount.TotalIncome = companyAccount.TotalIncome.Add(req.Amount)
        companyAccount.LastTransactionAt = timeNow()
        
        if err := tx.Save(&companyAccount).Error; err != nil {
            return err
        }
        
        // 5. 创建收款流水
        txn := &TransactionRecord{
            TransactionNo:   generateTransactionNo(),
            AccountID:       settlement.CompanyAccountID,
            AccountNo:       companyAccount.AccountNo,
            TransactionType: TransactionTypeSettlementReceive,
            BusinessType:    "settlement",
            Direction:       "income",
            Amount:          req.Amount,
            BeforeBalance:   beforeBalance,
            AfterBalance:    companyAccount.Balance,
            RelatedType:     "settlement",
            RelatedID:       settlement.ID,
            RelatedNo:       settlement.SettlementNo,
            PaymentMethodID: req.PaymentMethodID,
            PaymentAccount:  req.PaymentAccount,
            PaymentTime:     timeNow(),
            CounterpartyType: settlement.PartyType,
            CounterpartyID:   settlement.PartyID,
            CounterpartyName: settlement.PartyName,
            Description:     fmt.Sprintf("结算收款: %s", settlement.SettlementNo),
            Remark:          req.Remark,
            Status:          "success",
            OperatorID:      userID,
        }
        
        if err := tx.Create(txn).Error; err != nil {
            return err
        }
        
        // 6. 更新结算单
        settlement.PaidAmount = settlement.PaidAmount.Add(req.Amount)
        settlement.UnpaidAmount = settlement.UnpaidAmount.Sub(req.Amount)
        settlement.PaymentMethodID = req.PaymentMethodID
        settlement.PaymentAccount = req.PaymentAccount
        settlement.PaymentVoucher = req.PaymentVoucher
        settlement.PaymentTime = timeNow()
        
        // 判断是否全部收款
        if settlement.UnpaidAmount.LessThanOrEqual(decimal.Zero) {
            settlement.Status = "completed"
        } else {
            settlement.Status = "partial"
        }
        
        return tx.Save(&settlement).Error
    })
}

// 每月1号自动生成上月结算单
func AutoGenerateMonthlySettlement() {
    lastMonth := time.Now().AddDate(0, -1, 0)
    startDate := time.Date(lastMonth.Year(), lastMonth.Month(), 1, 0, 0, 0, 0, time.Local)
    endDate := startDate.AddDate(0, 1, -1)
    
    // 查询所有需要自动结算的客户
    var customers []Customer
    db.Where("auto_settlement = true AND settlement_cycle = 'monthly'").Find(&customers)
    
    for _, customer := range customers {
        req := &ReqCreateCustomerSettlement{
            CustomerID:            customer.ID,
            CustomerAccountID:     customer.AccountID,
            SettlementPeriodStart: startDate.Format("2006-01-02"),
            SettlementPeriodEnd:   endDate.Format("2006-01-02"),
            Description:           fmt.Sprintf("%d年%d月结算", lastMonth.Year(), lastMonth.Month()),
        }
        
        _, err := settlementBiz.CreateCustomerSettlement(db, req, 1) // 系统用户
        if err != nil {
            log.Error("自动生成结算单失败", "customer_id", customer.ID, "error", err)
        }
    }
}

// 批量生成结算单
func (b *SettlementBiz) BatchGenerate(db *gorm.DB, requests []ReqCreateCustomerSettlement, userID uint) ([]string, error) {
    settlementNos := make([]string, 0, len(requests))
    
    for _, req := range requests {
        resp, err := b.CreateCustomerSettlement(db, &req, userID)
        if err != nil {
            log.Error("批量生成结算单失败", "customer_id", req.CustomerID, "error", err)
            continue
        }
        settlementNos = append(settlementNos, resp.SettlementNo)
    }
    
    return settlementNos, nil
}
