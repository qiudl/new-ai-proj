<Form.Item

#### 2.2 快速排序功能

```tsx
// 菜单排序快捷按钮
const SortButtons: React.FC<{ record: MenuItem }> = ({ record }) => (
  <Space.Compact>
    <Tooltip title="上移">
      <Button
        type="text"
        size="small"
        icon={<ArrowUpOutlined />}
        onClick={() => handleMoveMenu(record, 'up')}
      />
    </Tooltip>
    <Tooltip title="下移">
      <Button
        type="text"
        size="small"
        icon={<ArrowDownOutlined />}
        onClick={() => handleMoveMenu(record, 'down')}
      />
    </Tooltip>
  </Space.Compact>
);

// 优化后的排序逻辑
const handleMoveMenu = async (record: MenuItem, direction: 'up' | 'down') => {
  try {
    // 获取同级菜单
    const siblings = menuList.filter(m => m.parent_id === record.parent_id);
    const currentIndex = siblings.findIndex(m => m.id === record.id);
    
    if (direction === 'up' && currentIndex === 0) {
      message.warning('已经是第一个了');
      return;
    }
    
    if (direction === 'down' && currentIndex === siblings.length - 1) {
      message.warning('已经是最后一个了');
      return;
    }
    
    const targetIndex = direction === 'up' ? currentIndex - 1 : currentIndex + 1;
    const targetMenu = siblings[targetIndex];
    
    // 交换排序号
    await menuApi.swapMenuSort(record.id, targetMenu.id);
    
    message.success('排序调整成功');
    tableRef.current?.reload();
  } catch (error) {
    message.error('排序调整失败');
  }
};
```

      }
    },
  });
};

## 📋 完整的表单验证规则库

```tsx
// /src/utils/formRules.ts

export const FormRules = {
  // 必填规则
  required: (label: string) => ({
    required: true,
    message: `请输入${label}`
  }),

  // 用户名规则
  username: [
    { required: true, message: '请输入用户名' },
    { min: 3, max: 20, message: '用户名长度应在3-20字符之间' },
    { pattern: /^[a-zA-Z0-9_]+$/, message: '用户名只能包含字母、数字和下划线' }
  ],

  // 邮箱规则
  email: [
    { type: 'email' as const, message: '请输入有效的邮箱地址' }
  ],

  // 手机号规则
  phone: [
    { pattern: /^1[3-9]\d{9}$/, message: '请输入有效的11位手机号' }
  ],

  // 密码规则
  password: [
    { required: true, message: '请输入密码' },
    { min: 6, message: '密码至少6位' },
    { 
      pattern: /^(?=.*[a-zA-Z])(?=.*\d).+$/, 
      message: '密码必须包含字母和数字' 
    }
  ],

  // URL规则
  url: [
    { 
      pattern: /^https?:\/\/.+/, 
      message: '请输入有效的URL（以http://或https://开头）' 
    }
  ],

  // 编码规则（字母、数字、下划线、短横线）
  code: [
    { required: true, message: '请输入编码' },
    { 
      pattern: /^[a-zA-Z0-9_-]+$/, 
      message: '编码只能包含字母、数字、下划线和短横线' 
    }
  ],
};

// 使用示例
<Form.Item name="username" label="用户名" rules={FormRules.username}>
  <Input />
</Form.Item>
```
