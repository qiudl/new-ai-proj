**CreateExcelFile 示例**:
```go
func (g *fileGenerator) CreateExcelFile(ctx context.Context, data [][]interface{}, headers []string) (string, error) {
    f := excelize.NewFile()
    defer f.Close()

    sheet := "Sheet1"
    sheetIndex, _ := f.NewSheet(sheet)

    // 创建表头样式
    headerStyle, _ := f.NewStyle(&excelize.Style{
        Font: &excelize.Font{Bold: true, Size: 12},
        Fill: excelize.Fill{Type: "pattern", Color: []string{"#F0F0F0"}, Pattern: 1},
        Alignment: &excelize.Alignment{Horizontal: "center", Vertical: "center"},
    })

    // 写入表头
    for colIdx, header := range headers {
        cell, _ := excelize.CoordinatesToCellName(colIdx+1, 1)
        f.SetCellValue(sheet, cell, header)
        f.SetCellStyle(sheet, cell, cell, headerStyle)
    }

    // 写入数据
    for rowIdx, row := range data {
        for colIdx, value := range row {
            cell, _ := excelize.CoordinatesToCellName(colIdx+1, rowIdx+2)
            f.SetCellValue(sheet, cell, value)
        }
    }

    // 设置列宽并保存
    for colIdx := range headers {
        col, _ := excelize.ColumnNumberToName(colIdx + 1)
        f.SetColWidth(sheet, col, col, 15)
    }

    filePath := g.generateFileName("xlsx")
    f.SaveAs(filePath)
    return filePath, nil
}
```

**CreateExcelFileStreaming 特性**:
- 使用excelize的StreamWriter
- 无需一次性加载所有数据到内存
- DataProvider函数式接口，灵活获取数据
- 自动记录处理进度日志
- 适合5000+条记录的大数据量导出

**ParseExcelFile 用途**:
- 批量任务处理时解析用户上传的Excel
- 自动排除表头行
- 返回纯数据行（[][]string）
- 用于BatchProcessor

```go
func NewMinioClient(cfg config.OSSConfig) (Client, error) {
    // 初始化MinIO客户端
    client, err := minio.New(cfg.Endpoint, &minio.Options{
        Creds:  credentials.NewStaticV4(cfg.AccessKey, cfg.SecretKey, ""),
        Secure: cfg.UseSSL,
    })

    // 检查并创建Bucket
    ctx := context.Background()
    exists, _ := client.BucketExists(ctx, cfg.Bucket)
    if !exists {
        client.MakeBucket(ctx, cfg.Bucket, minio.MakeBucketOptions{})
    }

    return &minioClient{...}
}
```

**UploadFile实现**:
```go
func (c *minioClient) UploadFile(ctx context.Context, localPath, objectName string) (string, error) {
    // 获取MIME类型
    contentType := getContentType(localPath)
    
    // 上传文件
    info, err := c.client.FPutObject(ctx, c.bucketName, objectName, localPath, minio.PutObjectOptions{
        ContentType: contentType,
    })
    
    // 生成下载URL（默认24小时）
    downloadURL, _ := c.GetDownloadURL(ctx, objectName, 24*time.Hour)
    return downloadURL, nil
}
```

**GetDownloadURL实现**:
```go
func (c *minioClient) GetDownloadURL(ctx context.Context, objectName string, expires time.Duration) (string, error) {
    // 生成预签名URL
    presignedURL, err := c.client.PresignedGetObject(ctx, c.bucketName, objectName, expires, nil)
    return presignedURL.String(), nil
}
```

**Content-Type检测**:
```go
func getContentType(filePath string) string {
    ext := filepath.Ext(filePath)
    switch ext {
    case ".xlsx":
        return "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
    case ".csv":
        return "text/csv; charset=utf-8"
    case ".pdf":
        return "application/pdf"
    default:
        return "application/octet-stream"
    }
}
```

**doRequest统一请求处理**:
```go
func (c *httpClient) doRequest(ctx context.Context, method, url string, tenantID int64) ([]byte, error) {
    req, _ := http.NewRequestWithContext(ctx, method, url, nil)
    
    // 设置请求头
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("X-Tenant-ID", fmt.Sprintf("%d", tenantID))
    
    resp, _ := c.httpClient.Do(req)
    defer resp.Body.Close()
    
    body, _ := io.ReadAll(resp.Body)
    
    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("unexpected status code: %d", resp.StatusCode)
    }
    
    return body, nil
}
```

```go
type ServiceContext struct {
    Config          config.Config

func NewServiceContext(c config.Config) *ServiceContext {
    // 初始化文件生成器
    fileGen, _ := fileutil.NewFileGenerator(c.Export.TempDir)
    
    // 初始化OSS客户端（Mock或MinIO）
    var ossClient oss.Client
    if c.OSS.Type == "mock" {
        ossClient = oss.NewMockClient(c.OSS.Bucket)
    } else {
        ossClient, _ = oss.NewMinioClient(c.OSS)
    }
    
    // 初始化Business客户端（Mock或HTTP）
    var businessClient business.Client
    if c.BusinessService.BaseURL == "mock" {
        businessClient = business.NewMockClient()
    } else {
        businessClient = business.NewHTTPClient(c.BusinessService)
    }
    
    return &ServiceContext{
        Config:          c,
        FileGenerator:   fileGen,
        OSSClient:       ossClient,
        BusinessClient:  businessClient,
    }
}
```

### 1. 文件清理
- 所有生成的临时文件必须使用defer清理
