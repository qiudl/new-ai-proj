import (

func (suite *SpuIntegrationTestSuite) SetupSuite() {

func (suite *SpuIntegrationTestSuite) TearDownSuite() {

func (suite *SpuIntegrationTestSuite) SetupTest() {
    // 每个测试前清空数据
    suite.db.Exec("TRUNCATE TABLE spus, skus CASCADE")

func (suite *SpuIntegrationTestSuite) TestCreateSPU() {
    // 1. 创建品牌
    brand := &model.Brand{BrandName: "李宁"}
    err := suite.db.Create(brand).Error
    suite.NoError(err)
    
    // 2. 创建分类
    category := &model.Category{Name: "运动鞋"}
    err = suite.db.Create(category).Error
    suite.NoError(err)
    
    // 3. 创建SPU
    spu := &model.Spu{
        Code: "TEST-SPU-001",
        Name: "李宁运动鞋",
        BrandID: brand.ID,
        CategoryID: category.ID,
        RetailPrice: 299.00,
        Enabled: true,
    }
    err = suite.db.Create(spu).Error
    suite.NoError(err)
    suite.NotZero(spu.ID)
    
    // 4. 验证创建成功
    var found model.Spu
    err = suite.db.First(&found, spu.ID).Error
    suite.NoError(err)
    suite.Equal("TEST-SPU-001", found.Code)
}

func (suite *SpuIntegrationTestSuite) TestCreateSPUWithSKUs() {
    // 创建SPU并关联多个SKU
    spu := &model.Spu{
        Code: "TEST-SPU-002",
        Name: "李宁篮球鞋",
        RetailPrice: 399.00,
        Skus: []*model.Sku{
            {Code: "SKU-001", Size: "40", Color: "红色", MarketPrice: 399.00, Stock: 100},
            {Code: "SKU-002", Size: "41", Color: "红色", MarketPrice: 399.00, Stock: 100},
            {Code: "SKU-003", Size: "42", Color: "蓝色", MarketPrice: 399.00, Stock: 100},
        },
    }
    
    err := suite.db.Create(spu).Error
    suite.NoError(err)
    
    // 验证SKU创建
    var skuCount int64
    suite.db.Model(&model.Sku{}).Where("spu_id = ?", spu.ID).Count(&skuCount)
    suite.Equal(int64(3), skuCount)
}

func (suite *SpuIntegrationTestSuite) TestUpdateSPU() {
    // 创建SPU
    spu := &model.Spu{Code: "TEST-SPU-003", Name: "原始名称", RetailPrice: 299.00}
    suite.db.Create(spu)
    
    // 更新SPU
    spu.Name = "更新后名称"
    spu.RetailPrice = 399.00
    err := suite.db.Save(spu).Error
    suite.NoError(err)
    
    // 验证更新
    var found model.Spu
    suite.db.First(&found, spu.ID)
    suite.Equal("更新后名称", found.Name)
    suite.Equal(399.00, found.RetailPrice)
}

func (suite *SpuIntegrationTestSuite) TestSoftDelete() {
    // 创建SPU
    spu := &model.Spu{Code: "TEST-SPU-004", Name: "待删除产品", RetailPrice: 199.00}
    suite.db.Create(spu)
    
    // 软删除
    err := suite.db.Delete(spu).Error
    suite.NoError(err)
    
    // 验证软删除 - 正常查询找不到
    var found model.Spu
    err = suite.db.First(&found, spu.ID).Error
    suite.Error(err)
    suite.Equal(gorm.ErrRecordNotFound, err)
    
    // 验证软删除 - Unscoped能找到
    err = suite.db.Unscoped().First(&found, spu.ID).Error
    suite.NoError(err)
    suite.NotNil(found.DeletedAt)
}

func (suite *SpuIntegrationTestSuite) TestTenantIsolation() {
    // 在tenant_1 schema创建SPU
    suite.db.Exec("SET search_path TO tenant_1")
    spu1 := &model.Spu{Code: "TENANT1-SPU", Name: "租户1产品", RetailPrice: 299.00}
    suite.db.Create(spu1)
    
    // 切换到tenant_2 schema
    suite.db.Exec("SET search_path TO tenant_2")
    
    // 验证查询不到tenant_1的数据
    var found model.Spu
    err := suite.db.Where("code = ?", "TENANT1-SPU").First(&found).Error
    suite.Error(err)
    suite.Equal(gorm.ErrRecordNotFound, err)
}

func TestSpuIntegrationTestSuite(t *testing.T) {
    suite.Run(t, new(SpuIntegrationTestSuite))
}

func (suite *SpuAPITestSuite) TestCreateSPU_Success() {
    // 准备请求数据

## 5. 自动化测试

### 5.1 CI/CD集成

**GitHub Actions配置**: `.github/workflows/test.yml`

```yaml
name: Go Tests

on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main, develop ]

jobs:
  test:
    runs-on: ubuntu-latest
    
    services:
      postgres:
        image: postgres:14
        env:
          POSTGRES_DB: coolbuy_test
          POSTGRES_USER: test
          POSTGRES_PASSWORD: test123
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 5432:5432
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up Go
      uses: actions/setup-go@v4
      with:
        go-version: '1.21'
    
    - name: Install dependencies
      run: go mod download
    
    - name: Run migrations
      run: |
        cd business-service
        ./bin/migrate -database "postgres://test:test123@localhost:5432/coolbuy_test?sslmode=disable" -path migrations up
    
    - name: Run unit tests
      run: |
        cd business-service
        go test ./... -v -coverprofile=coverage.out
    
    - name: Run integration tests
      run: |
        cd business-service
        go test ./tests/integration/... -v
    
    - name: Upload coverage
      uses: codecov/codecov-action@v3
      with:
        files: ./business-service/coverage.out
        flags: unittests
        name: codecov-umbrella
    
    - name: Check coverage threshold
      run: |
        cd business-service
        go tool cover -func=coverage.out | grep total | awk '{print $3}' | sed 's/%//' | \
        awk '{if ($1 < 70) exit 1}'
```
