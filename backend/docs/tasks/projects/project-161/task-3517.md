# 订单同步功能设计文档

## 1. 概述

订单同步功能实现从外部电商平台拉取订单数据到内部系统，支持订单状态跟踪、物流信息同步等功能。

### 1.1 同步方向

- **Pull（拉取）**: 从外部平台获取订单数据（主要模式）
- 订单通常不支持Push，因为订单由买家在平台下单产生

### 1.2 同步类型

- **实时同步**: 通过Webhook接收平台订单通知
- **定时同步**: 定时轮询平台订单接口
- **补偿同步**: 处理漏单和异常订单

func NewOrderSyncBiz() *OrderSyncBiz {

// PullOrders 拉取订单
func (b *OrderSyncBiz) PullOrders(db *gorm.DB, platformID uint, shopID string, startTime, endTime time.Time) ([]uint, error) {
	// 获取平台配置
	var platform model.MallPlatform
	if err := db.First(&platform, platformID).Error; err != nil {
		return nil, fmt.Errorf("平台不存在: %w", err)
	}
	
	// 获取授权配置
	var authConfig model.MallAuthConfig
	if err := db.Where("platform_id = ? AND shop_id = ?", platformID, shopID).First(&authConfig).Error; err != nil {
		return nil, fmt.Errorf("授权配置不存在: %w", err)
	}
	
	// 从平台拉取订单列表
	externalOrders, err := b.fetchOrdersFromPlatform(&platform, &authConfig, startTime, endTime)
	if err != nil {
		return nil, fmt.Errorf("拉取订单失败: %w", err)
	}
	
	var syncedOrderIDs []uint
	
	// 逐个处理订单
	for _, extOrder := range externalOrders {
		orderID, err := b.ProcessExternalOrder(db, platformID, shopID, extOrder)
		if err != nil {
			log.Printf("处理订单失败 [外部订单号=%s]: %v", extOrder.OrderID, err)
			continue
		}
		syncedOrderIDs = append(syncedOrderIDs, orderID)
	}
	
	return syncedOrderIDs, nil
}

// ProcessExternalOrder 处理外部订单
func (b *OrderSyncBiz) ProcessExternalOrder(db *gorm.DB, platformID uint, shopID string, extOrder *ExternalOrder) (uint, error) {
	// 检查订单是否已存在
	var existingRecord model.OrderSyncRecord
	err := db.Where("external_order_id = ?", extOrder.OrderID).First(&existingRecord).Error
	
	if err == nil {
		// 订单已存在，更新状态
		return b.UpdateExistingOrder(db, &existingRecord, extOrder)
	}
	
	// 创建新订单同步记录
	return b.CreateNewOrder(db, platformID, shopID, extOrder)
}

// CreateNewOrder 创建新订单
func (b *OrderSyncBiz) CreateNewOrder(db *gorm.DB, platformID uint, shopID string, extOrder *ExternalOrder) (uint, error) {
	return db.Transaction(func(tx *gorm.DB) (uint, error) {
		// 转换订单明细
		itemsJSON, _ := json.Marshal(extOrder.Items)
		receiverJSON, _ := json.Marshal(extOrder.Receiver)
		rawDataJSON, _ := json.Marshal(extOrder)
		
		// 创建订单同步记录
		syncRecord := &model.OrderSyncRecord{
			PlatformID:        platformID,
			ShopID:            shopID,
			ExternalOrderID:   extOrder.OrderID,
			ExternalOrderTime: extOrder.OrderTime,
			ExternalBuyerID:   extOrder.BuyerID,
			ExternalBuyerNick: extOrder.BuyerNick,
			TotalAmount:       decimal.NewFromFloat(extOrder.TotalAmount),
			PaymentAmount:     decimal.NewFromFloat(extOrder.PaymentAmount),
			DiscountAmount:    decimal.NewFromFloat(extOrder.DiscountAmount),
			FreightAmount:     decimal.NewFromFloat(extOrder.FreightAmount),
			ExternalStatus:    extOrder.Status,
			SyncStatus:        "pending",
			OrderItems:        string(itemsJSON),
			ReceiverInfo:      string(receiverJSON),
			RawData:           string(rawDataJSON),
		}
		
		if err := tx.Create(syncRecord).Error; err != nil {
			return 0, err
		}
		
		// 转换为内部订单
		internalOrder, err := b.ConvertToInternalOrder(tx, syncRecord, extOrder)
		if err != nil {
			// 标记为失败
			tx.Model(syncRecord).Updates(map[string]interface{}{
				"sync_status":   "failed",
				"error_message": err.Error(),
			})
			return 0, err
		}
		
		// 更新同步记录
		now := time.Now()
		tx.Model(syncRecord).Updates(map[string]interface{}{
			"internal_order_id": internalOrder.ID,
			"internal_order_no": internalOrder.OrderNo,
			"internal_status":   internalOrder.Status,
			"sync_status":       "synced",
			"synced_at":         now,
		})
		
		// 记录日志
		b.logOrderSync(tx, syncRecord.ID, "success", "")
		
		return syncRecord.ID, nil
	})
}

// ConvertToInternalOrder 转换为内部订单
func (b *OrderSyncBiz) ConvertToInternalOrder(db *gorm.DB, syncRecord *model.OrderSyncRecord, extOrder *ExternalOrder) (*model.Order, error) {
	// 创建内部订单
	order := &model.Order{
		OrderNo:       b.generateOrderNo(),
		OrderSource:   "mall_" + syncRecord.PlatformID,
		OrderType:     "normal",
		TotalAmount:   syncRecord.TotalAmount,
		PaymentAmount: syncRecord.PaymentAmount,
		Status:        b.mapOrderStatus(extOrder.Status),
		// ... 其他字段映射
	}
	
	if err := db.Create(order).Error; err != nil {
		return nil, err
	}
	
	// 创建订单明细
	for _, item := range extOrder.Items {
		// 查找商品映射
		var mapping model.ProductSyncMapping
		err := db.Where("platform_id = ? AND external_sku_id = ?", 
			syncRecord.PlatformID, item.SkuID).First(&mapping).Error
		
		if err != nil {
			log.Printf("未找到商品映射 [外部SKU=%s]", item.SkuID)
			continue
		}
		
		// 创建订单明细
		orderItem := &model.OrderItem{
			OrderID:    order.ID,
			SkuID:      mapping.SkuID,
			SpuID:      mapping.SpuID,
			Quantity:   item.Quantity,
			Price:      decimal.NewFromFloat(item.Price),
			TotalPrice: decimal.NewFromFloat(item.TotalPrice),
		}
		
		db.Create(orderItem)
	}
	
	return order, nil
}

// UpdateExistingOrder 更新已有订单
func (b *OrderSyncBiz) UpdateExistingOrder(db *gorm.DB, syncRecord *model.OrderSyncRecord, extOrder *ExternalOrder) (uint, error) {
	return db.Transaction(func(tx *gorm.DB) (uint, error) {
		// 检查状态是否变化
		if syncRecord.ExternalStatus == extOrder.Status {
			return syncRecord.ID, nil // 无变化，直接返回
		}
		
		// 更新同步记录
		rawDataJSON, _ := json.Marshal(extOrder)
		updates := map[string]interface{}{
			"external_status": extOrder.Status,
			"raw_data":        string(rawDataJSON),
		}
		
		if err := tx.Model(syncRecord).Updates(updates).Error; err != nil {
			return 0, err
		}
		
		// 更新内部订单状态
		if syncRecord.InternalOrderID > 0 {
			internalStatus := b.mapOrderStatus(extOrder.Status)
			tx.Model(&model.Order{}).
				Where("id = ?", syncRecord.InternalOrderID).
				Update("status", internalStatus)
			
			// 更新内部状态字段
			tx.Model(syncRecord).Update("internal_status", internalStatus)
		}
		
		// 记录日志
		b.logOrderSync(tx, syncRecord.ID, "success", fmt.Sprintf("状态更新: %s -> %s", syncRecord.ExternalStatus, extOrder.Status))
		
		return syncRecord.ID, nil
	})
}

	// 不同平台的状态映射

		"WAIT_SELLER_STOCK_OUT": "paid",
		"SENT":             "shipped",
		"FINISHED":         "completed",
		"CANCELED":         "cancelled",

		"CONFIRMED":        "paid",
		"SHIPPED":          "shipped",
		"FINISHED":         "completed",
		"CANCELLED":        "cancelled",
	}

// CreateScheduledSyncTask 创建定时同步任务
func (b *OrderSyncBiz) CreateScheduledSyncTask(db *gorm.DB, req *api.ReqCreateOrderSyncTask) (*model.SyncTask, error) {
	task := &model.SyncTask{
		TaskName:      req.TaskName,
		TaskType:      "order",
		PlatformID:    req.PlatformID,
		ShopID:        req.ShopID,
		ScheduleType:  req.ScheduleType,
		ScheduleCron:  req.ScheduleCron,
		NextRunAt:     &req.NextRunAt,
		Status:        "pending",
		MaxRetryCount: 3,
		CreatedBy:     req.CreatedBy,
		CreatedByName: req.CreatedByName,
	}
	
	// 配置同步参数
	configJSON, _ := json.Marshal(map[string]interface{}{
		"sync_hours": req.SyncHours,  // 同步最近N小时的订单
		"batch_size": req.BatchSize,  // 每批处理数量
	})
	task.TaskConfig = string(configJSON)
	
	if err := db.Create(task).Error; err != nil {
		return nil, err
	}
	
	return task, nil
}

// ExecuteSyncTask 执行同步任务
func (b *OrderSyncBiz) ExecuteSyncTask(db *gorm.DB, taskID uint) error {
	var task model.SyncTask
	if err := db.First(&task, taskID).Error; err != nil {
		return err
	}
	
	// 更新任务状态
	now := time.Now()
	db.Model(&task).Updates(map[string]interface{}{
		"status":     "running",
		"started_at": now,
	})
	
	// 解析任务配置
	var config map[string]interface{}
	json.Unmarshal([]byte(task.TaskConfig), &config)
	
	syncHours := int(config["sync_hours"].(float64))
	startTime := time.Now().Add(-time.Duration(syncHours) * time.Hour)
	endTime := time.Now()
	
	// 执行同步
	orderIDs, err := b.PullOrders(db, task.PlatformID, task.ShopID, startTime, endTime)
	
	// 更新任务结果
	completedAt := time.Now()
	executionTime := int(completedAt.Sub(now).Seconds())
	
	updates := map[string]interface{}{
		"status":         "completed",
		"completed_at":   completedAt,
		"execution_time": executionTime,
		"success_count":  len(orderIDs),
	}
	
	if err != nil {
		updates["status"] = "failed"
		updates["error_message"] = err.Error()
		updates["failed_count"] = 1
	}
	
	// 计算下次执行时间
	if task.ScheduleType == "interval" {
		nextRun := completedAt.Add(time.Duration(task.SyncInterval) * time.Second)
		updates["next_run_at"] = nextRun
	}
	
	db.Model(&task).Updates(updates)
	
	return err
}

// HandleWebhook 处理Webhook通知
func (b *OrderSyncBiz) HandleWebhook(db *gorm.DB, platformCode, shopID string, payload []byte) error {
	// 验证签名
	if err := b.verifyWebhookSignature(platformCode, payload); err != nil {
		return fmt.Errorf("签名验证失败: %w", err)
	}
	
	// 解析Webhook数据
	var webhookData WebhookData
	if err := json.Unmarshal(payload, &webhookData); err != nil {
		return err
	}
	
	// 根据事件类型处理
	switch webhookData.EventType {
	case "order_create":
		return b.handleOrderCreate(db, platformCode, shopID, webhookData.Data)
	case "order_update":
		return b.handleOrderUpdate(db, platformCode, shopID, webhookData.Data)
	case "order_cancel":
		return b.handleOrderCancel(db, platformCode, shopID, webhookData.Data)
	default:
		log.Printf("未知的Webhook事件类型: %s", webhookData.EventType)
		return nil
	}
}

// handleOrderCreate 处理订单创建事件
func (b *OrderSyncBiz) handleOrderCreate(db *gorm.DB, platformCode, shopID string, data map[string]interface{}) error {
	// 获取平台ID
	var platform model.MallPlatform
	db.Where("platform_code = ?", platformCode).First(&platform)
	
	// 转换为标准订单结构
	extOrder := b.parseWebhookOrder(data)
	
	// 处理订单
	_, err := b.ProcessExternalOrder(db, platform.ID, shopID, extOrder)
	return err
}

// DetectMissingOrders 检测漏单
func (b *OrderSyncBiz) DetectMissingOrders(db *gorm.DB, platformID uint, shopID string, checkDate time.Time) ([]string, error) {
	// 获取该日期的所有外部订单号
	externalOrders, err := b.fetchOrdersFromPlatform(platformID, shopID, checkDate, checkDate.Add(24*time.Hour))
	if err != nil {
		return nil, err
	}
	
	externalOrderIDs := make([]string, len(externalOrders))
	for i, order := range externalOrders {
		externalOrderIDs[i] = order.OrderID
	}
	
	// 查询已同步的订单
	var syncedRecords []model.OrderSyncRecord
	db.Where("platform_id = ? AND shop_id = ? AND DATE(external_order_time) = ?",
		platformID, shopID, checkDate.Format("2006-01-02")).
		Select("external_order_id").
		Find(&syncedRecords)
	
	syncedMap := make(map[string]bool)
	for _, record := range syncedRecords {
		syncedMap[record.ExternalOrderID] = true
	}
	
	// 找出漏单
	var missingOrders []string
	for _, orderID := range externalOrderIDs {
		if !syncedMap[orderID] {
			missingOrders = append(missingOrders, orderID)
		}
	}
	
	return missingOrders, nil
}

// CompensateOrders 补偿漏单
func (b *OrderSyncBiz) CompensateOrders(db *gorm.DB, platformID uint, shopID string, orderIDs []string) error {
	for _, orderID := range orderIDs {
		// 获取订单详情
		extOrder, err := b.fetchOrderDetail(platformID, shopID, orderID)
		if err != nil {
			log.Printf("获取订单详情失败 [订单号=%s]: %v", orderID, err)
			continue
		}
		
		// 处理订单
		if _, err := b.ProcessExternalOrder(db, platformID, shopID, extOrder); err != nil {
			log.Printf("补偿订单失败 [订单号=%s]: %v", orderID, err)
		}
	}
	
	return nil
}

// fetchOrdersFromTaobao 从淘宝拉取订单
func (b *OrderSyncBiz) fetchOrdersFromTaobao(platform *model.MallPlatform, authConfig *model.MallAuthConfig, startTime, endTime time.Time) ([]*ExternalOrder, error) {
	client := NewTaobaoClient(platform, authConfig)
	
	// 调用淘宝订单查询API
	response, err := client.GetOrders(map[string]interface{}{
		"start_created": startTime.Format("2006-01-02 15:04:05"),
		"end_created":   endTime.Format("2006-01-02 15:04:05"),
		"page_size":     100,
	})
	
	if err != nil {
		return nil, err
	}
	
	// 转换为标准格式
	var orders []*ExternalOrder
	for _, tbOrder := range response.Orders {
		order := &ExternalOrder{
			OrderID:       tbOrder.Tid,
			OrderTime:     tbOrder.Created,
			BuyerID:       tbOrder.BuyerID,
			BuyerNick:     tbOrder.BuyerNick,
			TotalAmount:   tbOrder.TotalFee,
			PaymentAmount: tbOrder.Payment,
			Status:        tbOrder.Status,
		}
		
		// 转换订单明细
		for _, tbItem := range tbOrder.Orders {
			order.Items = append(order.Items, ExternalOrderItem{
				SkuID:      tbItem.SkuID,
				Title:      tbItem.Title,
				Quantity:   tbItem.Num,
				Price:      tbItem.Price,
				TotalPrice: tbItem.TotalFee,
			})
		}
		
		orders = append(orders, order)
	}
	
	return orders, nil
}

// PullOrdersBatch 批量拉取订单（分页）
func (b *OrderSyncBiz) PullOrdersBatch(db *gorm.DB, platformID uint, shopID string, startTime, endTime time.Time, batchSize int) error {
	page := 1
	
	for {
		// 分页拉取
		orders, hasMore, err := b.fetchOrdersWithPagination(platformID, shopID, startTime, endTime, page, batchSize)
		if err != nil {
			return err
		}
		
		// 处理当前批次
		for _, order := range orders {
			b.ProcessExternalOrder(db, platformID, shopID, order)
		}
		
		if !hasMore {
			break
		}
		
		page++
		time.Sleep(1 * time.Second) // 避免请求过快
	}
	
	return nil
}
