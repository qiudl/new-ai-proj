# 商品同步功能设计文档

## 1. 概述

商品同步功能实现内部商品数据与外部电商平台商品数据的双向同步，包括商品基本信息、规格、图片、描述等内容的同步。

### 1.1 同步方向

- **Pull（拉取）**: 从外部平台获取商品数据，更新到内部系统
- **Push（推送）**: 将内部商品数据推送到外部平台

### 1.2 同步模式

- **全量同步**: 同步所有商品数据
- **增量同步**: 仅同步变更的商品数据
- **单品同步**: 同步指定的单个商品

```go
package biz

import (

// ProductMappingBiz 商品映射业务逻辑
type ProductMappingBiz struct{}

// NewProductMappingBiz 创建商品映射业务逻辑
func NewProductMappingBiz() *ProductMappingBiz {
	return &ProductMappingBiz{}

// CreateMapping 创建商品映射
func (b *ProductMappingBiz) CreateMapping(db *gorm.DB, req *api.ReqCreateProductMapping) (*model.ProductSyncMapping, error) {
	// 检查是否已存在映射
	var existingMapping model.ProductSyncMapping
	err := db.Where("platform_id = ? AND shop_id = ? AND external_sku_id = ?", 
		req.PlatformID, req.ShopID, req.ExternalSkuID).
		First(&existingMapping).Error
	
	if err == nil {
		return nil, fmt.Errorf("商品映射已存在")
	}
	
	// 获取内部SKU信息
	var sku model.Sku
	if err := db.Preload("Spu").First(&sku, req.SkuID).Error; err != nil {
		return nil, fmt.Errorf("SKU不存在: %w", err)
	}
	
	mapping := &model.ProductSyncMapping{
		SpuID:         sku.SpuID,
		SkuID:         req.SkuID,
		SpuCode:       sku.Spu.Code,
		SkuCode:       sku.Code,
		PlatformID:    req.PlatformID,
		ShopID:        req.ShopID,
		ExternalSpuID: req.ExternalSpuID,
		ExternalSkuID: req.ExternalSkuID,
		MappingType:   "sku",
		SyncMode:      req.SyncMode,
		EnableSync:    req.EnableSync,
		SyncPrice:     req.SyncPrice,
		SyncInventory: req.SyncInventory,
		SyncImages:    req.SyncImages,
		SyncDesc:      req.SyncDesc,
		Status:        "active",
	}
	
	if err := db.Create(mapping).Error; err != nil {
		return nil, fmt.Errorf("创建映射失败: %w", err)
	}
	
	return mapping, nil
}

// BatchCreateMappings 批量创建映射
func (b *ProductMappingBiz) BatchCreateMappings(db *gorm.DB, req *api.ReqBatchCreateMappings) ([]uint, error) {
	var mappingIDs []uint
	
	err := db.Transaction(func(tx *gorm.DB) error {
		for _, item := range req.Mappings {
			mapping, err := b.CreateMapping(tx, item)
			if err != nil {
				return err
			}
			mappingIDs = append(mappingIDs, mapping.ID)
		}
		return nil
	})
	
	if err != nil {
		return nil, err
	}
	
	return mappingIDs, nil
}
```

```go
func (b *ProductMappingBiz) AutoMatchMapping(db *gorm.DB, platformID uint, shopID string) ([]model.ProductSyncMapping, error) {

	// 获取外部平台商品列表
	externalProducts, err := b.fetchExternalProducts(platformID, shopID)
	if err != nil {
		return nil, err
	}
	
	for _, extProduct := range externalProducts {
		// 尝试通过条码匹配

		err := db.Where("eanupc = ?", extProduct.Barcode).First(&sku).Error
		if err != nil {
			// 尝试通过款号匹配
			err = db.Where("style_number = ?", extProduct.StyleNumber).First(&sku).Error
			if err != nil {
				continue // 无法匹配，跳过
			}
		}
		
		// 创建映射
		mapping := model.ProductSyncMapping{
			SpuID:         sku.SpuID,
			SkuID:         sku.ID,
			SkuCode:       sku.Code,
			PlatformID:    platformID,
			ShopID:        shopID,

			Status:        "active",
		}

			mappings = append(mappings, mapping)
		}
	}
```

## 3. 商品数据拉取（Pull）

### 3.1 拉取外部商品数据

```go
package biz

import (
	"encoding/json"
	"time"
)

// ProductSyncBiz 商品同步业务逻辑
type ProductSyncBiz struct{}

// NewProductSyncBiz 创建商品同步业务逻辑
func NewProductSyncBiz() *ProductSyncBiz {
	return &ProductSyncBiz{}
}

// PullProduct 从外部平台拉取商品数据
func (b *ProductSyncBiz) PullProduct(db *gorm.DB, mappingID uint) error {
	// 获取映射关系
	var mapping model.ProductSyncMapping
	if err := db.Preload("Platform").Preload("Sku").First(&mapping, mappingID).Error; err != nil {
		return fmt.Errorf("映射不存在: %w", err)
	}
	
	// 从外部平台获取商品数据
	externalProduct, err := b.fetchProductFromPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID)
	if err != nil {
		b.logSyncError(db, mappingID, "pull", err)
		return err
	}
	
	// 开始事务更新
	return db.Transaction(func(tx *gorm.DB) error {
		// 更新SKU基本信息
		updates := make(map[string]interface{})
		
		if mapping.SyncPrice && externalProduct.Price > 0 {
			updates["market_price"] = externalProduct.Price
		}
		
		if mapping.SyncInventory && externalProduct.Stock >= 0 {
			updates["stock"] = externalProduct.Stock
		}
		
		if len(updates) > 0 {
			if err := tx.Model(&mapping.Sku).Updates(updates).Error; err != nil {
				return err
			}
		}
		
		// 更新映射状态
		tx.Model(&mapping).Updates(map[string]interface{}{
			"last_sync_at":     time.Now(),
			"last_sync_status": "success",
		})
		
		// 记录同步日志
		b.logSyncSuccess(tx, mappingID, "pull", externalProduct)
		
		return nil
	})
}

// PullProductBatch 批量拉取商品数据
func (b *ProductSyncBiz) PullProductBatch(db *gorm.DB, platformID uint, shopID string) error {
	// 获取所有启用同步的映射
	var mappings []model.ProductSyncMapping
	err := db.Where("platform_id = ? AND shop_id = ? AND enable_sync = ? AND status = ?",
		platformID, shopID, true, "active").
		Find(&mappings).Error
	if err != nil {
		return err
	}
	
	successCount := 0
	failedCount := 0
	
	for _, mapping := range mappings {
		if err := b.PullProduct(db, mapping.ID); err != nil {
			failedCount++
			continue
		}
		successCount++
	}
	
	log.Printf("批量拉取完成: 成功=%d, 失败=%d", successCount, failedCount)
	return nil
}
```

### 4.1 推送商品到外部平台

```go
// PushProduct 推送商品到外部平台
func (b *ProductSyncBiz) PushProduct(db *gorm.DB, mappingID uint) error {
	// 获取映射和SKU数据
	var mapping model.ProductSyncMapping
	if err := db.Preload("Platform").Preload("Sku.Spu").First(&mapping, mappingID).Error; err != nil {
		return fmt.Errorf("映射不存在: %w", err)
	}
	
	// 构建推送数据
	pushData := b.buildPushData(&mapping)
	
	// 推送到外部平台
	err := b.pushProductToPlatform(mapping.Platform, mapping.ShopID, pushData)
	if err != nil {
		b.logSyncError(db, mappingID, "push", err)
		return err
	}
	
	// 更新同步状态
	db.Model(&mapping).Updates(map[string]interface{}{
		"last_sync_at":     time.Now(),
		"last_sync_status": "success",
	})
	
	// 记录日志
	b.logSyncSuccess(db, mappingID, "push", pushData)
	
	return nil
}

// buildPushData 构建推送数据
func (b *ProductSyncBiz) buildPushData(mapping *model.ProductSyncMapping) map[string]interface{} {
	data := make(map[string]interface{})
	
	sku := mapping.Sku
	spu := sku.Spu
	
	// 基本信息
	data["external_sku_id"] = mapping.ExternalSkuID
	data["title"] = spu.Name
	data["outer_id"] = sku.Code
	
	// 价格
	if mapping.SyncPrice {
		data["price"] = sku.MarketPrice
	}
	
	// 库存
	if mapping.SyncInventory {
		data["stock"] = sku.Stock
	}
	
	// 图片
	if mapping.SyncImages && spu.MainImage != "" {
		data["images"] = []string{spu.MainImage}
	}
	
	// 描述
	if mapping.SyncDesc {
		data["description"] = spu.Description
	}
	
	return data
}

// pushProductToPlatform 推送到平台
func (b *ProductSyncBiz) pushProductToPlatform(platform *model.MallPlatform, shopID string, data map[string]interface{}) error {
	switch platform.PlatformCode {
	case "taobao", "tmall":
		return b.pushToTaobao(platform, shopID, data)
	case "jd":
		return b.pushToJD(platform, shopID, data)
	case "pdd":
		return b.pushToPDD(platform, shopID, data)
	default:
		return fmt.Errorf("不支持的平台: %s", platform.PlatformCode)
	}
}

// pushToTaobao 推送到淘宝/天猫
func (b *ProductSyncBiz) pushToTaobao(platform *model.MallPlatform, shopID string, data map[string]interface{}) error {
	apiClient := NewTaobaoClient(platform)
	
	// 调用淘宝商品更新API
	_, err := apiClient.UpdateItem(data)
	if err != nil {
		return fmt.Errorf("推送失败: %w", err)
	}
	
	return nil
}
```

### 5.1 检测变更

```go
// DetectChanges 检测商品变更
func (b *ProductSyncBiz) DetectChanges(db *gorm.DB, mappingID uint) (bool, []string, error) {
	var mapping model.ProductSyncMapping
	if err := db.Preload("Sku").First(&mapping, mappingID).Error; err != nil {
		return false, nil, err
	}
	
	// 获取外部商品数据
	externalProduct, err := b.fetchProductFromPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID)
	if err != nil {
		return false, nil, err
	}
	
	var changes []string
	hasChanges := false
	
	// 比较价格
	if mapping.SyncPrice && mapping.Sku.MarketPrice != externalProduct.Price {
		changes = append(changes, fmt.Sprintf("价格: %.2f -> %.2f", mapping.Sku.MarketPrice, externalProduct.Price))
		hasChanges = true
	}
	
	// 比较库存
	if mapping.SyncInventory && mapping.Sku.Stock != externalProduct.Stock {
		changes = append(changes, fmt.Sprintf("库存: %d -> %d", mapping.Sku.Stock, externalProduct.Stock))
		hasChanges = true
	}
	
	return hasChanges, changes, nil
}

// IncrementalSync 增量同步
func (b *ProductSyncBiz) IncrementalSync(db *gorm.DB, platformID uint, shopID string) error {
	var mappings []model.ProductSyncMapping
	db.Where("platform_id = ? AND shop_id = ? AND enable_sync = ?", platformID, shopID, true).
		Find(&mappings)
	
	for _, mapping := range mappings {
		hasChanges, changes, err := b.DetectChanges(db, mapping.ID)
		if err != nil {
			continue
		}
		
		if hasChanges {
			log.Printf("检测到变更 [映射ID=%d]: %v", mapping.ID, changes)
			b.PullProduct(db, mapping.ID)
		}
	}
	
	return nil
}
```

## 6. 冲突处理

### 6.1 冲突检测与解决

```go
// ConflictResolutionStrategy 冲突解决策略
type ConflictResolutionStrategy string

const (
	// StrategyLocalFirst 本地优先
	StrategyLocalFirst ConflictResolutionStrategy = "local_first"
	// StrategyRemoteFirst 远程优先
	StrategyRemoteFirst ConflictResolutionStrategy = "remote_first"
	// StrategyManual 手动处理
	StrategyManual ConflictResolutionStrategy = "manual"
	// StrategyNewest 最新优先
	StrategyNewest ConflictResolutionStrategy = "newest"
)

// ResolveConflict 解决同步冲突
func (b *ProductSyncBiz) ResolveConflict(db *gorm.DB, mappingID uint, strategy ConflictResolutionStrategy) error {
	var mapping model.ProductSyncMapping
	db.Preload("Sku").First(&mapping, mappingID)
	
	// 获取外部数据
	externalProduct, err := b.fetchProductFromPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID)
	if err != nil {
		return err
	}
	
	switch strategy {
	case StrategyLocalFirst:
		// 推送本地数据到平台
		return b.PushProduct(db, mappingID)
		
	case StrategyRemoteFirst:
		// 拉取平台数据到本地
		return b.PullProduct(db, mappingID)
		
	case StrategyNewest:
		// 比较更新时间
		if mapping.Sku.UpdatedAt.After(externalProduct.UpdatedAt) {
			return b.PushProduct(db, mappingID)
		}
		return b.PullProduct(db, mappingID)
		
	case StrategyManual:
		// 创建冲突记录，等待人工处理
		return b.createConflictRecord(db, mappingID, externalProduct)
		
	default:
		return fmt.Errorf("不支持的冲突解决策略: %s", strategy)
	}
}
```

```go
// SyncWithRetry 带重试的同步
func (b *ProductSyncBiz) SyncWithRetry(db *gorm.DB, mappingID uint, direction string, maxRetries int) error {
	var lastErr error
	
	for i := 0; i < maxRetries; i++ {
		var err error
		if direction == "pull" {
			err = b.PullProduct(db, mappingID)
		} else {
			err = b.PushProduct(db, mappingID)
		}
		
		if err == nil {
			return nil // 成功
		}
		
		lastErr = err
		log.Printf("同步失败，重试 %d/%d: %v", i+1, maxRetries, err)
		
		// 指数退避
		time.Sleep(time.Duration(1<<uint(i)) * time.Second)
	}
	
	return fmt.Errorf("同步失败，已重试%d次: %w", maxRetries, lastErr)
}
```

### 9.1 并发同步

```go
// ConcurrentSync 并发同步多个商品
func (b *ProductSyncBiz) ConcurrentSync(db *gorm.DB, mappingIDs []uint, workers int) error {
	jobs := make(chan uint, len(mappingIDs))
	results := make(chan error, len(mappingIDs))
	
	// 启动worker
	for w := 0; w < workers; w++ {
		go func() {
			for mappingID := range jobs {
				err := b.PullProduct(db, mappingID)
				results <- err
			}
		}()
	}
	
	// 发送任务
	for _, id := range mappingIDs {
		jobs <- id
	}
	close(jobs)
	
	// 收集结果
	var errors []error
	for i := 0; i < len(mappingIDs); i++ {
		if err := <-results; err != nil {
			errors = append(errors, err)
		}
	}
	
	if len(errors) > 0 {
		return fmt.Errorf("部分同步失败: %d个错误", len(errors))
	}
	
	return nil
}
```

```go
import "golang.org/x/time/rate"

// RateLimitedSync 限流同步
func (b *ProductSyncBiz) RateLimitedSync(db *gorm.DB, mappingIDs []uint, rateLimit int) error {
	limiter := rate.NewLimiter(rate.Limit(rateLimit), rateLimit)
	
	for _, mappingID := range mappingIDs {
		// 等待限流器允许
		if err := limiter.Wait(context.Background()); err != nil {
			return err
		}
		
		// 执行同步
		if err := b.PullProduct(db, mappingID); err != nil {
			log.Printf("同步失败 [ID=%d]: %v", mappingID, err)
		}
	}
	
	return nil
}
```
