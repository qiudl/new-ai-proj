# 业绩计算逻辑设计文档

## 1. 概述

本文档定义业绩和佣金的计算逻辑，包括订单业绩录入、佣金计算、统计汇总等核心算法。

## 2. 业绩录入

### 2.1 订单完成触发业绩录入

```go
// RecordPerformance 记录业绩
func (b *PerformanceBiz) RecordPerformance(db *gorm.DB, order *model.Order) error {
	return db.Transaction(func(tx *gorm.DB) error {
		// 计算业绩金额
		salesAmount := order.TotalAmount
		costAmount := b.calculateCostAmount(tx, order)
		profitAmount := salesAmount.Sub(costAmount)
		
		// 获取统计周期
		statPeriod := order.CompletedAt.Format("2006-01-02")
		statYear := order.CompletedAt.Year()
		statMonth := int(order.CompletedAt.Month())
		_, statWeek := order.CompletedAt.ISOWeek()
		
		// 创建业绩记录
		record := &model.PerformanceRecord{
			UserID:            order.SalespersonID,
			UserName:          order.SalespersonName,
			UserRole:          "salesperson",
			OrderID:           order.ID,
			OrderNo:           order.OrderNo,
			OrderDate:         *order.CompletedAt,
			SalesAmount:       salesAmount,
			CostAmount:        costAmount,
			ProfitAmount:      profitAmount,
			DiscountAmount:    order.DiscountAmount,
			PerformanceType:   "sales",
			PerformanceWeight: decimal.NewFromFloat(1.00),
			StatPeriod:        statPeriod,
			StatYear:          statYear,
			StatMonth:         statMonth,
			StatWeek:          statWeek,
			Status:            "confirmed",
		}
		
		if err := tx.Create(record).Error; err != nil {
			return fmt.Errorf("创建业绩记录失败: %w", err)
		}
		
		// 触发佣金计算
		if err := b.calculateCommission(tx, record); err != nil {
			return fmt.Errorf("计算佣金失败: %w", err)
		}
		
		return nil
	})
}
```

## 3. 佣金计算

### 3.1 核心计算逻辑

```go
// calculateCommission 计算佣金
func (b *PerformanceBiz) calculateCommission(tx *gorm.DB, record *model.PerformanceRecord) error {
	// 1. 获取适用的佣金模板
	template, err := b.getApplicableTemplate(tx, record.UserID, record.UserRole)
	if err != nil {
		return fmt.Errorf("获取佣金模板失败: %w", err)
	}
	
	// 2. 根据模板类型获取计算基数
	baseAmount := b.getBaseAmount(record, template.TemplateType)
	
	// 3. 查找匹配的规则
	var rules []model.CommissionRule
	tx.Where("template_id = ? AND status = ?", template.ID, "active").
		Order("priority DESC").
		Find(&rules)
	
	var matchedRule *model.CommissionRule
	for _, rule := range rules {
		if rule.IsApplicable(baseAmount) {
			matchedRule = &rule
			break
		}
	}
	
	if matchedRule == nil {
		return errors.New("没有匹配的佣金规则")
	}
	
	// 4. 计算佣金金额
	var commissionAmount, bonusAmount decimal.Decimal
	
	switch template.CalculateMethod {
	case "percentage":
		commissionAmount = baseAmount.Mul(matchedRule.CommissionRate)
	case "fixed":
		commissionAmount = matchedRule.FixedAmount
	case "tiered":
		commissionAmount = b.calculateTieredCommission(baseAmount, rules)
	}
	
	bonusAmount = matchedRule.BonusAmount
	totalAmount := commissionAmount.Add(bonusAmount)
	
	// 5. 创建佣金记录
	commissionRecord := &model.CommissionRecord{
		UserID:              record.UserID,
		UserName:            record.UserName,
		PerformanceRecordID: record.ID,
		OrderID:             record.OrderID,
		OrderNo:             record.OrderNo,
		TemplateID:          template.ID,
		TemplateCode:        template.TemplateCode,
		RuleID:              matchedRule.ID,
		BaseAmount:          baseAmount,
		CommissionRate:      matchedRule.CommissionRate,
		CommissionAmount:    commissionAmount,
		BonusAmount:         bonusAmount,
		TotalAmount:         totalAmount,
		StatPeriod:          record.StatPeriod,
		StatYear:            record.StatYear,
		StatMonth:           record.StatMonth,
		Status:              "pending",
		CalculatedAt:        time.Now(),
	}
	
	if err := tx.Create(commissionRecord).Error; err != nil {
		return fmt.Errorf("创建佣金记录失败: %w", err)
	}
	
	// 6. 更新统计表
	return b.updateStatistics(tx, record.UserID, record.StatPeriod)
}
```

### 3.2 阶梯佣金计算

```go
// calculateTieredCommission 计算阶梯佣金
func (b *PerformanceBiz) calculateTieredCommission(amount decimal.Decimal, rules []model.CommissionRule) decimal.Decimal {
	totalCommission := decimal.Zero
	remainingAmount := amount
	
	// 按MinAmount排序
	sort.Slice(rules, func(i, j int) bool {
		return rules[i].MinAmount.LessThan(rules[j].MinAmount)
	})
	
	for i, rule := range rules {
		if remainingAmount.IsZero() {
			break
		}
		
		// 计算当前阶梯的金额
		var tierAmount decimal.Decimal
		if rule.MaxAmount.IsZero() {
			// 无上限，使用剩余全部金额
			tierAmount = remainingAmount
		} else {
			tierMax := rule.MaxAmount.Sub(rule.MinAmount)
			if remainingAmount.LessThanOrEqual(tierMax) {
				tierAmount = remainingAmount
			} else {
				tierAmount = tierMax
			}
		}
		
		// 计算该阶梯的佣金
		tierCommission := tierAmount.Mul(rule.CommissionRate)
		totalCommission = totalCommission.Add(tierCommission)
		
		remainingAmount = remainingAmount.Sub(tierAmount)
	}
	
	return totalCommission
}
```

### 3.3 获取计算基数

```go
// getBaseAmount 根据模板类型获取计算基数
func (b *PerformanceBiz) getBaseAmount(record *model.PerformanceRecord, templateType string) decimal.Decimal {
	switch templateType {
	case "sales":
		return record.SalesAmount
	case "profit":
		return record.ProfitAmount
	case "quantity":
		return decimal.NewFromInt(1) // 按单计算
	default:
		return record.SalesAmount
	}
}
```

## 4. 统计更新

```go
// updateStatistics 更新业绩统计
func (b *PerformanceBiz) updateStatistics(tx *gorm.DB, userID uint, statPeriod string) error {
	// 查询或创建统计记录
	var stats model.PerformanceStatistics
	result := tx.Where("user_id = ? AND stat_period = ?", userID, statPeriod).
		First(&stats)
	
	if result.Error == gorm.ErrRecordNotFound {
		// 创建新记录
		stats = model.PerformanceStatistics{
			UserID:     userID,
			StatPeriod: statPeriod,
		}
	}
	
	// 汇总业绩数据
	var performance struct {
		OrderCount   int64
		SalesAmount  decimal.Decimal
		CostAmount   decimal.Decimal
		ProfitAmount decimal.Decimal
	}
	
	tx.Model(&model.PerformanceRecord{}).
		Where("user_id = ? AND stat_period = ? AND status = ?", userID, statPeriod, "confirmed").
		Select("COUNT(*) as order_count, SUM(sales_amount) as sales_amount, SUM(cost_amount) as cost_amount, SUM(profit_amount) as profit_amount").
		Scan(&performance)
	
	// 汇总佣金数据
	var commission struct {
		CommissionAmount decimal.Decimal
		BonusAmount      decimal.Decimal
		TotalIncome      decimal.Decimal
	}
	
	tx.Model(&model.CommissionRecord{}).
		Where("user_id = ? AND stat_period = ? AND status IN ?", userID, statPeriod, []string{"pending", "confirmed", "paid"}).
		Select("SUM(commission_amount) as commission_amount, SUM(bonus_amount) as bonus_amount, SUM(total_amount) as total_income").
		Scan(&commission)
	
	// 更新统计
	stats.OrderCount = int(performance.OrderCount)
	stats.SalesAmount = performance.SalesAmount
	stats.CostAmount = performance.CostAmount
	stats.ProfitAmount = performance.ProfitAmount
	stats.CommissionAmount = commission.CommissionAmount
	stats.BonusAmount = commission.BonusAmount
	stats.TotalIncome = commission.TotalIncome
	stats.LastCalculatedAt = time.Now()
	
	if stats.ID == 0 {
		return tx.Create(&stats).Error
	} else {
		return tx.Save(&stats).Error
	}
}
```

## 5. 批量计算

```go
// BatchCalculateCommission 批量计算佣金
func (b *PerformanceBiz) BatchCalculateCommission(db *gorm.DB, statPeriod string) error {
	// 查询待计算的业绩记录
	var records []model.PerformanceRecord
	db.Where("stat_period = ? AND status = ?", statPeriod, "confirmed").
		Find(&records)
	
	for _, record := range records {
		// 检查是否已计算
		var count int64
		db.Model(&model.CommissionRecord{}).
			Where("performance_record_id = ?", record.ID).
			Count(&count)
		
		if count > 0 {
			continue // 已计算，跳过
		}
		
		// 计算佣金
		if err := b.calculateCommission(db, &record); err != nil {
			log.Printf("计算佣金失败 - Record ID: %d, Error: %v", record.ID, err)
			continue
		}
	}
	
	return nil
}
```
