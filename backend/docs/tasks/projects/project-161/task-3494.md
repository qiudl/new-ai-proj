# 物流发货数据库Schema设计文档

## 文档概述

本文档定义物流发货模块的完整数据库结构设计，包括物流单管理、发货明细、物流跟踪、物流商配置等核心数据表，采用PostgreSQL数据库，支持多租户Schema隔离。

## 多租户架构

### Schema隔离策略

```
PostgreSQL Database: paas_business
├── public (公共配置)
│   └── logistics_providers (物流商配置 - 跨租户共享)
├── tenant_1 (租户1)
│   ├── shipments
│   ├── shipment_items
│   ├── shipment_tracks
│   └── shipment_addresses
└── tenant_2 (租户2)
    ├── shipments
    ├── shipment_items
    ├── shipment_tracks
    └── shipment_addresses
```

**设计原则**:
- 物流商配置放在public schema，所有租户共享
- 业务数据按租户Schema隔离
- 通过中间件自动切换search_path

## 核心数据表设计

### 1. 物流单表 (shipments)

物流单是物流发货的核心实体，记录每次发货的完整信息。

```go
package model

import (
    "time"
    "github.com/shopspring/decimal"
    "gorm.io/gorm"
)

// Shipment 物流单模型
// 多租户通过PostgreSQL Schema隔离，无需TenantID字段
type Shipment struct {
    ID uint `gorm:"primaryKey;autoIncrement" json:"id"`
    
    // 基本信息
    ShipmentNo   string `gorm:"type:varchar(50);uniqueIndex;not null" json:"shipment_no"` // 物流单号
    OrderID      uint   `gorm:"index;not null" json:"order_id"`                           // 关联订单ID
    OrderNo      string `gorm:"type:varchar(50);index;not null" json:"order_no"`          // 订单编号
    
    // 物流信息
    ProviderID      uint   `gorm:"index;not null" json:"provider_id"`                    // 物流商ID
    ProviderCode    string `gorm:"type:varchar(50);not null" json:"provider_code"`       // 物流商编码
    ProviderName    string `gorm:"type:varchar(100);not null" json:"provider_name"`      // 物流商名称
    TrackingNo      string `gorm:"type:varchar(100);index" json:"tracking_no"`           // 物流追踪号
    TrackingURL     string `gorm:"type:varchar(500)" json:"tracking_url"`                // 物流追踪URL
    
    // 发货信息
    ShipmentType    string    `gorm:"type:varchar(20);not null;index" json:"shipment_type"` // 发货类型: full-全部发货, partial-部分发货
    ShipmentMethod  string    `gorm:"type:varchar(20)" json:"shipment_method"`              // 发货方式: express-快递, logistics-物流, self_pickup-自提
    WarehouseID     uint      `gorm:"index" json:"warehouse_id"`                            // 发货仓库ID
    WarehouseName   string    `gorm:"type:varchar(100)" json:"warehouse_name"`              // 仓库名称
    ShippedAt       time.Time `gorm:"index" json:"shipped_at"`                              // 发货时间
    
    // 地址信息
    SenderName      string `gorm:"type:varchar(100);not null" json:"sender_name"`       // 发件人姓名
    SenderPhone     string `gorm:"type:varchar(20);not null" json:"sender_phone"`       // 发件人电话
    SenderProvince  string `gorm:"type:varchar(50)" json:"sender_province"`             // 发件省份
    SenderCity      string `gorm:"type:varchar(50)" json:"sender_city"`                 // 发件城市
    SenderDistrict  string `gorm:"type:varchar(50)" json:"sender_district"`             // 发件区县
    SenderAddress   string `gorm:"type:varchar(500);not null" json:"sender_address"`    // 发件详细地址
    
    ReceiverName    string `gorm:"type:varchar(100);not null" json:"receiver_name"`     // 收件人姓名
    ReceiverPhone   string `gorm:"type:varchar(20);not null" json:"receiver_phone"`     // 收件人电话
    ReceiverProvince string `gorm:"type:varchar(50)" json:"receiver_province"`          // 收件省份
    ReceiverCity     string `gorm:"type:varchar(50)" json:"receiver_city"`              // 收件城市
    ReceiverDistrict string `gorm:"type:varchar(50)" json:"receiver_district"`          // 收件区县
    ReceiverAddress  string `gorm:"type:varchar(500);not null" json:"receiver_address"` // 收件详细地址
    
    // 费用信息
    FreightAmount  decimal.Decimal `gorm:"type:decimal(10,2);not null;default:0" json:"freight_amount"`   // 运费
    InsuranceAmount decimal.Decimal `gorm:"type:decimal(10,2);default:0" json:"insurance_amount"`         // 保价金额
    CODAmount      decimal.Decimal `gorm:"type:decimal(10,2);default:0" json:"cod_amount"`               // 代收货款金额
    
    // 包裹信息
    PackageCount   int     `gorm:"type:int;not null;default:1" json:"package_count"`       // 包裹数量
    TotalWeight    float64 `gorm:"type:decimal(10,2)" json:"total_weight"`                 // 总重量(kg)
    TotalVolume    float64 `gorm:"type:decimal(10,2)" json:"total_volume"`                 // 总体积(m³)
    
    // 状态信息
    Status         string     `gorm:"type:varchar(20);not null;default:'pending';index" json:"status"` // 状态: pending-待发货, shipped-已发货, in_transit-运输中, delivered-已送达, returned-已退回, cancelled-已取消
    CurrentLocation string    `gorm:"type:varchar(200)" json:"current_location"`                       // 当前位置
    DeliveredAt    *time.Time `json:"delivered_at"`                                                    // 签收时间
    SignedBy       string     `gorm:"type:varchar(100)" json:"signed_by"`                              // 签收人
    
    // 备注信息
    Remark         string `gorm:"type:text" json:"remark"`                     // 备注
    OperatorID     uint   `gorm:"index" json:"operator_id"`                    // 操作人ID
    OperatorName   string `gorm:"type:varchar(100)" json:"operator_name"`      // 操作人姓名
    
    // 时间字段
    CreatedAt time.Time      `json:"created_at"`
    UpdatedAt time.Time      `json:"updated_at"`
    DeletedAt gorm.DeletedAt `gorm:"index" json:"-"`
    
    // 关联关系
    Items  []ShipmentItem  `gorm:"foreignKey:ShipmentID" json:"items,omitempty"`   // 发货明细
    Tracks []ShipmentTrack `gorm:"foreignKey:ShipmentID" json:"tracks,omitempty"`  // 物流轨迹
}

// TableName 指定表名
func (Shipment) TableName() string {
    return "shipments"
}

// GetShipmentNo 获取物流单号
func (s *Shipment) GetShipmentNo() string {
    return s.ShipmentNo
}

// IsPartialShipment 是否部分发货
func (s *Shipment) IsPartialShipment() bool {
    return s.ShipmentType == "partial"
}

// CanCancel 是否可以取消
func (s *Shipment) CanCancel() bool {
    return s.Status == "pending" || s.Status == "shipped"
}
```

### 2. 物流明细表 (shipment_items)

记录每个物流单包含的商品明细。

```go
package model

// ShipmentItem 物流明细模型
type ShipmentItem struct {
    ID uint `gorm:"primaryKey;autoIncrement" json:"id"`
    
    // 关联信息
    ShipmentID    uint   `gorm:"index;not null" json:"shipment_id"`      // 物流单ID
    ShipmentNo    string `gorm:"type:varchar(50);index" json:"shipment_no"` // 物流单号
    OrderID       uint   `gorm:"index;not null" json:"order_id"`         // 订单ID
    OrderItemID   uint   `gorm:"index;not null" json:"order_item_id"`    // 订单明细ID
    
    // 商品信息
    SkuID         uint   `gorm:"index;not null" json:"sku_id"`           // SKU ID
    SkuCode       string `gorm:"type:varchar(100);not null" json:"sku_code"` // SKU编码
    SkuName       string `gorm:"type:varchar(200);not null" json:"sku_name"` // SKU名称
    ProductImage  string `gorm:"type:varchar(500)" json:"product_image"` // 商品图片
    
    // 规格信息
    Color         string `gorm:"type:varchar(100)" json:"color"`         // 颜色
    Size          string `gorm:"type:varchar(50)" json:"size"`           // 尺码
    
    // 数量信息
    OrderedQty    int `gorm:"type:int;not null" json:"ordered_qty"`     // 订单数量
    ShippedQty    int `gorm:"type:int;not null" json:"shipped_qty"`     // 本次发货数量
    
    // 价格信息
    UnitPrice     decimal.Decimal `gorm:"type:decimal(10,2);not null" json:"unit_price"` // 单价
    TotalAmount   decimal.Decimal `gorm:"type:decimal(10,2);not null" json:"total_amount"` // 总金额
    
    // 时间字段
    CreatedAt time.Time      `json:"created_at"`
    UpdatedAt time.Time      `json:"updated_at"`
    DeletedAt gorm.DeletedAt `gorm:"index" json:"-"`
    
    // 关联关系
    Shipment *Shipment `gorm:"foreignKey:ShipmentID" json:"shipment,omitempty"`
}

// TableName 指定表名
func (ShipmentItem) TableName() string {
    return "shipment_items"
}

// CalculateTotalAmount 计算总金额
func (si *ShipmentItem) CalculateTotalAmount() decimal.Decimal {
    return si.UnitPrice.Mul(decimal.NewFromInt(int64(si.ShippedQty)))
}
```

```sql
CREATE OR REPLACE FUNCTION update_shipment_totals()
RETURNS TRIGGER AS $$
BEGIN
    -- 更新物流单的总重量、总体积
    UPDATE shipments SET
        total_weight = (
            SELECT SUM(si.shipped_qty * sku.weight) 
            FROM shipment_items si
            JOIN skus sku ON si.sku_id = sku.id
            WHERE si.shipment_id = NEW.shipment_id
        ),
        total_volume = (
            SELECT SUM(si.shipped_qty * sku.length * sku.width * sku.height / 1000000) 
            FROM shipment_items si
            JOIN skus sku ON si.sku_id = sku.id
            WHERE si.shipment_id = NEW.shipment_id
        )
    WHERE id = NEW.shipment_id;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_shipment_totals
AFTER INSERT OR UPDATE ON shipment_items
FOR EACH ROW
EXECUTE FUNCTION update_shipment_totals();
```

**物流轨迹自动更新物流单状态**:
```sql
CREATE OR REPLACE FUNCTION update_shipment_status_from_track()
RETURNS TRIGGER AS $$
BEGIN
    -- 根据轨迹状态更新物流单状态
    IF NEW.track_status IN ('delivered', 'signed') THEN
        UPDATE shipments SET
            status = 'delivered',
            delivered_at = NEW.track_time,
            signed_by = NEW.operator_name,
            current_location = NEW.location
        WHERE id = NEW.shipment_id;
    ELSIF NEW.track_status = 'in_transit' THEN
        UPDATE shipments SET
            status = 'in_transit',
            current_location = NEW.location
        WHERE id = NEW.shipment_id AND status != 'delivered';
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_shipment_status
AFTER INSERT ON shipment_tracks
FOR EACH ROW
```

## 数据一致性保证

### 1. 事务处理

所有涉及多表操作的业务逻辑必须在事务中执行：

```go
// 创建物流单（事务示例）
func (s *ShipmentStore) Create(db *gorm.DB, shipment *Shipment, items []ShipmentItem) error {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 创建物流单
        if err := tx.Create(shipment).Error; err != nil {
            return err
        }
        
        // 2. 创建物流明细
        for i := range items {
            items[i].ShipmentID = shipment.ID
            items[i].ShipmentNo = shipment.ShipmentNo
        }
        if err := tx.Create(&items).Error; err != nil {
            return err
        }
        
        // 3. 更新订单发货状态
        if err := tx.Model(&Order{}).Where("id = ?", shipment.OrderID).
            Update("shipment_status", "partial_shipped").Error; err != nil {
            return err
        }
        
        // 4. 扣减库存
        for _, item := range items {
            if err := tx.Model(&Inventory{}).
                Where("sku_id = ? AND warehouse_id = ?", item.SkuID, shipment.WarehouseID).
                Update("quantity", gorm.Expr("quantity - ?", item.ShippedQty)).Error; err != nil {
                return err
            }
        }
        
        return nil
    })
}
```

### 2. 乐观锁

对于库存扣减等并发操作，使用version字段实现乐观锁：

```go
type Inventory struct {
    ID       uint
    Quantity int
    Version  int `gorm:"not null;default:0"` // 版本号
}

// 乐观锁扣减库存
func DeductInventory(db *gorm.DB, skuID uint, qty int) error {
    for i := 0; i < 3; i++ {  // 重试3次
        var inventory Inventory
        if err := db.Where("sku_id = ?", skuID).First(&inventory).Error; err != nil {
            return err
        }
        
        result := db.Model(&Inventory{}).
            Where("sku_id = ? AND version = ?", skuID, inventory.Version).
            Updates(map[string]interface{}{
                "quantity": gorm.Expr("quantity - ?", qty),
                "version":  gorm.Expr("version + 1"),
            })
        
        if result.Error != nil {
            return result.Error
        }
        
        if result.RowsAffected > 0 {
            return nil  // 成功
        }
        
        // 版本冲突，重试
        time.Sleep(time.Millisecond * 10)
    }
    
    return errors.New("库存扣减失败，并发冲突")
}
```
