// api/client.ts
const apiClient = axios.create({
  baseURL: API_BASE_URL,
  timeout: 10000,
});

apiClient.interceptors.request.use(async (config) => {
  const token = await getToken();
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

#### 3.3 Custom Hooks模块

**职责:**
- 业务逻辑封装
- 状态管理
- 副作用处理

**核心Hooks:**
- `useCamera` - 相机控制
- `usePermissions` - 权限管理
- `useQRCodeScan` - 扫码逻辑
- `useQRCodeConfirm` - 确认逻辑
- `useAuth` - 认证逻辑

**示例:**
```typescript
export function useQRCodeScan() {
  const [isScanning, setIsScanning] = useState(false);
  const [qrcodeId, setQrcodeId] = useState<string | null>(null);

  const scan = useCallback(async (id: string) => {
    setIsScanning(true);
    try {
      await QRCodeService.scan(id);
      setQrcodeId(id);
    } finally {
      setIsScanning(false);
    }
  }, []);

  return { isScanning, qrcodeId, scan };
}
```

### 4. 技术决策 ✅

#### 4.1 状态管理

**方案:** React Hooks

**理由:**
- 应用规模小
- Hooks足够
- 无需Redux

**状态类型:**
```typescript
interface ScanState {
  isScanning: boolean;
  qrcodeId: string | null;
  error: Error | null;
}

interface ConfirmState {
  isConfirming: boolean;
  userInfo: UserInfo | null;
  tenantInfo: TenantInfo | null;
}
```

---

#### 4.2 样式方案

**方案:** StyleSheet (React Native内置)

**理由:**
- 性能最优
- 类型支持好
- 零额外依赖

**组织方式:**
- 大组件: 独立styles.ts文件
- 小组件: 组件内StyleSheet.create

---

#### 4.3 错误处理

**策略:**
- 全局ErrorBoundary
- API统一拦截
- 用户友好提示

**实现:**
```typescript
class ErrorBoundary extends React.Component {
  componentDidCatch(error, info) {
    logger.error(error, info);
    // 上报错误
  }
}
```
