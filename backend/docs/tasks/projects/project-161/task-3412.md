        ID:               taskID,
        TenantID:         tenantID,
        UserID:           userID,
        FileName:         req.FileName,
        ExportType:       req.ExportType,
        ExportStatus:     "00", // 待处理
        FileFormat:       req.FileFormat,
        TotalCount:       totalCount,
        FilterConditions: jsonb(req.FilterConditions),
    }
    l.svcCtx.ExportTaskModel.Insert(l.ctx, task)
    
    // 5. 提交到Asynq队列
    payload := ExportTaskPayload{...}
    asynqTask := asynq.NewTask(TypeExportTask, payloadBytes)
    l.svcCtx.AsynqClient.Enqueue(asynqTask, asynq.Queue("export"))
    
    // 6. 返回响应
    return &types.CreateExportTaskResponse{
        TaskId:        taskID,
        EstimatedTime: calculateEstimate(totalCount),
        TotalCount:    totalCount,
    }

### 3. 中间件实现 ✅

#### 3.1 Auth中间件（JWT验证）

**文件**: `api/internal/middleware/auth_middleware.go`

**核心功能**:
```go
type AuthMiddleware struct {
    jwtSecret string
}

func (m *AuthMiddleware) Handle(next http.HandlerFunc) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        // 1. 提取 Authorization Header
        authHeader := r.Header.Get("Authorization")
        
        // 2. 解析 Bearer Token
        tokenString := extractBearerToken(authHeader)
        
        // 3. 验证 JWT Token
        token, err := jwt.ParseWithClaims(tokenString, &CustomClaims{}, func(token *jwt.Token) (interface{}, error) {
            return []byte(m.jwtSecret), nil
        })
        
        // 4. 提取Claims
        claims := token.Claims.(*CustomClaims)
        
        // 5. 验证必需字段
        if claims.UserID == 0 {
            return error("Missing user_id")
        }
        
        // 6. 存入Context
        ctx := r.Context()
        ctx = context.WithValue(ctx, "user_id", claims.UserID)
        ctx = context.WithValue(ctx, "tenant_id", claims.TenantID)
        ctx = context.WithValue(ctx, "username", claims.Username)
        
        // 7. 传递到下一个handler
        next(w, r.WithContext(ctx))
    }
}
```

**JWT Claims结构**:
```go
type CustomClaims struct {
    UserID   int64  `json:"user_id"`
    TenantID int64  `json:"tenant_id"`
    Username string `json:"username"`
    jwt.RegisteredClaims
}
```

**关键特性**:
- ✅ Bearer Token解析
- ✅ HMAC-SHA256签名验证
- ✅ 提取user_id, tenant_id, username
- ✅ 存入Context供Logic层使用

## 🔄 工作流程

### 创建导出任务流程
```
1. 客户端发送请求 → POST /api/v1/export-tasks + Bearer Token
2. Auth中间件验证JWT，提取user_id, tenant_id存入Context
3. Handler调用CreateExportTaskLogic
4. Logic层：
   - 生成UUID任务ID
   - 从Context获取租户ID和用户ID
   - 序列化过滤条件为JSONB
   - 调用ExportTaskModel.Insert()创建任务记录（status=00）
   - 构造Asynq任务Payload
   - 提交到Asynq的export队列
5. 返回响应（task_id, estimated_time, total_count）
6. Worker服务从队列中消费任务并处理
```

### 查询任务进度流程
```
1. 客户端发送请求 → GET /api/v1/export-tasks/:taskId + Bearer Token
2. Auth中间件验证JWT
3. Handler调用GetExportTaskLogic
4. Logic层：
   - 调用ExportTaskModel.FindOne()查询任务
   - 计算进度百分比 = processed_count / total_count * 100
   - 格式化时间字段
5. 返回任务详情（包含状态、进度、下载链接等）
```
