# 售后数据库Schema设计文档

## 1. 概述

本文档定义Coolbuy 3.0售后模块的数据库Schema设计，包括退货、退款、换货等售后业务场景的数据模型。

### 1.1 设计原则

- **多租户隔离**: 使用PostgreSQL Schema实现租户数据隔离
- **状态机管理**: 严格的售后单状态流转控制
- **审批流程**: 支持多级审批机制
- **财务对账**: 与资金流水模块无缝对接
- **物流跟踪**: 退货物流信息完整记录
- **操作审计**: 完整的操作日志记录

### 1.2 核心表结构

```
售后模块核心表:
├── after_sales (售后单主表)
├── after_sales_items (售后明细表)
├── after_sales_approvals (审批记录表)
├── after_sales_refunds (退款记录表)
├── after_sales_logistics (售后物流表)
├── after_sales_images (售后凭证图片表)
└── after_sales_logs (售后操作日志表)
```

## 2. 数据模型设计

### 2.1 售后单主表 (after_sales)

**表说明**: 售后单核心表，记录退货、退款、换货等售后申请

```go
package model

import (
	"time"
	"github.com/shopspring/decimal"
	"gorm.io/gorm"
)

// AfterSales 售后单模型
// 支持退货退款、仅退款、换货三种类型
// 多租户通过PostgreSQL Schema隔离，无需TenantID字段
type AfterSales struct {
	ID uint `gorm:"primaryKey;autoIncrement" json:"id"`
	
	// 基本信息
	AfterSalesNo   string `gorm:"type:varchar(50);uniqueIndex;not null" json:"after_sales_no"`     // 售后单号
	AfterSalesType string `gorm:"type:varchar(20);not null;index" json:"after_sales_type"`         // 售后类型: refund_only(仅退款), return_refund(退货退款), exchange(换货)
	OrderID        uint   `gorm:"index;not null" json:"order_id"`                                   // 原订单ID
	OrderNo        string `gorm:"type:varchar(50);index;not null" json:"order_no"`                 // 原订单号
	
	// 客户信息
	CustomerID   uint   `gorm:"index;not null" json:"customer_id"`                    // 客户ID
	CustomerName string `gorm:"type:varchar(100);not null" json:"customer_name"`      // 客户姓名
	CustomerPhone string `gorm:"type:varchar(20);not null" json:"customer_phone"`     // 客户手机
	
	// 售后原因
	ReasonType    string `gorm:"type:varchar(50);not null;index" json:"reason_type"`  // 原因类型: quality_issue(质量问题), size_mismatch(尺码不符), wrong_item(发错货), description_mismatch(描述不符), other(其他)
	ReasonDetail  string `gorm:"type:text" json:"reason_detail"`                       // 详细原因描述
	
	// 金额信息
	TotalAmount      decimal.Decimal `gorm:"type:decimal(10,2);not null" json:"total_amount"`        // 总金额
	RefundAmount     decimal.Decimal `gorm:"type:decimal(10,2);not null;default:0" json:"refund_amount"` // 退款金额
	RefundFreight    decimal.Decimal `gorm:"type:decimal(10,2);default:0" json:"refund_freight"`     // 退运费金额
	DeductionAmount  decimal.Decimal `gorm:"type:decimal(10,2);default:0" json:"deduction_amount"`   // 扣款金额（折旧、运费等）
	ActualRefundAmount decimal.Decimal `gorm:"type:decimal(10,2);default:0" json:"actual_refund_amount"` // 实际退款金额
	
	// 退货物流信息
	ReturnLogisticsProvider string    `gorm:"type:varchar(100)" json:"return_logistics_provider"` // 退货物流商
	ReturnTrackingNo        string    `gorm:"type:varchar(100);index" json:"return_tracking_no"`  // 退货物流单号
	ReturnShippedAt         *time.Time `json:"return_shipped_at"`                                  // 客户发货时间
	ReturnReceivedAt        *time.Time `json:"return_received_at"`                                 // 仓库收货时间
	
	// 换货信息（仅换货类型使用）
	ExchangeOrderID  *uint   `gorm:"index" json:"exchange_order_id"`                    // 换货新订单ID
	ExchangeOrderNo  string  `gorm:"type:varchar(50);index" json:"exchange_order_no"`   // 换货新订单号
	
	// 审批信息
	ApprovalStatus   string `gorm:"type:varchar(20);not null;default:'pending';index" json:"approval_status"` // 审批状态: pending(待审批), approved(已通过), rejected(已拒绝)
	ApproverID       *uint  `gorm:"index" json:"approver_id"`                           // 审批人ID
	ApproverName     string `gorm:"type:varchar(100)" json:"approver_name"`             // 审批人姓名
	ApprovedAt       *time.Time `json:"approved_at"`                                     // 审批时间
	ApprovalRemark   string `gorm:"type:text" json:"approval_remark"`                   // 审批备注
	
	// 退款信息
	RefundStatus     string     `gorm:"type:varchar(20);not null;default:'pending';index" json:"refund_status"` // 退款状态: pending(待退款), processing(退款中), success(已退款), failed(退款失败)
	RefundMethod     string     `gorm:"type:varchar(50)" json:"refund_method"`           // 退款方式: original(原路退回), bank_transfer(银行转账), cash(现金)
	RefundAccount    string     `gorm:"type:varchar(200)" json:"refund_account"`         // 退款账户信息
	RefundedAt       *time.Time `json:"refunded_at"`                                     // 退款时间
	RefundTransactionNo string  `gorm:"type:varchar(100);index" json:"refund_transaction_no"` // 退款流水号
	
	// 状态信息
	Status           string `gorm:"type:varchar(20);not null;default:'pending';index" json:"status"` // 售后单状态: pending(待审核), approved(已通过), rejected(已拒绝), returning(退货中), returned(已退货), refunding(退款中), completed(已完成), cancelled(已取消), closed(已关闭)
	
	// 时间节点
	AppliedAt        time.Time  `gorm:"not null" json:"applied_at"`                      // 申请时间
	ProcessedAt      *time.Time `json:"processed_at"`                                    // 处理完成时间
	CompletedAt      *time.Time `json:"completed_at"`                                    // 完成时间
	CancelledAt      *time.Time `json:"cancelled_at"`                                    // 取消时间
	
	// 其他信息
	Remark           string `gorm:"type:text" json:"remark"`                             // 备注
	InternalNotes    string `gorm:"type:text" json:"internal_notes"`                     // 内部备注
	ProcessorID      *uint  `gorm:"index" json:"processor_id"`                           // 处理人ID
	ProcessorName    string `gorm:"type:varchar(100)" json:"processor_name"`             // 处理人姓名
	
	CreatedAt        time.Time      `json:"created_at"`
	UpdatedAt        time.Time      `json:"updated_at"`
	DeletedAt        gorm.DeletedAt `gorm:"index" json:"-"`
	
	// 关联关系
	Items      []AfterSalesItem     `gorm:"foreignKey:AfterSalesID" json:"items,omitempty"`
	Approvals  []AfterSalesApproval `gorm:"foreignKey:AfterSalesID" json:"approvals,omitempty"`
	Refunds    []AfterSalesRefund   `gorm:"foreignKey:AfterSalesID" json:"refunds,omitempty"`
	Logistics  []AfterSalesLogistics `gorm:"foreignKey:AfterSalesID" json:"logistics,omitempty"`
	Images     []AfterSalesImage    `gorm:"foreignKey:AfterSalesID" json:"images,omitempty"`
	Logs       []AfterSalesLog      `gorm:"foreignKey:AfterSalesID" json:"logs,omitempty"`
}

// TableName 指定表名
func (AfterSales) TableName() string {
	return "after_sales"
}

// CanApprove 判断是否可以审批
func (a *AfterSales) CanApprove() bool {
	return a.Status == "pending" && a.ApprovalStatus == "pending"
}

// CanRefund 判断是否可以退款
func (a *AfterSales) CanRefund() bool {
	return a.ApprovalStatus == "approved" && a.RefundStatus == "pending"
}

// CanCancel 判断是否可以取消
func (a *AfterSales) CanCancel() bool {
	return a.Status == "pending" || a.Status == "approved"
}
```

### 4.2 状态变更审计触发器

```sql
-- 售后单状态变更自动记录日志
CREATE OR REPLACE FUNCTION log_after_sales_status_change()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status IS DISTINCT FROM NEW.status THEN
        INSERT INTO after_sales_logs (
            after_sales_id,
            after_sales_no,
            operation_type,
            operation_desc,
            before_status,
            after_status,
            operator_type
        ) VALUES (
            NEW.id,
            NEW.after_sales_no,
            'status_change',
            '售后单状态变更: ' || OLD.status || ' -> ' || NEW.status,
            OLD.status,
            NEW.status,
            'system'
        );
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER after_sales_status_change_log
    AFTER UPDATE ON after_sales
    FOR EACH ROW
    EXECUTE FUNCTION log_after_sales_status_change();
```

CREATE TABLE after_sales_archive (LIKE after_sales INCLUDING ALL);

INSERT INTO after_sales_archive 
SELECT * FROM after_sales 
WHERE status IN ('completed', 'closed') 
  AND completed_at < NOW() - INTERVAL '6 months';

DELETE FROM after_sales 
WHERE id IN (SELECT id FROM after_sales_archive);
