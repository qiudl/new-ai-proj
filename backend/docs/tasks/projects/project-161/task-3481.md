package integration

import (

func TestPaymentFlow_BalancePayment(t *testing.T) {

    defer test.CleanupTestDB(t, db)
    
    // 1. 创建付款方和收款方账户
    payerAccount := test.CreateTestAccount(t, db, "customer")
    payeeAccount := test.CreateTestAccount(t, db, "company")
    
    paymentBiz := biz.NewPaymentBiz()
    
    // 2. 创建支付订单
    req := &biz.ReqCreatePayment{
        BusinessType:    "订单支付",
        RelatedType:     "order",
        RelatedID:       100,
        RelatedNo:       "ORD202501060001",
        PayerAccountID:  payerAccount.ID,
        PayeeAccountID:  payeeAccount.ID,
        Amount:          "500.00",
        PaymentMethod:   "balance",
    }
    
    paymentOrder, paymentResp, err := paymentBiz.CreatePayment(db, req, 1)
    
    // 验证支付订单创建成功
    assert.NoError(t, err)
    assert.NotNil(t, paymentOrder)
    assert.Equal(t, "paid", paymentOrder.Status)
    
    // 3. 验证付款方账户余额减少
    var updatedPayer Account
    db.First(&updatedPayer, payerAccount.ID)
    expectedPayerBalance := decimal.NewFromInt(10000).Sub(decimal.NewFromFloat(500.00))
    assert.True(t, updatedPayer.Balance.Equal(expectedPayerBalance))
    
    // 4. 验证收款方账户余额增加
    var updatedPayee Account
    db.First(&updatedPayee, payeeAccount.ID)
    expectedPayeeBalance := decimal.NewFromInt(10000).Add(decimal.NewFromFloat(500.00))
    assert.True(t, updatedPayee.Balance.Equal(expectedPayeeBalance))
    
    // 5. 验证交易记录创建
    var transactions []TransactionRecord
    db.Where("account_id IN ?", []uint{payerAccount.ID, payeeAccount.ID}).
        Order("id").
        Find(&transactions)
    
    assert.Len(t, transactions, 2)
    
    // 验证付款方交易记录
    assert.Equal(t, "expense", transactions[0].Direction)
    assert.True(t, transactions[0].Amount.Equal(decimal.NewFromFloat(500.00)))
    
    // 验证收款方交易记录
    assert.Equal(t, "income", transactions[1].Direction)
    assert.True(t, transactions[1].Amount.Equal(decimal.NewFromFloat(500.00)))
    assert.Equal(t, transactions[0].TransactionNo, transactions[1].ParentTransactionNo)
}

type AccountOption func(*Account)

func WithAccountName(name string) AccountOption {
    return func(a *Account) {
        a.AccountName = name

func WithBalance(balance decimal.Decimal) AccountOption {

        a.AvailableBalance = balance.Sub(a.FrozenAmount)
