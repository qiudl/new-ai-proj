# 入库管理功能设计文档

## 1. 概述

### 1.1 文档目的

本文档详细描述仓库库存模块的入库管理功能设计，包括采购入库、退货入库、调拨入库、盘盈入库等核心业务流程的API接口设计、业务逻辑、质检流程、上架策略、异常处理和性能优化。

### 1.2 设计原则

- **流程规范**: 标准化入库流程，支持多种入库类型
- **数据准确**: 严格校验入库数据，确保库存准确性
- **可追溯**: 完整记录入库操作轨迹和库存变更历史
- **灵活扩展**: 支持自定义质检规则和上架策略
- **高并发**: 支持批量入库，响应时间 < 200ms (P95)

### 1.3 适用范围

- 仓库管理人员：执行日常入库操作
- 采购人员：查看采购入库进度
- 质检人员：执行入库质量检验
- 系统管理员：配置入库规则和策略

## 2. 入库业务场景

### 2.1 入库类型

#### 2.1.1 采购入库 (Purchase Inbound)

**业务描述**: 从供应商采购商品入库

**关键字段**:
- 采购单号（PO Number）
- 供应商信息
- 预期到货日期
- 采购价格

**业务流程**:
```
1. 创建采购入库单 → 2. 收货登记 → 3. 质检 → 4. 上架 → 5. 完成入库
```

#### 2.1.2 退货入库 (Return Inbound)

**业务描述**: 客户退货商品入库

**关键字段**:
- 原订单号
- 退货原因
- 退货类型（质量问题、客户原因等）
- 是否需要质检

**业务流程**:
```
1. 创建退货入库单 → 2. 退货接收 → 3. 质检 → 4. 判定处理（上架/报废） → 5. 完成入库
```

#### 2.1.3 调拨入库 (Transfer Inbound)

**业务描述**: 从其他仓库调拨商品入库

**关键字段**:
- 调拨单号
- 源仓库信息
- 调拨原因

**业务流程**:
```
1. 接收调拨通知 → 2. 收货确认 → 3. 上架 → 4. 完成入库
```

#### 2.1.4 盘盈入库 (Surplus Inbound)

**业务描述**: 盘点发现实际库存多于账面库存

**关键字段**:
- 盘点单号
- 盘盈数量
- 盘盈原因

**业务流程**:
```
1. 创建盘盈入库单 → 2. 审核 → 3. 上架 → 4. 完成入库
```

### 2.2 入库状态流转

```
待入库 (Pending) 
  ↓
收货中 (Receiving)
  ↓
质检中 (Inspecting) ← [如需质检]
  ↓
上架中 (Shelving)
  ↓
已完成 (Completed)
  ↓
已取消 (Cancelled) ← [异常取消]
```

```go
func (b *InboundBiz) CreatePurchaseInbound(db *gorm.DB, req *ReqCreatePurchaseInbound, userID uint) (*RespCreateInbound, error) {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 生成入库单号
        orderNumber := b.generateOrderNumber("PI") // PI-20250106-00001
        
        // 2. 计算总数量和总价
        totalQty := 0
        totalPrice := 0.0
        for _, item := range req.Items {
            totalQty += item.PlannedQty
            totalPrice += item.UnitPrice * float64(item.PlannedQty)
        }
        
        // 3. 创建入库单主记录
        order := &InboundOrder{
            OrderNumber:       orderNumber,
            InboundType:       "purchase",
            Status:            "pending",
            WarehouseID:       req.WarehouseID,
            ZoneID:            req.ZoneID,
            SourceType:        "supplier",
            SourceID:          &req.SupplierID,
            SourceOrderNo:     req.PurchaseOrderNo,
            PlannedDate:       req.PlannedDate,
            TotalQuantity:     totalQty,
            RequireInspection: req.RequireInspection,
            InspectionStatus:  b.getInspectionStatus(req.RequireInspection),
            Remark:            req.Remark,
            CreatedBy:         userID,
        }
        
        if err := tx.Create(order).Error; err != nil {
            return err
        }
        
        // 4. 创建入库单明细
        for _, itemReq := range req.Items {
            // 验证SKU存在
            var sku Sku
            if err := tx.First(&sku, itemReq.SkuID).Error; err != nil {
                return fmt.Errorf("SKU %d 不存在", itemReq.SkuID)
            }
            
            item := &InboundOrderItem{
                InboundOrderID:   order.ID,
                SkuID:            itemReq.SkuID,
                SkuCode:          sku.Code,
                SkuName:          sku.Name,
                BatchNumber:      itemReq.BatchNumber,
                ProductionDate:   itemReq.ProductionDate,
                ExpiryDate:       itemReq.ExpiryDate,
                PlannedQty:       itemReq.PlannedQty,
                UnitPrice:        itemReq.UnitPrice,
                TotalPrice:       itemReq.UnitPrice * float64(itemReq.PlannedQty),
                InspectionStatus: b.getInspectionStatus(req.RequireInspection),
                Remark:           itemReq.Remark,
            }
            
            if err := tx.Create(item).Error; err != nil {
                return err
            }
        }
        
        // 5. 记录操作日志
        log := &InboundOrderLog{
            InboundOrderID: order.ID,
            Action:         "created",
            ToStatus:       "pending",
            OperatorID:     userID,
            Description:    fmt.Sprintf("创建采购入库单，采购单号: %s", req.PurchaseOrderNo),
        }
        tx.Create(log)
        
        return &RespCreateInbound{
            ID:          order.ID,
            OrderNumber: order.OrderNumber,
            InboundType: order.InboundType,
            Status:      order.Status,
            CreatedAt:   order.CreatedAt,
        }, nil
    })
}

// 生成入库单号
func (b *InboundBiz) generateOrderNumber(prefix string) string {
    now := time.Now()
    dateStr := now.Format("20060102")
    
    // 查询当天最大序号
    var maxSeq int
    b.db.Model(&InboundOrder{}).
        Where("order_number LIKE ?", prefix+"-"+dateStr+"-%").
        Select("MAX(CAST(SUBSTRING(order_number, LENGTH(?)+1) AS UNSIGNED)) as max_seq", prefix+"-"+dateStr+"-").
        Scan(&maxSeq)
    
    seq := maxSeq + 1
    return fmt.Sprintf("%s-%s-%05d", prefix, dateStr, seq)
}

// 获取质检状态
func (b *InboundBiz) getInspectionStatus(requireInspection bool) string {
    if requireInspection {
        return "pending"
    }
    return "not_required"
}
```

```go
func (b *InboundBiz) StartReceiving(db *gorm.DB, orderID uint, req *ReqStartReceiving) error {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 查询入库单
        var order InboundOrder
        if err := tx.First(&order, orderID).Error; err != nil {
            return err
        }
        
        // 2. 验证状态
        if order.Status != "pending" {
            return errors.New("只有待入库状态的单据可以开始收货")
        }
        
        // 3. 更新状态
        updates := map[string]interface{}{
            "status":      "receiving",
            "received_by": req.ReceiverID,
        }
        if err := tx.Model(&order).Updates(updates).Error; err != nil {
            return err
        }
        
        // 4. 记录日志
        log := &InboundOrderLog{
            InboundOrderID: order.ID,
            Action:         "start_receiving",
            FromStatus:     "pending",
            ToStatus:       "receiving",
            OperatorID:     req.ReceiverID,
            Description:    "开始收货",
        }
        tx.Create(log)
        
        return nil
    })
}
```

**业务逻辑**:
```go
func (b *InboundBiz) ReceiveGoods(db *gorm.DB, orderID uint, req *ReqReceiveGoods, userID uint) (*RespReceiveGoods, error) {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 查询入库单
        var order InboundOrder
        if err := tx.Preload("Items").First(&order, orderID).Error; err != nil {
            return nil, err
        }
        
        // 2. 验证状态
        if order.Status != "receiving" {
            return nil, errors.New("入库单状态不正确")
        }
        
        // 3. 更新明细收货数量
        totalReceived := 0
        for _, receiveItem := range req.Items {
            var item InboundOrderItem
            if err := tx.First(&item, receiveItem.ItemID).Error; err != nil {
                return nil, err
            }
            
            // 验证是否属于当前入库单
            if item.InboundOrderID != orderID {
                return nil, errors.New("入库单明细不属于当前入库单")
            }
            
            // 更新收货数量
            item.ReceivedQty = receiveItem.ReceivedQty
            if err := tx.Save(&item).Error; err != nil {
                return nil, err
            }
            
            totalReceived += receiveItem.ReceivedQty
        }
        
        // 4. 更新入库单收货数量
        order.ReceivedQty = totalReceived
        order.ReceivedDate = &time.Time{}
        *order.ReceivedDate = time.Now()
        
        // 5. 判断是否收货完成
        isComplete := totalReceived >= order.TotalQuantity
        nextStep := ""
        
        if isComplete {
            // 收货完成，进入下一流程
            if order.RequireInspection {
                order.Status = "inspecting"
                order.InspectionStatus = "pending"
                nextStep = "inspecting"
            } else {
                order.Status = "shelving"
                nextStep = "shelving"
                
                // 不需要质检，直接标记所有明细为合格
                tx.Model(&InboundOrderItem{}).
                    Where("inbound_order_id = ?", orderID).
                    Updates(map[string]interface{}{
                        "qualified_qty":     gorm.Expr("received_qty"),
                        "inspection_status": "passed",
                        "quality_grade":     "A",
                    })
            }
        }
        
        if err := tx.Save(&order).Error; err != nil {
            return nil, err
        }
        
        // 6. 记录日志
        log := &InboundOrderLog{
            InboundOrderID: order.ID,
            Action:         "receive_goods",
            FromStatus:     "receiving",
            ToStatus:       order.Status,
            OperatorID:     userID,
```

```go
func (b *InboundBiz) InspectGoods(db *gorm.DB, orderID uint, req *ReqInspectGoods, userID uint) (*RespInspectGoods, error) {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 查询入库单
        var order InboundOrder
        if err := tx.Preload("Items").First(&order, orderID).Error; err != nil {
            return nil, err
        }
        
        // 2. 验证状态
        if order.Status != "inspecting" {
            return nil, errors.New("入库单状态不正确")
        }
        
        // 3. 更新明细质检结果
        totalQualified := 0
        totalUnqualified := 0
        
        for _, inspectItem := range req.Items {
            var item InboundOrderItem
            if err := tx.First(&item, inspectItem.ItemID).Error; err != nil {
                return nil, err
            }
            
            // 验证数量
            if inspectItem.QualifiedQty+inspectItem.UnqualifiedQty != item.ReceivedQty {
                return nil, fmt.Errorf("明细 %d 的合格+不合格数量应等于收货数量", item.ID)
            }
            
            // 更新质检结果
            item.QualifiedQty = inspectItem.QualifiedQty
            item.UnqualifiedQty = inspectItem.UnqualifiedQty
            item.QualityGrade = inspectItem.QualityGrade
            item.UnqualifiedReason = inspectItem.UnqualifiedReason
            
            // 判定质检状态
            if inspectItem.UnqualifiedQty == 0 {
                item.InspectionStatus = "passed"
            } else if inspectItem.QualifiedQty == 0 {
                item.InspectionStatus = "failed"
            } else {
                item.InspectionStatus = "partial"
            }
            
            if err := tx.Save(&item).Error; err != nil {
                return nil, err
            }
            
            totalQualified += inspectItem.QualifiedQty
            totalUnqualified += inspectItem.UnqualifiedQty
        }
        
        // 4. 更新入库单质检汇总
        order.QualifiedQty = totalQualified
        order.UnqualifiedQty = totalUnqualified
        order.InspectedBy = &userID
        
        // 5. 判断是否质检完成
        isComplete := (totalQualified + totalUnqualified) >= order.ReceivedQty
        
        if isComplete {
            // 质检完成，进入上架流程
            order.Status = "shelving"
            order.InspectionStatus = "passed"
        }
        
        if err := tx.Save(&order).Error; err != nil {
            return nil, err
        }
        
        // 6. 记录日志
        log := &InboundOrderLog{
            InboundOrderID: order.ID,
            Action:         "inspect_goods",
            FromStatus:     "inspecting",
            ToStatus:       order.Status,
            OperatorID:     userID,
            Description:    fmt.Sprintf("质检完成: 合格%d, 不合格%d", totalQualified, totalUnqualified),
        }
        tx.Create(log)
        
        return &RespInspectGoods{
            OrderID:        order.ID,
            QualifiedQty:   totalQualified,
            UnqualifiedQty: totalUnqualified,
            IsComplete:     isComplete,
            NextStep:       "shelving",
        }, nil
    })
}
```

```go
func (b *InboundBiz) ShelveGoods(db *gorm.DB, orderID uint, req *ReqShelveGoods, userID uint) (*RespShelveGoods, error) {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 查询入库单
        var order InboundOrder
        if err := tx.Preload("Items").First(&order, orderID).Error; err != nil {
            return nil, err
        }
        
        // 2. 验证状态
        if order.Status != "shelving" {
            return nil, errors.New("入库单状态不正确")
        }
        
        totalShelved := 0
        
        // 3. 逐个处理上架明细
        for _, shelveItem := range req.Items {
            var item InboundOrderItem
            if err := tx.First(&item, shelveItem.ItemID).Error; err != nil {
                return nil, err
            }
            
            // 验证库位已分配
            if item.LocationID == nil {
                return nil, fmt.Errorf("明细 %d 未分配库位", item.ID)
            }
            
            // 验证上架数量
            maxQty := item.QualifiedQty // 只有合格品可上架
            if shelveItem.ShelvedQty > maxQty {
                return nil, fmt.Errorf("明细 %d 上架数量不能超过合格数量 %d", item.ID, maxQty)
            }
            
            // 更新明细上架数量
            item.ShelvedQty = shelveItem.ShelvedQty
            if err := tx.Save(&item).Error; err != nil {
                return nil, err
            }
            
            // 4. 更新库存表
            err := b.updateInventory(tx, &item, shelveItem.ShelvedQty)
            if err != nil {
                return nil, err
            }
            
            totalShelved += shelveItem.ShelvedQty
        }
        
        // 5. 更新入库单上架汇总
        order.ShelvedQty = totalShelved
        order.ShelvedBy = &userID
        
        // 6. 判断是否上架完成
        isComplete := totalShelved >= order.QualifiedQty
        
        if isComplete {
            order.Status = "completed"
            now := time.Now()
            order.CompletedDate = &now
        }
        
        if err := tx.Save(&order).Error; err != nil {
            return nil, err
        }
        
        // 7. 记录日志
        log := &InboundOrderLog{
            InboundOrderID: order.ID,
            Action:         "shelve_goods",
            FromStatus:     "shelving",
            ToStatus:       order.Status,
            OperatorID:     userID,
            Description:    fmt.Sprintf("上架完成: %d件", totalShelved),
        }
        tx.Create(log)
        
        return &RespShelveGoods{
            OrderID:    order.ID,
            ShelvedQty: totalShelved,
            IsComplete: isComplete,
        }, nil
    })
}

// 更新库存
func (b *InboundBiz) updateInventory(tx *gorm.DB, item *InboundOrderItem, quantity int) error {
    // 查询或创建库存记录
    var inventory Inventory
    err := tx.Where(map[string]interface{}{
        "warehouse_id": item.InboundOrder.WarehouseID,
        "sku_id":       item.SkuID,
        "location_id":  item.LocationID,
        "batch_number": item.BatchNumber,
    }).FirstOrCreate(&inventory, Inventory{
        WarehouseID:      item.InboundOrder.WarehouseID,
        SkuID:            item.SkuID,
        LocationID:       item.LocationID,
        BatchNumber:      item.BatchNumber,
        ProductionDate:   item.ProductionDate,
        ExpiryDate:       item.ExpiryDate,
        Quantity:         0,
        AvailableQuantity: 0,
    }).Error
    
    if err != nil {
        return err
    }
    
    // 增加库存
    inventory.Quantity += quantity
    inventory.AvailableQuantity += quantity
    
    return tx.Save(&inventory).Error
}
```

## 5. 上架策略

### 5.1 自动分配库位策略

```go
// 自动分配库位（按优先级）
func (b *InboundBiz) AutoAssignLocation(db *gorm.DB, warehouseID, skuID uint, quantity int) (*WarehouseLocation, error) {
    // 策略1: 同SKU集中存放（查找已有库位）
    var existingLocation WarehouseLocation
    err := db.Joins("JOIN inventories ON inventories.location_id = warehouse_locations.id").
        Where("inventories.warehouse_id = ? AND inventories.sku_id = ? AND inventories.available_quantity > 0", warehouseID, skuID).
        Order("inventories.quantity DESC"). // 优先填满库位
        First(&existingLocation).Error
    
    if err == nil {
        // 检查库位容量
        if b.checkLocationCapacity(&existingLocation, quantity) {
            return &existingLocation, nil
        }
    }
    
    // 策略2: 分配空闲库位
    var emptyLocation WarehouseLocation
    err = db.Where("warehouse_id = ? AND is_enabled = true", warehouseID).
        Where("id NOT IN (SELECT DISTINCT location_id FROM inventories WHERE location_id IS NOT NULL)").
        Order("code ASC").
        First(&emptyLocation).Error
    
    if err == nil {
        return &emptyLocation, nil
    }
    
    return nil, errors.New("无可用库位")
}

// 检查库位容量
func (b *InboundBiz) checkLocationCapacity(location *WarehouseLocation, quantity int) bool {
    if location.MaxCapacity == nil {
        return true // 未设置容量限制
    }
    
    // 查询当前库位已用容量
    var currentQty int
    b.db.Model(&Inventory{}).
        Where("location_id = ?", location.ID).
        Select("SUM(quantity)").
        Scan(&currentQty)
    
    return (currentQty + quantity) <= *location.MaxCapacity
}
```

```go
// 上架失败回滚
func (b *InboundBiz) rollbackShelve(tx *gorm.DB, orderID uint) error {
    // 1. 删除已创建的库存记录
    var items []InboundOrderItem
    tx.Where("inbound_order_id = ?", orderID).Find(&items)
    
    for _, item := range items {
        if item.ShelvedQty > 0 {
            // 减少库存

                Where("warehouse_id = ? AND sku_id = ? AND location_id = ? AND batch_number = ?",

                UpdateColumn("quantity", gorm.Expr("quantity - ?", item.ShelvedQty)).
                UpdateColumn("available_quantity", gorm.Expr("available_quantity - ?", item.ShelvedQty))
        }
    }
    
    // 2. 重置入库单状态
    tx.Model(&InboundOrder{}).Where("id = ?", orderID).
        Updates(map[string]interface{}{
            "status":      "inspecting",
            "shelved_qty": 0,
        })
    
    return nil
```
