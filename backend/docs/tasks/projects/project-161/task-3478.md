# 支付收款功能设计文档

## 1. 功能概述

支付收款功能是资金流水模块的核心业务功能，负责处理系统中所有的资金收付场景，包括订单支付、退款、充值、提现等。本文档详细定义支付收款的完整流程、支付方式管理、回调处理、对账等功能。

### 1.1 核心功能

- **订单收款**：处理客户订单支付
- **客户充值**：客户账户余额充值
- **退款处理**：订单退款、充值退款
- **提现管理**：客户账户提现
- **支付方式管理**：管理各类支付渠道
- **支付回调**：处理第三方支付回调
- **对账功能**：与第三方支付平台对账

### 1.2 支付方式

| 支付方式 | 类型 | 提供商 | 应用场景 |
|---------|------|--------|---------|
| 余额支付 | balance | 内部 | 使用账户余额支付 |
| 现金支付 | cash | 线下 | 线下门店现金收款 |
| 银行转账 | bank_transfer | 线下 | 对公转账 |
| 支付宝 | alipay | 线上 | 线上订单支付 |
| 微信支付 | wechat | 线上 | 线上订单支付 |
| 银联支付 | unionpay | 线上 | 银行卡支付 |

```go
func (b *PaymentBiz) CreatePaymentOrder(db *gorm.DB, req *ReqCreatePaymentOrder, userID uint) (*RespCreatePaymentOrder, error) {
    return db.Transaction(func(tx *gorm.DB) (*RespCreatePaymentOrder, error) {
        // 1. 查询支付方式
        var paymentMethod PaymentMethod
        if err := tx.First(&paymentMethod, req.PaymentMethodID).Error; err != nil {
            return nil, errors.New("支付方式不存在")
        }
        
        if !paymentMethod.IsEnabled {
            return nil, errors.New("支付方式已禁用")
        }
        
        // 2. 检查金额限制
        if !paymentMethod.SingleLimit.IsZero() {
            if req.Amount.GreaterThan(paymentMethod.SingleLimit) {
                return nil, errors.New("超过单笔限额")
            }
        }
        
        // 3. 查询付款账户
        var payerAccount Account
        if err := tx.First(&payerAccount, req.PayerAccountID).Error; err != nil {
            return nil, errors.New("付款账户不存在")
        }
        
        if payerAccount.Status != "active" {
            return nil, errors.New("付款账户状态异常")
        }
        
        // 4. 余额支付需检查余额
        if paymentMethod.Code == "balance" {
            if payerAccount.AvailableBalance.LessThan(req.Amount) {
                return nil, errors.New("账户余额不足")
            }
        }
        
        // 5. 生成支付单号
        paymentOrderNo := generatePaymentOrderNo()
        
        // 6. 创建支付订单
        paymentOrder := &PaymentOrder{
            PaymentOrderNo:  paymentOrderNo,
            RelatedType:     req.RelatedType,
            RelatedID:       req.RelatedID,
            RelatedNo:       req.RelatedNo,
            PayerAccountID:  req.PayerAccountID,
            PayeeAccountID:  req.PayeeAccountID,
            Amount:          req.Amount,
            PaymentMethodID: req.PaymentMethodID,
            PaymentMethod:   paymentMethod.Code,
            Status:          "pending",
            Description:     req.Description,
            ReturnURL:       req.ReturnURL,
            NotifyURL:       req.NotifyURL,
            ExpireTime:      time.Now().Add(30 * time.Minute), // 30分钟过期
            CreatorID:       userID,
        }
        
        if err := tx.Create(paymentOrder).Error; err != nil {
            return nil, err
        }
        
        // 7. 根据支付方式处理
        var paymentInfo interface{}
        var err error
        
        switch paymentMethod.Code {
        case "balance":
            // 余额支付直接扣款
            paymentInfo, err = b.processBalancePayment(tx, paymentOrder)
        case "alipay":
            // 调用支付宝接口
            paymentInfo, err = b.createAlipayOrder(paymentOrder)
        case "wechat":
            // 调用微信支付接口
            paymentInfo, err = b.createWechatOrder(paymentOrder)
        case "cash", "bank_transfer":
            // 线下支付，等待确认
            paymentInfo = map[string]string{
                "message": "请线下完成支付，财务确认后订单生效",
            }
        default:
            return nil, errors.New("不支持的支付方式")
        }
        
        if err != nil {
            return nil, err
        }
        
        return &RespCreatePaymentOrder{
            PaymentOrderNo: paymentOrderNo,
            Amount:         req.Amount,
            Status:         paymentOrder.Status,
            PaymentInfo:    paymentInfo,
            ExpireTime:     paymentOrder.ExpireTime,
        }, nil
    })
}
```

```go
func (b *PaymentBiz) processBalancePayment(tx *gorm.DB, paymentOrder *PaymentOrder) (interface{}, error) {
    // 1. 锁定付款账户
    var payerAccount Account
    if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
        First(&payerAccount, paymentOrder.PayerAccountID).Error; err != nil {
        return nil, err
    }
    
    // 2. 检查余额
    if payerAccount.AvailableBalance.LessThan(paymentOrder.Amount) {
        return nil, errors.New("账户余额不足")
    }
    
    // 3. 扣减付款账户余额
    beforeBalance := payerAccount.Balance
    payerAccount.Balance = payerAccount.Balance.Sub(paymentOrder.Amount)
    payerAccount.AvailableBalance = payerAccount.AvailableBalance.Sub(paymentOrder.Amount)
    payerAccount.TotalExpense = payerAccount.TotalExpense.Add(paymentOrder.Amount)
    payerAccount.LastTransactionAt = timeNow()
    
    if err := tx.Save(&payerAccount).Error; err != nil {
        return nil, err
    }
    
    // 4. 创建付款流水
    paymentTxn := &TransactionRecord{
        TransactionNo:   generateTransactionNo(),
        AccountID:       paymentOrder.PayerAccountID,
        AccountNo:       payerAccount.AccountNo,
        TransactionType: TransactionTypeOrderPayment,
        BusinessType:    paymentOrder.RelatedType,
        Direction:       "expense",
        Amount:          paymentOrder.Amount,
        BeforeBalance:   beforeBalance,
        AfterBalance:    payerAccount.Balance,
        RelatedType:     paymentOrder.RelatedType,
        RelatedID:       paymentOrder.RelatedID,
        RelatedNo:       paymentOrder.RelatedNo,
        PaymentMethodID: paymentOrder.PaymentMethodID,
        Description:     fmt.Sprintf("支付订单: %s", paymentOrder.RelatedNo),
        Status:          "success",
        OperatorID:      paymentOrder.CreatorID,
    }
    
    if err := tx.Create(paymentTxn).Error; err != nil {
        return nil, err
    }
    
    // 5. 增加收款账户余额
    var payeeAccount Account
    if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
        First(&payeeAccount, paymentOrder.PayeeAccountID).Error; err != nil {
        return nil, err
    }
    
    beforeBalance = payeeAccount.Balance
    payeeAccount.Balance = payeeAccount.Balance.Add(paymentOrder.Amount)
    payeeAccount.AvailableBalance = payeeAccount.AvailableBalance.Add(paymentOrder.Amount)
    payeeAccount.TotalIncome = payeeAccount.TotalIncome.Add(paymentOrder.Amount)
    payeeAccount.LastTransactionAt = timeNow()
    
    if err := tx.Save(&payeeAccount).Error; err != nil {
        return nil, err
    }
    
    // 6. 创建收款流水
    receiveTxn := &TransactionRecord{
        TransactionNo:       generateTransactionNo(),
        ParentTransactionNo: paymentTxn.TransactionNo,
        AccountID:           paymentOrder.PayeeAccountID,
        AccountNo:           payeeAccount.AccountNo,
        TransactionType:     TransactionTypeOrderPayment,
        BusinessType:        paymentOrder.RelatedType,
        Direction:           "income",
        Amount:              paymentOrder.Amount,
        BeforeBalance:       beforeBalance,
        AfterBalance:        payeeAccount.Balance,
        RelatedType:         paymentOrder.RelatedType,
        RelatedID:           paymentOrder.RelatedID,
        RelatedNo:           paymentOrder.RelatedNo,
        PaymentMethodID:     paymentOrder.PaymentMethodID,
        CounterpartyType:    payerAccount.OwnerType,
        CounterpartyID:      payerAccount.OwnerID,
        CounterpartyName:    payerAccount.AccountName,
        Description:         fmt.Sprintf("收款: %s", paymentOrder.RelatedNo),
        Status:              "success",
        OperatorID:          paymentOrder.CreatorID,
    }
    
    if err := tx.Create(receiveTxn).Error; err != nil {
        return nil, err
    }
    
    // 7. 更新支付订单状态
    paymentOrder.Status = "paid"
    paymentOrder.PaidAt = timeNow()
    paymentOrder.PaymentTransactionNo = paymentTxn.TransactionNo
    
    if err := tx.Save(paymentOrder).Error; err != nil {
        return nil, err
    }
    
    return map[string]string{
        "message": "支付成功",
        "transaction_no": paymentTxn.TransactionNo,
    }, nil
}
```

```go
func (b *PaymentBiz) createAlipayOrder(paymentOrder *PaymentOrder) (interface{}, error) {
    // 调用支付宝SDK创建订单
    alipayClient := alipay.New(config.AlipayAppID, config.AlipayPrivateKey, false)
    
    // 构建支付参数
    var p = alipay.TradePreCreate{
        Trade: alipay.Trade{
            Subject:     paymentOrder.Description,
            OutTradeNo:  paymentOrder.PaymentOrderNo,
            TotalAmount: paymentOrder.Amount.String(),
            NotifyURL:   paymentOrder.NotifyURL,
        },
    }
    
    // 发起请求
    resp, err := alipayClient.TradePreCreate(p)
    if err != nil {
        return nil, err
    }
    
    if !resp.IsSuccess() {
        return nil, errors.New(resp.Content.Msg)
    }
    
    // 保存支付宝订单号
    paymentOrder.ThirdPartyOrderNo = resp.Content.OutTradeNo
    db.Save(paymentOrder)
    
    return map[string]interface{}{
        "qr_code": resp.Content.QRCode, // 二维码URL
        "order_no": resp.Content.OutTradeNo,
    }, nil
}
```

```go
func (b *PaymentBiz) createWechatOrder(paymentOrder *PaymentOrder) (interface{}, error) {
    // 调用微信支付SDK
    client := wechat.NewClient(config.WechatAppID, config.WechatMchID, config.WechatAPIKey)
    
    // 构建支付参数
    params := map[string]string{
        "body":            paymentOrder.Description,
        "out_trade_no":    paymentOrder.PaymentOrderNo,
        "total_fee":       paymentOrder.Amount.Mul(decimal.NewFromInt(100)).String(), // 分
        "spbill_create_ip": "127.0.0.1",
        "notify_url":      paymentOrder.NotifyURL,
        "trade_type":      "NATIVE", // 扫码支付
    }
    
    // 统一下单
    resp, err := client.UnifiedOrder(params)
    if err != nil {
        return nil, err
    }
    
    if resp["return_code"] != "SUCCESS" {
        return nil, errors.New(resp["return_msg"])
    }
    
    // 保存微信订单号
    paymentOrder.ThirdPartyOrderNo = resp["prepay_id"]
    db.Save(paymentOrder)
    
    return map[string]interface{}{
        "code_url": resp["code_url"], // 二维码URL
        "prepay_id": resp["prepay_id"],
    }, nil
}
```

```go
func (h *PaymentHandler) AlipayNotify(c *gin.Context) {
    // 1. 验证签名
    if err := alipay.VerifySign(c.Request); err != nil {
        c.String(http.StatusOK, "fail")
        return
    }
    
    // 2. 解析回调参数
    outTradeNo := c.PostForm("out_trade_no")
    tradeNo := c.PostForm("trade_no")
    tradeStatus := c.PostForm("trade_status")
    
    // 3. 查询支付订单
    var paymentOrder PaymentOrder
    if err := db.Where("payment_order_no = ?", outTradeNo).First(&paymentOrder).Error; err != nil {
        c.String(http.StatusOK, "fail")
        return
    }
    
    // 4. 幂等性检查
    if paymentOrder.Status == "paid" {
        c.String(http.StatusOK, "success")
        return
    }
    
    // 5. 处理支付结果
    if tradeStatus == "TRADE_SUCCESS" || tradeStatus == "TRADE_FINISHED" {
        if err := b.handlePaymentSuccess(db, &paymentOrder, tradeNo); err != nil {
            c.String(http.StatusOK, "fail")
            return
        }
    } else if tradeStatus == "TRADE_CLOSED" {
        paymentOrder.Status = "closed"
        db.Save(&paymentOrder)
    }
    
    c.String(http.StatusOK, "success")
}
```

```go
func (b *PaymentBiz) handlePaymentSuccess(db *gorm.DB, paymentOrder *PaymentOrder, thirdPartyTxnNo string) error {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 锁定支付订单
        var order PaymentOrder

            First(&order, paymentOrder.ID).Error; err != nil {
            return err
        }
        
        // 2. 幂等性检查
        if order.Status == "paid" {
            return nil
        }
        
        // 3. 增加收款账户余额
        var payeeAccount Account
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&payeeAccount, order.PayeeAccountID).Error; err != nil {
            return err
        }
        
        beforeBalance := payeeAccount.Balance
        payeeAccount.Balance = payeeAccount.Balance.Add(order.Amount)
        payeeAccount.AvailableBalance = payeeAccount.AvailableBalance.Add(order.Amount)
        payeeAccount.TotalIncome = payeeAccount.TotalIncome.Add(order.Amount)
        payeeAccount.LastTransactionAt = timeNow()
        
        if err := tx.Save(&payeeAccount).Error; err != nil {
            return err
        }
        
        // 4. 创建收款流水
        txn := &TransactionRecord{
            TransactionNo:   generateTransactionNo(),
            AccountID:       order.PayeeAccountID,
            AccountNo:       payeeAccount.AccountNo,
            TransactionType: TransactionTypeOrderPayment,
            BusinessType:    order.RelatedType,
            Direction:       "income",
            Amount:          order.Amount,
            BeforeBalance:   beforeBalance,
            AfterBalance:    payeeAccount.Balance,
            RelatedType:     order.RelatedType,
            RelatedID:       order.RelatedID,
            RelatedNo:       order.RelatedNo,
            PaymentMethodID: order.PaymentMethodID,
            PaymentAccount:  thirdPartyTxnNo,
            PaymentTime:     timeNow(),
            Description:     fmt.Sprintf("收款: %s", order.RelatedNo),
            Status:          "success",
            OperatorID:      order.CreatorID,
        }
        
        if err := tx.Create(txn).Error; err != nil {
            return err
        }
        
        // 5. 更新支付订单
        order.Status = "paid"
        order.PaidAt = timeNow()
        order.ThirdPartyTransactionNo = thirdPartyTxnNo
        order.PaymentTransactionNo = txn.TransactionNo
        
        return tx.Save(&order).Error
    })
}
```

```go
func (b *RefundBiz) Create(db *gorm.DB, req *ReqCreateRefund, userID uint) (*RespCreateRefund, error) {
    return db.Transaction(func(tx *gorm.DB) (*RespCreateRefund, error) {
        // 1. 查询原支付记录
        var paymentOrder PaymentOrder
        if err := tx.Where("related_type = ? AND related_id = ?", 
            req.RelatedType, req.RelatedID).First(&paymentOrder).Error; err != nil {
            return nil, errors.New("未找到原支付记录")
        }
        
        if paymentOrder.Status != "paid" {
            return nil, errors.New("原订单未支付，无法退款")
        }
        
        // 2. 检查退款金额
        if req.RefundAmount.GreaterThan(paymentOrder.Amount) {
            return nil, errors.New("退款金额超过支付金额")
        }
        
        // 3. 检查已退款金额
        var totalRefunded decimal.Decimal
        tx.Model(&RefundOrder{}).
            Where("payment_order_id = ? AND status = 'success'", paymentOrder.ID).
            Select("COALESCE(SUM(refund_amount), 0)").
            Scan(&totalRefunded)
        
        if totalRefunded.Add(req.RefundAmount).GreaterThan(paymentOrder.Amount) {
            return nil, errors.New("退款总额超过支付金额")
        }
        
        // 4. 创建退款订单
        refundNo := generateRefundNo()
        refund := &RefundOrder{
            RefundNo:         refundNo,
            PaymentOrderID:   paymentOrder.ID,
            PaymentOrderNo:   paymentOrder.PaymentOrderNo,
            RelatedType:      req.RelatedType,
            RelatedID:        req.RelatedID,
            RelatedNo:        req.RelatedNo,
            RefundAccountID:  req.RefundAccountID,
            RefundAmount:     req.RefundAmount,
            PaymentMethodID:  paymentOrder.PaymentMethodID,
            PaymentMethod:    paymentOrder.PaymentMethod,
            Reason:           req.Reason,
            Remark:           req.Remark,
            Status:           "pending",
            CreatorID:        userID,
        }
        
        if err := tx.Create(refund).Error; err != nil {
            return nil, err
        }
        
        // 5. 根据支付方式处理退款
        var err error
        switch paymentOrder.PaymentMethod {
        case "balance":
            // 余额支付直接退款
            err = b.processBalanceRefund(tx, refund)
        case "alipay":
            // 调用支付宝退款接口
            err = b.processAlipayRefund(tx, refund)
        case "wechat":
            // 调用微信退款接口
            err = b.processWechatRefund(tx, refund)
        default:
            // 线下退款，需要财务确认
            refund.Status = "pending_approval"
            tx.Save(refund)
        }
        
        if err != nil {
            return nil, err
        }
        
        return &RespCreateRefund{
            RefundNo:     refundNo,
            RefundAmount: req.RefundAmount,
            Status:       refund.Status,
        }, nil
    })
}
```

```go
func (b *RefundBiz) processBalanceRefund(tx *gorm.DB, refund *RefundOrder) error {
    // 1. 锁定退款账户
    var refundAccount Account
    if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
        First(&refundAccount, refund.RefundAccountID).Error; err != nil {
        return err
    }
    
    // 2. 增加账户余额
    beforeBalance := refundAccount.Balance
    refundAccount.Balance = refundAccount.Balance.Add(refund.RefundAmount)
    refundAccount.AvailableBalance = refundAccount.AvailableBalance.Add(refund.RefundAmount)
    refundAccount.TotalIncome = refundAccount.TotalIncome.Add(refund.RefundAmount)
    refundAccount.LastTransactionAt = timeNow()
    
    if err := tx.Save(&refundAccount).Error; err != nil {
        return err
    }
    
    // 3. 创建退款流水
    txn := &TransactionRecord{
        TransactionNo:   generateTransactionNo(),
        AccountID:       refund.RefundAccountID,
        AccountNo:       refundAccount.AccountNo,
        TransactionType: TransactionTypeOrderRefund,
        BusinessType:    refund.RelatedType,
        Direction:       "income",
        Amount:          refund.RefundAmount,
        BeforeBalance:   beforeBalance,
        AfterBalance:    refundAccount.Balance,
        RelatedType:     refund.RelatedType,
        RelatedID:       refund.RelatedID,
        RelatedNo:       refund.RelatedNo,
        PaymentMethodID: refund.PaymentMethodID,
        Description:     fmt.Sprintf("退款: %s", refund.RelatedNo),
        Remark:          refund.Reason,
        Status:          "success",
        OperatorID:      refund.CreatorID,
    }
    
    if err := tx.Create(txn).Error; err != nil {
        return err
    }
    
    // 4. 更新退款订单
    refund.Status = "success"
    refund.RefundedAt = timeNow()
    refund.RefundTransactionNo = txn.TransactionNo
    
    return tx.Save(refund).Error
}
```

```go
    return db.Transaction(func(tx *gorm.DB) (*RespCreateWithdrawal, error) {
        // 1. 锁定账户
        var account Account
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&account, req.AccountID).Error; err != nil {
            return nil, errors.New("账户不存在")
        }
        
        if account.Status != "active" {
            return nil, errors.New("账户状态异常")
        }
        
        // 2. 检查余额
        if account.AvailableBalance.LessThan(req.Amount) {
            return nil, errors.New("可用余额不足")
        }
        
        // 3. 冻结提现金额
        account.FrozenAmount = account.FrozenAmount.Add(req.Amount)
        account.AvailableBalance = account.AvailableBalance.Sub(req.Amount)
        
        if err := tx.Save(&account).Error; err != nil {
            return nil, err
        }
        
        // 4. 创建提现订单
        withdrawalNo := generateWithdrawalNo()
        withdrawal := &WithdrawalOrder{
            WithdrawalNo:  withdrawalNo,
            AccountID:     req.AccountID,
            Amount:        req.Amount,
            BankName:      req.BankName,
            BankAccount:   req.BankAccount,
            AccountHolder: req.AccountHolder,
            Status:        "pending_approval",
            Remark:        req.Remark,
            CreatorID:     userID,
        }
        
        if err := tx.Create(withdrawal).Error; err != nil {
            return nil, err
        }
        
        return &RespCreateWithdrawal{
            WithdrawalNo: withdrawalNo,
            Amount:       req.Amount,
            Status:       withdrawal.Status,
        }, nil
    })
```

```go
func (b *WithdrawalBiz) Approve(db *gorm.DB, withdrawalID uint, req *ReqApproveWithdrawal, auditorID uint) error {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 锁定提现订单
        var withdrawal WithdrawalOrder
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&withdrawal, withdrawalID).Error; err != nil {
            return err
        }
        
        if withdrawal.Status != "pending_approval" {
            return errors.New("订单状态不允许审核")
        }
        
        // 2. 锁定账户
        var account Account
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&account, withdrawal.AccountID).Error; err != nil {
            return err
        }
        
        if req.Approved {
            // 审核通过，扣减余额
            beforeBalance := account.Balance
            account.Balance = account.Balance.Sub(withdrawal.Amount)
            account.FrozenAmount = account.FrozenAmount.Sub(withdrawal.Amount)
            account.TotalExpense = account.TotalExpense.Add(withdrawal.Amount)
            account.LastTransactionAt = timeNow()
            
            if err := tx.Save(&account).Error; err != nil {
                return err
            }
            
            // 创建支出流水
            txn := &TransactionRecord{
                TransactionNo:   generateTransactionNo(),
                AccountID:       withdrawal.AccountID,
                AccountNo:       account.AccountNo,
                TransactionType: TransactionTypeWithdraw,
                BusinessType:    "withdrawal",
                Direction:       "expense",
                Amount:          withdrawal.Amount,
                BeforeBalance:   beforeBalance,
                AfterBalance:    account.Balance,
                RelatedType:     "withdrawal",
                RelatedID:       withdrawal.ID,
                RelatedNo:       withdrawal.WithdrawalNo,
                Description:     "提现",
                Remark:          req.Remark,
                Status:          "success",
                OperatorID:      auditorID,
            }
            
            if err := tx.Create(txn).Error; err != nil {
                return err
            }
            
            withdrawal.Status = "approved"
            withdrawal.WithdrawalTransactionNo = txn.TransactionNo
        } else {
            // 审核拒绝，解冻金额
            account.FrozenAmount = account.FrozenAmount.Sub(withdrawal.Amount)
            account.AvailableBalance = account.AvailableBalance.Add(withdrawal.Amount)
            
            if err := tx.Save(&account).Error; err != nil {
                return err
            }
            
            withdrawal.Status = "rejected"
        }
        
        withdrawal.AuditorID = auditorID
        withdrawal.AuditedAt = timeNow()
        withdrawal.AuditRemark = req.Remark
        
        return tx.Save(&withdrawal).Error
    })
}
```

```go
func (b *ReconciliationBiz) Create(db *gorm.DB, req *ReqCreateReconciliation, userID uint) (*RespCreateReconciliation, error) {
    // 1. 查询时间范围内的所有交易
    var transactions []TransactionRecord
    if err := db.Where("payment_method_id = ? AND DATE(created_at) BETWEEN ? AND ? AND status = 'success'",
        req.PaymentMethodID, req.StartDate, req.EndDate).
        Find(&transactions).Error; err != nil {
        return nil, err
    }
    
    // 2. 统计数据
    var totalIncome, totalExpense decimal.Decimal
    incomeCount := 0
    expenseCount := 0
    
    for _, txn := range transactions {
        if txn.Direction == "income" {
            totalIncome = totalIncome.Add(txn.Amount)
            incomeCount++
        } else {
            totalExpense = totalExpense.Add(txn.Amount)
            expenseCount++
        }
    }
    
    // 3. 创建对账单
    reconciliationNo := generateReconciliationNo()
    reconciliation := &Reconciliation{
        ReconciliationNo: reconciliationNo,
        PaymentMethodID:  req.PaymentMethodID,
        StartDate:        req.StartDate,
        EndDate:          req.EndDate,
        TotalIncome:      totalIncome,
        TotalExpense:     totalExpense,
        IncomeCount:      incomeCount,
        ExpenseCount:     expenseCount,
        Status:           "pending",
        CreatorID:        userID,
    }
    
    if err := db.Create(reconciliation).Error; err != nil {
        return nil, err
    }
    
    return &RespCreateReconciliation{
        ReconciliationNo: reconciliationNo,
        TotalIncome:      totalIncome,
        TotalExpense:     totalExpense,
        TotalCount:       incomeCount + expenseCount,
    }, nil
}
```
