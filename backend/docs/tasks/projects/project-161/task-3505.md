# 售后物流管理设计文档

## 1. 概述

本文档定义Coolbuy 3.0售后模块的物流管理功能，包括退货物流、换货物流、物流跟踪等。

## 2. 物流类型

### 2.1 退货物流 (return)

客户将商品寄回仓库的物流。

### 2.2 换货退回物流 (exchange_return)

换货场景中，客户将原商品寄回仓库的物流。

### 2.3 换货发出物流 (exchange_send)

换货场景中，仓库将新商品发给客户的物流。

## 3. 物流跟踪

### 3.1 物流信息录入

```go
// TrackLogistics 物流跟踪
func (b *LogisticsBiz) TrackLogistics(db *gorm.DB, logisticsID uint) error {
	var logistics model.AfterSalesLogistics
	db.First(&logistics, logisticsID)
	
	// 调用物流商API获取最新轨迹
	tracks, err := b.getTrackingFromProvider(logistics.ProviderCode, logistics.TrackingNo)
	if err != nil {
		return fmt.Errorf("获取物流信息失败: %w", err)
	}
	
	// 更新物流状态
	if len(tracks) > 0 {
		latestTrack := tracks[0]
		
		updates := map[string]interface{}{
			"logistics_status": latestTrack.Status,
			"current_location": latestTrack.Location,
		}
		
		if latestTrack.Status == "delivered" {
			updates["delivered_at"] = latestTrack.Time
			updates["signed_by"] = latestTrack.Operator
		}
		
		db.Model(&logistics).Updates(updates)
		
		// 如果已签收，触发相应的业务流程
		if latestTrack.Status == "delivered" {
			b.handleLogisticsDelivered(db, &logistics)
		}
	}
	
	return nil
}

// handleLogisticsDelivered 处理物流签收
func (b *LogisticsBiz) handleLogisticsDelivered(db *gorm.DB, logistics *model.AfterSalesLogistics) error {
	switch logistics.LogisticsType {
	case "return", "exchange_return":
		// 退货已签收，触发收货质检流程
		return b.triggerReceiveReturn(db, logistics.AfterSalesID)
		
	case "exchange_send":
		// 换货已签收，完成换货流程
		return b.triggerCompleteExchange(db, logistics.AfterSalesID)
	}
	
	return nil
}
```

### 3.2 物流状态同步

```go
// SyncLogisticsStatus 同步物流状态（定时任务）
func (b *LogisticsBiz) SyncLogisticsStatus(db *gorm.DB) error {
	// 查询所有未完成的物流单
	var logisticsList []model.AfterSalesLogistics
	db.Where("logistics_status NOT IN ?", []string{"delivered", "cancelled"}).
		Find(&logisticsList)
	
	for _, logistics := range logisticsList {
		// 同步物流状态
		if err := b.TrackLogistics(db, logistics.ID); err != nil {
			log.Printf("同步物流状态失败: %v", err)
			continue
		}
		
		// 避免频繁调用API
		time.Sleep(100 * time.Millisecond)
	}
	
	return nil
}
```

```go
// ReturnAddress 退货地址配置
type ReturnAddress struct {
	ID          uint   `json:"id"`
	WarehouseID uint   `json:"warehouse_id"` // 仓库ID
	Name        string `json:"name"`         // 收货人
	Phone       string `json:"phone"`        // 电话
	Province    string `json:"province"`     // 省
	City        string `json:"city"`         // 市
	District    string `json:"district"`     // 区
	Address     string `json:"address"`      // 详细地址
	IsDefault   bool   `json:"is_default"`   // 是否默认
}

// GetReturnAddress 获取退货地址
func (b *LogisticsBiz) GetReturnAddress(db *gorm.DB, warehouseID uint) (*ReturnAddress, error) {
	var address ReturnAddress
	
	if warehouseID > 0 {
		// 获取指定仓库的退货地址
		db.Where("warehouse_id = ?", warehouseID).First(&address)
	} else {
		// 获取默认退货地址
		db.Where("is_default = ?", true).First(&address)
	}
	
	return &address, nil
}
```

## 5. 物流费用管理

```go
// CalculateReturnFreight 计算退货运费
func (b *LogisticsBiz) CalculateReturnFreight(afterSales *model.AfterSales) (decimal.Decimal, error) {
	// 根据售后原因判断运费承担方
	switch afterSales.ReasonType {
	case "quality_issue", "wrong_item":
		// 质量问题、发错货 - 公司承担运费
		return decimal.Zero, nil
		
	case "size_mismatch", "not_needed", "other":
		// 尺码不符、不想要了 - 客户承担运费
		return decimal.NewFromFloat(10.00), nil // 固定运费
		
	default:
		return decimal.NewFromFloat(10.00), nil
	}
}

// RefundFreight 退运费
func (b *LogisticsBiz) RefundFreight(db *gorm.DB, afterSalesID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var afterSales model.AfterSales
		tx.First(&afterSales, afterSalesID)
		
		// 计算应退运费
		freight, _ := b.CalculateReturnFreight(&afterSales)
		
		if freight.IsZero() {
			// 无需退运费
			return nil
		}
		
		// 更新售后单退运费金额
		tx.Model(&afterSales).Update("refund_freight", freight)
		
		// 增加实际退款金额
		tx.Model(&afterSales).Update("actual_refund_amount", 
			gorm.Expr("actual_refund_amount + ?", freight))
		
		return nil
	})
}
```

## 6. 物流异常处理

```go
// LogisticsException 物流异常类型
const (
	ExceptionLost       = "lost"        // 包裹丢失
	ExceptionDamaged    = "damaged"     // 包裹损坏
	ExceptionRefused    = "refused"     // 拒收
	ExceptionReturned   = "returned"    // 退回
	ExceptionTimeout    = "timeout"     // 超时未签收
)

// HandleLogisticsException 处理物流异常
func (b *LogisticsBiz) HandleLogisticsException(db *gorm.DB, logisticsID uint, exceptionType, remark string) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var logistics model.AfterSalesLogistics
		tx.First(&logistics, logisticsID)
		
		// 更新物流状态
		tx.Model(&logistics).Updates(map[string]interface{}{
			"logistics_status": "exception",
			"remark":           fmt.Sprintf("[%s] %s", exceptionType, remark),
		})
		
		// 创建售后日志
		log := &model.AfterSalesLog{
			AfterSalesID:  logistics.AfterSalesID,
			AfterSalesNo:  logistics.AfterSalesNo,
			OperationType: "logistics_exception",
			OperationDesc: fmt.Sprintf("物流异常: %s - %s", exceptionType, remark),
			OperatorType:  "system",
		}
		tx.Create(log)
		
		// 根据异常类型进行处理
		switch exceptionType {
		case ExceptionLost:
			// 包裹丢失 - 直接退款
			return b.processLostPackage(tx, logistics.AfterSalesID)
			
		case ExceptionDamaged:
			// 包裹损坏 - 记录并等待客服处理
			return nil
			
		case ExceptionRefused:
			// 拒收 - 关闭售后
			return b.closeAfterSales(tx, logistics.AfterSalesID, "客户拒收")
		}
		
		return nil
	})
}
```

```go
// 物流商状态 → 系统状态映射
var LogisticsStatusMap = map[string]map[string]string{
	"sf_express": {
		"收件":   "picked_up",
		"在途中":  "in_transit",
		"派件中":  "delivering",
		"已签收":  "delivered",
		"已取消":  "cancelled",
		"退回":   "returned",
	},
	"yto": {
		"已揽收":  "picked_up",
		"运输中":  "in_transit",
		"派送中":  "delivering",
		"已签收":  "delivered",
		"问题件":  "exception",
	},
}

// MapLogisticsStatus 映射物流状态
func MapLogisticsStatus(provider, providerStatus string) string {
	if statusMap, ok := LogisticsStatusMap[provider]; ok {
		if status, ok := statusMap[providerStatus]; ok {
			return status
		}
	}
	return "unknown"
}
```
