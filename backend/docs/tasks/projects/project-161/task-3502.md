# 退货退款功能设计文档

## 1. 概述

本文档定义Coolbuy 3.0售后模块的退货退款业务逻辑，包括申请创建、审批流程、退货质检、退款处理等核心功能。

### 1.1 业务场景

- **仅退款**: 不需要退货，直接退款（如多付款、价格保护等）
- **退货退款**: 需要退货后退款（如质量问题、尺码不符等）
- **部分退货**: 订单中部分商品退货

### 1.2 业务流程

```
客户申请 → 审批 → [需退货] → 客户发货 → 仓库收货 → 质检 → 退款 → 完成
                 ↓
              [仅退款] → 直接退款 → 完成
```

## 2. 核心功能设计

### 2.1 创建售后申请

**功能说明**: 客户提交售后申请，系统验证并创建售后单

**接口定义**:
```go
// ReqCreateAfterSales 创建售后申请请求
type ReqCreateAfterSales struct {
	OrderID        uint                   `json:"order_id" binding:"required"`        // 订单ID
	AfterSalesType string                 `json:"after_sales_type" binding:"required,oneof=refund_only return_refund exchange"` // 售后类型
	ReasonType     string                 `json:"reason_type" binding:"required"`     // 原因类型
	ReasonDetail   string                 `json:"reason_detail"`                      // 详细原因
	Items          []ReqAfterSalesItem    `json:"items" binding:"required,min=1"`     // 退货商品
	Images         []string               `json:"images"`                             // 凭证图片URL

}

type ReqAfterSalesItem struct {

}

// RespCreateAfterSales 创建售后申请响应
type RespCreateAfterSales struct {
```

**业务逻辑**:
```go
package biz

import (
	"errors"
	"fmt"
	"time"
	"github.com/shopspring/decimal"
	"gorm.io/gorm"
	"gorm.io/gorm/clause"
)

type AfterSalesBiz struct {
	afterSalesStore *store.AfterSalesStore
	orderStore      *store.OrderStore
	inventoryStore  *store.InventoryStore
}

func NewAfterSalesBiz() *AfterSalesBiz {
	return &AfterSalesBiz{
		afterSalesStore: store.NewAfterSalesStore(),
		orderStore:      store.NewOrderStore(),
		inventoryStore:  store.NewInventoryStore(),
	}
}

// Create 创建售后申请
func (b *AfterSalesBiz) Create(db *gorm.DB, req *ReqCreateAfterSales, customerID uint) (*RespCreateAfterSales, error) {
	var resp *RespCreateAfterSales
	
	err := db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询订单信息（加锁）
		order, err := b.orderStore.GetByIDForUpdate(tx, req.OrderID)
		if err != nil {
			return fmt.Errorf("订单不存在: %w", err)
		}
		
		// 2. 验证订单所属
		if order.CustomerID != customerID {
			return errors.New("无权操作该订单")
		}
		
		// 3. 验证订单状态（必须已完成或已发货）
		if order.Status != "completed" && order.Status != "shipped" {
			return errors.New("订单状态不允许申请售后")
		}
		
		// 4. 查询订单明细并验证
		var totalRefundAmount decimal.Decimal
		var afterSalesItems []model.AfterSalesItem
		
		for _, itemReq := range req.Items {
			orderItem, err := b.orderStore.GetOrderItem(tx, itemReq.OrderItemID)
			if err != nil {
				return fmt.Errorf("订单明细不存在: %w", err)
			}
			
			// 验证订单明细属于该订单
			if orderItem.OrderID != req.OrderID {
				return errors.New("订单明细不匹配")
			}
			
			// 验证退货数量
			if itemReq.Quantity > orderItem.Quantity - orderItem.ReturnedQty {
				return fmt.Errorf("SKU %s 退货数量超出可退货数量", orderItem.SkuCode)
			}
			
			// 计算退款金额（按比例）
			itemRefundAmount := orderItem.UnitPrice.Mul(decimal.NewFromInt(int64(itemReq.Quantity)))
			totalRefundAmount = totalRefundAmount.Add(itemRefundAmount)
			
			// 构建售后明细
			afterSalesItems = append(afterSalesItems, model.AfterSalesItem{
				OrderID:      req.OrderID,
				OrderItemID:  itemReq.OrderItemID,
				SpuID:        orderItem.SpuID,
				SkuID:        orderItem.SkuID,
				SkuCode:      orderItem.SkuCode,
				SkuName:      orderItem.SkuName,
				SpuName:      orderItem.SpuName,
				Size:         orderItem.Size,
				Color:        orderItem.Color,
				Quantity:     itemReq.Quantity,
				UnitPrice:    orderItem.UnitPrice,
				TotalPrice:   itemRefundAmount,
				RefundAmount: itemRefundAmount,
				ImageURL:     orderItem.ImageURL,
			})
		}
		
		// 5. 生成售后单号
		afterSalesNo := b.generateAfterSalesNo()
		
		// 6. 创建售后单
		afterSales := &model.AfterSales{
			AfterSalesNo:   afterSalesNo,
			AfterSalesType: req.AfterSalesType,
			OrderID:        req.OrderID,
			OrderNo:        order.OrderNo,
			CustomerID:     customerID,
			CustomerName:   order.CustomerName,
			CustomerPhone:  req.ContactPhone,
			ReasonType:     req.ReasonType,
			ReasonDetail:   req.ReasonDetail,
			TotalAmount:    totalRefundAmount,
			RefundAmount:   totalRefundAmount,
			ActualRefundAmount: totalRefundAmount, // 初始为全额，审批时可调整
			RefundAccount:  req.RefundAccount,
			Status:         "pending",
			ApprovalStatus: "pending",
			RefundStatus:   "pending",
			AppliedAt:      time.Now(),
		}
		
		// 仅退款类型，设置退款方式为原路退回
		if req.AfterSalesType == "refund_only" {
			afterSales.RefundMethod = "original"
		}
		
		if err := tx.Create(afterSales).Error; err != nil {
			return fmt.Errorf("创建售后单失败: %w", err)
		}
		
		// 7. 创建售后明细
		for i := range afterSalesItems {
			afterSalesItems[i].AfterSalesID = afterSales.ID
			afterSalesItems[i].AfterSalesNo = afterSales.AfterSalesNo
		}
		if err := tx.Create(&afterSalesItems).Error; err != nil {
			return fmt.Errorf("创建售后明细失败: %w", err)
		}
		
		// 8. 保存凭证图片
		if len(req.Images) > 0 {
			var images []model.AfterSalesImage
			for i, imageURL := range req.Images {
				images = append(images, model.AfterSalesImage{
					AfterSalesID: afterSales.ID,
					AfterSalesNo: afterSales.AfterSalesNo,
					ImageType:    "quality_issue",
					ImageURL:     imageURL,
					SortOrder:    i + 1,
				})
			}
			if err := tx.Create(&images).Error; err != nil {
				return fmt.Errorf("保存凭证图片失败: %w", err)
			}
		}
		
		// 9. 创建操作日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSales.ID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: "create",
			OperationDesc: fmt.Sprintf("客户提交%s申请", b.getAfterSalesTypeDesc(req.AfterSalesType)),
			OperatorID:    customerID,
			OperatorName:  order.CustomerName,
			OperatorType:  "customer",
			AfterStatus:   "pending",
		}
		if err := tx.Create(log).Error; err != nil {
			return fmt.Errorf("创建操作日志失败: %w", err)
		}
		
		// 10. 更新订单售后状态
		if err := tx.Model(&model.Order{}).Where("id = ?", req.OrderID).
			Update("after_sales_status", "in_progress").Error; err != nil {
			return fmt.Errorf("更新订单售后状态失败: %w", err)
		}
		
		resp = &RespCreateAfterSales{
			AfterSalesID: afterSales.ID,
			AfterSalesNo: afterSales.AfterSalesNo,
			Status:       afterSales.Status,
		}
		
		return nil
	})
	
	return resp, err
}

// generateAfterSalesNo 生成售后单号
func (b *AfterSalesBiz) generateAfterSalesNo() string {
	// 格式: AS + YYYYMMDD + 8位随机数
	return fmt.Sprintf("AS%s%08d", time.Now().Format("20060102"), time.Now().UnixNano()%100000000)
}

// getAfterSalesTypeDesc 获取售后类型描述
func (b *AfterSalesBiz) getAfterSalesTypeDesc(afterSalesType string) string {
	switch afterSalesType {
	case "refund_only":
		return "仅退款"
	case "return_refund":
		return "退货退款"
	case "exchange":
		return "换货"
	default:
		return "售后"
	}
}
```

```go
// ReqApproveAfterSales 审批售后申请请求
type ReqApproveAfterSales struct {
	Action         string          `json:"action" binding:"required,oneof=approve reject"` // 审批动作
	ApprovalRemark string          `json:"approval_remark"`                                 // 审批意见
	ApprovedAmount *decimal.Decimal `json:"approved_amount"`                               // 批准退款金额（可调整）
	DeductionAmount *decimal.Decimal `json:"deduction_amount"`                             // 扣款金额
	DeductionReason string          `json:"deduction_reason"`                              // 扣款原因
}
```

**业务逻辑**:
```go
// Approve 审批售后申请
func (b *AfterSalesBiz) Approve(db *gorm.DB, afterSalesID uint, req *ReqApproveAfterSales, approverID uint, approverName string) error {
	return db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询售后单（加锁）
		var afterSales model.AfterSales
		if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
			Preload("Items").
			First(&afterSales, afterSalesID).Error; err != nil {
			return fmt.Errorf("售后单不存在: %w", err)
		}
		
		// 2. 验证状态
		if !afterSales.CanApprove() {
			return errors.New("售后单当前状态不允许审批")
		}
		
		now := time.Now()
		updates := map[string]interface{}{
			"approver_id":     approverID,
			"approver_name":   approverName,
			"approved_at":     now,
			"approval_remark": req.ApprovalRemark,
			"approval_status": req.Action == "approve" ? "approved" : "rejected",
		}
		
		if req.Action == "approve" {
			// 审批通过
			updates["status"] = "approved"
			
			// 调整退款金额
			if req.ApprovedAmount != nil && !req.ApprovedAmount.IsZero() {
				updates["actual_refund_amount"] = *req.ApprovedAmount
			}
			if req.DeductionAmount != nil && !req.DeductionAmount.IsZero() {
				updates["deduction_amount"] = *req.DeductionAmount
				updates["actual_refund_amount"] = afterSales.RefundAmount.Sub(*req.DeductionAmount)
			}
			
			// 仅退款类型，直接设置为待退款状态
			if afterSales.AfterSalesType == "refund_only" {
				updates["refund_status"] = "pending"
			}
		} else {
			// 审批拒绝
			updates["status"] = "rejected"
		}
		
		// 3. 更新售后单
		if err := tx.Model(&afterSales).Updates(updates).Error; err != nil {
			return fmt.Errorf("更新售后单失败: %w", err)
		}
		
		// 4. 创建审批记录
		approval := &model.AfterSalesApproval{
			AfterSalesID:    afterSalesID,
			AfterSalesNo:    afterSales.AfterSalesNo,
			ApprovalLevel:   1,
			ApproverID:      approverID,
			ApproverName:    approverName,
			ApprovalStatus:  req.Action == "approve" ? "approved" : "rejected",
			ApprovalRemark:  req.ApprovalRemark,
			ApprovedAt:      &now,
			RequestedAmount: afterSales.RefundAmount,
		}
		
		if req.ApprovedAmount != nil {
			approval.ApprovedAmount = *req.ApprovedAmount
		} else {
			approval.ApprovedAmount = afterSales.RefundAmount
		}
		
		if req.DeductionAmount != nil {
			approval.DeductionAmount = *req.DeductionAmount
		}
		
		if err := tx.Create(approval).Error; err != nil {
			return fmt.Errorf("创建审批记录失败: %w", err)
		}
		
		// 5. 创建操作日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSalesID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: req.Action,
			OperationDesc: fmt.Sprintf("售后申请审批%s", map[string]string{"approve": "通过", "reject": "拒绝"}[req.Action]),
			OperatorID:    approverID,
			OperatorName:  approverName,
			OperatorType:  "staff",
			BeforeStatus:  "pending",
			AfterStatus:   updates["status"].(string),
		}
		if err := tx.Create(log).Error; err != nil {
			return fmt.Errorf("创建操作日志失败: %w", err)
		}
		
		// 6. 如果拒绝，更新订单售后状态
		if req.Action == "reject" {
			// 检查是否还有其他进行中的售后单
			var count int64
			tx.Model(&model.AfterSales{}).
				Where("order_id = ? AND id != ? AND status NOT IN ?", 
					afterSales.OrderID, afterSalesID, []string{"rejected", "cancelled", "closed"}).
				Count(&count)
			
			if count == 0 {
				tx.Model(&model.Order{}).Where("id = ?", afterSales.OrderID).
					Update("after_sales_status", "none")
			}
		}
		
		return nil
	})
}
```

```go
// SubmitReturnLogistics 提交退货物流信息
func (b *AfterSalesBiz) SubmitReturnLogistics(db *gorm.DB, afterSalesID uint, req *ReqSubmitReturnLogistics, customerID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询售后单（加锁）
		var afterSales model.AfterSales
		if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
			First(&afterSales, afterSalesID).Error; err != nil {
			return fmt.Errorf("售后单不存在: %w", err)
		}
		
		// 2. 验证权限和状态
		if afterSales.CustomerID != customerID {
			return errors.New("无权操作该售后单")
		}
		if afterSales.Status != "approved" {
			return errors.New("售后单状态不允许提交物流")
		}
		if afterSales.AfterSalesType == "refund_only" {
			return errors.New("仅退款类型无需提交物流")
		}
		
		now := time.Now()
		
		// 3. 更新售后单状态
		updates := map[string]interface{}{
			"return_logistics_provider": req.LogisticsProvider,
			"return_tracking_no":        req.TrackingNo,
			"return_shipped_at":         now,
			"status":                    "returning",
		}
		if err := tx.Model(&afterSales).Updates(updates).Error; err != nil {
			return fmt.Errorf("更新售后单失败: %w", err)
		}
		
		// 4. 创建物流记录
		// 获取收货地址（仓库地址）
		warehouse, _ := b.getWarehouseAddress(tx, afterSales.OrderID)
		
		logistics := &model.AfterSalesLogistics{
			AfterSalesID:    afterSalesID,
			AfterSalesNo:    afterSales.AfterSalesNo,
			LogisticsType:   "return",
			ProviderCode:    req.LogisticsProvider,
			ProviderName:    req.LogisticsProvider,
			TrackingNo:      req.TrackingNo,
			SenderName:      req.SenderName,
			SenderPhone:     req.SenderPhone,
			SenderAddress:   req.SenderAddress,
			ReceiverName:    warehouse.Name,
			ReceiverPhone:   warehouse.Phone,
			ReceiverAddress: warehouse.Address,
			LogisticsStatus: "in_transit",
			ShippedAt:       &now,
		}
		if err := tx.Create(logistics).Error; err != nil {
			return fmt.Errorf("创建物流记录失败: %w", err)
		}
		
		// 5. 创建操作日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSalesID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: "submit_logistics",
			OperationDesc: fmt.Sprintf("客户已发货，物流单号: %s", req.TrackingNo),
			OperatorID:    customerID,
			OperatorName:  afterSales.CustomerName,
			OperatorType:  "customer",
			BeforeStatus:  "approved",
			AfterStatus:   "returning",
		}
		if err := tx.Create(log).Error; err != nil {
			return fmt.Errorf("创建操作日志失败: %w", err)
		}
		
		return nil
	})
}
```

**业务逻辑**:
```go
// ReceiveReturn 仓库收货与质检
func (b *AfterSalesBiz) ReceiveReturn(db *gorm.DB, afterSalesID uint, req *ReqReceiveReturn, receiverID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询售后单（加锁）
		var afterSales model.AfterSales
		if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
			Preload("Items").
			First(&afterSales, afterSalesID).Error; err != nil {
			return fmt.Errorf("售后单不存在: %w", err)
		}
		
		// 2. 验证状态
		if afterSales.Status != "returning" {
			return errors.New("售后单状态不允许收货")
		}
		
		now := time.Now()
		allPassed := true
		
		// 3. 更新质检结果
		for _, itemReq := range req.Items {
			updates := map[string]interface{}{
				"quality_check_status": itemReq.QualityCheckStatus,
				"quality_check_remark": itemReq.QualityCheckRemark,
				"quality_checker_id":   receiverID,
				"quality_checked_at":   now,
			}
			
			if err := tx.Model(&model.AfterSalesItem{}).
				Where("id = ? AND after_sales_id = ?", itemReq.AfterSalesItemID, afterSalesID).
				Updates(updates).Error; err != nil {
				return fmt.Errorf("更新质检结果失败: %w", err)
			}
			
			if itemReq.QualityCheckStatus == "failed" {
				allPassed = false
			}
		}
		
		// 4. 更新售后单状态
		updates := map[string]interface{}{
			"return_received_at": now,
			"status":             "returned",
		}
		
		// 如果质检全部通过，设置为待退款
		if allPassed {
			updates["refund_status"] = "pending"
		} else {
			// 如果有质检不通过的，需要重新审批或扣款
			updates["status"] = "approval_pending"
		}
		
		if err := tx.Model(&afterSales).Updates(updates).Error; err != nil {
			return fmt.Errorf("更新售后单失败: %w", err)
		}
		
		// 5. 恢复库存（质检通过的商品）
		for _, item := range afterSales.Items {
			for _, itemReq := range req.Items {
				if item.ID == itemReq.AfterSalesItemID && itemReq.QualityCheckStatus == "passed" {
					// 获取原发货仓库
					warehouseID, _ := b.getOrderWarehouse(tx, afterSales.OrderID)
					if err := b.inventoryStore.AddInventory(tx, item.SkuID, warehouseID, item.Quantity); err != nil {
						return fmt.Errorf("恢复库存失败: %w", err)
					}
				}
			}
		}
		
		// 6. 更新订单明细退货数量
		for _, item := range afterSales.Items {
			for _, itemReq := range req.Items {
				if item.ID == itemReq.AfterSalesItemID && itemReq.QualityCheckStatus == "passed" {
					if err := tx.Model(&model.OrderItem{}).
						Where("id = ?", item.OrderItemID).
						Update("returned_qty", gorm.Expr("returned_qty + ?", item.Quantity)).Error; err != nil {
						return fmt.Errorf("更新订单明细退货数量失败: %w", err)
					}
				}
			}
		}
		
		// 7. 更新物流状态
		if err := tx.Model(&model.AfterSalesLogistics{}).
			Where("after_sales_id = ? AND logistics_type = ?", afterSalesID, "return").
			Updates(map[string]interface{}{
				"logistics_status": "delivered",
				"delivered_at":     now,
				"signed_by":        req.ReceivedBy,
			}).Error; err != nil {
			return fmt.Errorf("更新物流状态失败: %w", err)
		}
		
		// 8. 创建操作日志
		log := &model.AfterSalesLog{
```

```go
// ProcessRefund 处理退款
func (b *AfterSalesBiz) ProcessRefund(db *gorm.DB, afterSalesID uint, req *ReqProcessRefund, operatorID uint, operatorName string) error {
	return db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询售后单（加锁）
		var afterSales model.AfterSales
		if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
			First(&afterSales, afterSalesID).Error; err != nil {
			return fmt.Errorf("售后单不存在: %w", err)
		}
		
		// 2. 验证状态
		if !afterSales.CanRefund() {
			return errors.New("售后单当前状态不允许退款")
		}
		
		// 3. 验证退款金额
		if req.RefundAmount.GreaterThan(afterSales.ActualRefundAmount) {
			return errors.New("退款金额超出应退金额")
		}
		
		now := time.Now()
		
		// 4. 调用支付平台退款接口
		transactionNo, err := b.callPaymentRefund(afterSales.OrderNo, req.RefundAmount, req.PaymentPlatform)
		if err != nil {
			return fmt.Errorf("调用支付平台退款失败: %w", err)
		}
		
		// 5. 创建退款记录
		refundNo := b.generateRefundNo()
		refund := &model.AfterSalesRefund{
			AfterSalesID:    afterSalesID,
			AfterSalesNo:    afterSales.AfterSalesNo,
			RefundNo:        refundNo,
			RefundType:      "full",
			RefundAmount:    req.RefundAmount,
			RefundMethod:    req.RefundMethod,
			RefundAccount:   req.RefundAccount,
			RefundStatus:    "success",
			RefundedAt:      &now,
			TransactionNo:   transactionNo,
			PaymentPlatform: req.PaymentPlatform,
			OperatorID:      operatorID,
			OperatorName:    operatorName,
			Remark:          req.Remark,
		}
		if err := tx.Create(refund).Error; err != nil {
			return fmt.Errorf("创建退款记录失败: %w", err)
		}
		
		// 6. 更新售后单状态
		updates := map[string]interface{}{
			"refund_status":        "success",
			"refunded_at":          now,
			"refund_transaction_no": transactionNo,
			"status":               "completed",
			"completed_at":         now,
		}
		if err := tx.Model(&afterSales).Updates(updates).Error; err != nil {
			return fmt.Errorf("更新售后单失败: %w", err)
		}
		
		// 7. 创建资金流水记录（对接财务模块）
		if err := b.createFinancialTransaction(tx, &afterSales, refund); err != nil {
			return fmt.Errorf("创建资金流水失败: %w", err)
		}
		
		// 8. 更新订单退款金额
		if err := tx.Model(&model.Order{}).
			Where("id = ?", afterSales.OrderID).
			Update("refunded_amount", gorm.Expr("refunded_amount + ?", req.RefundAmount)).Error; err != nil {
			return fmt.Errorf("更新订单退款金额失败: %w", err)
		}
		
		// 9. 检查订单是否全部退款，更新售后状态
		var order model.Order
		tx.First(&order, afterSales.OrderID)
		if order.RefundedAmount.GreaterThanOrEqual(order.TotalAmount) {
			tx.Model(&model.Order{}).Where("id = ?", afterSales.OrderID).
				Update("after_sales_status", "all_refunded")
		} else {
			tx.Model(&model.Order{}).Where("id = ?", afterSales.OrderID).
				Update("after_sales_status", "partial_refunded")
		}
		
		// 10. 创建操作日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSalesID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: "refund",
			OperationDesc: fmt.Sprintf("退款成功，金额: ¥%.2f，流水号: %s", req.RefundAmount, transactionNo),
			OperatorID:    operatorID,
			OperatorName:  operatorName,
			OperatorType:  "staff",
			BeforeStatus:  "returned",
			AfterStatus:   "completed",
		}
		if err := tx.Create(log).Error; err != nil {
			return fmt.Errorf("创建操作日志失败: %w", err)
		}
		
		return nil
	})
}
```

```go
// Cancel 取消售后申请
func (b *AfterSalesBiz) Cancel(db *gorm.DB, afterSalesID uint, customerID uint, reason string) error {
	return db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询售后单（加锁）
		var afterSales model.AfterSales
		if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
			First(&afterSales, afterSalesID).Error; err != nil {
			return fmt.Errorf("售后单不存在: %w", err)
		}
		
		// 2. 验证权限和状态
		if afterSales.CustomerID != customerID {
			return errors.New("无权操作该售后单")
		}
		if !afterSales.CanCancel() {
			return errors.New("售后单当前状态不允许取消")
		}
		
		now := time.Now()
		
		// 3. 更新售后单状态
		updates := map[string]interface{}{
			"status":      "cancelled",
			"cancelled_at": now,
			"remark":      reason,
		}
		if err := tx.Model(&afterSales).Updates(updates).Error; err != nil {
			return fmt.Errorf("更新售后单失败: %w", err)
		}
		
		// 4. 创建操作日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSalesID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: "cancel",
			OperationDesc: fmt.Sprintf("客户取消售后申请: %s", reason),
			OperatorID:    customerID,
			OperatorName:  afterSales.CustomerName,
			OperatorType:  "customer",
			BeforeStatus:  afterSales.Status,
			AfterStatus:   "cancelled",
		}
		if err := tx.Create(log).Error; err != nil {
			return fmt.Errorf("创建操作日志失败: %w", err)
		}
		
		// 5. 检查是否还有其他进行中的售后单，更新订单售后状态
		var count int64
		tx.Model(&model.AfterSales{}).
			Where("order_id = ? AND id != ? AND status NOT IN ?", 
				afterSales.OrderID, afterSalesID, []string{"rejected", "cancelled", "closed"}).
			Count(&count)
		
		if count == 0 {
			tx.Model(&model.Order{}).Where("id = ?", afterSales.OrderID).
				Update("after_sales_status", "none")
		}
		
		return nil
	})
}
```
