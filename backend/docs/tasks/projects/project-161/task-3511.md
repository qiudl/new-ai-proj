# 奖金发放流程设计文档

## 1. 概述

本文档定义奖金发放的完整流程，包括发放批次创建、审批、发放、确认等环节。

## 2. 发放流程

```
创建发放批次 → 确认发放明细 → 审批 → 执行发放 → 完成
```

## 3. 核心功能

### 3.1 创建发放批次

```go
// CreateDistribution 创建发放批次
func (b *BonusBiz) CreateDistribution(db *gorm.DB, req *ReqCreateDistribution) (*model.BonusDistribution, error) {
	return db.Transaction(func(tx *gorm.DB) (*model.BonusDistribution, error) {
		// 查询待发放的佣金记录
		var records []model.CommissionRecord
		tx.Where("stat_period = ? AND status = ?", req.StatPeriod, "pending").
			Find(&records)
		
		if len(records) == 0 {
			return nil, errors.New("没有待发放的佣金记录")
		}
		
		// 计算总金额和人数
		totalAmount := decimal.Zero
		userMap := make(map[uint]bool)
		for _, record := range records {
			totalAmount = totalAmount.Add(record.TotalAmount)
			userMap[record.UserID] = true
		}
		
		// 生成发放批次号
		distributionNo := fmt.Sprintf("DIST%s%04d", time.Now().Format("200601"), rand.Intn(10000))
		
		// 创建发放批次
		distribution := &model.BonusDistribution{
			DistributionNo:   distributionNo,
			DistributionName: req.DistributionName,
			StatPeriod:       req.StatPeriod,
			StatYear:         req.StatYear,
			StatMonth:        req.StatMonth,
			TotalAmount:      totalAmount,
			UserCount:        len(userMap),
			Status:           "draft",
		}
		
		if err := tx.Create(distribution).Error; err != nil {
			return nil, fmt.Errorf("创建发放批次失败: %w", err)
		}
		
		// 关联佣金记录
		tx.Model(&model.CommissionRecord{}).
			Where("stat_period = ? AND status = ?", req.StatPeriod, "pending").
			Update("distribution_id", distribution.ID)
		
		return distribution, nil
	})
}
```

### 3.2 确认发放批次

```go
// ConfirmDistribution 确认发放批次
func (b *BonusBiz) ConfirmDistribution(db *gorm.DB, distributionID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var distribution model.BonusDistribution
		tx.First(&distribution, distributionID)
		
		if distribution.Status != "draft" {
			return errors.New("只有草稿状态的批次可以确认")
		}
		
		// 更新状态
		tx.Model(&distribution).Update("status", "confirmed")
		
		// 更新佣金记录状态
		tx.Model(&model.CommissionRecord{}).
			Where("distribution_id = ?", distributionID).
			Update("status", "confirmed")
		
		return nil
	})
}
```

### 3.3 执行发放

```go
// ExecuteDistribution 执行发放
func (b *BonusBiz) ExecuteDistribution(db *gorm.DB, distributionID uint, operatorID uint, operatorName string) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var distribution model.BonusDistribution
		tx.Preload("CommissionRecords").First(&distribution, distributionID)
		
		if distribution.Status != "confirmed" {
			return errors.New("只有已确认的批次可以发放")
		}
		
		// 更新批次状态
		now := time.Now()
		tx.Model(&distribution).Updates(map[string]interface{}{
			"status":               "processing",
			"distributed_by":       operatorID,
			"distributed_by_name":  operatorName,
			"distributed_at":       now,
		})
		
		// 调用支付接口发放奖金（这里模拟）
		for _, record := range distribution.CommissionRecords {
			// 实际对接支付系统
			err := b.payToUser(record.UserID, record.TotalAmount)
			if err != nil {
				log.Printf("发放失败 - User: %d, Amount: %s, Error: %v", 
					record.UserID, record.TotalAmount, err)
				continue
			}
			
			// 更新记录状态
			tx.Model(&record).Update("status", "paid")
		}
		
		// 更新批次状态为已完成
		tx.Model(&distribution).Update("status", "completed")
		
		return nil
	})
}
```

## 4. 发放状态流转

```
draft (草稿) → confirmed (已确认) → processing (发放中) → completed (已完成)
                                                    ↓
                                             cancelled (已取消)
```
