## 2. 核心功能

### 2.1 创建佣金模板

```go
// ReqCreateTemplate 创建模板请求
type ReqCreateTemplate struct {
	TemplateName    string          `json:"template_name" binding:"required"`
	TemplateType    string          `json:"template_type" binding:"required,oneof=sales profit quantity mixed"`
	CalculateMethod string          `json:"calculate_method" binding:"required,oneof=percentage fixed tiered"`
	Description     string          `json:"description"`
	ApplicableRole  string          `json:"applicable_role"`
	EffectiveFrom   time.Time       `json:"effective_from" binding:"required"`
	EffectiveTo     *time.Time      `json:"effective_to"`
	Rules           []ReqCreateRule `json:"rules" binding:"required,min=1"`
}

// ReqCreateRule 创建规则请求
type ReqCreateRule struct {
	RuleName       string          `json:"rule_name" binding:"required"`
	RuleType       string          `json:"rule_type" binding:"required"`
	MinAmount      decimal.Decimal `json:"min_amount"`
	MaxAmount      decimal.Decimal `json:"max_amount"`
	CommissionRate decimal.Decimal `json:"commission_rate"`
	FixedAmount    decimal.Decimal `json:"fixed_amount"`
	BonusAmount    decimal.Decimal `json:"bonus_amount"`
}

// CreateTemplate 创建佣金模板
func (b *CommissionBiz) CreateTemplate(db *gorm.DB, req *ReqCreateTemplate, creatorID uint, creatorName string) (*model.CommissionTemplate, error) {
	return db.Transaction(func(tx *gorm.DB) (*model.CommissionTemplate, error) {
		// 生成模板代码
		templateCode := fmt.Sprintf("TPL%s%04d", time.Now().Format("200601"), rand.Intn(10000))
		
		// 创建模板
		template := &model.CommissionTemplate{
			TemplateName:    req.TemplateName,
			TemplateCode:    templateCode,
			TemplateType:    req.TemplateType,
			CalculateMethod: req.CalculateMethod,
			Description:     req.Description,
			ApplicableRole:  req.ApplicableRole,
			EffectiveFrom:   req.EffectiveFrom,
			EffectiveTo:     req.EffectiveTo,
			Status:          "draft",
			CreatedBy:       creatorID,
			CreatedByName:   creatorName,
		}
		
		if err := tx.Create(template).Error; err != nil {
			return nil, fmt.Errorf("创建模板失败: %w", err)
		}
		
		// 创建规则
		for i, ruleReq := range req.Rules {
			rule := &model.CommissionRule{
				TemplateID:     template.ID,
				TemplateCode:   template.TemplateCode,
				RuleName:       ruleReq.RuleName,
				RuleType:       ruleReq.RuleType,
				Priority:       i + 1,
				MinAmount:      ruleReq.MinAmount,
				MaxAmount:      ruleReq.MaxAmount,
				CommissionRate: ruleReq.CommissionRate,
				FixedAmount:    ruleReq.FixedAmount,
				BonusAmount:    ruleReq.BonusAmount,
				Status:         "active",
			}
			
			if err := tx.Create(rule).Error; err != nil {
				return nil, fmt.Errorf("创建规则失败: %w", err)
			}
		}
		
		// 创建日志
		log := &model.CommissionLog{
			RelatedType:   "template",
			RelatedID:     template.ID,
			OperationType: "create",
			OperationDesc: fmt.Sprintf("创建佣金模板: %s", template.TemplateName),
			OperatorID:    creatorID,
			OperatorName:  creatorName,
		}
		tx.Create(log)
		
		return template, nil
	})
}
```

### 2.2 激活模板

```go
// ActivateTemplate 激活模板
func (b *CommissionBiz) ActivateTemplate(db *gorm.DB, templateID uint, operatorID uint, operatorName string) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var template model.CommissionTemplate
		if err := tx.First(&template, templateID).Error; err != nil {
			return fmt.Errorf("模板不存在: %w", err)
		}
		
		// 验证规则完整性
		var ruleCount int64
		tx.Model(&model.CommissionRule{}).Where("template_id = ? AND status = ?", templateID, "active").Count(&ruleCount)
		if ruleCount == 0 {
			return errors.New("模板没有有效规则，无法激活")
		}
		
		// 更新状态
		if err := tx.Model(&template).Update("status", "active").Error; err != nil {
			return fmt.Errorf("激活模板失败: %w", err)
		}
		
		// 创建日志
		log := &model.CommissionLog{
			RelatedType:   "template",
			RelatedID:     templateID,
			OperationType: "activate",
			OperationDesc: "激活佣金模板",
			OperatorID:    operatorID,
			OperatorName:  operatorName,
		}
		tx.Create(log)
		
		return nil
	})
}
```

### 2.3 模板查询

```go
// ListTemplates 查询模板列表
func (b *CommissionBiz) ListTemplates(db *gorm.DB, req *ReqListTemplates) (*RespListTemplates, error) {
	var templates []model.CommissionTemplate
	var total int64
	
	query := db.Model(&model.CommissionTemplate{})
	
	// 筛选条件
	if req.Status != "" {
		query = query.Where("status = ?", req.Status)
	}
	if req.TemplateType != "" {
		query = query.Where("template_type = ?", req.TemplateType)
	}
	if req.ApplicableRole != "" {
		query = query.Where("applicable_role = ?", req.ApplicableRole)
	}
	
	// 总数
	query.Count(&total)
	
	// 分页
	offset := (req.Page - 1) * req.PageSize
	query.Offset(offset).Limit(req.PageSize).
		Preload("Rules").
		Order("created_at DESC").
		Find(&templates)
	
	return &RespListTemplates{
		Total: total,
		Data:  templates,
	}, nil
}
```
