```go
    }
    
    err := shipment.Validate()
    assert.NoError(t, err)

func TestShipment_CanCancel(t *testing.T) {
    tests := []struct {
        status  string
        canCancel bool
    }{
        {"pending", true},
        {"shipped", true},
        {"delivered", false},
    }
    
    for _, tt := range tests {
        shipment := &Shipment{Status: tt.status}
        assert.Equal(t, tt.canCancel, shipment.CanCancel())
    }
}
```
