#### 3.1 API客户端配置

```typescript
// src/api/client.ts
import axios from 'axios';
import AsyncStorage from '@react-native-async-storage/async-storage';

const API_BASE_URL = 'http://your-api-server.com/api/v1';

const apiClient = axios.create({
  baseURL: API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器 - 添加Token
apiClient.interceptors.request.use(
  async (config) => {
    const token = await AsyncStorage.getItem('access_token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// 响应拦截器 - 处理错误
apiClient.interceptors.response.use(
  (response) => response.data,
  (error) => {
    if (error.response?.status === 401) {
      // Token过期，清除本地存储并跳转登录
      AsyncStorage.clear();
    }
    return Promise.reject(error);
  }
);

export default apiClient;
```

#### 3.2 认证API封装

```typescript
// src/api/auth.ts
import apiClient from './client';

export interface QRCodeScanRequest {
  qrcode_id: string;
}

export interface QRCodeScanResponse {
  user_info: {
    user_id: number;
    username: string;
    tenant_id: number;
    tenant_code: string;
    tenant_name: string;
  };
}

export interface QRCodeConfirmRequest {
  qrcode_id: string;
  action: 'confirm' | 'cancel';
}

export interface QRCodeConfirmResponse {
  message: string;
}

export const authApi = {
  /**
   * 扫描二维码
   */
  scanQRCode: (qrcodeId: string) =>
    apiClient.post<QRCodeScanResponse>('/auth/qrcode/scan', {
      qrcode_id: qrcodeId,
    }),

  /**
   * 确认/取消登录
   */
  confirmQRCode: (qrcodeId: string, action: 'confirm' | 'cancel') =>
    apiClient.post<QRCodeConfirmResponse>('/auth/qrcode/confirm', {
      qrcode_id: qrcodeId,
      action,
    }),
};
```

#### 3.3 扫码页面组件

```typescript
// src/screens/QRCodeScan/index.tsx
import React, { useState, useEffect } from 'react';
import { View, Text, StyleSheet, Alert } from 'react-native';
import { Camera, useCameraDevices } from 'react-native-vision-camera';
import { useNavigation } from '@react-navigation/native';
import { authApi } from '../../api/auth';

const QRCodeScanScreen: React.FC = () => {
  const [hasPermission, setHasPermission] = useState(false);
  const devices = useCameraDevices();
  const device = devices.back;
  const navigation = useNavigation();

  useEffect(() => {
    // 请求摄像头权限
    (async () => {
      const status = await Camera.requestCameraPermission();
      setHasPermission(status === 'authorized');
    })();
  }, []);

  const handleBarCodeScanned = async ({ data }: { data: string }) => {
    try {
      // 解析二维码URL，提取qrcode_id
      // 格式：paas://login?qrcode_id={id}
      const url = new URL(data);
      const qrcodeId = url.searchParams.get('qrcode_id');

      if (!qrcodeId) {
        Alert.alert('错误', '无效的二维码');
        return;
      }

      // 调用扫码API
      const response = await authApi.scanQRCode(qrcodeId);

      // 导航到确认页面
      navigation.navigate('ConfirmLogin', {
        qrcodeId,
        userInfo: response.user_info,
      });
    } catch (error: any) {
      Alert.alert('扫码失败', error.message || '请重试');
    }
  };

  if (!hasPermission) {
    return (
      <View style={styles.container}>
        <Text>需要摄像头权限</Text>
      </View>
    );
  }

  if (!device) {
    return (
      <View style={styles.container}>
        <Text>未找到摄像头</Text>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      <Camera
        style={StyleSheet.absoluteFill}
        device={device}
        isActive={true}
        frameProcessor={(frame) => {
          'worklet';
          // 这里使用vision-camera的插件进行二维码识别
          // 实际项目中需要配置vision-camera-code-scanner
        }}
      />
      
      {/* 扫描框 */}
      <View style={styles.scanFrame}>
        <Text style={styles.hint}>将二维码放入框内</Text>
      </View>
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: '#000',
  },
  scanFrame: {
    position: 'absolute',
    top: '30%',
    left: '20%',
    width: '60%',
    height: '30%',
    borderWidth: 2,
    borderColor: '#fff',
    borderRadius: 12,
    justifyContent: 'flex-end',
    padding: 16,
  },
  hint: {
    color: '#fff',
    textAlign: 'center',
    fontSize: 14,
  },
});

export default QRCodeScanScreen;
```

// src/screens/ConfirmLogin/index.tsx
import React, { useState } from 'react';
import { View, Text, StyleSheet, TouchableOpacity, Alert } from 'react-native';
import { useRoute, useNavigation } from '@react-navigation/native';
import { authApi } from '../../api/auth';

const ConfirmLoginScreen: React.FC = () => {
  const route = useRoute();
  const navigation = useNavigation();
  const { qrcodeId, userInfo } = route.params as any;
  const [loading, setLoading] = useState(false);

  const handleConfirm = async () => {
    setLoading(true);
    try {
      await authApi.confirmQRCode(qrcodeId, 'confirm');
      
      Alert.alert('成功', '登录确认成功', [
        {
          text: '确定',
          onPress: () => navigation.goBack(),
        },
      ]);
    } catch (error: any) {
      Alert.alert('错误', error.message || '操作失败');
    } finally {
      setLoading(false);
    }
  };

  const handleCancel = async () => {
    setLoading(true);
    try {
      await authApi.confirmQRCode(qrcodeId, 'cancel');
      
      Alert.alert('已取消', '登录已取消', [
        {
          text: '确定',
          onPress: () => navigation.goBack(),
        },
      ]);
    } catch (error: any) {
      Alert.alert('错误', error.message || '操作失败');
    } finally {
      setLoading(false);
    }
  };

  return (
    <View style={styles.container}>
      <View style={styles.header}>
        <Text style={styles.title}>✓ 扫码成功</Text>
      </View>

      <View style={styles.userInfoCard}>
        <Text style={styles.username}>{userInfo.username}</Text>
        <Text style={styles.role}>平台管理员</Text>

};
