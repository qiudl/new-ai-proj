# 换货功能设计文档

## 1. 概述

本文档定义Coolbuy 3.0售后模块的换货功能，包括换货申请、审批、物流、换货单生成等业务逻辑。

### 1.1 换货场景

- **尺码换货**: 尺码不合适，换其他尺码
- **颜色换货**: 颜色不满意，换其他颜色
- **款式换货**: 换成同类型其他款式
- **质量换货**: 质量问题换新品

### 1.2 换货流程

```
客户申请换货 → 审批 → 客户寄回 → 仓库收货质检 → 发出新货 → 客户签收 → 完成
```

## 2. 核心功能设计

### 2.1 创建换货申请

**接口定义**:

```go
// ReqCreateExchange 创建换货申请请求
type ReqCreateExchange struct {

}

// ReqExchangeItem 换货商品明细
type ReqExchangeItem struct {
	OrderItemID      uint `json:"order_item_id" binding:"required"`      // 原订单明细ID
	Quantity         int  `json:"quantity" binding:"required,min=1"`     // 换货数量
	ExchangeSkuID    uint `json:"exchange_sku_id" binding:"required"`    // 换货SKU ID
	ExchangeQuantity int  `json:"exchange_quantity" binding:"required,min=1"` // 换货数量
```

**业务逻辑**:
```go
// CreateExchange 创建换货申请
func (b *AfterSalesBiz) CreateExchange(db *gorm.DB, req *ReqCreateExchange, customerID uint) (*RespCreateAfterSales, error) {
	var resp *RespCreateAfterSales
	
	err := db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询订单信息
		order, err := b.orderStore.GetByID(tx, req.OrderID)
		if err != nil {
			return fmt.Errorf("订单不存在: %w", err)
		}
		
		// 2. 验证订单所属和状态
		if order.CustomerID != customerID {
			return errors.New("无权操作该订单")
		}
		if order.Status != "completed" && order.Status != "shipped" {
			return errors.New("订单状态不允许申请换货")
		}
		
		// 3. 验证换货商品和库存
		var totalAmount decimal.Decimal
		var afterSalesItems []model.AfterSalesItem
		
		for _, itemReq := range req.Items {
			// 查询原订单明细
			orderItem, err := b.orderStore.GetOrderItem(tx, itemReq.OrderItemID)
			if err != nil {
				return fmt.Errorf("订单明细不存在: %w", err)
			}
			
			// 验证数量
			if itemReq.Quantity > orderItem.Quantity - orderItem.ReturnedQty {
				return fmt.Errorf("SKU %s 换货数量超出可换货数量", orderItem.SkuCode)
			}
			
			// 查询换货SKU信息
			exchangeSku, err := b.skuStore.GetByID(tx, itemReq.ExchangeSkuID)
			if err != nil {
				return fmt.Errorf("换货SKU不存在: %w", err)
			}
			
			// 验证换货SKU库存
			inventory, err := b.inventoryStore.GetBySkuAndWarehouse(tx, itemReq.ExchangeSkuID, 1) // 默认仓库
			if inventory.AvailableQuantity < itemReq.ExchangeQuantity {
				return fmt.Errorf("换货SKU %s 库存不足", exchangeSku.Code)
			}
			
			// 计算价差（如果有）
			priceDiff := exchangeSku.MarketPrice.Sub(orderItem.UnitPrice).Mul(decimal.NewFromInt(int64(itemReq.ExchangeQuantity)))
			totalAmount = totalAmount.Add(priceDiff)
			
			// 构建售后明细
			afterSalesItems = append(afterSalesItems, model.AfterSalesItem{
				OrderID:          req.OrderID,
				OrderItemID:      itemReq.OrderItemID,
				SpuID:            orderItem.SpuID,
				SkuID:            orderItem.SkuID,
				SkuCode:          orderItem.SkuCode,
				SkuName:          orderItem.SkuName,
				SpuName:          orderItem.SpuName,
				Size:             orderItem.Size,
				Color:            orderItem.Color,
				Quantity:         itemReq.Quantity,
				UnitPrice:        orderItem.UnitPrice,
				TotalPrice:       orderItem.UnitPrice.Mul(decimal.NewFromInt(int64(itemReq.Quantity))),
				RefundAmount:     decimal.Zero,
				ExchangeSkuID:    &itemReq.ExchangeSkuID,
				ExchangeSkuCode:  exchangeSku.Code,
				ExchangeSkuName:  exchangeSku.Name,
				ExchangeQuantity: itemReq.ExchangeQuantity,
				ImageURL:         orderItem.ImageURL,
			})
		}
		
		// 4. 生成售后单号
		afterSalesNo := b.generateAfterSalesNo()
		
		// 5. 创建换货售后单
		afterSales := &model.AfterSales{
			AfterSalesNo:   afterSalesNo,
			AfterSalesType: "exchange",
			OrderID:        req.OrderID,
			OrderNo:        order.OrderNo,
			CustomerID:     customerID,
			CustomerName:   order.CustomerName,
			CustomerPhone:  req.ContactPhone,
			ReasonType:     req.ReasonType,
			ReasonDetail:   req.ReasonDetail,
			TotalAmount:    totalAmount, // 价差
			RefundAmount:   decimal.Zero,
			ActualRefundAmount: decimal.Zero,
			Status:         "pending",
			ApprovalStatus: "pending",
			RefundStatus:   "none",
			AppliedAt:      time.Now(),
		}
		
		if err := tx.Create(afterSales).Error; err != nil {
			return fmt.Errorf("创建售后单失败: %w", err)
		}
		
		// 6. 创建售后明细
		for i := range afterSalesItems {
			afterSalesItems[i].AfterSalesID = afterSales.ID
			afterSalesItems[i].AfterSalesNo = afterSales.AfterSalesNo
		}
		if err := tx.Create(&afterSalesItems).Error; err != nil {
			return fmt.Errorf("创建售后明细失败: %w", err)
		}
		
		// 7. 保存凭证图片
		if len(req.Images) > 0 {
			var images []model.AfterSalesImage
			for i, imageURL := range req.Images {
				images = append(images, model.AfterSalesImage{
					AfterSalesID: afterSales.ID,
					AfterSalesNo: afterSales.AfterSalesNo,
					ImageType:    "quality_issue",
					ImageURL:     imageURL,
					SortOrder:    i + 1,
				})
			}
			if err := tx.Create(&images).Error; err != nil {
				return fmt.Errorf("保存凭证图片失败: %w", err)
			}
		}
		
		// 8. 创建操作日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSales.ID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: "create",
			OperationDesc: "客户提交换货申请",
			OperatorID:    customerID,
			OperatorName:  order.CustomerName,
			OperatorType:  "customer",
			AfterStatus:   "pending",
		}
		if err := tx.Create(log).Error; err != nil {
			return fmt.Errorf("创建操作日志失败: %w", err)
		}
		
		// 9. 更新订单售后状态
		if err := tx.Model(&model.Order{}).Where("id = ?", req.OrderID).
			Update("after_sales_status", "in_progress").Error; err != nil {
			return fmt.Errorf("更新订单售后状态失败: %w", err)
		}
		
		resp = &RespCreateAfterSales{
			AfterSalesID: afterSales.ID,
			AfterSalesNo: afterSales.AfterSalesNo,
			Status:       afterSales.Status,
		}
		
		return nil
	})
	
	return resp, err
}
```

### 2.2 换货收货与质检

**业务逻辑**:
```go
// ReceiveExchangeReturn 换货收货与质检
func (b *AfterSalesBiz) ReceiveExchangeReturn(db *gorm.DB, afterSalesID uint, req *ReqReceiveReturn, receiverID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询售后单（加锁）
		var afterSales model.AfterSales
		if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
			Preload("Items").
			First(&afterSales, afterSalesID).Error; err != nil {
			return fmt.Errorf("售后单不存在: %w", err)
		}
		
		// 2. 验证是换货类型
		if afterSales.AfterSalesType != "exchange" {
			return errors.New("非换货类型售后单")
		}
		if afterSales.Status != "returning" {
			return errors.New("售后单状态不允许收货")
		}
		
		now := time.Now()
		allPassed := true
		
		// 3. 更新质检结果
		for _, itemReq := range req.Items {
			updates := map[string]interface{}{
				"quality_check_status": itemReq.QualityCheckStatus,
				"quality_check_remark": itemReq.QualityCheckRemark,
				"quality_checker_id":   receiverID,
				"quality_checked_at":   now,
			}
			
			if err := tx.Model(&model.AfterSalesItem{}).
				Where("id = ? AND after_sales_id = ?", itemReq.AfterSalesItemID, afterSalesID).
				Updates(updates).Error; err != nil {
				return fmt.Errorf("更新质检结果失败: %w", err)
			}
			
			if itemReq.QualityCheckStatus == "failed" {
				allPassed = false
			}
		}
		
		// 4. 更新售后单状态
		updates := map[string]interface{}{
			"return_received_at": now,
			"status":             "returned",
		}
		
		if err := tx.Model(&afterSales).Updates(updates).Error; err != nil {
			return fmt.Errorf("更新售后单失败: %w", err)
		}
		
		// 5. 如果质检通过，创建换货新订单
		if allPassed {
			exchangeOrder, err := b.createExchangeOrder(tx, &afterSales)
			if err != nil {
				return fmt.Errorf("创建换货订单失败: %w", err)
			}
			
			// 更新售后单的换货订单信息
			tx.Model(&afterSales).Updates(map[string]interface{}{
				"exchange_order_id": exchangeOrder.ID,
				"exchange_order_no": exchangeOrder.OrderNo,
			})
		}
		
		// 6. 创建操作日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSalesID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: "receive_return",
			OperationDesc: fmt.Sprintf("换货商品已收货，质检结果: %s", map[bool]string{true: "全部通过", false: "部分不通过"}[allPassed]),
			OperatorID:    receiverID,
			OperatorName:  req.ReceivedBy,
			OperatorType:  "staff",
			BeforeStatus:  "returning",
			AfterStatus:   "returned",
		}
		if err := tx.Create(log).Error; err != nil {
			return fmt.Errorf("创建操作日志失败: %w", err)
		}
		
		return nil
	})
}

// createExchangeOrder 创建换货订单
func (b *AfterSalesBiz) createExchangeOrder(tx *gorm.DB, afterSales *model.AfterSales) (*model.Order, error) {
	// 1. 查询原订单
	originalOrder, err := b.orderStore.GetByID(tx, afterSales.OrderID)
	if err != nil {
		return nil, fmt.Errorf("查询原订单失败: %w", err)
	}
	
	// 2. 生成新订单号
	newOrderNo := fmt.Sprintf("EX%s%08d", time.Now().Format("20060102"), time.Now().UnixNano()%100000000)
	
	// 3. 创建换货订单
	exchangeOrder := &model.Order{
		OrderNo:      newOrderNo,
		OrderType:    "exchange",
		CustomerID:   afterSales.CustomerID,
		CustomerName: afterSales.CustomerName,
		CustomerPhone: afterSales.CustomerPhone,
		ReceiverName: originalOrder.ReceiverName,
		ReceiverPhone: originalOrder.ReceiverPhone,
		ReceiverAddress: originalOrder.ReceiverAddress,
		TotalAmount:  decimal.Zero, // 换货订单金额为0（或价差）
		PaidAmount:   decimal.Zero,
		PaymentMethod: "exchange",
		PaymentStatus: "paid",
		Status:       "paid",
		CreatedAt:    time.Now(),
	}
	
	if err := tx.Create(exchangeOrder).Error; err != nil {
		return nil, fmt.Errorf("创建换货订单失败: %w", err)
	}
	
	// 4. 创建订单明细（换货新商品）
	for _, item := range afterSales.Items {
		if item.ExchangeSkuID == nil {
			continue
		}
		
		exchangeSku, _ := b.skuStore.GetByID(tx, *item.ExchangeSkuID)
		
		orderItem := &model.OrderItem{
			OrderID:   exchangeOrder.ID,
			OrderNo:   exchangeOrder.OrderNo,
			SpuID:     exchangeSku.SpuID,
			SkuID:     *item.ExchangeSkuID,
			SkuCode:   item.ExchangeSkuCode,
			SkuName:   item.ExchangeSkuName,
			Quantity:  item.ExchangeQuantity,
			UnitPrice: exchangeSku.MarketPrice,
			TotalPrice: exchangeSku.MarketPrice.Mul(decimal.NewFromInt(int64(item.ExchangeQuantity))),
		}
		
		if err := tx.Create(orderItem).Error; err != nil {
			return nil, fmt.Errorf("创建订单明细失败: %w", err)
		}
		
		// 扣减库存
		if err := b.inventoryStore.DeductInventory(tx, *item.ExchangeSkuID, 1, item.ExchangeQuantity); err != nil {
			return nil, fmt.Errorf("扣减库存失败: %w", err)
		}
	}
	
	// 5. 自动创建发货单
	// ... 省略发货逻辑
	
	return exchangeOrder, nil
}
```

```go
// CompleteExchange 完成换货（新货已签收）
func (b *AfterSalesBiz) CompleteExchange(db *gorm.DB, afterSalesID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		// 1. 查询售后单
		var afterSales model.AfterSales
		if err := tx.First(&afterSales, afterSalesID).Error; err != nil {
			return fmt.Errorf("售后单不存在: %w", err)
		}
		
		// 2. 验证类型和状态
		if afterSales.AfterSalesType != "exchange" {
			return errors.New("非换货类型售后单")
		}
		
		now := time.Now()
		
		// 3. 更新售后单状态
		updates := map[string]interface{}{
			"status":       "completed",
			"completed_at": now,
		}
		if err := tx.Model(&afterSales).Updates(updates).Error; err != nil {
			return fmt.Errorf("更新售后单失败: %w", err)
		}
		
		// 4. 更新换货订单状态
		if afterSales.ExchangeOrderID != nil {
			tx.Model(&model.Order{}).Where("id = ?", *afterSales.ExchangeOrderID).
				Update("status", "completed")
		}
		
		// 5. 更新原订单售后状态
		tx.Model(&model.Order{}).Where("id = ?", afterSales.OrderID).
			Update("after_sales_status", "exchanged")
		
		// 6. 创建操作日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSalesID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: "complete",
			OperationDesc: "换货已完成",
			OperatorType:  "system",
			BeforeStatus:  "returned",
			AfterStatus:   "completed",
		}
		if err := tx.Create(log).Error; err != nil {
			return fmt.Errorf("创建操作日志失败: %w", err)
		}
		
		return nil
	})
}
```

## 3. 换货与退货的区别

| 特性 | 退货退款 | 换货 |
|-----|---------|------|
| 退款 | 需要退款 | 不退款（除非有价差） |
| 库存 | 恢复原SKU库存 | 不恢复原SKU库存，扣减新SKU库存 |
| 新订单 | 无 | 创建换货订单 |
| 物流 | 单程（客户→仓库） | 双程（客户→仓库→客户） |
| 完成条件 | 退款成功 | 新货签收 |

## 4. 换货状态流转

```
pending (待审核)
    ├─ approve → approved (已通过)
    │              └─ returning (客户寄回)
    │                  └─ returned (仓库收货)
    │                      └─ quality_check (质检)
    │                          ├─ passed → create_exchange_order (创建换货订单)
    │                          │             └─ shipping (发出新货)
    │                          │                 └─ delivered (客户签收)
    │                          │                     └─ completed (完成)
    │                          └─ failed → approval_pending
    ├─ reject → rejected (已拒绝)
    └─ cancel → cancelled (已取消)
```

## 5. 总结

本文档定义了换货功能的核心业务逻辑，包括：

- ✅ **换货申请**: 验证商品和库存
- ✅ **换货收货**: 质检流程
- ✅ **换货订单**: 自动创建新订单
- ✅ **库存管理**: 不恢复原SKU，扣减新SKU
- ✅ **双程物流**: 退货物流 + 发货物流
- ✅ **状态管理**: 完整的状态流转

换货与退货的主要区别在于需要创建新订单和发出新货，业务流程更复杂。
