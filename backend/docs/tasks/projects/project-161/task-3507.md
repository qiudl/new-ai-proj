package model

func TestAfterSales_CanApprove(t *testing.T) {
	tests := []struct {
		name   string
		status string
		approvalStatus string
		want   bool
	}{
		{"待审核状态可以审批", "pending", "pending", true},
		{"已审批状态不能审批", "approved", "approved", false},
		{"已拒绝状态不能审批", "rejected", "rejected", false},

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			as := &AfterSales{
				Status:         tt.status,
				ApprovalStatus: tt.approvalStatus,
			}
			assert.Equal(t, tt.want, as.CanApprove())
		})
	}
}

func TestAfterSales_CanRefund(t *testing.T) {
	as := &AfterSales{
		ApprovalStatus: "approved",
		RefundStatus:   "pending",
	}
	assert.True(t, as.CanRefund())
	
	as.RefundStatus = "success"
	assert.False(t, as.CanRefund())
}

## 3. 集成测试

### 3.1 完整退货退款流程测试

```go
// after_sales_flow_test.go
func TestAfterSalesFlow_ReturnRefund(t *testing.T) {
	db := setupTestDB(t)
	defer cleanupTestDB(t, db)
	
	biz := NewAfterSalesBiz()
	
	// 1. 创建订单
	order := createTestOrder(t, db)
	order.Status = "completed"
	db.Save(order)
	
	// 2. 创建售后申请
	afterSales := createTestAfterSales(t, db, order.ID)
	assert.Equal(t, "pending", afterSales.Status)
	
	// 3. 审批通过
	approveReq := &ReqApproveAfterSales{
		Action: "approve",
		ApprovalRemark: "同意",
	}
	err := biz.Approve(db, afterSales.ID, approveReq, 100, "审批员")
	assert.NoError(t, err)
	
	// 4. 客户提交物流信息
	logisticsReq := &ReqSubmitReturnLogistics{
		LogisticsProvider: "顺丰速运",
		TrackingNo:        "SF1234567890",
		SenderName:        "张三",
		SenderPhone:       "13800138000",
		SenderAddress:     "北京市朝阳区xxx",
	}
	err = biz.SubmitReturnLogistics(db, afterSales.ID, logisticsReq, order.CustomerID)
	assert.NoError(t, err)
	
	// 验证状态
	db.First(&afterSales, afterSales.ID)
	assert.Equal(t, "returning", afterSales.Status)
	
	// 5. 仓库收货质检
	receiveReq := &ReqReceiveReturn{
		ReceivedBy: "仓管员",
		Items: []ReqReturnQualityCheck{
			{
				AfterSalesItemID:   afterSales.Items[0].ID,
				QualityCheckStatus: "passed",
			},
		},
	}
	err = biz.ReceiveReturn(db, afterSales.ID, receiveReq, 200)
	assert.NoError(t, err)
	
	// 验证状态和库存
	db.First(&afterSales, afterSales.ID)
	assert.Equal(t, "returned", afterSales.Status)
	
	// 6. 退款
	refundReq := &ReqProcessRefund{
		RefundMethod:    "original",
		RefundAmount:    afterSales.ActualRefundAmount,
		PaymentPlatform: "alipay",
	}
	err = biz.ProcessRefund(db, afterSales.ID, refundReq, 300, "财务员")
	assert.NoError(t, err)
	
	// 验证最终状态
	db.First(&afterSales, afterSales.ID)
	assert.Equal(t, "completed", afterSales.Status)
	assert.Equal(t, "success", afterSales.RefundStatus)
	assert.NotNil(t, afterSales.CompletedAt)
}
```

### 5.2 并发测试

```go
func TestConcurrentApproval(t *testing.T) {
	db := setupTestDB(t)
	defer cleanupTestDB(t, db)
	
	biz := NewAfterSalesBiz()
	afterSales := createTestAfterSales(t, db)
	
	// 模拟多个审批人同时审批
	var wg sync.WaitGroup
	errors := make(chan error, 3)
	
	for i := 0; i < 3; i++ {
		wg.Add(1)
		go func(approverID int) {
			defer wg.Done()
			
			req := &ReqApproveAfterSales{
				Action: "approve",
				ApprovalRemark: fmt.Sprintf("审批人%d", approverID),
			}
			
			err := biz.Approve(db, afterSales.ID, req, uint(approverID), fmt.Sprintf("审批员%d", approverID))
			if err != nil {
				errors <- err
			}
		}(i)
	}
	
	wg.Wait()
	close(errors)
	
	// 应该只有一个审批成功，其他两个报错
	errorCount := 0
	for err := range errors {
		if err != nil {
			errorCount++
		}
	}
	
	assert.Equal(t, 2, errorCount, "应该有2个并发审批失败")
}
```
