func TestPriceSyncBiz_ApplyStrategy(t *testing.T) {
	biz := NewPriceSyncBiz()
	
	t.Run("固定价格策略", func(t *testing.T) {
		strategy := &PriceStrategy{
			StrategyType: "fixed",
			FixedPrice:   decimal.NewFromFloat(199.00),
		}
		
		basePrice := decimal.NewFromFloat(299.00)
		finalPrice, err := biz.ApplyStrategy(basePrice, strategy)
		
		assert.NoError(t, err)
		assert.Equal(t, "199.00", finalPrice.String())
	})
	
	t.Run("百分比加价策略", func(t *testing.T) {
		strategy := &PriceStrategy{
			StrategyType: "markup",
			MarkupType:   "percentage",
			MarkupValue:  decimal.NewFromFloat(10.00), // 加价10%
		}
		
		basePrice := decimal.NewFromFloat(100.00)
		finalPrice, err := biz.ApplyStrategy(basePrice, strategy)
		
		assert.NoError(t, err)
		assert.Equal(t, "110.00", finalPrice.String())
	})
	
	t.Run("价格取整到x.9", func(t *testing.T) {
		strategy := &PriceStrategy{
			StrategyType: "markup",
			MarkupType:   "amount",
			MarkupValue:  decimal.NewFromFloat(20.00),
			RoundingType: "unit",
			RoundingUnit: decimal.NewFromFloat(0.90),
		}
		
		basePrice := decimal.NewFromFloat(100.00)
		finalPrice, err := biz.ApplyStrategy(basePrice, strategy)
		
		assert.NoError(t, err)
		assert.Equal(t, "120.90", finalPrice.String())
	})

### 5.1 并发推送测试

```go
func TestConcurrentInventoryPush(t *testing.T) {
	db := setupTestDB()
	defer teardownTestDB(db)
	
	// 创建100个商品映射
	for i := 1; i <= 100; i++ {
		createTestMapping(db, uint(i), fmt.Sprintf("ext_sku_%03d", i), uint(i))
	}
	
	biz := NewInventorySyncBiz()
	
	// 并发推送
	concurrency := 10
	mappingIDs := make([]uint, 100)
	for i := range mappingIDs {
		mappingIDs[i] = uint(i + 1)
	}
	
	start := time.Now()
	err := biz.ConcurrentSync(db, mappingIDs, concurrency)
	duration := time.Since(start)
	
	assert.NoError(t, err)
	t.Logf("并发推送100个商品耗时: %v", duration)
	
	// 验证所有推送成功
	var successCount int64
	db.Model(&model.InventorySyncRecord{}).
		Where("sync_status = ?", "success").
		Count(&successCount)
	
	assert.Equal(t, int64(100), successCount)
}
```
