```go
// TestInboundFlow_Integration 测试完整入库流程
func TestInboundFlow_Integration(t *testing.T) {
    db := setupTestDB()
    defer cleanupTestDB(db)
    
    // 1. 创建采购入库单
    inboundBiz := NewInboundBiz()
    req := &ReqCreatePurchaseInbound{
        WarehouseID:       1,
        SupplierID:        10,
        PurchaseOrderNo:   "PO-001",
        RequireInspection: true,
        Items: []PurchaseInboundItem{
            {SkuID: 1, PlannedQty: 100, UnitPrice: 50.00},
        },
    }
    
    resp, err := inboundBiz.CreatePurchaseInbound(db, req, 1)
    assert.NoError(t, err)
    assert.NotNil(t, resp)
    
    orderID := resp.ID
    
    // 2. 开始收货
    err = inboundBiz.StartReceiving(db, orderID, &ReqStartReceiving{ReceiverID: 2})
    assert.NoError(t, err)
    
    // 3. 记录收货数量
    receiveResp, err := inboundBiz.ReceiveGoods(db, orderID, &ReqReceiveGoods{
        Items: []ReceiveItem{{ItemID: 1, ReceivedQty: 100}},
    }, 2)
    assert.NoError(t, err)
    assert.True(t, receiveResp.IsComplete)
    
    // 4. 质检
    inspectResp, err := inboundBiz.InspectGoods(db, orderID, &ReqInspectGoods{
        Items: []InspectItem{{ItemID: 1, QualifiedQty: 95, UnqualifiedQty: 5, QualityGrade: "A"}},
    }, 3)
    assert.NoError(t, err)
    assert.Equal(t, 95, inspectResp.QualifiedQty)
    
    // 5. 上架
    err = inboundBiz.AssignLocations(db, orderID, &ReqAssignLocations{
        Items: []AssignLocationItem{{ItemID: 1, LocationID: 1}},
    }, 4)
    assert.NoError(t, err)
    
    shelveResp, err := inboundBiz.ShelveGoods(db, orderID, &ReqShelveGoods{
        Items: []ShelveItem{{ItemID: 1, ShelvedQty: 95}},
    }, 4)
    assert.NoError(t, err)
    assert.True(t, shelveResp.IsComplete)
    
    // 6. 验证库存已增加
    var inventory Inventory
    db.Where("sku_id = ? AND warehouse_id = ?", 1, 1).First(&inventory)
    assert.Equal(t, 95, inventory.Quantity)
    assert.Equal(t, 95, inventory.AvailableQuantity)
}
```

### 4.2 并发测试

```go
// TestInventoryReservation_Concurrency 测试并发库存预留
func TestInventoryReservation_Concurrency(t *testing.T) {
    db := setupTestDB()
    defer cleanupTestDB(db)
    
    // 创建库存：总量100
    inventory := &Inventory{
        WarehouseID:       1,
        SkuID:             1,
        Quantity:          100,
        AvailableQuantity: 100,
    }
    db.Create(inventory)
    
    // 并发预留：10个goroutine，每个预留20件（总需求200件，应该只有5个成功）
    concurrency := 10
    reserveQty := 20
    
    var wg sync.WaitGroup
    successCount := int32(0)
    failureCount := int32(0)
    
    for i := 0; i < concurrency; i++ {
        wg.Add(1)
        go func(index int) {
            defer wg.Done()
            
            // 创建出库单
            outbound := &OutboundOrder{
                OrderNumber:   fmt.Sprintf("OB-%03d", index),
                WarehouseID:   1,
                Status:        "pending",
                TotalQuantity: reserveQty,
            }
            db.Create(outbound)
            
            item := &OutboundOrderItem{
                OutboundOrderID: outbound.ID,
                SkuID:           1,
                PlannedQty:      reserveQty,
            }
            db.Create(item)
            
            // 尝试预留
            biz := NewOutboundBiz()
            resp, err := biz.ReserveInventory(db, outbound.ID, &ReqReserveInventory{}, 1)
            
            if err == nil && resp.IsComplete {
                atomic.AddInt32(&successCount, 1)
            } else {
                atomic.AddInt32(&failureCount, 1)
            }
        }(i)
    }
    
    wg.Wait()
    
    // 验证结果：应该只有5个成功（100 / 20 = 5）
    assert.Equal(t, int32(5), successCount)
    assert.Equal(t, int32(5), failureCount)
    
    // 验证库存：全部预留完毕
    db.First(inventory, inventory.ID)
    assert.Equal(t, 0, inventory.AvailableQuantity)
    assert.Equal(t, 100, inventory.ReservedQuantity)
}
```
