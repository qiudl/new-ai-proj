### 2. 索引优化

#### 2.1 复合索引

```sql
-- 列表查询优化索引（租户 + 状态 + 排序）
CREATE INDEX idx_product_skus_list
ON public.product_skus(tenant_id, status, sort_order DESC, id DESC)
WHERE deleted_at IS NULL;
```

#### 2.2 部分索引（Partial Index）

```sql
-- 在售商品索引
CREATE INDEX idx_product_skus_on_sale
ON public.product_skus(tenant_id, is_on_sale)
WHERE is_on_sale = true AND deleted_at IS NULL;

-- 低库存预警索引
CREATE INDEX idx_product_skus_low_stock
ON public.product_skus(tenant_id, available_stock)
```

### 3. 触发器设计

#### 3.1 库存变动自动记录触发器

```sql
CREATE OR REPLACE FUNCTION public.log_sku_stock_change()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.stock IS DISTINCT FROM NEW.stock THEN
        INSERT INTO public.sku_stock_logs (
            tenant_id, sku_id, sku_code, operation_type,
            quantity_change, stock_before, stock_after,
            operator_id, created_at
        ) VALUES (
            NEW.tenant_id, NEW.id, NEW.sku_code, 'ADJUST',
            NEW.stock - OLD.stock, OLD.stock, NEW.stock,
            NEW.updated_by, CURRENT_TIMESTAMP
        );
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_log_sku_stock_change
AFTER UPDATE ON public.product_skus
FOR EACH ROW
EXECUTE FUNCTION public.log_sku_stock_change();
```
