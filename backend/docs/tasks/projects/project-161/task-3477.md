# 流水记录功能设计文档

## 1. 功能概述

流水记录功能是资金流水模块的核心，负责记录系统中所有的资金变动，提供完整的审计追踪能力。每一笔资金的流入流出都会生成对应的交易流水记录，确保财务数据的完整性和可追溯性。

### 1.1 核心价值

- **完整审计**：记录所有资金变动，支持完整的审计追踪
- **实时查询**：支持多维度实时查询交易流水
- **对账依据**：作为账户对账的核心依据
- **报表基础**：为各类财务报表提供数据支撑
- **风控监控**：实时监控异常交易，支持风险预警

### 1.2 业务场景

| 场景 | 交易类型 | 描述 |
|------|---------|------|
| 订单收款 | order_payment | 客户下单支付产生收入流水 |
| 订单退款 | order_refund | 订单退款产生支出流水 |
| 客户充值 | recharge | 客户账户充值 |
| 提现 | withdraw | 客户账户提现 |
| 采购付款 | purchase_payment | 向供应商支付采购款 |
| 结算收款 | settlement_receive | 收到客户结算款 |
| 结算付款 | settlement_payment | 向供应商支付结算款 |
| 佣金支付 | commission | 支付销售佣金 |
| 账户调整 | adjustment | 财务人员手动调整 |
| 内部转账 | transfer_in/transfer_out | 账户间资金转移 |

```go
func (b *TransactionRecordBiz) List(db *gorm.DB, req *ReqListTransactionRecords) (*RespListTransactionRecords, error) {
    query := db.Model(&TransactionRecord{}).
        Preload("Account")
    
    // 账户筛选
    if req.AccountID > 0 {
        query = query.Where("account_id = ?", req.AccountID)
    }
    if req.AccountNo != "" {
        query = query.Where("account_no = ?", req.AccountNo)
    }
    
    // 交易类型筛选
    if req.TransactionType != "" {
        query = query.Where("transaction_type = ?", req.TransactionType)
    }
    if req.BusinessType != "" {
        query = query.Where("business_type = ?", req.BusinessType)
    }
    if req.Direction != "" {
        query = query.Where("direction = ?", req.Direction)
    }
    
    // 状态筛选
    if req.Status != "" {
        query = query.Where("status = ?", req.Status)
    }
    
    // 关联业务筛选
    if req.RelatedType != "" {
        query = query.Where("related_type = ?", req.RelatedType)
    }
    if req.RelatedID > 0 {
        query = query.Where("related_id = ?", req.RelatedID)
    }
    if req.RelatedNo != "" {
        query = query.Where("related_no = ?", req.RelatedNo)
    }
    
    // 对方筛选
    if req.CounterpartyType != "" {
        query = query.Where("counterparty_type = ?", req.CounterpartyType)
    }
    if req.CounterpartyID > 0 {
        query = query.Where("counterparty_id = ?", req.CounterpartyID)
    }
    if req.CounterpartyName != "" {
        query = query.Where("counterparty_name LIKE ?", "%"+req.CounterpartyName+"%")
    }
    
    // 金额范围
    if !req.MinAmount.IsZero() {
        query = query.Where("amount >= ?", req.MinAmount)
    }
    if !req.MaxAmount.IsZero() {
        query = query.Where("amount <= ?", req.MaxAmount)
    }
    
    // 时间范围
    if req.StartTime != "" {
        startTime, err := time.Parse("2006-01-02 15:04:05", req.StartTime)
        if err == nil {
            query = query.Where("created_at >= ?", startTime)
        }
    }
    if req.EndTime != "" {
        endTime, err := time.Parse("2006-01-02 15:04:05", req.EndTime)
        if err == nil {
            query = query.Where("created_at <= ?", endTime)
        }
    }
    
    // 统计总数
    var total int64
    query.Count(&total)
    
    // 排序
    sortBy := "created_at"
    if req.SortBy != "" {
        sortBy = req.SortBy
    }
    sortOrder := "DESC"
    if req.SortOrder == "asc" {
        sortOrder = "ASC"
    }
    query = query.Order(fmt.Sprintf("%s %s", sortBy, sortOrder))
    
    // 分页
    offset := (req.Page - 1) * req.PageSize
    query = query.Offset(offset).Limit(req.PageSize)
    
    var records []TransactionRecord
    if err := query.Find(&records).Error; err != nil {
        return nil, err
    }
    
    // 转换为VO
    data := make([]TransactionRecordVO, len(records))
    for i, record := range records {
        data[i] = b.toVO(&record)
    }
    
    return &RespListTransactionRecords{
        Data:     data,
        Total:    total,
        Page:     req.Page,
        PageSize: req.PageSize,
    }, nil
}
```

#### 2.4.4 业务逻辑

```go
func (b *TransactionRecordBiz) Statistics(db *gorm.DB, req *ReqTransactionStatistics) (*RespTransactionStatistics, error) {
    query := db.Model(&TransactionRecord{})
    
    // 账户筛选
    if req.AccountID > 0 {
        query = query.Where("account_id = ?", req.AccountID)
    }
    
    // 时间范围
    if req.StartDate != "" {
        query = query.Where("DATE(created_at) >= ?", req.StartDate)
    }
    if req.EndDate != "" {
        query = query.Where("DATE(created_at) <= ?", req.EndDate)
    }
    
    // 总体统计
    var stats struct {
        TotalCount   int
        SuccessCount int
        FailedCount  int
        TotalIncome  decimal.Decimal
        TotalExpense decimal.Decimal
        MaxIncome    decimal.Decimal
        MaxExpense   decimal.Decimal
    }
    
    query.Select(`
        COUNT(*) as total_count,
        SUM(CASE WHEN status = 'success' THEN 1 ELSE 0 END) as success_count,
        SUM(CASE WHEN status = 'failed' THEN 1 ELSE 0 END) as failed_count,
        COALESCE(SUM(CASE WHEN direction = 'income' AND status = 'success' THEN amount ELSE 0 END), 0) as total_income,
        COALESCE(SUM(CASE WHEN direction = 'expense' AND status = 'success' THEN amount ELSE 0 END), 0) as total_expense,
        COALESCE(MAX(CASE WHEN direction = 'income' THEN amount END), 0) as max_income,
        COALESCE(MAX(CASE WHEN direction = 'expense' THEN amount END), 0) as max_expense
    `).Scan(&stats)
    
    resp := &RespTransactionStatistics{
        TotalCount:   stats.TotalCount,
        TotalIncome:  stats.TotalIncome,
        TotalExpense: stats.TotalExpense,
        NetAmount:    stats.TotalIncome.Sub(stats.TotalExpense),
        SuccessCount: stats.SuccessCount,
        FailedCount:  stats.FailedCount,
        MaxIncome:    stats.MaxIncome,
        MaxExpense:   stats.MaxExpense,
    }
    
    if stats.TotalCount > 0 {
        resp.SuccessRate = float64(stats.SuccessCount) / float64(stats.TotalCount) * 100
        resp.AvgIncome = stats.TotalIncome.Div(decimal.NewFromInt(int64(stats.SuccessCount)))
        resp.AvgExpense = stats.TotalExpense.Div(decimal.NewFromInt(int64(stats.SuccessCount)))
    }
    
    // 分组统计
    if req.GroupBy != "" {
        var groups []StatGroup
        
        groupByClause := req.GroupBy
        if req.GroupBy == "day" {
            groupByClause = "DATE(created_at)"
        } else if req.GroupBy == "month" {
            groupByClause = "DATE_TRUNC('month', created_at)"
        }
        
        query.Select(fmt.Sprintf(`
            %s as key,
            COUNT(*) as count,
            COALESCE(SUM(CASE WHEN direction = 'income' AND status = 'success' THEN amount ELSE 0 END), 0) as total_income,
            COALESCE(SUM(CASE WHEN direction = 'expense' AND status = 'success' THEN amount ELSE 0 END), 0) as total_expense
        `, groupByClause)).
        Group(groupByClause).
        Scan(&groups)
        
        for i := range groups {
            groups[i].NetAmount = groups[i].TotalIncome.Sub(groups[i].TotalExpense)
        }
        
        resp.Groups = groups
    }
    
    return resp, nil
}
```

```go
func (b *TransactionRecordBiz) Create(db *gorm.DB, req *ReqCreateTransaction, operatorID uint) (*TransactionRecord, error) {
    return db.Transaction(func(tx *gorm.DB) (*TransactionRecord, error) {
        // 1. 查询账户并锁定
        var account Account
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&account, req.AccountID).Error; err != nil {
            return nil, err
        }
        
        // 2. 检查账户状态
        if account.Status != "active" {
            return nil, errors.New("账户状态异常")
        }
        
        // 3. 检查余额（支出时）
        if req.Direction == "expense" {
            if account.AvailableBalance.LessThan(req.Amount) {
                return nil, errors.New("账户余额不足")
            }
        }
        
        // 4. 计算新余额
        beforeBalance := account.Balance
        var afterBalance decimal.Decimal
        
        if req.Direction == "income" {
            afterBalance = beforeBalance.Add(req.Amount)
        } else {
            afterBalance = beforeBalance.Sub(req.Amount)
        }
        
        // 5. 生成流水号
        transactionNo := generateTransactionNo()
        
        // 6. 创建交易流水
        transaction := &TransactionRecord{
            TransactionNo:       transactionNo,
            AccountID:           req.AccountID,
            AccountNo:           account.AccountNo,
            TransactionType:     req.TransactionType,
            BusinessType:        req.BusinessType,
            Direction:           req.Direction,
            Amount:              req.Amount,
            BeforeBalance:       beforeBalance,
            AfterBalance:        afterBalance,
            RelatedType:         req.RelatedType,
            RelatedID:           req.RelatedID,
            RelatedNo:           req.RelatedNo,
            PaymentMethodID:     req.PaymentMethodID,
            PaymentAccount:      req.PaymentAccount,
            PaymentTime:         timeNow(),
            CounterpartyType:    req.CounterpartyType,
            CounterpartyID:      req.CounterpartyID,
            CounterpartyName:    req.CounterpartyName,
            CounterpartyAccount: req.CounterpartyAccount,
            Description:         req.Description,
            Remark:              req.Remark,
            Status:              "success",
            OperatorID:          operatorID,
        }
        
        if err := tx.Create(transaction).Error; err != nil {
            return nil, err
        }
        
        // 7. 更新账户余额
        account.Balance = afterBalance
        account.AvailableBalance = afterBalance.Sub(account.FrozenAmount)
        
        if req.Direction == "income" {
            account.TotalIncome = account.TotalIncome.Add(req.Amount)
        } else {
            account.TotalExpense = account.TotalExpense.Add(req.Amount)
        }
        
        account.LastTransactionAt = timeNow()
        
        if err := tx.Save(&account).Error; err != nil {
            return nil, err
        }
        
        return transaction, nil
    })
}
```

```go
func (b *TransactionRecordBiz) Cancel(db *gorm.DB, transactionID uint, req *ReqCancelTransaction, operatorID uint) error {
    return db.Transaction(func(tx *gorm.DB) error {
        var transaction TransactionRecord
        
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&transaction, transactionID).Error; err != nil {
            return err
        }
        
        // 只有pending或failed状态的流水可以取消
        if transaction.Status != "pending" && transaction.Status != "failed" {
            return errors.New("流水状态不允许取消")
        }
        
        // 如果是success状态，需要冲正（创建反向流水）
        if transaction.Status == "success" {
            return errors.New("已成功的流水无法取消，请使用冲正功能")
        }
        
        // 更新状态
        transaction.Status = "cancelled"
        transaction.Remark = fmt.Sprintf("%s. 取消原因: %s", transaction.Remark, req.Reason)
        
        return tx.Save(&transaction).Error
    })
}
```

### 3.3 冲正交易

对于已成功的交易，需要通过冲正操作（创建反向流水）来撤销。

#### 3.3.1 接口定义

```
POST /api/v1/transaction-records/:id/reverse
```

#### 3.3.2 业务逻辑

```go
func (b *TransactionRecordBiz) Reverse(db *gorm.DB, transactionID uint, req *ReqReverseTransaction, operatorID uint) (*TransactionRecord, error) {
    return db.Transaction(func(tx *gorm.DB) (*TransactionRecord, error) {
        // 1. 查询原交易
        var originalTxn TransactionRecord
        if err := tx.First(&originalTxn, transactionID).Error; err != nil {
            return nil, err
        }
        
        // 2. 检查是否可以冲正
        if originalTxn.Status != "success" {
            return nil, errors.New("只有成功的交易可以冲正")
        }
        
        // 3. 检查是否已冲正
        var existingReverse int64
        tx.Model(&TransactionRecord{}).
            Where("parent_transaction_no = ?", originalTxn.TransactionNo).
            Count(&existingReverse)
        if existingReverse > 0 {
            return nil, errors.New("该交易已被冲正")
        }
        
        // 4. 锁定账户
        var account Account
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            First(&account, originalTxn.AccountID).Error; err != nil {
            return nil, err
        }
        
        // 5. 反向方向
        reverseDirection := "income"
        if originalTxn.Direction == "income" {
            reverseDirection = "expense"
        }
        
        // 6. 检查余额（如果是反向支出）
        if reverseDirection == "expense" {
            if account.AvailableBalance.LessThan(originalTxn.Amount) {
                return nil, errors.New("账户余额不足，无法冲正")
            }
        }
        
        // 7. 计算新余额
        beforeBalance := account.Balance
        var afterBalance decimal.Decimal
        
        if reverseDirection == "income" {
            afterBalance = beforeBalance.Add(originalTxn.Amount)
        } else {
            afterBalance = beforeBalance.Sub(originalTxn.Amount)
        }
        
        // 8. 创建冲正流水
        reverseTransactionNo := generateTransactionNo()
        reverseTxn := &TransactionRecord{
            TransactionNo:       reverseTransactionNo,
            ParentTransactionNo: originalTxn.TransactionNo,
            AccountID:           originalTxn.AccountID,
            AccountNo:           originalTxn.AccountNo,
            TransactionType:     originalTxn.TransactionType,
            BusinessType:        originalTxn.BusinessType,
            Direction:           reverseDirection,
            Amount:              originalTxn.Amount,
            BeforeBalance:       beforeBalance,
            AfterBalance:        afterBalance,
            RelatedType:         originalTxn.RelatedType,
            RelatedID:           originalTxn.RelatedID,
            RelatedNo:           originalTxn.RelatedNo,
            Description:         fmt.Sprintf("冲正交易: %s. 原因: %s", originalTxn.TransactionNo, req.Reason),
            Remark:              req.Remark,
            Status:              "success",
            OperatorID:          operatorID,
        }
        
        if err := tx.Create(reverseTxn).Error; err != nil {
            return nil, err
        }
        
        // 9. 更新账户余额
        account.Balance = afterBalance
        account.AvailableBalance = afterBalance.Sub(account.FrozenAmount)
        
        if reverseDirection == "income" {
            account.TotalIncome = account.TotalIncome.Add(originalTxn.Amount)
        } else {
            account.TotalExpense = account.TotalExpense.Add(originalTxn.Amount)
        }
        
        account.LastTransactionAt = timeNow()
        
        if err := tx.Save(&account).Error; err != nil {
            return nil, err
        }
        
        return reverseTxn, nil
    })
}
```

```go
        Where("account_id = ?", req.AccountID)
    
    // 使用上一页最后一条记录的ID作为游标
    if req.Cursor > 0 {
        query = query.Where("id < ?", req.Cursor)
    }
    
    query = query.Order("id DESC").Limit(req.PageSize)
    
    var records []TransactionRecord
    if err := query.Find(&records).Error; err != nil {
        return nil, err
    }
    
    // 下一页游标
    var nextCursor uint
    if len(records) > 0 {
        nextCursor = records[len(records)-1].ID
    }
    
    return &RespListTransactionRecords{
        Data:       records,
        NextCursor: nextCursor,
        HasMore:    len(records) == req.PageSize,
    }, nil
}
```

```go
func (b *TransactionRecordBiz) GetRecentTransactions(db *gorm.DB, accountID uint, limit int) ([]TransactionRecord, error) {
    cacheKey := fmt.Sprintf("recent_transactions:%d", accountID)
    
    // 尝试从Redis获取
    var records []TransactionRecord
    cached, err := redis.Get(cacheKey).Result()
    if err == nil {
        json.Unmarshal([]byte(cached), &records)
        return records, nil
    }
    
    // 从数据库查询
    if err := db.Where("account_id = ?", accountID).
        Order("created_at DESC").
        Limit(limit).
        Find(&records).Error; err != nil {
        return nil, err
    }
    
    // 写入缓存（1分钟过期）
    data, _ := json.Marshal(records)
    redis.Set(cacheKey, data, 1*time.Minute)
    
    return records, nil
}
```

### 6.1 余额一致性校验

定期检查账户余额与流水记录是否一致：

```go
func (b *TransactionRecordBiz) ValidateAccountBalance(db *gorm.DB, accountID uint) error {
    var account Account
    if err := db.First(&account, accountID).Error; err != nil {
        return err
    }
    
    // 计算流水累计金额
    var calculatedBalance struct {
        Income  decimal.Decimal
        Expense decimal.Decimal
    }
    
    db.Model(&TransactionRecord{}).
        Where("account_id = ? AND status = 'success'", accountID).
        Select(`
            COALESCE(SUM(CASE WHEN direction = 'income' THEN amount ELSE 0 END), 0) as income,
            COALESCE(SUM(CASE WHEN direction = 'expense' THEN amount ELSE 0 END), 0) as expense
        `).
        Scan(&calculatedBalance)
    
    expectedBalance := calculatedBalance.Income.Sub(calculatedBalance.Expense)
    
    // 比较差异
    diff := account.Balance.Sub(expectedBalance).Abs()
    if diff.GreaterThan(decimal.NewFromFloat(0.01)) { // 允许0.01的误差
        return fmt.Errorf("账户余额不一致: 账户=%s, 流水计算=%s, 差异=%s",
            account.Balance.String(),
            expectedBalance.String(),
            diff.String())
    }
    
    return nil
}
```
