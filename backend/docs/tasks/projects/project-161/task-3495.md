# 物流单管理功能设计文档

## 功能概述

物流单管理是物流发货模块的核心功能，提供物流单的创建、查询、修改、取消等完整生命周期管理，支持多维度查询、状态跟踪、批量操作等功能。

## API接口设计

```go
type ReqCreateShipment struct {
    // 订单信息
    OrderID         uint     `json:"order_id" binding:"required"`
    OrderItemIDs    []uint   `json:"order_item_ids" binding:"required,min=1"` // 发货的订单明细ID列表

    // 发货明细

    // 备注
    Remark          string `json:"remark"`
}
```

```go
// internal/logistics/biz/shipment_biz.go
package biz

import (
    "fmt"
    "time"
    "github.com/shopspring/decimal"
    "gorm.io/gorm"
    "gorm.io/gorm/clause"
)

type ShipmentBiz struct {
    shipmentStore *store.ShipmentStore
    orderStore    *store.OrderStore
    inventoryStore *store.InventoryStore
}

func NewShipmentBiz() *ShipmentBiz {
    return &ShipmentBiz{
        shipmentStore: store.NewShipmentStore(),
        orderStore:    store.NewOrderStore(),
        inventoryStore: store.NewInventoryStore(),
    }
}

// Create 创建物流单
func (b *ShipmentBiz) Create(db *gorm.DB, req *ReqCreateShipment, userID uint) (*RespCreateShipment, error) {
    return db.Transaction(func(tx *gorm.DB) (*RespCreateShipment, error) {
        // 1. 查询订单信息
        order, err := b.orderStore.GetByID(tx, req.OrderID)
        if err != nil {
            return nil, fmt.Errorf("订单不存在: %w", err)
        }
        
        // 验证订单状态
        if order.Status != "paid" && order.Status != "partial_shipped" {
            return nil, errors.New("订单状态不允许发货")
        }
        
        // 2. 查询物流商信息
        provider, err := b.getLogisticsProvider(tx, req.ProviderID)
        if err != nil {
            return nil, err
        }
        
        // 3. 获取发货地址和收货地址
        senderAddr, err := b.getSenderAddress(tx, req.SenderAddressID, req.WarehouseID)
        if err != nil {
            return nil, err
        }
        
        receiverAddr, err := b.getReceiverAddress(tx, req.ReceiverAddressID, order.CustomerID)
        if err != nil {
            return nil, err
        }
        
        // 4. 验证发货明细
        var totalAmount decimal.Decimal
        shipmentItems := make([]model.ShipmentItem, 0, len(req.Items))
        
        for _, itemReq := range req.Items {
            // 查询订单明细
            orderItem, err := b.orderStore.GetOrderItem(tx, itemReq.OrderItemID)
            if err != nil {
                return nil, fmt.Errorf("订单明细不存在: %w", err)
            }
            
            // 验证发货数量
            if itemReq.ShippedQty > orderItem.Quantity - orderItem.ShippedQty {
                return nil, fmt.Errorf("SKU %s 发货数量超出可发货数量", orderItem.SkuCode)
            }
            
            // 验证库存
            inventory, err := b.inventoryStore.GetBySkuAndWarehouse(tx, itemReq.SkuID, req.WarehouseID)
            if err != nil {
                return nil, fmt.Errorf("库存不存在: %w", err)
            }
            
            if inventory.AvailableQuantity < itemReq.ShippedQty {
                return nil, fmt.Errorf("SKU %s 库存不足", orderItem.SkuCode)
            }
            
            // 构建发货明细
            itemAmount := orderItem.UnitPrice.Mul(decimal.NewFromInt(int64(itemReq.ShippedQty)))
            totalAmount = totalAmount.Add(itemAmount)
            
            shipmentItems = append(shipmentItems, model.ShipmentItem{
                OrderID:      req.OrderID,
                OrderItemID:  itemReq.OrderItemID,
                SkuID:        itemReq.SkuID,
                SkuCode:      orderItem.SkuCode,
                SkuName:      orderItem.SkuName,
                Color:        orderItem.Color,
                Size:         orderItem.Size,
                OrderedQty:   orderItem.Quantity,
                ShippedQty:   itemReq.ShippedQty,
                UnitPrice:    orderItem.UnitPrice,
                TotalAmount:  itemAmount,
            })
        }
        
        // 5. 生成物流单号
        shipmentNo := b.generateShipmentNo()
        
        // 6. 创建物流单
        freightAmount, _ := decimal.NewFromString(req.FreightAmount)
        insuranceAmount, _ := decimal.NewFromString(req.InsuranceAmount)
        codAmount, _ := decimal.NewFromString(req.CODAmount)
        
        shipment := &model.Shipment{
            ShipmentNo:      shipmentNo,
            OrderID:         req.OrderID,
            OrderNo:         order.OrderNo,
            ProviderID:      provider.ID,
            ProviderCode:    provider.Code,
            ProviderName:    provider.Name,
            ShipmentType:    req.ShipmentType,
            ShipmentMethod:  "express",
            WarehouseID:     req.WarehouseID,
            ShippedAt:       time.Now(),
            
            // 发件人信息
            SenderName:      senderAddr.ContactName,
            SenderPhone:     senderAddr.ContactPhone,
            SenderProvince:  senderAddr.Province,
            SenderCity:      senderAddr.City,
            SenderDistrict:  senderAddr.District,
            SenderAddress:   senderAddr.Address,
            
            // 收件人信息
            ReceiverName:    receiverAddr.ContactName,
            ReceiverPhone:   receiverAddr.ContactPhone,
            ReceiverProvince: receiverAddr.Province,
            ReceiverCity:    receiverAddr.City,
            ReceiverDistrict: receiverAddr.District,
            ReceiverAddress: receiverAddr.Address,
            
            // 费用
            FreightAmount:   freightAmount,
            InsuranceAmount: insuranceAmount,
            CODAmount:       codAmount,
            
            // 状态
            Status:          "shipped",
            PackageCount:    1,
            
            // 操作人
            OperatorID:      userID,
            Remark:          req.Remark,
        }
        
        if err := tx.Create(shipment).Error; err != nil {
            return nil, fmt.Errorf("创建物流单失败: %w", err)
        }
        
        // 7. 创建物流明细
        for i := range shipmentItems {
            shipmentItems[i].ShipmentID = shipment.ID
            shipmentItems[i].ShipmentNo = shipment.ShipmentNo
            shipmentItems[i].OrderID = req.OrderID
        }
        
        if err := tx.Create(&shipmentItems).Error; err != nil {
            return nil, fmt.Errorf("创建物流明细失败: %w", err)
        }
        
        // 8. 扣减库存
        for _, item := range shipmentItems {
            if err := b.inventoryStore.DeductInventory(tx, item.SkuID, req.WarehouseID, item.ShippedQty); err != nil {
                return nil, fmt.Errorf("扣减库存失败: %w", err)
            }
        }
        
        // 9. 更新订单发货状态
        totalShippedQty := 0
        totalOrderQty := 0
        
        for _, item := range shipmentItems {
            totalShippedQty += item.ShippedQty
            totalOrderQty += item.OrderedQty
        }
        
        shipmentStatus := "partial_shipped"
        if totalShippedQty >= totalOrderQty {
            shipmentStatus = "shipped"
        }
        
        if err := tx.Model(&model.Order{}).Where("id = ?", req.OrderID).
            Update("shipment_status", shipmentStatus).Error; err != nil {
            return nil, err
        }
        
        // 10. 更新订单明细发货数量
        for _, item := range shipmentItems {
            if err := tx.Model(&model.OrderItem{}).Where("id = ?", item.OrderItemID).
                Update("shipped_qty", gorm.Expr("shipped_qty + ?", item.ShippedQty)).Error; err != nil {
                return nil, err
            }
        }
        
        // 11. 调用物流商API获取运单号（异步或同步）
        trackingNo, err := b.createShipmentWithProvider(provider, shipment)
        if err != nil {
            // 记录错误，但不回滚事务
            tx.Model(shipment).Update("remark", fmt.Sprintf("获取运单号失败: %v", err))
        } else {
            tx.Model(shipment).Updates(map[string]interface{}{
                "tracking_no":  trackingNo,
                "tracking_url": b.getTrackingURL(provider.Code, trackingNo),
            })
        }
        
        // 12. 创建初始物流轨迹
        track := &model.ShipmentTrack{
            ShipmentID:   shipment.ID,
            ShipmentNo:   shipment.ShipmentNo,
            TrackingNo:   trackingNo,
            TrackStatus:  "shipped",
            TrackTime:    time.Now(),
            Location:     senderAddr.City,
            Description:  "商品已发货",
            DataSource:   "manual",
        }
        tx.Create(track)
        
        return &RespCreateShipment{
            ShipmentID:   shipment.ID,
            ShipmentNo:   shipment.ShipmentNo,
            TrackingNo:   trackingNo,
            ProviderName: provider.Name,
            Message:      "物流单创建成功",
        }, nil
    })
}

// generateShipmentNo 生成物流单号
// 格式: SH + 年月日 + 8位流水号
func (b *ShipmentBiz) generateShipmentNo() string {
    now := time.Now()
    prefix := fmt.Sprintf("SH%s", now.Format("20060102"))
    
    // 使用Redis INCR生成流水号，保证唯一性
    // 这里简化处理，实际应使用分布式ID生成器
    seq := generateSequence() // 8位流水号
    
    return fmt.Sprintf("%s%08d", prefix, seq)
}
```

```go
// List 查询物流单列表
func (b *ShipmentBiz) List(db *gorm.DB, req *ReqListShipment) (*RespListShipment, error) {
    // 构建查询条件
    query := db.Model(&model.Shipment{})
    
    // 基本筛选
    if req.ShipmentNo != "" {
        query = query.Where("shipment_no = ?", req.ShipmentNo)
    }
    if req.OrderNo != "" {
        query = query.Where("order_no LIKE ?", "%"+req.OrderNo+"%")
    }
    if req.TrackingNo != "" {
        query = query.Where("tracking_no = ?", req.TrackingNo)
    }
    if req.Status != "" {
        query = query.Where("status = ?", req.Status)
    }
    if req.ProviderID > 0 {
        query = query.Where("provider_id = ?", req.ProviderID)
    }
    if req.WarehouseID > 0 {
        query = query.Where("warehouse_id = ?", req.WarehouseID)
    }
    
    // 时间范围
    if req.ShippedStart != "" {
        query = query.Where("DATE(shipped_at) >= ?", req.ShippedStart)
    }
    if req.ShippedEnd != "" {
        query = query.Where("DATE(shipped_at) <= ?", req.ShippedEnd)
    }
    
    // 收件人信息
    if req.ReceiverName != "" {
        query = query.Where("receiver_name LIKE ?", "%"+req.ReceiverName+"%")
    }
    if req.ReceiverPhone != "" {
        query = query.Where("receiver_phone = ?", req.ReceiverPhone)
    }
    
    // 统计总数
    var total int64
    if err := query.Count(&total).Error; err != nil {
        return nil, err
    }
    
    // 排序
    sortBy := "created_at"
    if req.SortBy != "" {
        sortBy = req.SortBy
    }
    sortOrder := "DESC"
    if req.SortOrder == "asc" {
        sortOrder = "ASC"
    }
    query = query.Order(fmt.Sprintf("%s %s", sortBy, sortOrder))
    
    // 分页
    offset := (req.Page - 1) * req.PageSize
    query = query.Limit(req.PageSize).Offset(offset)
    
    // 查询数据
    var shipments []model.Shipment
    if err := query.Find(&shipments).Error; err != nil {
        return nil, err
    }
    
    // 查询明细数量
    shipmentIDs := make([]uint, len(shipments))
    for i, s := range shipments {
        shipmentIDs[i] = s.ID
    }
    
    var itemCounts []struct {
        ShipmentID uint
        Count      int
    }
    
    db.Model(&model.ShipmentItem{}).
        Select("shipment_id, COUNT(*) as count").
        Where("shipment_id IN ?", shipmentIDs).
        Group("shipment_id").
        Scan(&itemCounts)
    
    itemCountMap := make(map[uint]int)
    for _, ic := range itemCounts {
        itemCountMap[ic.ShipmentID] = ic.Count
    }
    
    // 构建响应
    list := make([]ShipmentVO, len(shipments))
    for i, s := range shipments {
        list[i] = ShipmentVO{
            ID:              s.ID,
            ShipmentNo:      s.ShipmentNo,
            OrderNo:         s.OrderNo,
            ProviderName:    s.ProviderName,
            TrackingNo:      s.TrackingNo,
            Status:          s.Status,
            ReceiverName:    s.ReceiverName,
            ReceiverPhone:   s.ReceiverPhone,
            ReceiverAddress: s.ReceiverProvince + s.ReceiverCity + s.ReceiverDistrict + s.ReceiverAddress,
            ItemCount:       itemCountMap[s.ID],
            FreightAmount:   s.FreightAmount.String(),
            ShippedAt:       s.ShippedAt,
            CreatedAt:       s.CreatedAt,
        }
    }
    
    return &RespListShipment{
        List: list,
        Pagination: Pagination{
            Total:      int(total),
            Page:       req.Page,
            PageSize:   req.PageSize,
            TotalPages: int(math.Ceil(float64(total) / float64(req.PageSize))),
        },
    }, nil
}
```

### 4. 取消物流单

```go
// Cancel 取消物流单
func (b *ShipmentBiz) Cancel(db *gorm.DB, shipmentID uint, reason string, userID uint) error {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 查询物流单
        var shipment model.Shipment
        if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).
            Preload("Items").
            First(&shipment, shipmentID).Error; err != nil {
            return err
        }
        
        // 2. 验证是否可取消
        if !shipment.CanCancel() {
            return errors.New("当前状态不允许取消")
        }
        
        // 3. 更新物流单状态
        if err := tx.Model(&shipment).Updates(map[string]interface{}{
            "status": "cancelled",
            "remark": reason,
        }).Error; err != nil {
            return err
        }
        
        // 4. 恢复库存
        for _, item := range shipment.Items {
            if err := b.inventoryStore.AddInventory(tx, item.SkuID, shipment.WarehouseID, item.ShippedQty); err != nil {
                return err
            }
        }
        
        // 5. 更新订单发货数量
        for _, item := range shipment.Items {
            if err := tx.Model(&model.OrderItem{}).Where("id = ?", item.OrderItemID).
                Update("shipped_qty", gorm.Expr("shipped_qty - ?", item.ShippedQty)).Error; err != nil {
                return err
            }
        }
        
        // 6. 更新订单发货状态
        var orderItems []model.OrderItem
        tx.Where("order_id = ?", shipment.OrderID).Find(&orderItems)
        
        allShipped := true
        anyShipped := false
        for _, item := range orderItems {
            if item.ShippedQty < item.Quantity {
                allShipped = false
            }
            if item.ShippedQty > 0 {
                anyShipped = true
            }
        }
        
        newStatus := "paid"
        if allShipped {
            newStatus = "shipped"
        } else if anyShipped {
            newStatus = "partial_shipped"
        }
        
        if err := tx.Model(&model.Order{}).Where("id = ?", shipment.OrderID).
            Update("shipment_status", newStatus).Error; err != nil {
            return err
        }
        
        // 7. 调用物流商API取消运单（如果支持）
        // b.cancelShipmentWithProvider(shipment)
        
        return nil
    })
}
```

```go
// 缓存物流单详情
func (b *ShipmentBiz) GetDetailWithCache(db *gorm.DB, shipmentID uint) (*RespShipmentDetail, error) {
    cacheKey := fmt.Sprintf("shipment:detail:%d", shipmentID)
    
    // 1. 尝试从缓存获取
    var cached RespShipmentDetail
    if err := redis.Get(cacheKey, &cached); err == nil {
        return &cached, nil
    }
    
    // 2. 从数据库查询
    detail, err := b.GetDetail(db, shipmentID)
    if err != nil {
        return nil, err
    }
    
    // 3. 写入缓存
    redis.Set(cacheKey, detail, 5*time.Minute)
    
    return detail, nil
}
```
