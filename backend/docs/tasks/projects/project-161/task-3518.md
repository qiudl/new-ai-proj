# 库存同步功能设计文档

## 1. 概述

库存同步功能实现内部库存与外部电商平台库存的双向同步，确保库存数据的一致性，防止超卖和库存积压。

### 1.1 同步方向

- **Push（推送）**: 将内部库存推送到外部平台（主要模式）
- **Pull（拉取）**: 从外部平台获取库存数据

### 1.2 同步触发方式

- **自动同步**: 内部库存变化时自动推送
- **定时同步**: 定时全量或增量同步
- **手动同步**: 手动触发指定商品同步

```go
package biz

import (
	"fmt"

// InventorySyncBiz 库存同步业务逻辑
type InventorySyncBiz struct{}

// NewInventorySyncBiz 创建库存同步业务逻辑
func NewInventorySyncBiz() *InventorySyncBiz {
	return &InventorySyncBiz{}
}

// PushInventory 推送库存到外部平台
func (b *InventorySyncBiz) PushInventory(db *gorm.DB, mappingID uint, operatorID uint, operatorName string) error {
	// 获取商品映射
	var mapping model.ProductSyncMapping
	if err := db.Preload("Platform").Preload("Sku").First(&mapping, mappingID).Error; err != nil {
		return fmt.Errorf("映射不存在: %w", err)
	}
	
	// 检查是否启用库存同步
	if !mapping.SyncInventory {
		return fmt.Errorf("该映射未启用库存同步")
	}
	
	// 获取当前库存
	currentStock := mapping.Sku.Stock
	
	// 推送到外部平台
	err := b.pushStockToPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID, currentStock)
	if err != nil {
		// 记录失败
		b.recordSyncFailure(db, &mapping, currentStock, "push", err, operatorID, operatorName)
		return err
	}
	
	// 记录成功
	b.recordSyncSuccess(db, &mapping, 0, currentStock, "push", operatorID, operatorName)
	
	return nil
}

// PushInventoryBatch 批量推送库存
func (b *InventorySyncBiz) PushInventoryBatch(db *gorm.DB, platformID uint, shopID string, operatorID uint, operatorName string) error {
	// 获取所有启用库存同步的映射
	var mappings []model.ProductSyncMapping
	err := db.Where("platform_id = ? AND shop_id = ? AND sync_inventory = ? AND status = ?",
		platformID, shopID, true, "active").
		Preload("Sku").
		Find(&mappings).Error
	
	if err != nil {
		return err
	}
	
	successCount := 0
	failedCount := 0
	
	for _, mapping := range mappings {
		if err := b.PushInventory(db, mapping.ID, operatorID, operatorName); err != nil {
			failedCount++
			log.Printf("推送库存失败 [映射ID=%d]: %v", mapping.ID, err)
			continue
		}
		successCount++
	}
	
	log.Printf("批量推送完成: 成功=%d, 失败=%d", successCount, failedCount)
	return nil
}
```

```go
// OnInventoryChanged 库存变化事件处理
// 在SKU库存更新后自动触发
func (b *InventorySyncBiz) OnInventoryChanged(db *gorm.DB, skuID uint, oldStock, newStock int) error {
	// 查找该SKU的所有映射
	var mappings []model.ProductSyncMapping
	db.Where("sku_id = ? AND sync_inventory = ? AND enable_sync = ? AND status = ?",
		skuID, true, true, "active").
		Preload("Platform").
		Find(&mappings)
	
	// 逐个推送
	for _, mapping := range mappings {
		go func(m model.ProductSyncMapping) {
			err := b.pushStockToPlatform(m.Platform, m.ShopID, m.ExternalSkuID, newStock)
			if err != nil {
				log.Printf("自动推送库存失败 [映射ID=%d]: %v", m.ID, err)
				b.recordSyncFailure(db, &m, newStock, "push", err, 0, "系统自动")
			} else {
				b.recordSyncSuccess(db, &m, oldStock, newStock, "push", 0, "系统自动")
			}
		}(mapping)
	}
	
	return nil
}

// 在SKU模型中添加Hook
// AfterUpdate Hook
func (s *Sku) AfterUpdate(tx *gorm.DB) error {
	// 检查库存是否变化
	if tx.Statement.Changed("Stock") {
		oldStock := 0 // 从tx.Statement获取旧值
		newStock := s.Stock
		
		// 触发库存同步
		syncBiz := NewInventorySyncBiz()
		go syncBiz.OnInventoryChanged(tx, s.ID, oldStock, newStock)
	}
	
	return nil
}
```

```go
// PullInventory 从外部平台拉取库存
func (b *InventorySyncBiz) PullInventory(db *gorm.DB, mappingID uint, operatorID uint, operatorName string) error {
	// 获取映射
	var mapping model.ProductSyncMapping
	if err := db.Preload("Platform").Preload("Sku").First(&mapping, mappingID).Error; err != nil {
		return fmt.Errorf("映射不存在: %w", err)
	}
	
	// 从平台获取库存
	externalStock, err := b.fetchStockFromPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID)
	if err != nil {
		b.recordSyncFailure(db, &mapping, 0, "pull", err, operatorID, operatorName)
		return err
	}
	
	// 更新本地库存
	oldStock := mapping.Sku.Stock
	
	err = db.Model(&mapping.Sku).Update("stock", externalStock).Error
	if err != nil {
		return err
	}
	
	// 记录同步成功
	b.recordSyncSuccess(db, &mapping, oldStock, externalStock, "pull", operatorID, operatorName)
	
	return nil
}

// PullInventoryBatch 批量拉取库存
func (b *InventorySyncBiz) PullInventoryBatch(db *gorm.DB, platformID uint, shopID string, operatorID uint, operatorName string) error {
	var mappings []model.ProductSyncMapping
	db.Where("platform_id = ? AND shop_id = ? AND sync_inventory = ? AND enable_sync = ?",
		platformID, shopID, true, true).
		Preload("Sku").
		Find(&mappings)
	
	successCount := 0
	failedCount := 0
	
	for _, mapping := range mappings {
		if err := b.PullInventory(db, mapping.ID, operatorID, operatorName); err != nil {
			failedCount++
			continue
		}
		successCount++
	}
	
	log.Printf("批量拉取库存完成: 成功=%d, 失败=%d", successCount, failedCount)
	return nil
}
```

```go
// pushToTaobao 推送库存到淘宝/天猫
func (b *InventorySyncBiz) pushToTaobao(platform *model.MallPlatform, shopID, skuID string, stock int) error {
	client := NewTaobaoClient(platform)
	
	// 调用淘宝库存更新API

		"quantity": stock,
		"type": 1, // 1=全量更新
	})
	
	if err != nil {
		return fmt.Errorf("淘宝库存更新失败: %w", err)
	}
	
	return nil
}

	case "taobao", "tmall":
		return b.fetchFromTaobao(platform, shopID, externalSkuID)
	case "jd":
		return b.fetchFromJD(platform, shopID, externalSkuID)
	case "pdd":
		return b.fetchFromPDD(platform, shopID, externalSkuID)
	default:
		return 0, fmt.Errorf("不支持的平台: %s", platform.PlatformCode)
	}

// fetchFromTaobao 从淘宝获取库存
```

## 5. 库存预留机制

### 5.1 库存预留

```go
// ReserveInventory 预留库存
// 订单创建时预留库存，待支付后扣减
func (b *InventorySyncBiz) ReserveInventory(db *gorm.DB, skuID uint, quantity int, orderID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var sku model.Sku
		if err := tx.First(&sku, skuID).Error; err != nil {
			return err
		}
		
		// 检查库存是否充足
		if sku.Stock < quantity {
			return fmt.Errorf("库存不足: 当前库存=%d, 需要=%d", sku.Stock, quantity)
		}
		
		// 扣减库存
		if err := tx.Model(&sku).Update("stock", gorm.Expr("stock - ?", quantity)).Error; err != nil {
			return err
		}
		
		// 创建库存预留记录
		reservation := &model.InventoryReservation{
			SkuID:      skuID,
			OrderID:    orderID,
			Quantity:   quantity,
			Status:     "reserved",
			ExpireAt:   time.Now().Add(30 * time.Minute), // 30分钟后过期
		}
		
		return tx.Create(reservation).Error
	})
}

// ConfirmReservation 确认预留（订单支付成功）
func (b *InventorySyncBiz) ConfirmReservation(db *gorm.DB, orderID uint) error {
	return db.Model(&model.InventoryReservation{}).
		Where("order_id = ? AND status = ?", orderID, "reserved").
		Update("status", "confirmed").Error
}

// ReleaseReservation 释放预留（订单取消）
func (b *InventorySyncBiz) ReleaseReservation(db *gorm.DB, orderID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var reservations []model.InventoryReservation
		tx.Where("order_id = ? AND status = ?", orderID, "reserved").
			Find(&reservations)
		
		for _, reservation := range reservations {
			// 恢复库存
			tx.Model(&model.Sku{}).
				Where("id = ?", reservation.SkuID).
				Update("stock", gorm.Expr("stock + ?", reservation.Quantity))
			
			// 更新预留状态
			tx.Model(&reservation).Update("status", "released")
		}
		
		return nil
	})
}
```

```go
// CheckSafetyStock 检查安全库存
func (b *InventorySyncBiz) CheckSafetyStock(db *gorm.DB, skuID uint) (bool, error) {
	var sku model.Sku
	if err := db.First(&sku, skuID).Error; err != nil {
		return false, err
	}
	
	// 低于预警阈值
	if sku.Stock <= sku.StockAlertThreshold {
		// 发送预警通知
		b.sendStockAlert(skuID, sku.Stock, sku.StockAlertThreshold)
		return false, nil
	}
	
	return true, nil
}

// AdjustMallStock 根据安全库存调整商城库存
// 在推送到商城时，保留一定的安全库存
func (b *InventorySyncBiz) AdjustMallStock(actualStock, safetyStock int) int {
	mallStock := actualStock - safetyStock
	if mallStock < 0 {
		return 0
	}
	return mallStock
}

// PushInventoryWithSafety 带安全库存的推送
func (b *InventorySyncBiz) PushInventoryWithSafety(db *gorm.DB, mappingID uint, safetyStock int, operatorID uint, operatorName string) error {
	var mapping model.ProductSyncMapping
	db.Preload("Platform").Preload("Sku").First(&mapping, mappingID)
	
	// 计算可推送库存
	mallStock := b.AdjustMallStock(mapping.Sku.Stock, safetyStock)
	
	// 推送调整后的库存
	err := b.pushStockToPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID, mallStock)
	if err != nil {
		return err
	}
	
	// 记录同步
	b.recordSyncSuccess(db, &mapping, 0, mallStock, "push", operatorID, operatorName)
	
	return nil
}
```

```go
// ReconcileInventory 库存对账
func (b *InventorySyncBiz) ReconcileInventory(db *gorm.DB, platformID uint, shopID string) ([]InventoryDiff, error) {
	// 获取所有映射
	var mappings []model.ProductSyncMapping
	db.Where("platform_id = ? AND shop_id = ? AND sync_inventory = ?", platformID, shopID, true).
		Preload("Sku").
		Find(&mappings)
	
	var diffs []InventoryDiff
	
	for _, mapping := range mappings {
		// 获取外部库存
		externalStock, err := b.fetchStockFromPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID)
		if err != nil {
			log.Printf("获取外部库存失败 [映射ID=%d]: %v", mapping.ID, err)
			continue
		}
		
		// 比较库存
		localStock := mapping.Sku.Stock
		if localStock != externalStock {
			diffs = append(diffs, InventoryDiff{
				MappingID:     mapping.ID,
				SkuCode:       mapping.SkuCode,
				LocalStock:    localStock,
				ExternalStock: externalStock,
				Difference:    localStock - externalStock,
			})
		}
	}
	
	return diffs, nil
}

// InventoryDiff 库存差异
type InventoryDiff struct {
	MappingID     uint
	SkuCode       string
	LocalStock    int
	ExternalStock int
	Difference    int
}

// FixInventoryDiff 修复库存差异
func (b *InventorySyncBiz) FixInventoryDiff(db *gorm.DB, diff *InventoryDiff, strategy string) error {
	switch strategy {
	case "push":
		// 以本地为准，推送到平台
		return b.PushInventory(db, diff.MappingID, 0, "系统对账")
	case "pull":
		// 以平台为准，拉取到本地
		return b.PullInventory(db, diff.MappingID, 0, "系统对账")
	default:
		return fmt.Errorf("不支持的修复策略: %s", strategy)
	}
}
```

```go
func (b *InventorySyncBiz) recordSyncSuccess(db *gorm.DB, mapping *model.ProductSyncMapping, beforeStock, afterStock int, direction string, operatorID uint, operatorName string) error {
	record := &model.InventorySyncRecord{

		SyncedAt:      time.Now(),
		OperatorID:    operatorID,
		OperatorName:  operatorName,
		TriggerType:   b.getTriggerType(operatorID),
	}
	
	return db.Create(record).Error
}

// recordSyncFailure 记录同步失败
```

### 9.1 批量推送优化

```go
// BatchPushWithGrouping 分组批量推送
// 将相同平台的SKU分组，减少API调用次数
func (b *InventorySyncBiz) BatchPushWithGrouping(db *gorm.DB, mappingIDs []uint, operatorID uint, operatorName string) error {
	// 获取所有映射
	var mappings []model.ProductSyncMapping
	db.Where("id IN ?", mappingIDs).
		Preload("Platform").
		Preload("Sku").
		Find(&mappings)
	
	// 按平台分组
	platformGroups := make(map[uint][]model.ProductSyncMapping)
	for _, mapping := range mappings {
		platformGroups[mapping.PlatformID] = append(platformGroups[mapping.PlatformID], mapping)
	}
	
	// 逐平台处理
	for platformID, group := range platformGroups {
		// 调用平台批量接口
		if err := b.batchPushToPlatform(platformID, group); err != nil {
			log.Printf("批量推送失败 [平台ID=%d]: %v", platformID, err)
			continue
		}
		
		// 记录同步结果
		for _, mapping := range group {
			b.recordSyncSuccess(db, &mapping, 0, mapping.Sku.Stock, "push", operatorID, operatorName)
		}
	}
	
	return nil
}
```

```go
import "golang.org/x/time/rate"

// RateLimitedPush 限流推送
func (b *InventorySyncBiz) RateLimitedPush(db *gorm.DB, mappingIDs []uint, rateLimit int) error {
	limiter := rate.NewLimiter(rate.Limit(rateLimit), rateLimit)
	
	for _, mappingID := range mappingIDs {
		if err := limiter.Wait(context.Background()); err != nil {
			return err
		}
		
		if err := b.PushInventory(db, mappingID, 0, "系统自动"); err != nil {
			log.Printf("推送失败 [映射ID=%d]: %v", mappingID, err)
		}
	}
	
	return nil
}
```
