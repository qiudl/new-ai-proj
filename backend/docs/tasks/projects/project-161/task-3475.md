# 资金流水模块数据库Schema设计文档

## 1. 概述

### 1.1 设计目标

资金流水模块是Coolbuy 3.0系统的核心财务管理模块，负责管理所有与资金相关的业务流程。本文档定义了完整的数据库Schema设计，确保：

- **资金安全**：通过事务机制和余额校验确保资金数据准确无误
- **流水完整**：记录所有资金变动，支持完整的审计追踪
- **多维度查询**：支持按账户、时间、类型等多维度查询统计
- **高并发支持**：通过乐观锁和悲观锁机制保障并发安全
- **多租户隔离**：采用PostgreSQL Schema隔离，确保租户数据安全

### 1.2 技术选型

```yaml
数据库: PostgreSQL 14+
ORM框架: GORM v2
多租户策略: PostgreSQL Schema隔离
字符集: UTF-8
时区: Asia/Shanghai (北京时间)
精度策略:
  金额字段: DECIMAL(18,2) - 支持千万亿级金额，2位小数
  汇率字段: DECIMAL(10,6) - 6位小数精度
```

### 1.3 核心实体关系

```
┌─────────────────┐
│   Accounts      │  账户主表
│   (账户)        │
└────────┬────────┘
         │
         │ 1:N
         ▼
┌─────────────────┐       ┌──────────────────┐
│ TransactionRecords│◄─────┤ PaymentMethods   │
│   (交易流水)      │       │  (支付方式)      │
└────────┬────────┘       └──────────────────┘
         │
         │ 1:N
         ▼
┌─────────────────┐       ┌──────────────────┐
│  Settlements    │       │ AccountSnapshots │
│   (结算记录)     │       │  (余额快照)      │
└─────────────────┘       └──────────────────┘
```

## 2. 多租户架构设计

### 2.1 Schema隔离策略

采用PostgreSQL Schema实现多租户数据隔离：

```sql
-- 公共Schema (存储全局配置)
CREATE SCHEMA IF NOT EXISTS public;

-- 租户Schema (每个租户独立Schema)
CREATE SCHEMA IF NOT EXISTS tenant_1;
CREATE SCHEMA IF NOT EXISTS tenant_2;
CREATE SCHEMA IF NOT EXISTS tenant_N;
```

### 2.2 连接池配置

```go
// 中间件自动切换租户Schema
func TenantMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        tenantID := c.GetHeader("X-Tenant-ID")
        schemaName := fmt.Sprintf("tenant_%s", tenantID)
        
        // 切换到租户Schema
        db.Exec(fmt.Sprintf("SET search_path TO %s", schemaName))
        
        c.Set("db", db)
        c.Next()
    }
}
```

## 3. 核心表设计

### 3.1 账户表 (accounts)

账户是资金流水的核心实体，用于管理客户、供应商、公司内部等各类账户。

#### 3.1.1 表结构

```sql
CREATE TABLE accounts (
    id                  BIGSERIAL PRIMARY KEY,
    account_no          VARCHAR(50) UNIQUE NOT NULL,        -- 账户编号
    account_name        VARCHAR(200) NOT NULL,              -- 账户名称
    account_type        VARCHAR(20) NOT NULL,               -- 账户类型: customer/supplier/company/employee
    owner_type          VARCHAR(20) NOT NULL,               -- 所有者类型: customer/supplier/employee
    owner_id            BIGINT NOT NULL,                    -- 所有者ID
    
    -- 余额信息
    balance             DECIMAL(18,2) NOT NULL DEFAULT 0,   -- 当前余额
    frozen_amount       DECIMAL(18,2) NOT NULL DEFAULT 0,   -- 冻结金额
    available_balance   DECIMAL(18,2) NOT NULL DEFAULT 0,   -- 可用余额 = balance - frozen_amount
    credit_limit        DECIMAL(18,2) DEFAULT 0,            -- 信用额度
    
    -- 统计信息
    total_income        DECIMAL(18,2) NOT NULL DEFAULT 0,   -- 累计收入
    total_expense       DECIMAL(18,2) NOT NULL DEFAULT 0,   -- 累计支出
    last_transaction_at TIMESTAMP,                          -- 最后交易时间
    
    -- 状态管理
    status              VARCHAR(20) NOT NULL DEFAULT 'active', -- 状态: active/frozen/closed
    currency            VARCHAR(10) NOT NULL DEFAULT 'CNY',     -- 币种
    
    -- 风控信息
    risk_level          VARCHAR(20) DEFAULT 'normal',        -- 风险等级: low/normal/high
    daily_limit         DECIMAL(18,2),                       -- 日交易限额
    monthly_limit       DECIMAL(18,2),                       -- 月交易限额
    
    -- 审计字段
    version             INTEGER NOT NULL DEFAULT 0,          -- 乐观锁版本号
    created_at          TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at          TIMESTAMP,
    
    -- 索引
    INDEX idx_account_no (account_no),
    INDEX idx_owner (owner_type, owner_id),
    INDEX idx_account_type (account_type),
    INDEX idx_status (status),
    INDEX idx_deleted_at (deleted_at)
);

COMMENT ON TABLE accounts IS '账户表';
COMMENT ON COLUMN accounts.balance IS '当前余额';
COMMENT ON COLUMN accounts.frozen_amount IS '冻结金额（待结算/待审核等）';
COMMENT ON COLUMN accounts.available_balance IS '可用余额 = balance - frozen_amount';
COMMENT ON COLUMN accounts.version IS '乐观锁版本号，用于并发控制';
```

### 3.2 交易流水表 (transaction_records)

记录所有资金变动，是财务审计的核心数据。

#### 3.2.1 表结构

```sql
CREATE TABLE transaction_records (
    id                  BIGSERIAL PRIMARY KEY,
    transaction_no      VARCHAR(50) UNIQUE NOT NULL,        -- 交易流水号
    parent_transaction_no VARCHAR(50),                       -- 父交易流水号（用于关联原交易）
    
    -- 账户信息
    account_id          BIGINT NOT NULL,                    -- 账户ID
    account_no          VARCHAR(50) NOT NULL,               -- 账户编号（冗余）
    
    -- 交易基本信息
    transaction_type    VARCHAR(30) NOT NULL,               -- 交易类型
    business_type       VARCHAR(30) NOT NULL,               -- 业务类型
    direction           VARCHAR(10) NOT NULL,               -- 方向: income/expense
    
    -- 金额信息
    amount              DECIMAL(18,2) NOT NULL,             -- 交易金额
    before_balance      DECIMAL(18,2) NOT NULL,             -- 交易前余额
    after_balance       DECIMAL(18,2) NOT NULL,             -- 交易后余额
    
    -- 关联信息
    related_type        VARCHAR(30),                        -- 关联业务类型: order/settlement/refund等
    related_id          BIGINT,                             -- 关联业务ID
    related_no          VARCHAR(50),                        -- 关联业务单号
    
    -- 支付信息
    payment_method_id   BIGINT,                             -- 支付方式ID
    payment_account     VARCHAR(100),                       -- 支付账号
    payment_time        TIMESTAMP,                          -- 支付时间
    
    -- 对方信息
    counterparty_type   VARCHAR(20),                        -- 对方类型: customer/supplier/company
    counterparty_id     BIGINT,                             -- 对方ID
    counterparty_name   VARCHAR(200),                       -- 对方名称
    counterparty_account VARCHAR(50),                       -- 对方账户编号
    
    -- 交易状态
    status              VARCHAR(20) NOT NULL DEFAULT 'pending', -- 状态: pending/success/failed/cancelled
    
    -- 描述信息
    description         TEXT,                               -- 交易描述
    remark              TEXT,                               -- 备注
    
    -- 审核信息
    auditor_id          BIGINT,                             -- 审核人ID
    audited_at          TIMESTAMP,                          -- 审核时间
    audit_remark        TEXT,                               -- 审核备注
    
    -- 审计字段
    operator_id         BIGINT NOT NULL,                    -- 操作人ID
    created_at          TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at          TIMESTAMP,
    
    -- 索引
    INDEX idx_transaction_no (transaction_no),
    INDEX idx_account_id (account_id),
    INDEX idx_account_no (account_no),
    INDEX idx_transaction_type (transaction_type),
    INDEX idx_business_type (business_type),
    INDEX idx_related (related_type, related_id),
    INDEX idx_related_no (related_no),
    INDEX idx_status (status),
    INDEX idx_payment_time (payment_time),
    INDEX idx_created_at (created_at),
    INDEX idx_deleted_at (deleted_at),
    
    -- 外键
    FOREIGN KEY (account_id) REFERENCES accounts(id)
);

COMMENT ON TABLE transaction_records IS '交易流水表';
COMMENT ON COLUMN transaction_records.direction IS '资金方向: income(收入)/expense(支出)';
COMMENT ON COLUMN transaction_records.before_balance IS '交易前账户余额';
COMMENT ON COLUMN transaction_records.after_balance IS '交易后账户余额';
```

### 3.3 支付方式表 (payment_methods)

管理系统支持的所有支付方式。

```sql
CREATE TABLE payment_methods (

    -- 限额设置
    single_limit    DECIMAL(18,2),                          -- 单笔限额
    daily_limit     DECIMAL(18,2),                          -- 日限额
    monthly_limit   DECIMAL(18,2),                          -- 月限额
    
    -- 状态管理

    deleted_at      TIMESTAMP,
    
    INDEX idx_code (code),
    INDEX idx_type (type),
    INDEX idx_enabled (is_enabled),
```

### 3.4 结算记录表 (settlements)

管理与客户、供应商之间的结算流程。

```sql
CREATE TABLE settlements (
    id                  BIGSERIAL PRIMARY KEY,
    settlement_no       VARCHAR(50) UNIQUE NOT NULL,        -- 结算单号
    settlement_type     VARCHAR(20) NOT NULL,               -- 结算类型: customer/supplier
    
    -- 结算对象
    party_type          VARCHAR(20) NOT NULL,               -- 对方类型: customer/supplier
    party_id            BIGINT NOT NULL,                    -- 对方ID
    party_name          VARCHAR(200) NOT NULL,              -- 对方名称
    party_account_id    BIGINT NOT NULL,                    -- 对方账户ID
    
    -- 公司账户
    company_account_id  BIGINT NOT NULL,                    -- 公司账户ID
    
    -- 结算周期
    settlement_period_start DATE NOT NULL,                  -- 结算周期开始
    settlement_period_end   DATE NOT NULL,                  -- 结算周期结束
    
    -- 金额信息
    total_amount        DECIMAL(18,2) NOT NULL,             -- 结算总金额
    paid_amount         DECIMAL(18,2) NOT NULL DEFAULT 0,   -- 已支付金额
    unpaid_amount       DECIMAL(18,2) NOT NULL,             -- 未支付金额
    
    -- 费用明细
    order_amount        DECIMAL(18,2),                      -- 订单金额
    refund_amount       DECIMAL(18,2),                      -- 退款金额
    commission_amount   DECIMAL(18,2),                      -- 佣金
    adjustment_amount   DECIMAL(18,2),                      -- 调整金额
    
    -- 状态管理
    status              VARCHAR(20) NOT NULL DEFAULT 'pending', -- 状态: pending/partial/completed/cancelled
    
    -- 审核信息
    auditor_id          BIGINT,
    audited_at          TIMESTAMP,
    audit_remark        TEXT,
    
    -- 支付信息
    payment_method_id   BIGINT,
    payment_account     VARCHAR(100),
    payment_voucher     VARCHAR(200),                       -- 支付凭证
    payment_time        TIMESTAMP,
    
    -- 描述信息
    description         TEXT,
    remark              TEXT,
    
    -- 审计字段
    creator_id          BIGINT NOT NULL,
    created_at          TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at          TIMESTAMP,
    
    -- 索引
    INDEX idx_settlement_no (settlement_no),
    INDEX idx_party (party_type, party_id),
    INDEX idx_settlement_type (settlement_type),
    INDEX idx_status (status),
    INDEX idx_period (settlement_period_start, settlement_period_end),
    INDEX idx_payment_time (payment_time),
    INDEX idx_deleted_at (deleted_at),
    
    -- 外键
    FOREIGN KEY (party_account_id) REFERENCES accounts(id),
    FOREIGN KEY (company_account_id) REFERENCES accounts(id)
);

COMMENT ON TABLE settlements IS '结算记录表';
```

### 4.1 业务规则约束

```sql
-- 1. 账户余额约束：可用余额 = 余额 - 冻结金额
ALTER TABLE accounts ADD CONSTRAINT chk_available_balance 
CHECK (available_balance = balance - frozen_amount);

-- 2. 余额非负约束
ALTER TABLE accounts ADD CONSTRAINT chk_balance_positive 
CHECK (balance >= 0);

-- 3. 交易金额非负约束
ALTER TABLE transaction_records ADD CONSTRAINT chk_amount_positive 
CHECK (amount >= 0);

-- 4. 结算金额一致性约束：已支付 + 未支付 = 总金额
ALTER TABLE settlements ADD CONSTRAINT chk_settlement_amount 
CHECK (paid_amount + unpaid_amount = total_amount);
```

### 4.2 触发器

#### 4.2.1 自动更新账户余额快照

```sql
-- 每日生成账户余额快照的存储过程
CREATE OR REPLACE FUNCTION generate_daily_account_snapshots()
RETURNS void AS $$
DECLARE
    snapshot_date DATE := CURRENT_DATE;
BEGIN
    INSERT INTO account_balance_snapshots (
        account_id,
        account_no,
        snapshot_date,
        balance,
        frozen_amount,
        available_balance,
        total_income,
        total_expense,
        daily_income,
        daily_expense,
        transaction_count,
        created_at
    )
    SELECT 
        a.id,
        a.account_no,
        snapshot_date,
        a.balance,
        a.frozen_amount,
        a.available_balance,
        a.total_income,
        a.total_expense,
        COALESCE(daily_stats.daily_income, 0),
        COALESCE(daily_stats.daily_expense, 0),
        COALESCE(daily_stats.transaction_count, 0),
        NOW()
    FROM accounts a
    LEFT JOIN (
        SELECT 
            account_id,
            SUM(CASE WHEN direction = 'income' THEN amount ELSE 0 END) as daily_income,
            SUM(CASE WHEN direction = 'expense' THEN amount ELSE 0 END) as daily_expense,
            COUNT(*) as transaction_count
        FROM transaction_records
        WHERE DATE(created_at) = snapshot_date
            AND status = 'success'
            AND deleted_at IS NULL
        GROUP BY account_id
    ) daily_stats ON a.id = daily_stats.account_id
    WHERE a.deleted_at IS NULL
    ON CONFLICT (account_id, snapshot_date) DO UPDATE SET
        balance = EXCLUDED.balance,
        frozen_amount = EXCLUDED.frozen_amount,
        available_balance = EXCLUDED.available_balance,
        total_income = EXCLUDED.total_income,
        total_expense = EXCLUDED.total_expense,
        daily_income = EXCLUDED.daily_income,
        daily_expense = EXCLUDED.daily_expense,
        transaction_count = EXCLUDED.transaction_count;
END;
$$ LANGUAGE plpgsql;
```
