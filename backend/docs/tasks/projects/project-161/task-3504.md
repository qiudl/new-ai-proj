# 售后审批流程设计文档

## 1. 概述

本文档定义Coolbuy 3.0售后模块的审批流程设计，支持多级审批、条件审批、自动审批等场景。

## 2. 审批策略

### 2.1 审批规则配置

```go
// ApprovalRule 审批规则配置
type ApprovalRule struct {
	// 金额阈值
	AmountThresholds []AmountThreshold `json:"amount_thresholds"`
	
	// 自动审批条件
	AutoApprovalConditions AutoApprovalCondition `json:"auto_approval_conditions"`
}

// AmountThreshold 金额阈值审批规则
type AmountThreshold struct {
	MinAmount      decimal.Decimal `json:"min_amount"`       // 最小金额
	MaxAmount      decimal.Decimal `json:"max_amount"`       // 最大金额（0表示无限）
	ApprovalLevels int             `json:"approval_levels"`  // 需要审批级别数
	Approvers      []uint          `json:"approvers"`        // 审批人ID列表
}

// AutoApprovalCondition 自动审批条件
type AutoApprovalCondition struct {
	Enabled            bool            `json:"enabled"`              // 是否启用
	MaxAmount          decimal.Decimal `json:"max_amount"`           // 最大金额
	AllowedReasonTypes []string        `json:"allowed_reason_types"` // 允许的原因类型
	MaxDaysAfterOrder  int             `json:"max_days_after_order"` // 订单完成后最大天数
}
```

### 2.2 自动审批逻辑

```go
// CheckAutoApproval 检查是否满足自动审批条件
func (b *ApprovalBiz) CheckAutoApproval(db *gorm.DB, afterSales *model.AfterSales) (bool, string) {
	// 加载审批规则
	rule := b.loadApprovalRule()
	
	if !rule.AutoApprovalConditions.Enabled {
		return false, "自动审批未启用"
	}
	
	// 1. 检查金额
	if afterSales.RefundAmount.GreaterThan(rule.AutoApprovalConditions.MaxAmount) {
		return false, fmt.Sprintf("退款金额超过自动审批限额(%.2f)", rule.AutoApprovalConditions.MaxAmount)
	}
	
	// 2. 检查原因类型
	allowed := false
	for _, reasonType := range rule.AutoApprovalConditions.AllowedReasonTypes {
		if afterSales.ReasonType == reasonType {
			allowed = true
			break
		}
	}
	if !allowed {
		return false, "退货原因不在自动审批范围内"
	}
	
	// 3. 检查订单完成时间
	var order model.Order
	db.First(&order, afterSales.OrderID)
	if order.CompletedAt == nil {
		return false, "订单未完成"
	}
	
	daysSinceCompleted := int(time.Since(*order.CompletedAt).Hours() / 24)
	if daysSinceCompleted > rule.AutoApprovalConditions.MaxDaysAfterOrder {
		return false, fmt.Sprintf("订单完成已超过%d天", rule.AutoApprovalConditions.MaxDaysAfterOrder)
	}
	
	return true, "满足自动审批条件"
}

// AutoApprove 自动审批
func (b *ApprovalBiz) AutoApprove(db *gorm.DB, afterSalesID uint) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var afterSales model.AfterSales
		tx.First(&afterSales, afterSalesID)
		
		// 执行自动审批
		now := time.Now()
		updates := map[string]interface{}{
			"approver_id":     0,
			"approver_name":   "系统自动审批",
			"approved_at":     now,
			"approval_remark": "满足自动审批条件，系统自动通过",
			"approval_status": "approved",
			"status":          "approved",
		}
		
		if afterSales.AfterSalesType == "refund_only" {
			updates["refund_status"] = "pending"
		}
		
		tx.Model(&afterSales).Updates(updates)
		
		// 创建审批记录
		approval := &model.AfterSalesApproval{
			AfterSalesID:    afterSalesID,
			AfterSalesNo:    afterSales.AfterSalesNo,
			ApprovalLevel:   1,
			ApproverID:      0,
			ApproverName:    "系统自动审批",
			ApprovalStatus:  "approved",
			ApprovalRemark:  "满足自动审批条件，系统自动通过",
			ApprovedAt:      &now,
			RequestedAmount: afterSales.RefundAmount,
			ApprovedAmount:  afterSales.RefundAmount,
		}
		tx.Create(approval)
		
		// 创建日志
		log := &model.AfterSalesLog{
			AfterSalesID:  afterSalesID,
			AfterSalesNo:  afterSales.AfterSalesNo,
			OperationType: "auto_approve",
			OperationDesc: "系统自动审批通过",
			OperatorType:  "system",
			BeforeStatus:  "pending",
			AfterStatus:   "approved",
		}
		tx.Create(log)
		
		return nil
	})
}
```

### 2.3 多级审批流程

```go
// GetRequiredApprovalLevels 获取需要的审批级别数
func (b *ApprovalBiz) GetRequiredApprovalLevels(amount decimal.Decimal) int {
	rule := b.loadApprovalRule()
	
	for _, threshold := range rule.AmountThresholds {
		if amount.GreaterThanOrEqual(threshold.MinAmount) {
			if threshold.MaxAmount.IsZero() || amount.LessThan(threshold.MaxAmount) {
				return threshold.ApprovalLevels
			}
		}
	}
	
	return 1 // 默认1级审批
}

// SubmitForNextLevel 提交下一级审批
func (b *ApprovalBiz) SubmitForNextLevel(db *gorm.DB, afterSalesID uint, currentLevel int) error {
	return db.Transaction(func(tx *gorm.DB) error {
		var afterSales model.AfterSales
		tx.First(&afterSales, afterSalesID)
		
		requiredLevels := b.GetRequiredApprovalLevels(afterSales.RefundAmount)
		
		if currentLevel >= requiredLevels {
			// 所有级别审批完成，设置为已通过
			updates := map[string]interface{}{
				"approval_status": "approved",
				"status":          "approved",
			}
			
			if afterSales.AfterSalesType == "refund_only" {
				updates["refund_status"] = "pending"
			}
			
			tx.Model(&afterSales).Updates(updates)
			
			// 创建日志
			log := &model.AfterSalesLog{
				AfterSalesID:  afterSalesID,
				AfterSalesNo:  afterSales.AfterSalesNo,
				OperationType: "all_approved",
				OperationDesc: fmt.Sprintf("所有%d级审批已完成", requiredLevels),
				OperatorType:  "system",
				AfterStatus:   "approved",
			}
			tx.Create(log)
		} else {
			// 进入下一级审批
			log := &model.AfterSalesLog{
				AfterSalesID:  afterSalesID,
				AfterSalesNo:  afterSales.AfterSalesNo,
				OperationType: "next_level",
				OperationDesc: fmt.Sprintf("进入第%d级审批", currentLevel+1),
				OperatorType:  "system",
			}
			tx.Create(log)
		}
		
		return nil
	})
}
```

## 3. 审批流程图

```
售后申请创建
    ↓
检查自动审批条件
    ├─ 满足 → 自动审批通过 → approved
    └─ 不满足 → 进入人工审批
                  ↓
            确定审批级别(1/2/3级)
                  ↓
            第1级审批
                  ├─ 通过 → [需要更多级别] → 第2级审批 → ... → approved
                  │         └─ [无需更多级别] → approved
                  └─ 拒绝 → rejected
```

## 4. 审批权限管理

```go
// ApprovalPermission 审批权限
type ApprovalPermission struct {
	UserID       uint            `json:"user_id"`        // 用户ID
	MaxAmount    decimal.Decimal `json:"max_amount"`     // 最大可审批金额
	CanReject    bool            `json:"can_reject"`     // 是否可以拒绝
	CanAdjust    bool            `json:"can_adjust"`     // 是否可以调整金额
	ApprovalLevel int            `json:"approval_level"` // 可审批级别
}

// CheckApprovalPermission 检查审批权限
func (b *ApprovalBiz) CheckApprovalPermission(userID uint, afterSales *model.AfterSales, level int) error {
	permission := b.getApprovalPermission(userID)
	
	if permission == nil {
		return errors.New("无审批权限")
	}
	
	if level > permission.ApprovalLevel {
		return fmt.Errorf("无权审批第%d级申请", level)
	}
	
	if afterSales.RefundAmount.GreaterThan(permission.MaxAmount) {
		return fmt.Errorf("退款金额超过审批权限(%.2f)", permission.MaxAmount)
	}
	
	return nil
}
```

## 5. 审批通知

```go
// SendApprovalNotification 发送审批通知
func (b *ApprovalBiz) SendApprovalNotification(afterSales *model.AfterSales, level int) error {
	rule := b.loadApprovalRule()
	
	// 获取该级别的审批人
	var approvers []uint
	for _, threshold := range rule.AmountThresholds {
		if afterSales.RefundAmount.GreaterThanOrEqual(threshold.MinAmount) {
			if threshold.MaxAmount.IsZero() || afterSales.RefundAmount.LessThan(threshold.MaxAmount) {
				approvers = threshold.Approvers
				break
			}
		}
	}
	
	// 发送通知（邮件、短信、站内信）
	for _, approverID := range approvers {
		// 发送通知逻辑
		b.sendNotificationToUser(approverID, afterSales)
	}
	
	return nil
}
```

## 6. 总结
