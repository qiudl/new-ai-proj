# 价格同步功能设计文档

## 1. 概述

价格同步功能实现内部商品价格与外部电商平台价格的双向同步，支持多种价格类型（市场价、会员价等）的同步管理。

### 1.1 同步方向

- **Push（推送）**: 将内部价格推送到外部平台（主要模式）
- **Pull（拉取）**: 从外部平台获取价格数据

### 1.2 价格类型

- **市场价（Market Price）**: 商品标价
- **会员价（Member Price）**: 会员专享价
- **成本价（Cost Price）**: 采购成本价（不对外同步）

```go
	return &PriceSyncBiz{}

// PushPrice 推送价格到外部平台
func (b *PriceSyncBiz) PushPrice(db *gorm.DB, mappingID uint, priceType string, operatorID uint, operatorName string) error {
	// 获取商品映射
	var mapping model.ProductSyncMapping
	if err := db.Preload("Platform").Preload("Sku").First(&mapping, mappingID).Error; err != nil {
		return fmt.Errorf("映射不存在: %w", err)
	}
	
	// 检查是否启用价格同步
	if !mapping.SyncPrice {
		return fmt.Errorf("该映射未启用价格同步")
	}
	
	// 获取要同步的价格
	price, err := b.getPrice(&mapping, priceType)
	if err != nil {
		return err
	}
	
	// 推送到外部平台
	err = b.pushPriceToPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID, price, priceType)
	if err != nil {
		// 记录失败
		b.recordSyncFailure(db, &mapping, price, priceType, "push", err, operatorID, operatorName)
		return err
	}
	
	// 记录成功
	b.recordSyncSuccess(db, &mapping, decimal.Zero, price, priceType, "push", operatorID, operatorName)
	
	return nil
}

// getPrice 获取指定类型的价格
func (b *PriceSyncBiz) getPrice(mapping *model.ProductSyncMapping, priceType string) (decimal.Decimal, error) {
	switch priceType {
	case "market":
		return mapping.Sku.MarketPrice, nil
	case "member":
		return mapping.Sku.MemberPrice, nil
	case "cost":
		return decimal.Zero, fmt.Errorf("成本价不允许同步")
	default:
		return decimal.Zero, fmt.Errorf("不支持的价格类型: %s", priceType)
	}
}

// PushPriceBatch 批量推送价格
func (b *PriceSyncBiz) PushPriceBatch(db *gorm.DB, platformID uint, shopID string, priceType string, operatorID uint, operatorName string) error {
	// 获取所有启用价格同步的映射
	var mappings []model.ProductSyncMapping
	err := db.Where("platform_id = ? AND shop_id = ? AND sync_price = ? AND status = ?",
		platformID, shopID, true, "active").
		Preload("Sku").
		Find(&mappings).Error
	
	if err != nil {
		return err
	}
	
	successCount := 0
	failedCount := 0
	
	for _, mapping := range mappings {
		if err := b.PushPrice(db, mapping.ID, priceType, operatorID, operatorName); err != nil {
			failedCount++
			log.Printf("推送价格失败 [映射ID=%d]: %v", mapping.ID, err)
			continue
		}
		successCount++
	}
	
	log.Printf("批量推送价格完成: 成功=%d, 失败=%d", successCount, failedCount)
	return nil
}
```

```go
// OnPriceChanged 价格变化事件处理
func (b *PriceSyncBiz) OnPriceChanged(db *gorm.DB, skuID uint, priceType string, oldPrice, newPrice decimal.Decimal) error {
	// 查找该SKU的所有映射
	var mappings []model.ProductSyncMapping
	db.Where("sku_id = ? AND sync_price = ? AND enable_sync = ? AND status = ?",
		skuID, true, true, "active").
		Preload("Platform").
		Find(&mappings)
	
	// 逐个推送
	for _, mapping := range mappings {
		go func(m model.ProductSyncMapping) {
			err := b.pushPriceToPlatform(m.Platform, m.ShopID, m.ExternalSkuID, newPrice, priceType)
			if err != nil {
				log.Printf("自动推送价格失败 [映射ID=%d]: %v", m.ID, err)
				b.recordSyncFailure(db, &m, newPrice, priceType, "push", err, 0, "系统自动")
			} else {
				b.recordSyncSuccess(db, &m, oldPrice, newPrice, priceType, "push", 0, "系统自动")
			}
		}(mapping)
	}
	
	return nil
}

// SKU模型Hook
func (s *Sku) AfterUpdate(tx *gorm.DB) error {
	priceSyncBiz := NewPriceSyncBiz()
	
	// 检查市场价是否变化
	if tx.Statement.Changed("MarketPrice") {
		oldPrice := decimal.Zero // 从tx.Statement获取旧值
		go priceSyncBiz.OnPriceChanged(tx, s.ID, "market", oldPrice, s.MarketPrice)
	}
	
	// 检查会员价是否变化
	if tx.Statement.Changed("MemberPrice") {
		oldPrice := decimal.Zero
		go priceSyncBiz.OnPriceChanged(tx, s.ID, "member", oldPrice, s.MemberPrice)
	}
	
	return nil
}
```

```go
func (b *PriceSyncBiz) PullPrice(db *gorm.DB, mappingID uint, operatorID uint, operatorName string) error {
	// 获取映射
	var mapping model.ProductSyncMapping
	if err := db.Preload("Platform").Preload("Sku").First(&mapping, mappingID).Error; err != nil {
		return fmt.Errorf("映射不存在: %w", err)
	}
	
	// 从平台获取价格
	externalPrice, err := b.fetchPriceFromPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID)
	if err != nil {
		b.recordSyncFailure(db, &mapping, decimal.Zero, "market", "pull", err, operatorID, operatorName)
		return err
	}
	
	// 更新本地价格
	oldPrice := mapping.Sku.MarketPrice
	
	err = db.Model(&mapping.Sku).Update("market_price", externalPrice).Error
	if err != nil {
		return err
	}
	
	// 记录同步成功
	b.recordSyncSuccess(db, &mapping, oldPrice, externalPrice, "market", "pull", operatorID, operatorName)
	
	return nil
}

// PullPriceBatch 批量拉取价格

		platformID, shopID, true, true).
		Preload("Sku").
		Find(&mappings)
	
	successCount := 0
	failedCount := 0
	
	for _, mapping := range mappings {
		if err := b.PullPrice(db, mapping.ID, operatorID, operatorName); err != nil {
			failedCount++
			continue
		}
		successCount++
	}
	
	log.Printf("批量拉取价格完成: 成功=%d, 失败=%d", successCount, failedCount)
	return nil
```

## 4. 价格策略管理

### 4.1 价格策略配置

```go
// PriceStrategy 价格策略
type PriceStrategy struct {
	ID uint `gorm:"primaryKey" json:"id"`
	
	Name        string `gorm:"type:varchar(100);not null" json:"name"`
	Description string `gorm:"type:text" json:"description"`
	
	// 策略类型
	StrategyType string `gorm:"type:varchar(20);not null" json:"strategy_type"` // fixed, markup, discount
	
	// 固定价格策略
	FixedPrice decimal.Decimal `gorm:"type:decimal(10,2)" json:"fixed_price"`
	
	// 加价策略
	MarkupType    string          `gorm:"type:varchar(20)" json:"markup_type"`    // percentage, amount
	MarkupValue   decimal.Decimal `gorm:"type:decimal(10,2)" json:"markup_value"` // 加价值
	
	// 折扣策略
	DiscountType  string          `gorm:"type:varchar(20)" json:"discount_type"`  // percentage, amount
	DiscountValue decimal.Decimal `gorm:"type:decimal(10,2)" json:"discount_value"` // 折扣值
	
	// 价格取整
	RoundingType string          `gorm:"type:varchar(20)" json:"rounding_type"` // none, up, down, nearest
	RoundingUnit decimal.Decimal `gorm:"type:decimal(10,2)" json:"rounding_unit"` // 取整单位（如0.9表示x.9）
	
	// 价格区间限制
	MinPrice decimal.Decimal `gorm:"type:decimal(10,2)" json:"min_price"`
	MaxPrice decimal.Decimal `gorm:"type:decimal(10,2)" json:"max_price"`
	
	Status    string    `gorm:"type:varchar(20);default:'active'" json:"status"`
	CreatedAt time.Time `json:"created_at"`
	UpdatedAt time.Time `json:"updated_at"`
}

// ApplyStrategy 应用价格策略
func (b *PriceSyncBiz) ApplyStrategy(basePrice decimal.Decimal, strategy *PriceStrategy) (decimal.Decimal, error) {
	var finalPrice decimal.Decimal
	
	switch strategy.StrategyType {
	case "fixed":
		// 固定价格
		finalPrice = strategy.FixedPrice
		
	case "markup":
		// 加价策略
		if strategy.MarkupType == "percentage" {
			// 百分比加价
			markup := basePrice.Mul(strategy.MarkupValue).Div(decimal.NewFromInt(100))
			finalPrice = basePrice.Add(markup)
		} else {
			// 固定金额加价
			finalPrice = basePrice.Add(strategy.MarkupValue)
		}
		
	case "discount":
		// 折扣策略
		if strategy.DiscountType == "percentage" {
			// 百分比折扣
			discount := basePrice.Mul(strategy.DiscountValue).Div(decimal.NewFromInt(100))
			finalPrice = basePrice.Sub(discount)
		} else {
			// 固定金额折扣
			finalPrice = basePrice.Sub(strategy.DiscountValue)
		}
		
	default:
		return decimal.Zero, fmt.Errorf("不支持的策略类型: %s", strategy.StrategyType)
	}
	
	// 应用取整
	finalPrice = b.applyRounding(finalPrice, strategy)
	
	// 应用价格区间限制
	if !strategy.MinPrice.IsZero() && finalPrice.LessThan(strategy.MinPrice) {
		finalPrice = strategy.MinPrice
	}
	if !strategy.MaxPrice.IsZero() && finalPrice.GreaterThan(strategy.MaxPrice) {
		finalPrice = strategy.MaxPrice
	}
	
	return finalPrice, nil
}

// applyRounding 应用价格取整
func (b *PriceSyncBiz) applyRounding(price decimal.Decimal, strategy *PriceStrategy) decimal.Decimal {
	switch strategy.RoundingType {
	case "up":
		// 向上取整
		return price.Ceil()
	case "down":
		// 向下取整
		return price.Floor()
	case "nearest":
		// 四舍五入
		return price.Round(0)
	case "unit":
		// 按单位取整（如取到x.9）
		if !strategy.RoundingUnit.IsZero() {
			intPart := price.Floor()
			return intPart.Add(strategy.RoundingUnit)
		}
		return price
	default:
		return price
	}
}
```

```go
// PushPriceWithStrategy 应用策略后推送价格

	var mapping model.ProductSyncMapping

		return err
	}

	b.recordSyncSuccess(db, &mapping, basePrice, finalPrice, "market", "push", operatorID, operatorName)
	
	return nil
}
```

### 6.1 价格异常检测

```go
// DetectPriceAnomaly 检测价格异常
func (b *PriceSyncBiz) DetectPriceAnomaly(db *gorm.DB, skuID uint, newPrice decimal.Decimal) (bool, string, error) {
	var sku model.Sku
	if err := db.First(&sku, skuID).Error; err != nil {
		return false, "", err
	}
	
	oldPrice := sku.MarketPrice
	
	// 计算价格变化百分比
	if oldPrice.IsZero() {
		return false, "", nil
	}
	
	changePercent := newPrice.Sub(oldPrice).Div(oldPrice).Mul(decimal.NewFromInt(100))
	
	// 价格上涨超过30%
	if changePercent.GreaterThan(decimal.NewFromInt(30)) {
		return true, fmt.Sprintf("价格上涨%.2f%%，超过30%%阈值", changePercent), nil
	}
	
	// 价格下降超过50%
	if changePercent.LessThan(decimal.NewFromInt(-50)) {
		return true, fmt.Sprintf("价格下降%.2f%%，超过50%%阈值", changePercent), nil
	}
	
	// 价格低于成本价
	if newPrice.LessThan(sku.CostPrice) {
		return true, "价格低于成本价", nil
	}
	
	return false, "", nil
}

// ValidatePrice 价格校验
func (b *PriceSyncBiz) ValidatePrice(db *gorm.DB, mappingID uint, price decimal.Decimal) error {
	var mapping model.ProductSyncMapping
	db.Preload("Sku").First(&mapping, mappingID)
	
	// 检查价格是否为负
	if price.LessThanOrEqual(decimal.Zero) {
		return fmt.Errorf("价格不能为0或负数")
	}
	
	// 检测价格异常
	isAnomaly, reason, err := b.DetectPriceAnomaly(db, mapping.SkuID, price)
	if err != nil {
		return err
	}
	
	if isAnomaly {
		// 发送预警通知
		b.sendPriceAlert(mapping.SkuID, price, reason)
		return fmt.Errorf("价格异常: %s", reason)
	}
	
	return nil
}
```

```go
// ReconcilePrice 价格对账
func (b *PriceSyncBiz) ReconcilePrice(db *gorm.DB, platformID uint, shopID string) ([]PriceDiff, error) {
	// 获取所有映射
	var mappings []model.ProductSyncMapping
	db.Where("platform_id = ? AND shop_id = ? AND sync_price = ?", platformID, shopID, true).
		Preload("Sku").
		Find(&mappings)
	
	var diffs []PriceDiff
	
	for _, mapping := range mappings {
		// 获取外部价格
		externalPrice, err := b.fetchPriceFromPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID)
		if err != nil {
			log.Printf("获取外部价格失败 [映射ID=%d]: %v", mapping.ID, err)
			continue
		}
		
		// 比较价格
		localPrice := mapping.Sku.MarketPrice
		if !localPrice.Equal(externalPrice) {
			diffs = append(diffs, PriceDiff{
				MappingID:     mapping.ID,
				SkuCode:       mapping.SkuCode,
				LocalPrice:    localPrice,
				ExternalPrice: externalPrice,
				Difference:    localPrice.Sub(externalPrice),
			})
		}
	}
	
	return diffs, nil
}

// PriceDiff 价格差异
type PriceDiff struct {
	MappingID     uint
	SkuCode       string
	LocalPrice    decimal.Decimal
	ExternalPrice decimal.Decimal
	Difference    decimal.Decimal
}

// FixPriceDiff 修复价格差异
func (b *PriceSyncBiz) FixPriceDiff(db *gorm.DB, diff *PriceDiff, strategy string) error {
	switch strategy {
	case "push":
		// 以本地为准，推送到平台
		return b.PushPrice(db, diff.MappingID, "market", 0, "系统对账")
	case "pull":
		// 以平台为准，拉取到本地
		return b.PullPrice(db, diff.MappingID, 0, "系统对账")
	default:
		return fmt.Errorf("不支持的修复策略: %s", strategy)
	}
}
```

```go
// CreatePromotion 创建促销
func (b *PriceSyncBiz) CreatePromotion(db *gorm.DB, req *api.ReqCreatePromotion) (*PromotionPrice, error) {
	promotion := &PromotionPrice{
		MappingID:      req.MappingID,
		SkuID:          req.SkuID,
		OriginalPrice:  req.OriginalPrice,
		PromotionPrice: req.PromotionPrice,
		StartTime:      req.StartTime,
		EndTime:        req.EndTime,
		Status:         "active",
	}
	
	if err := db.Create(promotion).Error; err != nil {
		return nil, err
	}
	
	// 推送促销价格到平台
	go b.syncPromotionPrice(db, promotion)
	
	return promotion, nil
}

// syncPromotionPrice 同步促销价格
func (b *PriceSyncBiz) syncPromotionPrice(db *gorm.DB, promotion *PromotionPrice) error {
	var mapping model.ProductSyncMapping
	db.Preload("Platform").First(&mapping, promotion.MappingID)
	
	// 在促销开始时间推送促销价
	time.Sleep(time.Until(promotion.StartTime))
	b.pushPriceToPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID, promotion.PromotionPrice, "market")
	
	// 在促销结束时间恢复原价
	time.Sleep(time.Until(promotion.EndTime))
	b.pushPriceToPlatform(mapping.Platform, mapping.ShopID, mapping.ExternalSkuID, promotion.OriginalPrice, "market")
	
	return nil
}
```

```go
func (b *PriceSyncBiz) recordSyncFailure(db *gorm.DB, mapping *model.ProductSyncMapping, price decimal.Decimal, priceType, direction string, err error, operatorID uint, operatorName string) error {

		PriceType:     priceType,
		SyncDirection: direction,
		SyncStatus:    "failed",

		ErrorMessage:  err.Error(),
```

```go
// BatchPushWithGrouping 分组批量推送价格
func (b *PriceSyncBiz) BatchPushWithGrouping(db *gorm.DB, mappingIDs []uint, priceType string, operatorID uint, operatorName string) error {
	// 获取所有映射
	var mappings []model.ProductSyncMapping
	db.Where("id IN ?", mappingIDs).
		Preload("Platform").
		Preload("Sku").
		Find(&mappings)
	
	// 按平台分组
	platformGroups := make(map[uint][]model.ProductSyncMapping)
	for _, mapping := range mappings {
		platformGroups[mapping.PlatformID] = append(platformGroups[mapping.PlatformID], mapping)
	}
	
	// 逐平台处理
	for platformID, group := range platformGroups {
		// 调用平台批量接口
		if err := b.batchPushPriceToPlatform(platformID, group, priceType); err != nil {
			log.Printf("批量推送价格失败 [平台ID=%d]: %v", platformID, err)
			continue
		}
		
		// 记录同步结果
		for _, mapping := range group {
			price, _ := b.getPrice(&mapping, priceType)
			b.recordSyncSuccess(db, &mapping, decimal.Zero, price, priceType, "push", operatorID, operatorName)
		}
	}
	
	return nil
}
```
