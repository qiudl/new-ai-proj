# 业绩奖金数据库Schema设计文档

## 1. 概述

本文档定义Coolbuy 3.0业绩奖金模块的数据库Schema设计，包括佣金模板、业绩记录、奖金发放等核心数据结构。

### 1.1 设计原则

- **多租户隔离**: 使用PostgreSQL Schema实现租户数据隔离
- **灵活配置**: 支持多种佣金计算规则和模板
- **精确计算**: 使用decimal类型确保金额计算精度
- **审计追踪**: 完整的业绩和佣金记录
- **性能优化**: 合理的索引和分区策略
- **统计聚合**: 支持多维度业绩统计

### 1.2 核心表结构

```
业绩奖金模块核心表:
├── commission_templates (佣金模板表)
├── commission_rules (佣金规则表)
├── performance_records (业绩记录表)
├── commission_records (佣金记录表)
├── bonus_distributions (奖金发放表)
├── performance_statistics (业绩统计表)
└── commission_logs (佣金操作日志表)
```

## 2. 数据模型设计

### 2.1 佣金模板表 (commission_templates)

**表说明**: 佣金模板配置表，定义不同的佣金计算方案

```go
package model

import (
	"time"
	"github.com/shopspring/decimal"
	"gorm.io/gorm"
)

// CommissionTemplate 佣金模板模型
type CommissionTemplate struct {
	ID uint `gorm:"primaryKey;autoIncrement" json:"id"`
	
	// 基本信息
	TemplateName   string `gorm:"type:varchar(100);not null;index" json:"template_name"`       // 模板名称
	TemplateCode   string `gorm:"type:varchar(50);uniqueIndex;not null" json:"template_code"`  // 模板代码
	TemplateType   string `gorm:"type:varchar(20);not null;index" json:"template_type"`        // 模板类型: sales(销售额), profit(利润), quantity(数量), mixed(混合)
	Description    string `gorm:"type:text" json:"description"`                                 // 描述
	
	// 计算方式
	CalculateMethod string `gorm:"type:varchar(20);not null" json:"calculate_method"` // 计算方式: percentage(百分比), fixed(固定金额), tiered(阶梯)
	
	// 适用范围
	ApplicableRole  string `gorm:"type:varchar(50)" json:"applicable_role"`  // 适用角色: salesperson(销售), manager(经理), director(总监)
	ApplicableLevel string `gorm:"type:varchar(50)" json:"applicable_level"` // 适用级别
	
	// 时间范围
	EffectiveFrom   time.Time  `gorm:"not null" json:"effective_from"`  // 生效开始时间
	EffectiveTo     *time.Time `json:"effective_to"`                    // 生效结束时间
	
	// 状态
	Status          string `gorm:"type:varchar(20);not null;default:'active';index" json:"status"` // 状态: active(启用), inactive(停用), draft(草稿)
	IsDefault       bool   `gorm:"default:false" json:"is_default"`                                 // 是否默认模板
	
	// 配置JSON
	ConfigJSON      string `gorm:"type:jsonb" json:"config_json"` // 模板配置(JSON格式)
	
	// 审计信息
	CreatedBy       uint   `gorm:"index" json:"created_by"`       // 创建人ID
	CreatedByName   string `gorm:"type:varchar(100)" json:"created_by_name"` // 创建人姓名
	UpdatedBy       uint   `gorm:"index" json:"updated_by"`       // 更新人ID
	UpdatedByName   string `gorm:"type:varchar(100)" json:"updated_by_name"` // 更新人姓名
	
	CreatedAt       time.Time      `json:"created_at"`
	UpdatedAt       time.Time      `json:"updated_at"`
	DeletedAt       gorm.DeletedAt `gorm:"index" json:"-"`
	
	// 关联关系
	Rules []CommissionRule `gorm:"foreignKey:TemplateID" json:"rules,omitempty"`
}

// TableName 指定表名
func (CommissionTemplate) TableName() string {
	return "commission_templates"
}

// IsActive 判断模板是否有效
func (t *CommissionTemplate) IsActive() bool {
	now := time.Now()
	if t.Status != "active" {
		return false
	}
	if now.Before(t.EffectiveFrom) {
		return false
	}
	if t.EffectiveTo != nil && now.After(*t.EffectiveTo) {
		return false
	}
	return true
}
```

### 2.2 佣金规则表 (commission_rules)

**表说明**: 佣金计算规则明细表

```go
// CommissionRule 佣金规则模型
type CommissionRule struct {
	ID uint `gorm:"primaryKey;autoIncrement" json:"id"`
	
	TemplateID      uint   `gorm:"index;not null" json:"template_id"`      // 模板ID
	TemplateCode    string `gorm:"type:varchar(50);index" json:"template_code"` // 模板代码
	
	// 规则信息
	RuleName        string          `gorm:"type:varchar(100);not null" json:"rule_name"`    // 规则名称
	RuleType        string          `gorm:"type:varchar(20);not null" json:"rule_type"`     // 规则类型: tier(阶梯), threshold(门槛), bonus(奖励)
	Priority        int             `gorm:"default:0" json:"priority"`                       // 优先级（数字越大优先级越高）
	
	// 条件配置
	MinAmount       decimal.Decimal `gorm:"type:decimal(15,2);default:0" json:"min_amount"` // 最小金额/数量
	MaxAmount       decimal.Decimal `gorm:"type:decimal(15,2);default:0" json:"max_amount"` // 最大金额/数量（0表示无上限）
	
	// 佣金计算
	CommissionRate  decimal.Decimal `gorm:"type:decimal(10,4);default:0" json:"commission_rate"`  // 佣金比例
	FixedAmount     decimal.Decimal `gorm:"type:decimal(10,2);default:0" json:"fixed_amount"`     // 固定金额
	BonusAmount     decimal.Decimal `gorm:"type:decimal(10,2);default:0" json:"bonus_amount"`     // 奖励金额
	
	// 附加条件JSON
	ConditionsJSON  string `gorm:"type:jsonb" json:"conditions_json"` // 附加条件配置
	
	Status          string `gorm:"type:varchar(20);not null;default:'active'" json:"status"` // 状态
	
	CreatedAt       time.Time      `json:"created_at"`
	UpdatedAt       time.Time      `json:"updated_at"`
	DeletedAt       gorm.DeletedAt `gorm:"index" json:"-"`
	
	// 关联关系
	Template *CommissionTemplate `gorm:"foreignKey:TemplateID" json:"template,omitempty"`
}

// TableName 指定表名
func (CommissionRule) TableName() string {
	return "commission_rules"
}

// IsApplicable 判断规则是否适用于给定金额
func (r *CommissionRule) IsApplicable(amount decimal.Decimal) bool {
	if amount.LessThan(r.MinAmount) {
		return false
	}
	if !r.MaxAmount.IsZero() && amount.GreaterThan(r.MaxAmount) {
		return false
	}
	return true
}
```

**SQL DDL**:
```sql

    id SERIAL PRIMARY KEY,
    template_id INTEGER NOT NULL,
    template_code VARCHAR(50),

    min_amount DECIMAL(15,2) DEFAULT 0,
    max_amount DECIMAL(15,2) DEFAULT 0,
    commission_rate DECIMAL(10,4) DEFAULT 0,
    fixed_amount DECIMAL(10,2) DEFAULT 0,
    bonus_amount DECIMAL(10,2) DEFAULT 0,
    conditions_json JSONB,
    status VARCHAR(20) NOT NULL DEFAULT 'active',
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    
    FOREIGN KEY (template_id) REFERENCES commission_templates(id),
    INDEX idx_commission_rules_template_id (template_id),
    INDEX idx_commission_rules_deleted_at (deleted_at)
);

CREATE INDEX idx_commission_rules_conditions ON commission_rules USING GIN (conditions_json);

COMMENT ON TABLE commission_rules IS '佣金规则表';
```

```go
	UserRole        string `gorm:"type:varchar(50);index" json:"user_role"`     // 员工角色
	DepartmentID    uint   `gorm:"index" json:"department_id"`             // 部门ID
	DepartmentName  string `gorm:"type:varchar(100)" json:"department_name"`    // 部门名称
	
	// 订单信息
	OrderID         uint   `gorm:"index;not null" json:"order_id"`         // 订单ID
	OrderNo         string `gorm:"type:varchar(50);index;not null" json:"order_no"` // 订单号
	OrderDate       time.Time `gorm:"index;not null" json:"order_date"`     // 订单日期
	
	// 业绩金额
	SalesAmount     decimal.Decimal `gorm:"type:decimal(15,2);not null" json:"sales_amount"`     // 销售额
	CostAmount      decimal.Decimal `gorm:"type:decimal(15,2);not null" json:"cost_amount"`      // 成本额
	ProfitAmount    decimal.Decimal `gorm:"type:decimal(15,2);not null" json:"profit_amount"`    // 利润额
	DiscountAmount  decimal.Decimal `gorm:"type:decimal(15,2);default:0" json:"discount_amount"` // 折扣额
	
	// 业绩类型
	PerformanceType string `gorm:"type:varchar(20);not null;index" json:"performance_type"` // 业绩类型: sales(销售), return(退货), exchange(换货)
	
	// 业绩权重
	PerformanceWeight decimal.Decimal `gorm:"type:decimal(5,2);default:1.00" json:"performance_weight"` // 业绩权重（默认1.00）
	
	// 统计周期
	StatPeriod      string `gorm:"type:varchar(20);index" json:"stat_period"`    // 统计周期: YYYY-MM-DD, YYYY-MM, YYYY-Www
	StatYear        int    `gorm:"index" json:"stat_year"`                       // 统计年份
	StatMonth       int    `gorm:"index" json:"stat_month"`                      // 统计月份
	StatWeek        int    `gorm:"index" json:"stat_week"`                       // 统计周次
	
	// 状态
	Status          string `gorm:"type:varchar(20);not null;default:'confirmed';index" json:"status"` // 状态: confirmed(已确认), cancelled(已取消), adjusted(已调整)
	
	// 备注
	Remark          string `gorm:"type:text" json:"remark"` // 备注
	
	CreatedAt       time.Time      `json:"created_at"`
	UpdatedAt       time.Time      `json:"updated_at"`
	DeletedAt       gorm.DeletedAt `gorm:"index" json:"-"`
	
	// 关联关系
	CommissionRecords []CommissionRecord `gorm:"foreignKey:PerformanceRecordID" json:"commission_records,omitempty"`
}

// TableName 指定表名
func (PerformanceRecord) TableName() string {
	return "performance_records"
}

// GetEffectiveAmount 获取有效业绩金额
func (p *PerformanceRecord) GetEffectiveAmount(amountType string) decimal.Decimal {
	amount := decimal.Zero
	switch amountType {
	case "sales":
		amount = p.SalesAmount
	case "profit":
		amount = p.ProfitAmount
	case "quantity":
		// 如果是数量类型，这里返回1作为权重
		amount = decimal.NewFromInt(1)
	}
	
	// 应用业绩权重
	return amount.Mul(p.PerformanceWeight)
}
```
