# 仓库管理功能设计文档

## 1. 概述

### 1.1 功能目标

本文档描述了Coolbuy 3.0仓库管理功能的详细设计，实现:

- ✅ **仓库全生命周期管理**: 创建、编辑、启用/停用、删除
- ✅ **仓库层级结构**: 仓库 → 库区 → 库位三级管理
- ✅ **仓库信息维护**: 基本信息、联系方式、区域划分
- ✅ **仓库状态管理**: 启用/停用、锁定/解锁
- ✅ **容量监控**: 实时监控仓库使用率
- ✅ **权限控制**: 基于角色的仓库访问控制

### 1.2 核心功能模块

| 模块 | 功能 | 优先级 |
|------|------|--------|
| 仓库管理 | 仓库CRUD、状态管理 | P0 |
| 库区管理 | 库区CRUD、容量管理 | P0 |
| 库位管理 | 库位CRUD、位置编码 | P0 |
| 仓库配置 | 业务规则、预警设置 | P1 |
| 仓库统计 | 使用率、容量分析 | P1 |
| 仓库监控 | 实时状态、告警 | P2 |

```go
func (b *WarehouseBiz) List(db *gorm.DB, req *ReqListWarehouse) (*RespListWarehouse, error) {
    var warehouses []*model.Warehouse
    var count int64
    
    query := db.Model(&model.Warehouse{})
    
    // 筛选条件
    if req.Code != "" {
        query = query.Where("code LIKE ?", "%"+req.Code+"%")
    }
    if req.Name != "" {
        query = query.Where("name LIKE ?", "%"+req.Name+"%")
    }
    if req.Type != "" {
        query = query.Where("warehouse_type = ?", req.Type)
    }
    if req.Status != nil {
        query = query.Where("status = ?", *req.Status)
    }
    if req.Province != "" {
        query = query.Where("province = ?", req.Province)
    }
    if req.City != "" {
        query = query.Where("city = ?", req.City)
    }
    
    // 统计总数
    query.Count(&count)
    
    // 排序
    sortBy := "created_at"
    if req.SortBy != "" {
        sortBy = req.SortBy
    }
    sortOrder := "DESC"
    if req.SortOrder == "asc" {
        sortOrder = "ASC"
    }
    query = query.Order(fmt.Sprintf("%s %s", sortBy, sortOrder))
    
    // 分页
    page := 1
    if req.Page > 0 {
        page = req.Page
    }
    pageSize := 20
    if req.PageSize > 0 {
        pageSize = req.PageSize
    }
    offset := (page - 1) * pageSize
    
    err := query.Offset(offset).Limit(pageSize).Find(&warehouses).Error
    if err != nil {
        return nil, err
    }
    
    // 转换为响应格式
    data := make([]*WarehouseSimple, 0, len(warehouses))
    for _, wh := range warehouses {
        data = append(data, b.convertToSimple(db, wh))
    }
    
    return &RespListWarehouse{
        Count: int(count),
        Data:  data,
    }, nil
}
```

```go
func (b *WarehouseBiz) Validate(req *ReqCreateWarehouse) error {

    var count int64

        return errors.New("仓库编码已存在")
    }

    if req.UsableArea > req.TotalArea {

    }
    
    // 3. 如果设置为默认仓库，需要取消其他默认仓库
    if req.IsDefault {
        // 将在事务中处理
    }
    
    // 4. 联系电话格式验证
    if req.ContactPhone != "" {
        if !isValidPhone(req.ContactPhone) {
            return errors.New("联系电话格式不正确")
        }
    }
    
    return nil
}
```

```go
func (b *WarehouseBiz) Create(db *gorm.DB, req *ReqCreateWarehouse) (*model.Warehouse, error) {
    // 参数校验
    if err := b.Validate(req); err != nil {
        return nil, err
    }
    
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 如果设置为默认仓库，取消其他默认仓库
        if req.IsDefault {
            tx.Model(&model.Warehouse{}).
                Where("is_default = ?", true).
                Update("is_default", false)
        }
        
        // 2. 创建仓库
        warehouse := &model.Warehouse{
            Code:          req.Code,
            Name:          req.Name,
            WarehouseType: req.Type,
            Status:        1, // 默认启用
            Province:      req.Province,
            City:          req.City,
            District:      req.District,
            Address:       req.Address,
            ContactPerson: req.ContactPerson,
            ContactPhone:  req.ContactPhone,
            TotalArea:     req.TotalArea,
            UsableArea:    req.UsableArea,
            IsDefault:     req.IsDefault,
            SortOrder:     req.SortOrder,
            Remark:        req.Remark,
        }
        
        if err := tx.Create(warehouse).Error; err != nil {
            return err
        }
        
        // 3. 创建默认库区
        if req.CreateDefaultZones && len(req.DefaultZoneCodes) > 0 {
            for i, code := range req.DefaultZoneCodes {
                zone := &model.WarehouseZone{
                    WarehouseID: warehouse.ID,
                    Code:        code,
                    Name:        code + "区",
                    ZoneType:    "storage",
                    Status:      1,
                    SortOrder:   i + 1,
                }
                if err := tx.Create(zone).Error; err != nil {
                    return err
                }
            }
        }
        
        return nil
    })
}
```

```go
func (b *WarehouseBiz) Update(db *gorm.DB, id uint, req *ReqUpdateWarehouse) error {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 检查仓库是否存在
        warehouse, err := b.store.FindByID(tx, id)
        if err != nil {
            return err
        }
        
        // 2. 如果设置为默认仓库
        if req.IsDefault != nil && *req.IsDefault {
            tx.Model(&model.Warehouse{}).
                Where("is_default = ? AND id != ?", true, id).
                Update("is_default", false)
        }
        
        // 3. 更新仓库信息
        updates := make(map[string]interface{})
        if req.Name != nil {
            updates["name"] = *req.Name
        }
        if req.Type != nil {
            updates["warehouse_type"] = *req.Type
        }
        if req.Status != nil {
            // 检查是否可以停用
            if *req.Status == 0 {
                if err := b.checkCanDisable(tx, id); err != nil {
                    return err
                }
            }
            updates["status"] = *req.Status
        }
        // ... 其他字段更新
        
        return tx.Model(warehouse).Updates(updates).Error
    })
}

// 检查是否可以停用仓库
func (b *WarehouseBiz) checkCanDisable(db *gorm.DB, id uint) error {
    // 1. 检查是否有未完成的出入库单
    var pendingOrders int64
    db.Model(&model.InboundOrder{}).
        Where("warehouse_id = ? AND status NOT IN ?", id, []string{"completed", "cancelled"}).
        Count(&pendingOrders)
    if pendingOrders > 0 {
        return errors.New("存在未完成的入库单，无法停用")
    }
    
    db.Model(&model.OutboundOrder{}).
        Where("warehouse_id = ? AND status NOT IN ?", id, []string{"shipped", "cancelled"}).
        Count(&pendingOrders)
    if pendingOrders > 0 {
        return errors.New("存在未完成的出库单，无法停用")
    }
    
    // 2. 检查是否有库存
    var inventoryCount int64
    db.Model(&model.Inventory{}).
        Where("warehouse_id = ? AND quantity > 0", id).
        Count(&inventoryCount)
    if inventoryCount > 0 {
        return errors.New("仓库中还有库存，无法停用")
    }
    
    return nil
}
```

### 2.5 删除仓库

#### 2.5.1 功能描述

软删除仓库，需要检查关联数据

#### 2.5.2 业务逻辑

```go
func (b *WarehouseBiz) Delete(db *gorm.DB, id uint) error {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 检查仓库是否存在
        warehouse, err := b.store.FindByID(tx, id)
        if err != nil {
            return err
        }
        
        // 2. 检查是否可以删除
        if err := b.checkCanDelete(tx, id); err != nil {
            return err
        }
        
        // 3. 软删除库位
        tx.Where("warehouse_id = ?", id).Delete(&model.WarehouseLocation{})
        
        // 4. 软删除库区
        tx.Where("warehouse_id = ?", id).Delete(&model.WarehouseZone{})
        
        // 5. 软删除仓库
        return tx.Delete(warehouse).Error
    })
}

func (b *WarehouseBiz) checkCanDelete(db *gorm.DB, id uint) error {
    // 1. 不能删除默认仓库
    var warehouse model.Warehouse
    db.First(&warehouse, id)
    if warehouse.IsDefault {
        return errors.New("不能删除默认仓库")
    }
    
    // 2. 检查是否有库存
    var inventoryCount int64
    db.Model(&model.Inventory{}).
        Where("warehouse_id = ?", id).
        Count(&inventoryCount)
    if inventoryCount > 0 {
        return errors.New("仓库中还有库存记录，无法删除")
    }
    
    // 3. 检查是否有出入库记录（近6个月）
    sixMonthsAgo := time.Now().AddDate(0, -6, 0)
    var recentOrders int64
    db.Model(&model.InboundOrder{}).
        Where("warehouse_id = ? AND created_at > ?", id, sixMonthsAgo).
        Count(&recentOrders)
    if recentOrders > 0 {
        return errors.New("仓库有近期出入库记录，无法删除")
    }
    
    return nil
}
```

```go
func (b *ZoneBiz) Create(db *gorm.DB, req *ReqCreateZone) (*model.WarehouseZone, error) {
    return db.Transaction(func(tx *gorm.DB) error {
        // 1. 检查仓库是否存在
        var warehouse model.Warehouse
        if err := tx.First(&warehouse, req.WarehouseID).Error; err != nil {
            return errors.New("仓库不存在")
        }
        
        // 2. 检查库区编码唯一性
        var count int64
        tx.Model(&model.WarehouseZone{}).
            Where("warehouse_id = ? AND code = ?", req.WarehouseID, req.Code).
            Count(&count)
        if count > 0 {
            return errors.New("库区编码已存在")
        }
        
        // 3. 创建库区
        zone := &model.WarehouseZone{
            WarehouseID:      req.WarehouseID,
            Code:             req.Code,
            Name:             req.Name,
            ZoneType:         req.Type,
            Status:           1,
            TemperatureRange: req.TemperatureRange,
            HumidityRange:    req.HumidityRange,
            Area:             req.Area,
            Capacity:         req.Capacity,
            SortOrder:        req.SortOrder,
            Remark:           req.Remark,
        }
        
        if err := tx.Create(zone).Error; err != nil {
            return err
        }
        
        // 4. 自动生成库位
        if req.AutoCreateLocations {
            if err := b.generateLocations(tx, zone.ID, req); err != nil {
                return err
            }
        }
        
        return nil
    })
}

// 自动生成库位
func (b *ZoneBiz) generateLocations(db *gorm.DB, zoneID uint, req *ReqCreateZone) error {
    var zone model.WarehouseZone
    db.First(&zone, zoneID)
    
    for row := 1; row <= req.RowCount; row++ {
        for col := 1; col <= req.ColumnCount; col++ {
            for level := 1; level <= req.LevelCount; level++ {
                // 生成库位编码: A-01-02-03
                code := fmt.Sprintf("%s-%02d-%02d-%02d", zone.Code, row, col, level)
                
                location := &model.WarehouseLocation{
                    WarehouseID:  req.WarehouseID,
                    ZoneID:       &zoneID,
                    Code:         code,
                    Name:         code,
                    LocationType: "shelf",
                    RowNumber:    fmt.Sprintf("%02d", row),
                    ColumnNumber: fmt.Sprintf("%02d", col),
                    LevelNumber:  fmt.Sprintf("%02d", level),
                    Status:       1,
                }
                
                if err := db.Create(location).Error; err != nil {
                    return err
                }
            }
        }
    }
    
    return nil
}
```

```go
func (b *LocationBiz) Lock(db *gorm.DB, id uint, req *ReqLockLocation) error {
    return db.Model(&model.WarehouseLocation{}).
        Where("id = ?", id).
        Updates(map[string]interface{}{
            "is_locked":     true,
            "locked_reason": req.Reason,
        }).Error
}

func (b *LocationBiz) Unlock(db *gorm.DB, id uint) error {
    return db.Model(&model.WarehouseLocation{}).
        Where("id = ?", id).
        Updates(map[string]interface{}{
            "is_locked":     false,
            "locked_reason": "",
        }).Error
}
```

## 6. 业务规则

### 6.1 仓库编码规则

```
格式: WH{类型代码}{地区代码}{序号}
示例:
- WHNBJ01: 普通仓-北京-01号
- WHBBJ01: 保税仓-北京-01号
- WHTSH01: 中转仓-上海-01号

类型代码:
- N: Normal (普通仓)
- B: Bonded (保税仓)
- T: Transit (中转仓)
- V: Virtual (虚拟仓)
```

### 6.2 库区编码规则

```
格式: {大写字母}
示例: A, B, C, D ...

特殊库区:
- R: Return区(退货区)
- D: Damaged区(残次品区)
```

### 6.3 库位编码规则

```
格式: {库区}-{排号}-{列号}-{层号}
示例: A-01-02-03 (A区1排2列3层)

编号说明:
- 排号: 01-99 (从入口开始往里数)
- 列号: 01-99 (从左到右)
- 层号: 01-99 (从下到上)
```

```go
// Redis缓存仓库列表（15分钟）
func (b *WarehouseBiz) GetWarehouseList(redis *redis.Client) ([]*model.Warehouse, error) {
    cacheKey := "warehouse:list"
    
    // 尝试从缓存获取
    cached, err := redis.Get(context.Background(), cacheKey).Result()
    if err == nil {
        var warehouses []*model.Warehouse
        json.Unmarshal([]byte(cached), &warehouses)
        return warehouses, nil
    }
    
    // 从数据库查询
    var warehouses []*model.Warehouse
    db.Find(&warehouses)
    
    // 写入缓存
    data, _ := json.Marshal(warehouses)
    redis.Set(context.Background(), cacheKey, data, 15*time.Minute)
    
    return warehouses, nil
}
```
