- 移除`<iframe>`、`<object>`、`<embed>`标签

- 移除`javascript:`和`data:`协议

#### 1.2 HTML实体转义


#### 1.3 XSS检测模式
```go
patterns := []string{
    `<script[^>]*>.*?</script>`,      // Script标签
    `\s+on\w+\s*=`,                   // 事件处理器
    `javascript:`,                     // JavaScript协议
    `data:text/html`,                  // Data协议
    `<iframe`,                         // Iframe注入
    `<(object|embed)`,                 // Object和embed
    `<meta[^>]*http-equiv\s*=\s*["']?refresh`, // Meta refresh
    `base64.*<script`,                 // Base64编码的脚本
    `<img[^>]*onerror`,               // 常见的XSS payload
    `<svg[^>]*onload`,
    `<body[^>]*onload`,
}
```

```go
// 检测敏感词
func (s *contentSecurityServiceImpl) CheckSensitiveWords(ctx context.Context, content string) (*SensitiveWordCheckResult, error) {
    lowerContent := strings.ToLower(content)
    
    for word := range s.sensitiveWordsMap {
        if strings.Contains(lowerContent, word) {
            result.HasSensitiveWord = true
            result.SensitiveWords = append(result.SensitiveWords, word)
            
            // 替换为 ***
            replacement := strings.Repeat("*", utf8.RuneCountInString(word))
            result.CleanedContent = strings.ReplaceAll(result.CleanedContent, word, replacement)
        }
    }
    
    return result, nil
}
```

#### 5.3 自动审核流程

```go
func (s *commentModerationServiceImpl) AutoModerate(ctx context.Context, comment *models.RequirementComment) (*ModerationResult, error) {
    // 1. 内容安全检查
    sanitized, err := s.contentSecurityService.SanitizeContent(ctx, comment.Content)
    
    // 2. 如果检测到XSS攻击 → 自动拒绝
    if sanitized.HasXSSAttempt {
        result.AutoRejected = true
        result.RejectionNote = "检测到潜在的XSS攻击尝试"
        result.ModerationFlag = "xss_attempt"
        comment.Status = string(models.RequirementCommentStatusRejected)
        return result, nil
    }
    
    // 3. 如果包含敏感词 → 待人工审核
    if sanitized.HasSensitiveWord {
        result.NeedsReview = true
        result.ModerationFlag = "sensitive_word"
        comment.Status = string(models.RequirementCommentStatusPendingReview)
        return result, nil
    }
    
    // 4. 如果没有问题 → 自动批准
    result.AutoApproved = true
    comment.Status = string(models.RequirementCommentStatusActive)
    return result, nil
}
```

```go
// sanitizeHTML 清理HTML标签
func (s *contentSecurityServiceImpl) sanitizeHTML(content string) (string, []string) {
    removedTags := make([]string, 0)
    
    // 移除script标签
    scriptRegex := regexp.MustCompile(`(?i)<script[^>]*>.*?</script>`)
    content = scriptRegex.ReplaceAllString(content, "")
    
    // 移除事件处理器
    eventHandlerRegex := regexp.MustCompile(`(?i)\s+on\w+\s*=\s*["'][^"']*["']`)
    content = eventHandlerRegex.ReplaceAllString(content, "")
    
    // 移除javascript:协议
    javascriptProtocolRegex := regexp.MustCompile(`(?i)javascript:`)
    content = javascriptProtocolRegex.ReplaceAllString(content, "")
    
    return content, removedTags
}
```

## 安全防护流程图

```
用户提交评论
    ↓
频率限制中间件
    ↓ (通过)
Handler接收请求
    ↓
内容安全验证
    ├─ XSS检测 → 如果检测到 → 拒绝
    ├─ 敏感词检查
    └─ 长度验证
    ↓ (通过)
内容清理
    ├─ 移除危险HTML标签
    ├─ 转义HTML实体
    └─ 替换敏感词为***
    ↓
自动审核
    ├─ 检测到XSS → 自动拒绝 → 返回错误
    ├─ 包含敏感词 → 待审核状态 → 创建评论（不可见）
    └─ 无问题 → 自动批准 → 创建评论（可见）
    ↓
数据库存储（参数化查询）
    ↓
触发通知
    ↓
返回结果
```
