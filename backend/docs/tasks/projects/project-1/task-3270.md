### 1. 动态人数统计
不依赖静态 `employee_count` 字段，使用SQL实时统计：
```sql
SELECT d.*, COUNT(DISTINCT eu.id) as actual_employee_count
FROM enterprise_departments d
LEFT JOIN enterprise_users eu ON d.id = eu.department_id 
  AND eu.deleted_at IS NULL AND eu.status = 'active'
GROUP BY d.id
```

const flattenTree = (tree: Department[]): Department[] => {
  const result: Department[] = [];
  const flatten = (nodes: Department[]) => {
    nodes.forEach(node => {
      result.push(node);

    });
